17/*
18This file creates a mesh that represents a rectangular volume entering a
19cylinder laterally, perpendicular to its axis.
20
21The program is divided into two parts:
22
231)The function "detailed_circle" is first defined, inspired by
24hyper_ball in the deal.II library. This function creates a 2D coarse mesh of a
25circle which is adapted to be connected with a rectangle on its side.
26
272)The core of the file creates the 3D pipe assembled with the waveguide
28*/

31template <int dim, int spacedim>
32void
33GridPipeAndWaveguide<dim, spacedim>::detailed_circle(Triangulation<2, 2> &tria,
34 const Point<2> &p,
35 const double &a,
36 const double &b,
37 const double r,
38 const double R,
39 const bool inner_circle)
40{
41 const auto embed_point = [](const double x, const double y) -> Point<2> {
42 return Point<2>(x, y);
43 };
44
45 const auto vertices = [&]() -> std::vector<Point<2>> {
46 if (inner_circle)
47 // Inner circle
48 return {
49 p + embed_point(-1., -1.) * (r / std::numbers::sqrt2),
50 p + embed_point(+1., -1.) * (r / std::numbers::sqrt2),
51 p + embed_point(-1., +1.) * (r / std::numbers::sqrt2),
52 p + embed_point(+1., +1.) * (r / std::numbers::sqrt2),
53 // The four points which make the inner square
54 p + embed_point(-1., +1.) * (R / std::numbers::sqrt2),
55 p + embed_point(+1., +1.) * (R / std::numbers::sqrt2),
56 p + embed_point(-1., -1.) * (R / std::numbers::sqrt2),
57 p + embed_point(+1., -1.) * (R / std::numbers::sqrt2),
58 // Two points linked with the rectangle on is left
59 embed_point(b, 0),
60 embed_point(b, a),
61 // The two points (0, r) and (0, -r) from the center of the circle
62 p + embed_point(0, -r),
63 p + embed_point(0, +r),
64 // The two points on the right side of the outer circle that lie along
65 // the width of the rectangle
66 p + embed_point(std::sqrt(r * r - a * a / 4), -a / 2),
67 p + embed_point(std::sqrt(r * r - a * a / 4), +a / 2),
68 // The other points where the width of the rectangle intersects the
69 // central square
70 p + embed_point(-(std::sqrt(R * R - a * a / 4)), +a / 2),
71 p + embed_point(-(std::sqrt(R * R - a * a / 4)), -a / 2),
72 p + embed_point(+(std::sqrt(R * R - a * a / 4)), -a / 2),
73 p + embed_point(+(std::sqrt(R * R - a * a / 4)), +a / 2),
74 // The two points of the central square on the circle's axis of symmetry
75 p + embed_point(0, -R),
76 p + embed_point(0, +R),
77 // The two points where the width of the rectangle intersects the axis
78 // of symmetry of the circle
79 p + embed_point(0, +a / 2),
80 p + embed_point(0, -a / 2)};
81
82 else
83 // Inner square
84 return {
85 p + embed_point(-1., -1.) * (r / std::numbers::sqrt2),
86 p + embed_point(+1., -1.) * (r / std::numbers::sqrt2),
87 p + embed_point(-1., +1.) * (r / std::numbers::sqrt2),
88 p + embed_point(+1., +1.) * (r / std::numbers::sqrt2),
89 // The other points that form the inner square
90 p + embed_point(-1., -1.) * R,
91 p + embed_point(+1., -1.) * R,
92 p + embed_point(-1., +1.) * R,
93 p + embed_point(+1., +1.) * R,
94 // The two points attached to the rectangle
95 embed_point(b, 0),
96 embed_point(b, a),
97 // The two points (0, r) and (0, -r) from the center of the circle
98 p + embed_point(0, -r),
99 p + embed_point(0, +r),
100 // The two points on the right side of the outer circle that lie along
101 // the width of the rectangle
102 p + embed_point(std::sqrt(r * r - a * a / 4), -a / 2),
103 p + embed_point(std::sqrt(r * r - a * a / 4), +a / 2),
104 // The other points where the width of the rectangle intersects the
105 // central square
106 p + embed_point(-R, +a / 2),
107 p + embed_point(-R, -a / 2),
108 p + embed_point(+R, -a / 2),
109 p + embed_point(+R, +a / 2),
110 // The two points of the central square on the circle's axis of symmetry
111 p + embed_point(0, -R),
112 p + embed_point(0, +R),
113 // The two points where the width of the rectangle intersects the axis
114 // of symmetry of the circle
115 p + embed_point(0, +a / 2),
116 p + embed_point(0, -a / 2)};
117 }();
118
119 std::vector<CellData<2>> cells(16, CellData<2>());
120 static constexpr int circle_cell_vertices[16][4] = {
121 // We start with the 6 squares in the center square
122 {6, 18, 15, 21},
123 {15, 21, 14, 20},
124 {14, 20, 4, 19},
125 {18, 7, 21, 16},
126 {21, 16, 20, 17},
127 {20, 17, 19, 5},
128 // We fill in the outer cells, starting from the top left
129 {9, 14, 2, 4},
130 {8, 15, 9, 14},
131 {0, 6, 8, 15},
132 {0, 10, 6, 18},
133 {10, 1, 18, 7},
134 {7, 1, 16, 12},
135 {16, 12, 17, 13},
136 {17, 13, 5, 3},
137 {19, 5, 11, 3},
138 {4, 19, 2, 11}};
139
140 for (unsigned int i = 0; i < 16; ++i)
141 {
142 for (unsigned int j = 0; j < 4; ++j)
143 cells[i].vertices[j] = circle_cell_vertices[i][j];
144 }
145
146 tria.create_triangulation(std::vector<Point<2>>(std::begin(vertices),
147 std::end(vertices)),
148 cells,
149 SubCellData()); // no boundary information
150 tria.set_all_manifold_ids_on_boundary(0);
151 tria.set_manifold(0, SphericalManifold<2, 2>(p));
152 tria.set_manifold(1, FlatManifold<2, 2>());
153}

179template <int dim, int spacedim>
180void
181GridPipeAndWaveguide<dim, spacedim>::make_grid(
182 Triangulation<dim, spacedim> &triangulation)
183{
184 /*
185 Two geometric conditions that need to be verified to prevent the generation of
186 inverted cells.
187
188 The first condition concerns the inner shape. To improve mesh quality, a
189 central rectangle of width equal to the waveguide width is introduced. After
190 extrusion, the cells within this rectangle are nearly hexahedral, which
191 minimizes aspect ratio. However, this rectangle imposes a geometric
192 constraint: the inner shape must be large enough to fully contain it,
193 otherwise inverted cells are generated.
194
195 Then, the inner shape have to be strictly inscribed in the outer shape.
196 */
197
198 // Tests when the inner shape is a circle
199 if (inner_circle)
200 {
201 AssertThrow(
202 inner_length > rectangle_width / 2,
203 ExcMessage(
204 "The inner shape is not bigger than the rectangle that crosses the cylinder along the x-axis. This situations leads to the generation of inverted cells"));
205
206 AssertThrow(
207 inner_length < outer_radius,
208 ExcMessage(
209 "The internal radius is greater than the external radius. This situations leads to the generation of inverted cells"));
210 }
211
212 // Tests when the inner shape is a square
213 else
214 {
215 AssertThrow(
216 inner_length > rectangle_width / 2,
217 ExcMessage(
218 "The square side is too small compared to the width of the waveguide. This situations leads to the generation of inverted cells"));
219
220 AssertThrow(
221 inner_length < outer_radius / (std::sqrt(2)),
222 ExcMessage(
223 "The inner square length is greater than the external radius. This situations leads to the generation of inverted cells"));
224 }
225
226 // Dimension Test
227 if constexpr (!(dim == 3 && spacedim == 3))
228 {
229 AssertThrow(false,
230 ExcMessage("GridPipeAndWaveguide is only supported in 3D."));
231 return;
232 }
233
234 else
235 {
236 // COARSE 2D PARTS OF THE MESH
237
238 // PART -I-:Creating the Shape of the Medium: The paving stone embedded in
239 // the Cylinder
240
241 // Creating the first shape: a rectangle (after extrusion, this rectangle
242 // will represent the paving stone). This rectangle is the reason why we
243 // divide the volume into three different pieces.
244 Triangulation<2> rect;
245 GridGenerator::hyper_rectangle(
246 rect, Point<2>(0.0, 0.0), Point<2>(rectangle_length, rectangle_width));
247
248 // Creating the second shape: a rough circle with two points located
249 // exactly at the right corners of the rectangle
250 Triangulation<2> disk;
251 // Position of the center of the circle
252 const Point<2> center(rectangle_length +
253 sqrt(outer_radius * outer_radius -
254 rectangle_width * rectangle_width / 4),
255 rectangle_width / 2);
256 detailed_circle(disk,
257 center,
258 rectangle_width,
259 rectangle_length,
260 outer_radius,
261 inner_length,
262 inner_circle);
263
264 Triangulation<2> merged;
265 // The last argument of the following function is
266 // false since it is useless to copy the manifolds ids
267 // because we will set them after
268 GridGenerator::merge_triangulations(rect, disk, merged, 1e-12, false);
269
270 // PART -II- Creating the bottom cylinder
271 Triangulation<2, 2> base;
272 detailed_circle(base,
273 center,
274 rectangle_width,
275 rectangle_length,
276 outer_radius,
277 inner_length,
278 inner_circle);
279
280 // PART -III- Creating the top cylinder
281 Triangulation<2, 2> top;
282 detailed_circle(top,
283 center,
284 rectangle_width,
285 rectangle_length,
286 outer_radius,
287 inner_length,
288 inner_circle);
289
290 // EXTRUDE
291
292 Triangulation<3> extruded_base;
293 Triangulation<3> extruded_middle;
294 Triangulation<3> extruded_top;
295 GridGenerator::extrude_triangulation(base,
296 bottom_resolution,
297 bottom_height,
298 extruded_base);
299 GridGenerator::extrude_triangulation(merged,
300 middle_resolution,
301 middle_height,
302 extruded_middle);
303 GridGenerator::extrude_triangulation(top,
304 top_resolution,
305 top_height,
306 extruded_top);
307
308 // MANIFOLD
309
310 // Lambda function that places the id of the cylindrical manifold in the
311 // inside of the cylinder
312 auto attach_cylindrical_manifold_id = [&](Triangulation<3> &tria,
313 double height) {
314 for (const auto &cell : tria.active_cell_iterators())
315 if (cell->center()[0] >= rectangle_length - 1e-6)
316 // Concerns only the cylinder because every cell of the
317 // waveguide verifies the condition : x >= rectangle_length
318 for (const auto f : cell->face_indices())
319 if (cell->face(f)->at_boundary())
320 {
321 const Point<3> face_center = cell->face(f)->center();
322 if (std::abs(face_center[2]) > 1e-10 &&
323 std::abs(face_center[2] - height) > 1e-10)
324 // Does not take into account the the flat faces
325 // at z=0 and z=height because we do not want a manifold
326 // applied there.
327 cell->set_all_manifold_ids(1);
328 }
329 };
330
331 attach_cylindrical_manifold_id(extruded_base, bottom_height);
332 attach_cylindrical_manifold_id(extruded_middle, middle_height);
333 attach_cylindrical_manifold_id(extruded_top, top_height);
334
335 // Now that we have the IDs in the right place, we can install the
336 // manifolds
337 const Point<3> center_3d(center[0], center[1], 0.0);
338 const Tensor<1, 3> axis({0.0, 0.0, 1.0});
339 extruded_base.set_manifold(1, CylindricalManifold<3>(axis, center_3d));
340 extruded_middle.set_manifold(1, CylindricalManifold<3>(axis, center_3d));
341 extruded_top.set_manifold(1, CylindricalManifold<3>(axis, center_3d));
342
343 // MERGING THE 3D MESHES
344
345 // Move the cylinders in the right positions to prepare the merge
346 Tensor<1, 3> shift_down;
347 shift_down[0] = 0.0;
348 shift_down[1] = 0.0;
349 shift_down[2] = -bottom_height; // The base_cylinder must have its base at
350 // -bottom_height
351 GridTools::shift(shift_down, extruded_base);
352
353
354 Tensor<1, 3> shift_up;
355 shift_up[0] = 0.0;
356 shift_up[1] = 0.0;
357 shift_up[2] = +middle_height; // The top_cylinder must have its base at
358 // +middle_height
359 GridTools::shift(shift_up, extruded_top);
360
361
362 // Merge the 3 parts of the mesh
363 std::vector<const Triangulation<3, 3> *> parts = {&extruded_base,
364 &extruded_middle,
365 &extruded_top};
366
367 GridGenerator::merge_triangulations(
368 parts,
369 triangulation,
370 1e-12,
371 true); // “true” is used here to preserve the cell IDs; this is useful
372 // for attaching the manifold right after
373
374 // Attach the manifolds that we previously set on each part
375 triangulation.set_manifold(1, CylindricalManifold<3>(axis, center_3d));
376
377
378 // BOUNDARY AND MATERIAL IDs
379
380 /*
381 //Boundary
382 0: MW Inlet (x=0)
383 1: Fluid Inlet (y = -bottom_height)
384 2: Fluid Outlet (y = middle_height + top_height)
385 3: PEC walls (every other wall)
386
387 //Material
388 0: Fluid part (x> rectangle_length)
389 1: Waveguide paving stone (x<= rectangle_length)
390 */
391
392 for (auto &cell : triangulation.active_cell_iterators())
393 {
394 // Boundary ids
395 for (const auto f : cell->face_indices())
396 {
397 if (cell->face(f)->at_boundary())
398 {
399 const Point<3> face_center = cell->face(f)->center();
400 if (face_center[0] <= 1e-6)
401 cell->face(f)->set_boundary_id(0); // MW Inlet
402 else if (face_center[2] <= -bottom_height + 1e-6)
403 cell->face(f)->set_boundary_id(1); // Fluid Inlet
404 else if (face_center[2] >= middle_height + top_height - 1e-6)
405 cell->face(f)->set_boundary_id(2); // Fluid Outlet
406 else
407 cell->face(f)->set_boundary_id(
408 3); // PEC Walls (Perfect Electric Conductors)
409 }
410 }
411 // Material ids
412 if (cell->center()[0] <= rectangle_length)
413 cell->set_material_id(1); // waveguide (Paving stone)
414 else
415 cell->set_material_id(0); // Fluid (cylinder)
416 }
417 }
418}
