27/**
28 * @brief Iterates over a vector of maps (pairs_in_contact) to see if the
29 * particles which were in contact in the last time step, are still in contact
30 * or not. If they are still in contact it will update the collision info,
31 * including tangential displacement, based on new properties of the particle
32 * pair, if they are not in contact anymore it will delete the pair from the
33 * pairs_in_contact and also its information from pairs_in_contact_info.
34 * Then it iterates over the contact candidates from broad search to see if
35 * they already exist in the pairs_in_contact or not, if they are not in the
36 * pairs_in_contact and have an overlap, the pair will be added to the
37 * pairs_in_contact and its contact information will be stored in the
38 * corresponding element of the pairs_in_contact_info.
39 *
40 * @param particle_container A container that is used to obtain iterators to
41 * particles using their ids.
42 * @param adjacent_particles A map of maps which stores all the required
43 * information for calculation of the contact force of particle pairs. Can
44 * either be for non-periodic or periodic contact types.
45 * @param contact_pair_candidates The output of broad search which shows
46 * contact pair candidates.
47 * @param neighborhood_threshold A value which defines the neighbor particles.
48 * @param periodic_offset_per_direction An array whose component d holds the
49 * signed periodic offsets associated with each periodic directions. If the
50 * given d direction is not periodic, the signed distance is 0.
51 *
52 */
53template <int dim, ContactType contact_type>
54void
55particle_particle_fine_search(
56 const typename DEM::dem_data_structures<dim>::particle_index_iterator_map
57 &particle_container,
58 adjacent_pairs_for_contact_t<dim, contact_type> &adjacent_particles,
59 const typename DEM::dem_data_structures<dim>::particle_particle_candidates
60 &contact_pair_candidates,
61 const double neighborhood_threshold,
62 const std::array<double, dim> &periodic_offset_per_direction = {});

65/**
66 * @brief Find the translation that brings the nearest periodic image of
67 * particle two to particle one. Since periodic directions are axis-aligned
68 * and independent, this translation is found directly, one direction at a
69 * time.
70 *
71 * @param particle_one_location Location of particle one.
72 * @param particle_two_real_location Real (non-translated) location of
73 * particle two.
74 * @param periodic_offset_per_direction An array whose component d holds
75 * the signed periodic offset of the domain along direction d (0 if d is not
76 * periodic).
77 *
78 * @return The nearest translation, and whether any periodic direction
79 * required a nonzero translation. A pair whose nearest image requires no
80 * translation on any periodic direction is not a periodic contact (it is
81 * already handled by the non-periodic contact types).
82 */
83template <int dim>
84std::pair<Tensor<1, dim>, bool>
85nearest_periodic_translation(
86 const Point<dim, double> &particle_one_location,
87 const Point<dim, double> &particle_two_real_location,
88 const std::array<double, dim> &periodic_offset_per_direction);
