17template <int dim>
18TimeHarmonicMaxwell<dim>::TimeHarmonicMaxwell(
19 MultiphysicsInterface<dim> *multiphysics_interface,
20 const SimulationParameters<dim> &p_simulation_parameters,
21 std::shared_ptr<parallel::DistributedTriangulationBase<dim>> p_triangulation,
22 std::shared_ptr<SimulationControl> p_simulation_control)
23 : AuxiliaryPhysics<dim, GlobalVectorType>()
24 , multiphysics(multiphysics_interface)
25 , computing_timer(p_triangulation->get_mpi_communicator(),
26 this->pcout,
27 TimerOutput::never,
28 TimerOutput::wall_times)
29 , simulation_parameters(p_simulation_parameters)
30 , triangulation(p_triangulation)
31 , simulation_control(std::move(p_simulation_control))
32 , dof_handler_trial_interior(
33 std::make_shared<DoFHandler<dim>>(*triangulation))
34 , dof_handler_trial_skeleton(
35 std::make_shared<DoFHandler<dim>>(*triangulation))
36 , dof_handler_test(std::make_shared<DoFHandler<dim>>(*triangulation))
37 , extractor_E_real(0)
38 , extractor_E_imag(dim)
39 , extractor_H_real(2 * dim)
40 , extractor_H_imag(3 * dim)
41{
42 this->pcout << std::setprecision(simulation_control->get_log_precision())
43 << std::scientific;
44
45 if (simulation_parameters.mesh.simplex)
46 {
47 // for simplex meshes
48 AssertThrow(
49 false,
50 ExcMessage(
51 "TimeHarmonicMaxwell solver not yet implemented for simplex meshes."));
52 }
53 else
54 {
55 AssertThrow(dim == 3, TimeHarmonicMaxwellDimensionNotSupported(dim));
56
57 AssertThrow(
58 simulation_parameters.fem_parameters.electromagnetics_trial_degree <
59 simulation_parameters.fem_parameters.electromagnetics_test_degree,
60 ExcMessage(
61 "The DPG method requires the test space to be of higher order than the trial space."));
62
63 // Usual case, for quad/hex meshes
64 fe_trial_interior = std::make_shared<FESystem<dim>>(
65 FE_DGQ<dim>(
66 simulation_parameters.fem_parameters.electromagnetics_trial_degree) ^
67 dim,
68 FE_DGQ<dim>(
69 simulation_parameters.fem_parameters.electromagnetics_trial_degree) ^
70 dim,
71 FE_DGQ<dim>(
72 simulation_parameters.fem_parameters.electromagnetics_trial_degree) ^
73 dim,
74 FE_DGQ<dim>(
75 simulation_parameters.fem_parameters.electromagnetics_trial_degree) ^
76 dim);
77 fe_trial_skeleton = std::make_shared<FESystem<dim>>(
78 FE_Nedelec<dim>(
79 simulation_parameters.fem_parameters.electromagnetics_trial_degree),
80 FE_Nedelec<dim>(
81 simulation_parameters.fem_parameters.electromagnetics_trial_degree),
82 FE_Nedelec<dim>(
83 simulation_parameters.fem_parameters.electromagnetics_trial_degree),
84 FE_Nedelec<dim>(
85 simulation_parameters.fem_parameters.electromagnetics_trial_degree));
86 fe_test = std::make_shared<FESystem<dim>>(
87 FE_Nedelec<dim>(
88 simulation_parameters.fem_parameters.electromagnetics_test_degree),
89 FE_Nedelec<dim>(
90 simulation_parameters.fem_parameters.electromagnetics_test_degree),
91 FE_Nedelec<dim>(
92 simulation_parameters.fem_parameters.electromagnetics_test_degree),
93 FE_Nedelec<dim>(
94 simulation_parameters.fem_parameters.electromagnetics_test_degree));
95 mapping = std::make_shared<MappingQ<dim>>(fe_trial_interior->degree);
96 cell_quadrature = std::make_shared<QGauss<dim>>(fe_test->degree + 1);
97 face_quadrature = std::make_shared<QGauss<dim - 1>>(fe_test->degree + 1);
98 }
99
100 // Initialize solutions shared_ptr
101 present_solution = std::make_shared<GlobalVectorType>();
102 present_solution_skeleton = std::make_shared<GlobalVectorType>();
103
104 // Allocate solution transfer
105 solution_transfer = std::make_shared<SolutionTransfer<dim, GlobalVectorType>>(
106 *dof_handler_trial_interior);
107
108 // We may need the temperature field for the physical properties. If so,
109 // we create the corresponding FEValues object to evaluate the temperature at
110 // the quadrature points and its container to store the values.
111 const PhysicalPropertiesManager &physical_properties_manager =
112 simulation_parameters.physical_properties_manager;
113 unsigned int number_of_material_ids =
114 physical_properties_manager.get_number_of_fluids() +
115 physical_properties_manager.get_number_of_solids() - 1;
116
117 // The DPG error estimator is only needed when it is the error estimator of
118 // the electromagnetic fields.
119 const auto &adaptation_variables =
120 this->simulation_parameters.mesh_adaptation.variables;
121 const auto electromagnetic_adaptation =
122 adaptation_variables.find(Variable::electromagnetic_fields);
123 needs_dpg_error_estimator =
124 (electromagnetic_adaptation != adaptation_variables.end()) &&
125 (electromagnetic_adaptation->second.error_estimator ==
126 Parameters::MultipleAdaptationParameters::ErrorEstimator::dpg);
127
128 needs_temperature = false;
129 for (unsigned int material_id = 0; material_id <= number_of_material_ids;
130 ++material_id)
131 {
132 needs_temperature =
133 (physical_properties_manager.get_electric_conductivity(0, material_id)
134 ->depends_on(field::temperature) ||
135 physical_properties_manager
136 .get_electric_permittivity_real(0, material_id)
137 ->depends_on(field::temperature) ||
138 physical_properties_manager
139 .get_electric_permittivity_imag(0, material_id)
140 ->depends_on(field::temperature) ||
141 physical_properties_manager
142 .get_magnetic_permeability_real(0, material_id)
143 ->depends_on(field::temperature) ||
144 physical_properties_manager
145 .get_magnetic_permeability_imag(0, material_id)
146 ->depends_on(field::temperature));
147
148 if (needs_temperature)
149 {
150 // If temperature-dependent electromagnetic properties are requested,
151 // heat transfer must be enabled.
152
153 const std::vector<PhysicsID> active_physics_ids =
154 multiphysics->get_active_physics();
155 AssertThrow(
156 std::ranges::find(active_physics_ids, PhysicsID::heat_transfer) !=
157 active_physics_ids.end(),
158 ExcMessage(
159 "User-defined temperature-dependent electromagnetic properties requested without heat transfer physics. Enable heat transfer in the multiphysics subsection."));
160 break;
161 }
162 }

331template <int dim>
332void
333TimeHarmonicMaxwell<dim>::compute_electromagnetic_scaling(
334 const PhysicalPropertiesManager &physical_properties_manager)
335{
336 // Define reusable variables for the computation of the scaling factor.
337 auto mpi_communicator = this->triangulation->get_mpi_communicator();
338 const auto &electromagnetic_parameters =
339 this->simulation_parameters.multiphysics.time_harmonic_maxwell_parameters;
340 constexpr double void_impedance =
341 4 * numbers::PI *
342 29.9792458; // Void impedance in Ohms, the unit conversion of this value is
343 // done when computing the scaling factor. The factor 29.9792458
344 // is the speed of light in cm/ns and is used to make it
345 // convenient when multiplying by the void permeability to
346 // obtain the void impedance in the desired units (e.g., Ohms).
347
348 // Initialize the amplitude. The default value is put to zero.
349 // If there are no waveguide inlets, the maximum of amplitude will be taken
350 // between 0 and a strictly positive value ensuring that the strictly positive
351 // value remains.
352 double max_port_electric_amplitude = 0.0;
353
354 // If there are waveguide inlets we need to compute their scaling from the
355 // input power provided by the user.
356 if (electromagnetic_parameters.number_of_waveguide_inlets > 0)
357 {
358 const auto &waveguide_powers = electromagnetic_parameters.waveguide_power;
359
360 // The following vector will store the modal power for each waveguide
361 // inlet boundary condition computed from integrating the Poynting vector
362 // of the incident wave over the inlet face.
363 std::vector<double> waveguide_modal_powers(
364 electromagnetic_parameters.number_of_waveguide_inlets, 0.);
365
366 // Resize the vector that will store the waveguide port electric field
367 // amplitudes. Those are obtain by computing the square root of the ratio
368 // between the input power provided by the user and the modal power
369 // computed from integrating the Poynting vector of the incident wave over
370 // the inlet face.
371 this->waveguide_ports_electric_amplitudes = std::vector<double>(
372 electromagnetic_parameters.number_of_waveguide_inlets, 0.);
373
374 // We need to define a FEFaceValues object to perform the integral of the
375 // Poynting vector on faces
376 FEFaceValues<dim> fe_face_values_trial_skeleton(*this->mapping,
377 *this->fe_trial_skeleton,
378 *this->face_quadrature,
379 update_quadrature_points |
380 update_normal_vectors |
381 update_JxW_values);
382 const unsigned int n_face_q_points =
383 fe_face_values_trial_skeleton.n_quadrature_points;
384 BoundaryConditions::BoundaryType bc_type;
385
386 // Containers for electromagnetic quantities at the quadrature points
387 Tensor<1, dim, std::complex<double>> E_inc;
388 Tensor<1, dim, std::complex<double>> H_inc;
389 std::vector<std::complex<double>> effective_electric_permittivities(
390 n_face_q_points);
391 std::vector<std::complex<double>> effective_magnetic_permeabilities(
392 n_face_q_points);
393 unsigned int material_id;
394 bool cell_material_needs_temperature = false;
395
396 // The temperature field is only needed when at least one electromagnetic
397 // property depends on temperature.
398
399 std::unique_ptr<FEFaceValues<dim>> fe_face_values_temperature;
400 const DoFHandler<dim> *dof_handler_temperature = nullptr;
401 const GlobalVectorType *temperature_solution = nullptr;
402 std::vector<double> temperature_face_values(n_face_q_points);
403 std::map<field, std::vector<double>> field_values_vector;
404
405 if (needs_temperature)
406 {
407 dof_handler_temperature =
408 &this->multiphysics->get_dof_handler(PhysicsID::heat_transfer);
409 fe_face_values_temperature = std::make_unique<FEFaceValues<dim>>(
410 *this->mapping,
411 dof_handler_temperature->get_fe(),
412 *this->face_quadrature,
413 update_values | update_quadrature_points);
414 temperature_solution =
415 &this->multiphysics->get_solution(PhysicsID::heat_transfer);
416 }
417
418
419 // Here we perform the Poynting vector integration over the waveguide
420 // inlet faces.
421 for (const auto &cell : triangulation->active_cell_iterators())
422 {
423 if (cell->is_locally_owned())
424 {
425 material_id = cell->material_id();
426
427 // We check if the physical properties depend on the temperature
428 // field. If so, we will need to evaluate the temperature field at
429 // the quadrature points.
430 cell_material_needs_temperature =
431 physical_properties_manager
432 .get_electric_conductivity(0, material_id)
433 ->depends_on(field::temperature) ||
434 physical_properties_manager
435 .get_electric_permittivity_real(0, material_id)
436 ->depends_on(field::temperature) ||
437 physical_properties_manager
438 .get_electric_permittivity_imag(0, material_id)
439 ->depends_on(field::temperature) ||
440 physical_properties_manager
441 .get_magnetic_permeability_real(0, material_id)
442 ->depends_on(field::temperature) ||
443 physical_properties_manager
444 .get_magnetic_permeability_imag(0, material_id)
445 ->depends_on(field::temperature);
446
447 for (const auto &face : cell->face_iterators())
448 {
449 if (!(face->at_boundary()))
450 continue;
451
452 fe_face_values_trial_skeleton.reinit(cell, face);
453
454 if (cell_material_needs_temperature)
455 {
456 const typename DoFHandler<dim>::active_cell_iterator
457 cell_temperature = cell->as_dof_handler_iterator(
458 *dof_handler_temperature);
459 fe_face_values_temperature->reinit(cell_temperature,
460 face);
461 fe_face_values_temperature->get_function_values(
462 *temperature_solution, temperature_face_values);
463 }
464 else
465 {
466 std::ranges::fill(temperature_face_values, 0.);
467 }
468
469 bc_type =
470 this->simulation_parameters
471 .boundary_conditions_time_harmonic_electromagnetics.type
472 .at(face->boundary_id());
473
474 // We only compute the modal power if its a port boundary
475 if (!(bc_type ==
476 BoundaryConditions::BoundaryType::waveguide_port))
477 continue;
478
479 // Get the index of the waveguide port boundary
480 // condition to apply the correct mode parameters
481 const unsigned int waveguide_port_index =
482 get_waveguide_port_index(electromagnetic_parameters,
483 face->boundary_id());
484
485 // We update the material properties for the current face
486 // quadrature points.
487 field_values_vector[field::temperature] =
488 temperature_face_values;
489
490 compute_effective_electromagnetic_properties(
491 physical_properties_manager,
492 field_values_vector,
493 material_id,
494 effective_electric_permittivities,
495 effective_magnetic_permeabilities);
496
497 // Loop over all face quadrature points
498 for (unsigned int q_point = 0; q_point < n_face_q_points;
499 ++q_point)
500 {
501 // Initialize reusable variables
502 const auto &position =
503 fe_face_values_trial_skeleton.quadrature_point(q_point);
504 const auto &normal =
505 fe_face_values_trial_skeleton.normal_vector(q_point);
506 const double JxW_face =
507 fe_face_values_trial_skeleton.JxW(q_point);
508
509 std::tie(E_inc, H_inc) =
510 compute_waveguide_port_incident_fields(
511 electromagnetic_parameters,
512 position,
513 normal,
514 effective_electric_permittivities[q_point],
515 effective_magnetic_permeabilities[q_point],
516 waveguide_port_index);
517
518 // The Poynting vector is given by S = 0.5 * Re(E
519 // x H*), where H* is the complex conjugate of H.
520 // The normal points outward from the domain, so
521 // the power entering the domain is given by the
522 // negative of the flux of the Poynting vector
523 // through the face (that's why we have a negative
524 // sign in front of the integral). The cross product
525 // is written explicitly in 3D, so it is only compiled
526 // for dim = 3 (the waveguide port fields are not
527 // defined in 2D).
528 if constexpr (dim == 3)
529 {
530 waveguide_modal_powers[waveguide_port_index] -=
531 0.5 *
532 std::real(
533 normal[0] * (E_inc[1] * std::conj(H_inc[2]) -
534 E_inc[2] * std::conj(H_inc[1])) +
535 normal[1] * (E_inc[2] * std::conj(H_inc[0]) -
536 E_inc[0] * std::conj(H_inc[2])) +
537 normal[2] * (E_inc[0] * std::conj(H_inc[1]) -
538 E_inc[1] * std::conj(H_inc[0]))) *
539 JxW_face;
540 }
541 else
542 {
543 AssertThrow(false,
544 TimeHarmonicMaxwellDimensionNotSupported(
545 dim));
546 }
547 }
548 }
549 }
550 }
551
552 // Now we need to aggregate from all the MPI processes to get the global
553 // modal power for each waveguide inlet
554 for (unsigned int inlets = 0;
555 inlets < electromagnetic_parameters.number_of_waveguide_inlets;
556 ++inlets)
557 {
558 // The power amplitude comes from the non-dimensionalization of power:
559 // P = 0.5 * int(ExH*)= E_0 * H_0 * P_modal. So we can isolate the
560 // electric field amplitude E_0. We don't need to multiply by the
561 // dimensionality because we assume user input in W and impedance in
562 // Ohms so the resulting units of this integral will always be
563 // Volt/length unit (of the mesh).
564 this->waveguide_ports_electric_amplitudes[inlets] =
565 std::sqrt(waveguide_powers[inlets] * void_impedance /
566 Utilities::MPI::sum(waveguide_modal_powers[inlets],
567 mpi_communicator));
568 }
569
570 max_port_electric_amplitude =
571 *std::ranges::max_element(this->waveguide_ports_electric_amplitudes);
572 }
573
574 // Now we need to define the electromagnetic scaling factor according to the
575 // type chosen by the user.
576 if (electromagnetic_parameters.electromagnetic_scaling_type ==
577 Parameters::ElectromagneticScalingType::electric_field)
578 {
579 // We assume that the electric field amplitude provided by the user is MKS
580 // (V/m).
581 this->electromagnetic_scaling =
582 electromagnetic_parameters.electric_field_amplitude;
583 }
584 else if (electromagnetic_parameters.electromagnetic_scaling_type ==
585 Parameters::ElectromagneticScalingType::magnetic_field)
586 {
587 // We assume that the magnetic field amplitude provided by the user is in
588 // MKS (A/m). We convert it to the electric field for convenience to not
589 // carry multiple conversion factors when applying the scaling to the
590 // solution.
591 this->electromagnetic_scaling =
592 void_impedance * electromagnetic_parameters.magnetic_field_amplitude;
593 }
594 else if (electromagnetic_parameters.electromagnetic_scaling_type ==
595 Parameters::ElectromagneticScalingType::power)
596 { // From the Poynting vector integration and assumptions made in the
597 // computation of the waveguide port excitation, the unit here are in V/L,
598 // with L being the length unit of the mesh. Therefore, we need to divide
599 // by the length unit to get the correct scaling factor in V/m.
600 this->electromagnetic_scaling =
601 max_port_electric_amplitude /
602 this->simulation_parameters.dimensionality.length;
603 }
604 else if (electromagnetic_parameters.electromagnetic_scaling_type ==
605 Parameters::ElectromagneticScalingType::none)
606 {
607 // The solution is not rescaled. If there are waveguide inlets, their
608 // relative powers are already accounted for in the excitation, which is
609 // scaled by the amplitude of each inlet relative to the maximum amplitude
610 // of all the inlets (see TimeHarmonicMaxwellAssemblerRobinBC). The
611 // strongest inlet is therefore excited with the unit-amplitude mode.
612 this->electromagnetic_scaling = 1.0;
613 }
614 else
615 {
616 AssertThrow(false, ExcMessage("Unknown electromagnetic scaling type."));
617 }
618}

620template <int dim>
621void
622TimeHarmonicMaxwell<dim>::scale_solution_components(
623 const DoFHandler<dim> &dof_handler,
624 GlobalVectorType &solution)
625{
626 const auto &fe = dof_handler.get_fe();
627
628 if (this->simulation_parameters.multiphysics.time_harmonic_maxwell_parameters
629 .electromagnetic_scaling_type ==
630 Parameters::ElectromagneticScalingType::none)
631 return;
632
633 constexpr double void_impedance = 4 * numbers::PI * 29.9792458;
634 // Here, everything is in MKS units, so we need to convert it to the user
635 // desired units.
636 const double electric_scale =
637 this->electromagnetic_scaling *
638 this->simulation_parameters.dimensionality.electric_amplitude_scaling;
639 const double magnetic_scale =
640 this->electromagnetic_scaling / void_impedance *
641 this->simulation_parameters.dimensionality.magnetic_amplitude_scaling;
642
643 const IndexSet &locally_owned_dofs = solution.locally_owned_elements();
644 std::vector<bool> dof_scaled(locally_owned_dofs.n_elements(), false);
645 std::vector<types::global_dof_index> local_dof_indices(fe.n_dofs_per_cell());
646
647 // We loop over all the cells and apply the correct scaling to the electric
648 // and magnetic field components of the solution. We need to do so because the
649 // solution is stored as a single vector but contains different physical
650 // fields (electric and magnetic) that need to be scaled differently and the
651 // system_to_base_index function which allows us to identify which
652 // component of the solution corresponds to which dof is based on the local
653 // dof index and not the global one. Therefore, we need to loop over all the
654 // cells and their local dofs to apply the correct scaling to each component
655 // of the solution.
656 for (const auto &cell : dof_handler.active_cell_iterators())
657 {
658 if (cell->is_locally_owned())
659 {
660 // Get the global dof indices for the current cell
661 cell->get_dof_indices(local_dof_indices);
662
663 // Loop over the local dofs of the cell
664 for (unsigned int local_dof = 0; local_dof < local_dof_indices.size();
665 ++local_dof)
666 {
667 // Get the global dof index corresponding to the local dof index
668 const types::global_dof_index global_dof =
669 local_dof_indices[local_dof];
670
671 // Check if the global dof index is locally owned. If not, we skip
672 // it since we only want to scale the locally owned part of the
673 // solution vector.
674 if (!locally_owned_dofs.is_element(global_dof))
675 continue;
676
677 const unsigned int locally_owned_position =
678 locally_owned_dofs.index_within_set(global_dof);
679
680 // Since we loop over all the cells, we will encounter each dof
681 // multiple times (once for each cell it belongs to). We only want
682 // to apply the scaling once for each dof, so we keep track of
683 // which dofs have already been scaled using the dof_scaled
684 // vector. If a dof has already been scaled, we skip it.
685 if (dof_scaled[locally_owned_position])
686 continue;
687
688 dof_scaled[locally_owned_position] = true;
689
690 // Get the base element of the current dof. Remember that the
691 // convention for the FE system we are using is [E_real, E_imag,
692 // H_real, H_imag].
693 const unsigned int base_element =
694 fe.system_to_base_index(local_dof).first.first;
695
696 if (base_element == 0 || base_element == 1)
697 solution[global_dof] *= electric_scale;
698 else if (base_element == 2 || base_element == 3)
699 solution[global_dof] *= magnetic_scale;
700 }
701 }
702 }

754template <int dim>
755std::vector<double>
756TimeHarmonicMaxwell<dim>::calculate_L2_error()
757{
758 auto mpi_communicator = this->triangulation->get_mpi_communicator();
759
760 // Interior L2 error
761 FEValues<dim> fe_values_trial_interior(*this->mapping,
762 *this->fe_trial_interior,
763 *this->cell_quadrature,
764 update_values |
765 update_quadrature_points |
766 update_JxW_values);
767
768 const unsigned int n_q_points = this->cell_quadrature->size();
769
770 // The exact solution will be defined by the user but will need to be on all
771 // possible fields of the ultraweak formulation so we need 4*dim components.
772 std::vector<Vector<double>> exact_solution_values(n_q_points,
773 Vector<double>(4 * dim));
774 auto &exact_solution_function =
775 simulation_parameters.analytical_solution->electromagnetics;
776
777 // When looping on each cell we will extract the different field
778 // solution obtained numerically. The containers used to store the
779 // interpolated solution at the quadrature points are declared below.
780 std::vector<Tensor<1, dim>> local_E_values_real(n_q_points);
781 std::vector<Tensor<1, dim>> local_E_values_imag(n_q_points);
782 std::vector<Tensor<1, dim>> local_H_values_real(n_q_points);
783 std::vector<Tensor<1, dim>> local_H_values_imag(n_q_points);
784
785 // We create variables that will store all the integration results we are
786 // interested in.
787 double L2_error_E_real = 0;
788 double L2_error_E_imag = 0;
789 double L2_error_H_real = 0;
790 double L2_error_H_imag = 0;
791
792 for (const auto &cell : dof_handler_trial_interior->active_cell_iterators())
793 {
794 if (cell->is_locally_owned())
795 {
796 fe_values_trial_interior.reinit(cell);
797
798 // Get the simulated solution at quadrature points
799 fe_values_trial_interior[extractor_E_real].get_function_values(
800 *present_solution, local_E_values_real);
801 fe_values_trial_interior[extractor_E_imag].get_function_values(
802 *present_solution, local_E_values_imag);
803 fe_values_trial_interior[extractor_H_real].get_function_values(
804 *present_solution, local_H_values_real);
805 fe_values_trial_interior[extractor_H_imag].get_function_values(
806 *present_solution, local_H_values_imag);
807
808 // Get the exact solution at quadrature points
809 exact_solution_function.vector_value_list(
810 fe_values_trial_interior.get_quadrature_points(),
811 exact_solution_values);
812
813 // Loop on quadrature points to compute the L2 error contributions
814 for (unsigned int q = 0; q < n_q_points; ++q)
815 {
816 const double JxW = fe_values_trial_interior.JxW(q);
817
818 // Loop on dimensions to compute the squared error
819 for (int d = 0; d < dim; ++d)
820 {
821 // E real part
822 L2_error_E_real +=
823 Utilities::fixed_power<2>(local_E_values_real[q][d] -
824 exact_solution_values[q][d]) *
825 JxW;
826
827 // E imag part
828 L2_error_E_imag += Utilities::fixed_power<2>(
829 local_E_values_imag[q][d] -
830 exact_solution_values[q][d + dim]) *
831 JxW;
832
833 // H real part
834 L2_error_H_real += Utilities::fixed_power<2>(
835 local_H_values_real[q][d] -
836 exact_solution_values[q][d + 2 * dim]) *
837 JxW;
838
839 // H imag part
840 L2_error_H_imag += Utilities::fixed_power<2>(
841 local_H_values_imag[q][d] -
842 exact_solution_values[q][d + 3 * dim]) *
843 JxW;
844 }
845 }
846 }
847 }
848 // Skeleton L2 error
849 // TODO
850
851 L2_error_E_real = Utilities::MPI::sum(L2_error_E_real, mpi_communicator);
852 L2_error_E_imag = Utilities::MPI::sum(L2_error_E_imag, mpi_communicator);
853 L2_error_H_real = Utilities::MPI::sum(L2_error_H_real, mpi_communicator);
854 L2_error_H_imag = Utilities::MPI::sum(L2_error_H_imag, mpi_communicator);
855
856 return {std::sqrt(L2_error_E_real),
857 std::sqrt(L2_error_E_imag),
858 std::sqrt(L2_error_H_real),
859 std::sqrt(L2_error_H_imag)};
860}

1210template <int dim>
1211void
1212TimeHarmonicMaxwell<dim>::compute_dpg_error(
1213 dealii::Vector<float> &estimated_error_per_cell)
1214{
1215 // For efficiency, the DPG error estimator is computed in the same loop as the
1216 // reconstruction of the interior solution, so we do not implement it here as
1217 // a separate loop. If it is activated, the estimated error per cell is
1218 // computed and stored in the member variable `local_estimated_error_per_cell`
1219 // during the reconstruction and then used for marking the cells for
1220 // refinement here.
1221 auto mpi_communicator = this->triangulation->get_mpi_communicator();
1222
1223 // Here we add a flag to be sure if the local_estimated_error_per_cell has
1224 // been computed before calling this function. If not, all entry of the
1225 // local_estimated_error_per_cell vector will still be 0.
1226 bool dpg_error_computed = false;
1227
1228 for (const auto &cell :
1229 this->dof_handler_trial_interior->active_cell_iterators())
1230 {
1231 if (cell->is_locally_owned())
1232 {
1233 if (this->local_estimated_error_per_cell[cell->active_cell_index()] !=
1234 0)
1235 dpg_error_computed = true;
1236
1237 const unsigned int cell_index = cell->active_cell_index();
1238 estimated_error_per_cell[cell_index] =
1239 this->local_estimated_error_per_cell[cell_index];
1240 }
1241 }
1242
1243 // Reduce the flag across all MPI ranks so that processes that don't own
1244 // any cells don't falsely triggers the assertion.
1245 const bool is_global_dpg_error_computed =
1246 Utilities::MPI::max(static_cast<unsigned int>(dpg_error_computed),
1247 mpi_communicator) != 0;
1248
1249 AssertThrow(

1316 // because the dynamic sparsity pattern is more expensive in terms of
1317 // memory consumption than the static sparsity pattern.

1328 /*keep_constrained_dofs = */ false);

1331 this->locally_owned_dofs_trial_skeleton,

1356 << this->dof_handler_trial_interior->n_dofs() << std::endl;

1359 // mapping, and solution. Note that we provide the interior dof_handler and
1360 // interior solution as it is the one useful for the other physics (we do not

1617 // The DPG method requires the use of skeleton elements and shape functions.
1618 // Because we want to use high-order Nedelec elements for the face, but dealii
1619 // does not support a FE_FaceNedelec, we hack our way around this by using
1620 // full cell elements, but freezing the interior dofs using the function
1621 // constrain_dof_to_zero. To do so, we create a container for all dof indices
1622 // and a container for the face dof indices and we loop on all the faces of
1623 // each cell to find which dofs are on the face and flag them. The remaining
1624 // dofs are then constrained to zero.
1625
1626 std::vector<types::global_dof_index> cell_dof_indices(
1627 this->fe_trial_skeleton->n_dofs_per_cell());
1628 std::vector<types::global_dof_index> face_dof_indices(
1629 this->fe_trial_skeleton->n_dofs_per_face());
1630 std::vector<bool> is_dof_on_face(this->fe_trial_skeleton->n_dofs_per_cell(),
1631 false);

1634 for (const auto &cell :
1635 this->dof_handler_trial_skeleton->active_cell_iterators())
1636 {

1638 {
1639 // Get all dof indices on the cell
1640 cell->get_dof_indices(cell_dof_indices);
1641
1642 // Loop on all the faces of the cell
1643 for (const auto &face : cell->face_iterators())
1644 {
1645 face->get_dof_indices(face_dof_indices);
1646
1647 // Loop on all dofs on the face
1648 for (const auto &face_dof : face_dof_indices)
1649 {
1650 // Find the first iterator in the cell dof indices that
1651 // matches the face dof (find where the face dof is in the
1652 // cell dof indices)
1653 const auto it = std::ranges::find(cell_dof_indices, face_dof);
1654
1655 // If the dof is on the face (find returns the second
1656 // iterator if no match is found), set the corresponding
1657 // flag to true
1658 if (it != cell_dof_indices.end())
1659 {
1660 is_dof_on_face[std::distance(cell_dof_indices.begin(),
1661 it)] = true;
1662 }
1663 }
1664 }
1665
1666 // Loop on all dofs on the cell and constrain the interior ones to
1667 // zero
1668 for (unsigned int index = 0; index < cell_dof_indices.size(); ++index)
1669 {
1670 // If the dof is not on a face, then it is an interior dof
1671 if (!is_dof_on_face[index])
1672 {
1673 this->nonzero_constraints.constrain_dof_to_zero(
1674 cell_dof_indices[index]);
1675 }
1676 }
1677 }
1678 }
1679 this->nonzero_constraints.make_consistent_in_parallel(
1680 this->locally_owned_dofs_trial_skeleton,
1681 this->locally_relevant_dofs_trial_skeleton,
1682 this->triangulation->get_mpi_communicator());
1683}

1685template <int dim>
1686void
1687TimeHarmonicMaxwell<dim>::solve_linear_system()
1688{
1689 TimerOutput::Scope t(this->computing_timer, "Solve linear system");
1690
1691 AssertThrow(
1692 simulation_parameters.linear_solver.at(PhysicsID::electromagnetics)
1693 .preconditioner == Parameters::LinearSolver::PreconditionerType::none,
1694 ExcMessage(
1695 "The time harmonic electromagnetism physics does not support a preconditioner. Please set: subsection linear solver -> subsection electromagnetics -> set preconditioner = none."));
1696
1697 // Define the linear solver tolerance
1698 const double absolute_residual =
1699 simulation_parameters.linear_solver.at(PhysicsID::electromagnetics)
1700 .minimum_residual;
1701 const double relative_residual =
1702 simulation_parameters.linear_solver.at(PhysicsID::electromagnetics)
1703 .relative_residual;
1704
1705 const double rescale_metric = this->get_residual_rescale_metric();
1706 const double rescaled_residual = this->system_rhs.l2_norm() / rescale_metric;
1707 const double linear_solver_tolerance =
1708 std::max(relative_residual * rescaled_residual, absolute_residual);
1709
1710 if (this->simulation_parameters.linear_solver.at(PhysicsID::electromagnetics)
1711 .verbosity != Parameters::Verbosity::quiet)
1712 {
1713 this->pcout << " -Tolerance of iterative solver is : "
1714 << linear_solver_tolerance << std::endl;
1715 }
1716
1717 // Set up the solver control
1718 SolverControl solver_control(this->dof_handler_trial_skeleton->n_dofs(),
1719 linear_solver_tolerance);
1720
1721 // Solve
1722 TrilinosWrappers::SolverCG solver(solver_control);
1723
1724 solver.solve(this->system_matrix,
1725 this->initial_guess_iterative_solver,
1726 this->system_rhs,
1727 *this->preconditioner);
1728
1729 if (simulation_parameters.linear_solver.at(PhysicsID::electromagnetics)
1730 .verbosity != Parameters::Verbosity::quiet)
1731 {
1732 this->pcout << " -CG iterative solver took : "
1733 << solver_control.last_step()
1734 << " steps to reach a residual norm of "
1735 << solver_control.last_value() / rescale_metric << std::endl;
1736 }
1737
1738 // Update the solution vector for the skeleton. We keep the solution in the
1739 // initial_guess_iterative_solver vector as it is used for the next iteration
1740 // of the solver and the present_solution_skeleton vector is used for
1741 // everything else (output, multiphysics coupling, etc.). The major difference
1742 // is that the present_solution_skeleton vector may be renormalized which
1743 // would break the iterative solver if it was used for the next iteration.
1744 nonzero_constraints.distribute(this->initial_guess_iterative_solver);
1745 *this->present_solution_skeleton = this->initial_guess_iterative_solver;
1746
1747 // Reconstruct the interior solution from the skeleton solution
1748 reconstruct_interior_solution();
1749
1750 // We need to apply the scaling to the interior solution as it is the one
1751 // used for the multiphysics coupling and output.
1752 scale_solution_components(*this->dof_handler_trial_interior,
1753 *this->present_solution);
1754
1755 // We also apply the scaling to the skeleton solution for
1756 // consistency, even if it is not used for the multiphysics coupling nor
1757 // outputted at the moment.
1758 scale_solution_components(*this->dof_handler_trial_skeleton,
1759 *this->present_solution_skeleton);
1760
1761 update_material_properties_dependencies();
1762}
1763
1764template <int dim>
1765bool
1766TimeHarmonicMaxwell<dim>::should_solve_auxiliary_physics()
1767{
1768 const Parameters::TimeHarmonicMaxwell<dim> &thm_parameters =
1769 this->simulation_parameters.multiphysics.time_harmonic_maxwell_parameters;
1770 // For steady simulations, each outer iteration (e.g., mesh adaptation
1771 // cycles) should solve electromagnetics regardless of time-coupling settings.
1772 if (this->simulation_control->get_assembly_method() ==
1773 Parameters::SimulationControl::TimeSteppingMethod::steady)
1774 {
1775 return true;
1776 }
1777
1778 // Always solve at the first step of the simulation (simulation start as 0
1779 // when the set up is performed, and then it is incremented before solving the
1780 // physics for the first time, so we want this function to return true in both
1781 // cases.
1782 if (this->simulation_control->get_iteration_number() <= 1)
1783 {
1784 return true;
1785 }
1786 else
1787 {
1788 switch (thm_parameters.time_coupling_strategy)
1789 {
1790 case Parameters::TimeHarmonicMaxwellCouplingStrategy::none:
1791 return false;
1792
1793
1794 case Parameters::TimeHarmonicMaxwellCouplingStrategy::iteration:
1795 // Solve only if we are at a multiple of the specified iteration
1796 // frequency. We subtract 1 since the step number starts at 1.
1797 if ((this->simulation_control->get_iteration_number() - 1) %
1798 thm_parameters.coupling_iteration ==
1799 0)
1800 return true;
1801 else
1802 return false;
1803 case Parameters::TimeHarmonicMaxwellCouplingStrategy::time:
1804 {
1805 // Solve only if the current time has passed a multiple of
1806 // the specified time frequency since the last time the
1807 // electromagnetics were solved. This is done by comparing the
1808 // floor of the current time divided by the time coupling
1809 // parameter to the floor of the previous time divided by the time
1810 // coupling parameter. If they are different, it means we have
1811 // passed a multiple of the time coupling parameter and we should
1812 // solve the electromagnetics.
1813 const int difference_in_time_steps = static_cast<int>(
1814 std::floor(this->simulation_control->get_current_time() /
1815 thm_parameters.coupling_time) -
1816 std::floor(this->simulation_control->get_previous_time() /
1817 thm_parameters.coupling_time));
1818 if (difference_in_time_steps == 1)
1819 return true;
1820 else if (difference_in_time_steps > 1)
1821 {
1822 AssertThrow(
1823 false,
1824 ExcMessage(
1825 "The time coupling strategy for the time-harmonic Maxwell solver is set to 'time' with a coupling time of " +
1826 std::to_string(thm_parameters.coupling_time) +
1827 ", but the simulation time has advanced by " +
1828 std::to_string(
1829 this->simulation_control->get_current_time() -
1830 this->simulation_control->get_previous_time()) +
1831 " since the last time the electromagnetics were solved. Please either reduce the simulation time step or increase the coupling time to avoid missing coupling points."));
1832 return true;
1833 }
1834 else
1835 return false;
1836 }
1837 case Parameters::TimeHarmonicMaxwellCouplingStrategy::threshold:
1838 {
1839 auto mpi_communicator = triangulation->get_mpi_communicator();
1840 // Initialize the temporary variables to determine if the physical
1841 // properties have changed by more than the specified threshold
1842 // since the last time the electromagnetics were solved.
1843 unsigned int material_id;
1844 const PhysicalPropertiesManager &physical_properties_manager =
1845 this->simulation_parameters.physical_properties_manager;
1846 std::vector<std::complex<double>>
1847 effective_electric_permittivities_last_solved;
1848 std::vector<std::complex<double>>
1849 effective_magnetic_permeabilities_last_solved;
1850 std::map<field, std::vector<double>>
1851 field_values_vector_last_solved;
1852 std::vector<double> temperature_last_solved_values;

2104 DPGCopyData &copy_data)
