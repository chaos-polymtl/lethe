71 /**
72 * @brief This class is the base class for all boundary conditions. It stores
73 * the general information that all boundary condition share.
74 * In Lethe, boundary conditions are identified with an id, a type and, in the
75 * special case of periodic boundary condition, a periodic matching id
76 * (periodic_id) and a periodic direction (0, 1 or 2).
77 */
78 class BoundaryConditions
79 {
80 public:
81 /// Map containing the boundary id and the boundary type
82 std::map<types::boundary_id, BoundaryType> type;
83
84 /// Map containing the penalization parameter for weak dirichlet BCs and
85 /// outlets for a corresponding id.
86 std::map<types::boundary_id, double> beta;
87
88 /// Map containing the boundary layer thickness tangent component parameter
89 /// for partial slip for a corresponding id Map containing Dirichlet BCs for
90 /// a corresponding boundary id
91 std::map<types::boundary_id, double> boundary_layer_thickness;
92
93 /// Number of boundary conditions
94 unsigned int number_of_boundary_conditions;
95
96 /// indicator for transient BCs
97 bool time_dependent;
98
99 /// Periodic boundary pairs, keyed by the principal periodic boundary id
100 Parameters::PeriodicBoundaries periodic_boundaries;
101 };
