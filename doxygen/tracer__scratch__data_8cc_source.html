9template <int dim>
10TracerScratchData<dim>::TracerScratchData(
11 const PhysicalPropertiesManager &properties_manager,
12 const FiniteElement<dim> &fe_tracer,
13 const Quadrature<dim> &quadrature,
14 const Quadrature<dim - 1> &face_quadrature,
15 const Mapping<dim> &mapping,
16 const FiniteElement<dim> &fe_fd,
17 const bool uses_dg)
18 : uses_dg(uses_dg)
19 , properties_manager(properties_manager)
20 , fe_values_tracer(mapping,
21 fe_tracer,
22 quadrature,
23 update_values | update_quadrature_points |
24 update_JxW_values | update_gradients |
25 (uses_dg ? update_default : update_hessians))
26 , fe_interface_values_tracer(mapping,
27 fe_tracer,
28 face_quadrature,
29 update_values | update_gradients |
30 update_quadrature_points | update_JxW_values |
31 update_normal_vectors)
32 , fe_values_fd(mapping, fe_fd, quadrature, update_values | update_gradients)
33 , fe_face_values_fd(mapping, fe_fd, face_quadrature, update_values)
34{
35 allocate();
36}

66template <int dim>
67void
68TracerScratchData<dim>::allocate()
69{
70 // Initialize size of arrays
71 this->n_q_points = fe_values_tracer.get_quadrature().size();
72 this->n_dofs = fe_values_tracer.get_fe().n_dofs_per_cell();
73
74 // Initialize arrays related to quadrature
75 this->JxW.reinit(n_q_points);
76
77 // Forcing term array
78 this->source = std::vector<double>(n_q_points);
79
80 // Initialize arrays related to velocity
81 this->velocities.first_vector_component = 0;
82 // Velocity
83 this->velocity_values = std::vector<Tensor<1, dim>>(n_q_points);
84 this->velocity_divergences = std::vector<double>(n_q_points);
85
86 // Tracer
87 this->tracer_values = std::vector<double>(n_q_points);
88 this->tracer_gradients = std::vector<Tensor<1, dim>>(n_q_points);
89 this->previous_tracer_gradients =
90 std::vector<Tensor<1, dim>>(this->n_q_points);
91 this->tracer_laplacians = std::vector<double>(n_q_points);
92 this->tracer_diffusivity = std::vector<double>(n_q_points);
93 this->tracer_diffusivity_0 = std::vector<double>(n_q_points);
94 this->tracer_diffusivity_1 = std::vector<double>(n_q_points);
95 this->tracer_reaction_prefactor = std::vector<double>(n_q_points);
96 this->grad_tracer_reaction_prefactor = std::vector<double>(n_q_points);
97
98 // Arrays related to the faces, which are only used by the DG assemblers
99 const unsigned int n_face_q_points =
100 fe_interface_values_tracer.get_quadrature().size();
101 const unsigned int max_n_interface_dofs = 2 * n_dofs;
102 this->tracer_diffusivity_face = std::vector<double>(n_face_q_points);
103 this->face_velocity_values = std::vector<Tensor<1, dim>>(n_face_q_points);
104 this->values_here = std::vector<double>(n_face_q_points);
105 this->values_there = std::vector<double>(n_face_q_points);
106 this->tracer_value_jump = std::vector<double>(n_face_q_points);
107 this->gradients_here = std::vector<Tensor<1, dim>>(n_face_q_points);
108 this->tracer_average_gradient = std::vector<Tensor<1, dim>>(n_face_q_points);
109 this->face_phi_here.reinit(n_face_q_points, max_n_interface_dofs);
110 this->face_phi_there.reinit(n_face_q_points, max_n_interface_dofs);
111 this->face_jump_phi.reinit(n_face_q_points, max_n_interface_dofs);
112 this->face_average_grad_phi_dot_n.reinit(n_face_q_points,
113 max_n_interface_dofs);
114
115 // Solid signed distance function
116 if (properties_manager.field_is_required(field::levelset))
117 this->sdf_values = std::vector<double>(n_q_points);
118
119 // Velocity for BDF schemes
120 this->previous_tracer_values =
121 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
122 std::vector<double>(n_q_points));
123
124 // Initialize arrays related to shape functions
125 // Velocity shape functions
126 this->phi.reinit(n_q_points, n_dofs);
127 this->grad_phi.reinit(n_q_points, n_dofs);
128 if (!uses_dg)
129 {
130 this->hess_phi.reinit(n_q_points, n_dofs);
131 this->laplacian_phi.reinit(n_q_points, n_dofs);
132 }
133
134 if (properties_manager.field_is_required(field::levelset))
135 fields.insert(
136 std::pair<field, std::vector<double>>(field::levelset, n_q_points));
137 if (properties_manager.field_is_required(field::tracer_concentration))
138 fields.insert(
139 std::pair<field, std::vector<double>>(field::tracer_concentration,
140 n_q_points));
141}

144template <int dim>
145template <typename VectorType>
146void
147TracerScratchData<dim>::reinit(
148 const typename DoFHandler<dim>::active_cell_iterator &cell,
149 const VectorType &current_solution,
150 const std::vector<VectorType> &previous_solutions,
151 const Function<dim> *source_function)
152{
153 this->fe_values_tracer.reinit(cell);
154
155 quadrature_points = this->fe_values_tracer.get_quadrature_points();
156 auto &fe_tracer = this->fe_values_tracer.get_fe();
157
158 source_function->value_list(quadrature_points, source);
159
160 // Compute cell diameter
161 double cell_measure =
162 compute_cell_measure_with_JxW(this->fe_values_tracer.get_JxW_values());
163 this->cell_size = compute_cell_diameter<dim>(cell_measure, fe_tracer.degree);
164
165 // Gather tracer (values, gradient and laplacian)
166 this->fe_values_tracer.get_function_values(current_solution,
167 this->tracer_values);
168 this->fe_values_tracer.get_function_gradients(current_solution,
169 this->tracer_gradients);
170
171 // The laplacian and the previous gradient are only used by the
172 // stabilization of the continuous formulation
173 if (!uses_dg)
174 {
175 this->fe_values_tracer.get_function_laplacians(current_solution,
176 this->tracer_laplacians);
177 this->fe_values_tracer.get_function_gradients(
178 previous_solutions[0], this->previous_tracer_gradients);
179 }
180
181 // Gather previous tracer values
182 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
183 {
184 this->fe_values_tracer.get_function_values(previous_solutions[p],
185 previous_tracer_values[p]);
186 }
187
188 for (unsigned int q = 0; q < n_q_points; ++q)
189 {
190 this->JxW[q] = this->fe_values_tracer.JxW(q);
191
192 for (unsigned int k = 0; k < n_dofs; ++k)
193 {
194 // Shape function
195 this->phi[q][k] = this->fe_values_tracer.shape_value(k, q);
196 this->grad_phi[q][k] = this->fe_values_tracer.shape_grad(k, q);
197 }
198
199 if (!uses_dg)
200 for (unsigned int k = 0; k < n_dofs; ++k)
201 {
202 this->hess_phi[q][k] = this->fe_values_tracer.shape_hessian(k, q);
203 this->laplacian_phi[q][k] = trace(this->hess_phi[q][k]);
204 }
205 }
206
207 boundary_index = 0;
208}

212void
213TracerScratchData<dim>::reinit_signed_distance(
214 const typename DoFHandler<dim>::active_cell_iterator &cell,
215 const std::shared_ptr<Shape<dim>> &immersed_solid_shape)
216{
217 if (properties_manager.field_is_required(field::levelset))
218 {
219 this->fe_values_tracer.reinit(cell);
220
221 quadrature_points = this->fe_values_tracer.get_quadrature_points();
222
223 Assert(
224 immersed_solid_shape != nullptr,
225 ExcMessage(
226 "Shape is required for tracer assembly, but the shape is a nullptr"));
227
228 for (unsigned int q = 0; q < n_q_points; q++)
229 sdf_values[q] =
230 immersed_solid_shape->value_with_cell_guess(quadrature_points[q],
231 cell);
232 }

263template <int dim>
264template <typename VectorType>
265void
266TracerScratchData<dim>::reinit_velocity(
267 const typename DoFHandler<dim>::active_cell_iterator &cell,
268 const VectorType &velocity_solution,
269 const Parameters::ALE<dim> &ale,
270 const std::shared_ptr<Functions::ParsedFunction<dim>> &drift_velocity)
271{
272 this->fe_values_fd.reinit(cell);
273
274 this->fe_values_fd[velocities].get_function_values(velocity_solution,
275 velocity_values);
276
277 this->fe_values_fd[velocities].get_function_divergences(velocity_solution,
278 velocity_divergences);
279
280 // Add the drift velocity to the velocity to account for tracer drift flux
281 // modeling
282 Tensor<1, dim> drift_velocity_tensor;
283 Vector<double> drift_velocity_vector(dim);
284
285 for (unsigned int q = 0; q < n_q_points; ++q)
286 {
287 drift_velocity->vector_value(quadrature_points[q], drift_velocity_vector);
288 for (int d = 0; d < dim; ++d)
289 drift_velocity_tensor[d] = drift_velocity_vector[d];
290
291 velocity_values[q] += drift_velocity_tensor;
292 }
293
294 if (!ale.enabled())
295 return;
296
297 // ALE enabled, so extract the ALE velocity and subtract it from the
298 // velocity obtained from the fluid dynamics
299 Tensor<1, dim> velocity_ale;
300 std::shared_ptr<Functions::ParsedFunction<dim>> velocity_ale_function =
301 ale.velocity;
302 Vector<double> velocity_ale_vector(dim);
303
304 for (unsigned int q = 0; q < n_q_points; ++q)
305 {
306 velocity_ale_function->vector_value(quadrature_points[q],
307 velocity_ale_vector);
308 for (int d = 0; d < dim; ++d)
309 velocity_ale[d] = velocity_ale_vector[d];
310
311 velocity_values[q] -= velocity_ale;
312 }
313}

316template <int dim>
317template <typename VectorType>
318void
319TracerScratchData<dim>::reinit_internal_face(
320 const typename DoFHandler<dim>::active_cell_iterator &cell,
321 const unsigned int &face_no,
322 const unsigned int &sub_face_no,
323 const typename DoFHandler<dim>::active_cell_iterator &neigh_cell,
324 const unsigned int &neigh_face_no,
325 const unsigned int &neigh_sub_face_no,
326 const VectorType &current_solution)
327{
328 fe_interface_values_tracer.reinit(
329 cell, face_no, sub_face_no, neigh_cell, neigh_face_no, neigh_sub_face_no);
330 face_quadrature_points = fe_interface_values_tracer.get_quadrature_points();
331
332 n_interface_dofs = fe_interface_values_tracer.n_current_interface_dofs();
333
334 const double extent_here = cell->measure() / cell->face(face_no)->measure();
335 const double extent_there =
336 neigh_cell->measure() / neigh_cell->face(neigh_face_no)->measure();
337
338 penalty_factor = get_penalty_factor(fe_values_tracer.get_fe().degree,
339 extent_here,
340 extent_there);
341
342
343 fe_interface_values_tracer.get_fe_face_values(0).get_function_values(
344 current_solution, values_here);
345 fe_interface_values_tracer.get_fe_face_values(1).get_function_values(
346 current_solution, values_there);
347
348 fe_interface_values_tracer.get_average_of_function_gradients(
349 current_solution, tracer_average_gradient);
350
351 const std::vector<Tensor<1, dim>> &normals =
352 fe_interface_values_tracer.get_normal_vectors();
353
354 for (unsigned int q = 0; q < face_quadrature_points.size(); ++q)
355 {
356 // The jump is defined as the value on cell 0 minus the value on cell 1
357 tracer_value_jump[q] = values_here[q] - values_there[q];
358
359 // Gather the shape function quantities once per quadrature point
360 // since the FEInterfaceValues accessors map the interface dof to the
361 // cell dofs at every call.
362 for (unsigned int i = 0; i < n_interface_dofs; ++i)
363 {
364 face_phi_here[q][i] =
365 fe_interface_values_tracer.shape_value(true, i, q);
366 face_phi_there[q][i] =
367 fe_interface_values_tracer.shape_value(false, i, q);
368 face_jump_phi[q][i] = face_phi_here[q][i] - face_phi_there[q][i];
369 face_average_grad_phi_dot_n[q][i] =
370 fe_interface_values_tracer.average_of_shape_gradients(i, q) *
371 normals[q];
372 }
373 }
374}

457template <int dim>
458void
459TracerScratchData<dim>::reinit_mortar(
460 const typename DoFHandler<dim>::active_cell_iterator &cell,
461 const Parameters::Mortar<dim> &mortar_parameters,
462 const double &time)
463{
464 // Get updated rotor angular velocity
465 mortar_parameters.rotor_angular_velocity->set_time(time);
466 const double rotor_angular_velocity =
467 mortar_parameters.rotor_angular_velocity->value(Point<dim>());
468
469 // Apply prescribed rotor angular velocity only at rotor cells
470 // (material_id = 1)
471 double omega = 0.0;
472 if (cell->material_id() == 1)
473 omega = rotor_angular_velocity;
474
475 // The mortar implementation is not defined for more than two
476 // materials. Throw when running debug mode as a security measure to prevent
477 // this case
478 Assert(
479 cell->material_id() < 2,
480 ExcMessage(
481 "The material id in a cell was identified as equal or greater than 2, however the mortar implementation is not defined for more than two materials."));
482
483 // Compute rotor linear velocity at quadrature points
484 rotor_linear_velocity_values = std::vector<Tensor<1, dim>>(this->n_q_points);
485 for (unsigned int q = 0; q < n_q_points; ++q)
486 {
487 // Shift quadrature point by the center of rotation
488 const auto p = fe_values_tracer.quadrature_point(q) -
489 mortar_parameters.center_of_rotation;
490
491 // Compute terms of u_ale
492 rotor_linear_velocity_values[q] =
493 LetheGridTools::angular_to_linear_velocity(
494 omega, p, mortar_parameters.rotation_axis_direction);
495
496 // Subtract u_ale from the advective velocity
497 velocity_values[q] -= rotor_linear_velocity_values[q];
498 }
499}

502template <int dim>
503void
504TracerScratchData<dim>::calculate_face_physical_properties()
505{
506 const auto diffusivity_model = properties_manager.get_tracer_diffusivity();
507
508 // If the tracer diffusivity depends on the level set, take this into
509 // account
510 if (properties_manager.field_is_required(field::levelset))
511 set_field_vector(field::levelset, sdf_values, fields);
512
513
514 diffusivity_model->vector_value(fields, tracer_diffusivity_face);
515}

518template <int dim>
519void
520TracerScratchData<dim>::calculate_physical_properties()
521{
522 if (properties_manager.field_is_required(field::levelset))
523 set_field_vector(field::levelset, this->sdf_values, this->fields);
524
525 if (properties_manager.field_is_required(field::tracer_concentration))
526 set_field_vector(field::tracer_concentration,
527 this->tracer_values,
528 this->fields);
529
530 switch (properties_manager.get_number_of_fluids())
531 {
532 case 1:
533 {
534 const auto diffusivity_model =
535 properties_manager.get_tracer_diffusivity();
536 diffusivity_model->vector_value(fields, tracer_diffusivity);
537 const auto reaction_prefactor_model =
538 properties_manager.get_tracer_reaction_prefactor();
539 reaction_prefactor_model->vector_value(fields,
540 tracer_reaction_prefactor);
541 reaction_prefactor_model->vector_jacobian(
542 fields,
543 field::tracer_concentration,
544 grad_tracer_reaction_prefactor);
545
546 break;
547 }
548 case 2:
549 {
550 throw std::runtime_error(
551 "Unsupported number of fluids (2) with the tracer physics.");
552 // In this case, we need both density and viscosity
553 const auto diffusivity_models =
554 properties_manager.get_tracer_diffusivity_vector();
555
556 diffusivity_models[0]->vector_value(fields, tracer_diffusivity_0);
557 diffusivity_models[1]->vector_value(fields, tracer_diffusivity_1);
558
559 // TODO Incomplete at the present time because the tracer CLS
560 // complete is not finished Blend the physical properties
561 // using the CLS field
562 // for (unsigned int q = 0; q < this->n_q_points; ++q)
563 // {
564 // tracer_diffusivity[q] =
565 // calculate_point_property(this->phase_values[q],
566 // this->density_0[q],
567 // this->density_1[q]);
568 // }
569 break;
570 }
571 default:
572 throw std::runtime_error("Unsupported number of fluids (>2)");
573 }
574}
