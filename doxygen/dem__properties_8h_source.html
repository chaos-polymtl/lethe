24 /**
25 * @brief Identifies which physics is solved for the DEM particles.
26 *
27 * Two solver types are currently supported:
28 * - dem: the particles only carry their kinematic properties.
29 * - dem_mp: DEM multiphysics simulation, in which the particles also carry
30 * thermal properties.
31 *
32 * This is independent of the presence of a fluid: it is the application
33 * which decides whether the particles are coupled to a CFD solver, and thus
34 * whether the CFD-DEM index layout of the ParticleHandler PropertyPool is
35 * used.
36 */

261 /**
262 * @brief Compare the position of a property in two different index sets.
263 *
264 * The arguments are taken as integers rather than as enumerators so that
265 * enumerators of two distinct index sets may be compared without triggering
266 * -Wenum-compare.
267 *
268 * @param[in] first_index Position of the property in the first index set.
269 * @param[in] second_index Position of the property in the second index set.
270 * @return Whether the property is stored at the same position in both sets.
271 */
272 constexpr bool
273 is_stored_at_the_same_index(const int first_index, const int second_index)
274 {
275 return first_index == second_index;
276 }

344 /**
345 * @brief Whether a properties index set carries thermal properties.
346 *
347 * This is true for the index sets used by the multiphysics solvers, which
348 * store the temperature (T) and the specific heat of every particle. It is
349 * used to enable, at compile time, the heat transfer between particles and
350 * the integration of their temperature.
351 *
352 * @tparam PropertiesIndex Index of the properties used within the ParticleHandler.
353 */
354 template <typename PropertiesIndex>
355 inline constexpr bool has_thermal_properties = false;
356
357 template <>
358 inline constexpr bool
359 has_thermal_properties<DEMMPProperties::PropertiesIndex> = true;
360
361 template <>
362 inline constexpr bool
363 has_thermal_properties<CFDDEMMPProperties::PropertiesIndex> = true;
364
365
366 /**
367 * @brief Whether a properties index set carries fluid coupling properties.
368 *
369 * This is true for the index sets used by the CFD-DEM solvers, which store
370 * the fluid-particle interaction forces and torques, the volumetric
371 * contribution and the momentum transfer coefficient of every particle.
372 *
373 * @tparam PropertiesIndex Index of the properties used within the ParticleHandler.
374 */
375 template <typename PropertiesIndex>
376 inline constexpr bool has_fluid_coupling_properties = false;
377
378 template <>
379 inline constexpr bool
380 has_fluid_coupling_properties<CFDDEMProperties::PropertiesIndex> = true;
381
382 template <>
383 inline constexpr bool
384 has_fluid_coupling_properties<CFDDEMMPProperties::PropertiesIndex> = true;

387 /**
388 * @brief The pure DEM index set matching a given index set.
389 *
390 * A CFD-DEM simulation is usually started from the checkpoint of a pure DEM
391 * simulation, whose particle handler carries fewer properties. This alias
392 * gives the index set that checkpoint must have been written with: a
393 * multiphysics CFD-DEM simulation is always restarted from a multiphysics
394 * DEM simulation, so that the temperature and the specific heat of the
395 * particles are carried over rather than reset.
396 *
397 * @tparam PropertiesIndex Index of the properties used within the ParticleHandler.
398 */
399 template <typename PropertiesIndex>
400 using matching_dem_properties_index =
401 std::conditional_t<has_thermal_properties<PropertiesIndex>,
402 DEMMPProperties::PropertiesIndex,
403 DEMProperties::PropertiesIndex>;
