26 CLSBoundaryConditions<dim>::declare_default_entry(

29 {
30 prm.declare_entry("type",
31 "none",
32 Patterns::Selection("none|dirichlet|periodic"),
33 "Type of boundary condition for CLS"
34 "Choices are <none|dirichlet|periodic>.");
35
36 prm.declare_entry("id",
37 Utilities::int_to_string(default_boundary_id, 2),
38 Patterns::List(Patterns::Integer()),
39 "Mesh id for boundary conditions");

41 Functions::ParsedFunction<dim> temporary_function;

44 prm.leave_subsection();

48 "periodic id",
49 "-1",
50 Patterns::Integer(),
51 "Mesh id for periodic face matching. Default entry is -1 to ensure that the periodic id is set by the user");
52
53 prm.declare_entry("periodic direction",
54 "0",
55 Patterns::Integer(),
56 "Direction for periodic boundary condition");

89 template <int dim>
90 void
91 CLSBoundaryConditions<dim>::parse_boundary(ParameterHandler &prm)
92 {
93 // Parse the list of boundary ids
94 std::vector<types::boundary_id> boundary_ids =
95 convert_string_to_vector<types::boundary_id>(prm, "id");
96
97 // Check if the list contains at least one boundary id
98 AssertThrow(
99 boundary_ids.size() > 0,
100 ExcMessage(
101 "A boundary id has not been set for one of the CLS boundary conditions. Please ensure that the id is set for every boundary condition."));
102
103 // Loop through all boundary ids to ensure that they are all non-negative
104 // and unique
105 for (const auto boundary_id : boundary_ids)
106 {
107 AssertThrow(this->type.find(boundary_id) == this->type.end(),
108 CLSBoundaryDuplicated(boundary_id));
109
110 if (auto const option = prm.get("type"); option == "none")
111 {
112 this->type[boundary_id] = BoundaryType::none;
113 }
114
115 if (auto const option = prm.get("type"); option == "dirichlet")
116 {
117 this->type[boundary_id] = BoundaryType::cls_dirichlet;
118 prm.enter_subsection("dirichlet");
119 phase_indicator[boundary_id] =
120 std::make_shared<Functions::ParsedFunction<dim>>();
121 phase_indicator[boundary_id]->parse_parameters(prm);
122 prm.leave_subsection();
123 }
124 if (auto const option = prm.get("type"); option == "periodic")
125 {
126 types::boundary_id periodic_boundary_id =
127 prm.get_integer("periodic id");
128
129 this->type[boundary_id] = BoundaryType::periodic;
130
131 // We attribute a periodic neighbor boundary type to the neighbor
132 // boundary to ensure that all boundaries have a defined type
133 this->type[periodic_boundary_id] = BoundaryType::periodic_neighbor;
134
135 // We store the periodic id and direction
136 this->periodic_boundaries[boundary_id] = {
137 .neighbor_id = periodic_boundary_id,
138 .direction = static_cast<unsigned int>(
139 prm.get_integer("periodic direction"))};
140 }
141 }
142 }

144 template <int dim>
145 void
146 CLSBoundaryConditions<dim>::parse_parameters(ParameterHandler &prm)
147 {
148 prm.enter_subsection("boundary conditions CLS");
149 {
150 this->number_of_boundary_conditions = prm.get_integer("number");
151 this->time_dependent = prm.get_bool("time dependent");
152
153 for (unsigned int n = 0; n < this->number_of_boundary_conditions; n++)
154 {
155 prm.enter_subsection("bc " + std::to_string(n));
156 {
157 parse_boundary(prm);
158 }
159 prm.leave_subsection();
160 }
161 }
162 prm.leave_subsection();
163 }
