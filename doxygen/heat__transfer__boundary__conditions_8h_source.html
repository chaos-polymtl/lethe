48 /**
49 * @brief This class manages the boundary conditions for Heat-Transfer solver
50 * It introduces the boundary functions and declares the boundary conditions
51 * coherently.
52 * The members "value", "h" and "Tinf" contain double used for bc calculation:
53 *
54 * - if bc type is "noflux", no flux is assembled at the boundary
55 *
56 * - if bc type is "temperature" (Dirichlet condition), "value" is the
57 * double passed to the deal.ii ConstantFunction
58 *
59 * - if bc type is "convection_radiation" (Robin condition), "h" is the
60 * convective heat transfer coefficient and "Tinf" is the
61 * environment temperature at the boundary, "emissivity" is the emissivity
62 * coefficient, and "Stefan-Boltzmann constant" is the Stefan-Boltzmann
63 * constant = 5.6703*10-8 \f$(W.m^{-2}.K^{-4})\f$. It is also possible to
64 * impose a heat flux using "heat_flux"
65 *
66 */
67 template <int dim>
68 class HeatTransferBoundaryConditions
69 : public BoundaryConditions<HeatTransferBoundaryType>
70 {
71 public:
72 /// Temperature values for Dirichlet boundary conditions
73 std::map<types::boundary_id,
74 std::shared_ptr<Functions::ParsedFunction<dim>>>
75 dirichlet_value;
76 /// Convective heat transfer coefficients for Robin conditions
77 std::map<types::boundary_id,
78 std::shared_ptr<Functions::ParsedFunction<dim>>>
79 h;
80 /// Environment temperatures for Robin conditions
81 std::map<types::boundary_id,
82 std::shared_ptr<Functions::ParsedFunction<dim>>>
83 Tinf;
84 /// Emissivity coefficients for radiation boundary conditions
85 std::map<types::boundary_id,
86 std::shared_ptr<Functions::ParsedFunction<dim>>>
87 emissivity;
88 /// Heat-flux values for flux boundary conditions
89 std::map<types::boundary_id,
90 std::shared_ptr<Functions::ParsedFunction<dim>>>
91 heat_flux_bc;
92 /// Stefan-Boltzmann constant used in radiation boundary conditions
93 double Stefan_Boltzmann_constant;
94
95 /**
96 * @brief Declare the default parameters for one heat transfer boundary condition.
