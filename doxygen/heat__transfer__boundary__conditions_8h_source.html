27 /**
28 * @brief This class manages the boundary conditions for Heat-Transfer solver
29 * It introduces the boundary functions and declares the boundary conditions
30 * coherently.
31 * The members "value", "h" and "Tinf" contain double used for bc calculation:
32 *
33 * - if bc type is "noflux", no flux is assembled at the boundary
34 *
35 * - if bc type is "temperature" (Dirichlet condition), "value" is the
36 * double passed to the deal.ii ConstantFunction
37 *
38 * - if bc type is "convection-radiation-flux" (Robin condition), "h" is the
39 * convective heat transfer coefficient and "Tinf" is the
40 * environment temperature at the boundary, "emissivity" is the emissivity
41 * coefficient, and "Stefan-Boltzmann constant" is the Stefan-Boltzmann
42 * constant = 5.6703*10-8 \f$(W.m^{-2}.K^{-4})\f$. It is also possible to
43 * impose a heat flux using "heat_flux"
44 *
45 */
46 template <int dim>
47 class HeatTransferBoundaryConditions : public BoundaryConditions
48 {
49 public:
50 /// Temperature values for Dirichlet boundary conditions
51 std::map<types::boundary_id,
52 std::shared_ptr<Functions::ParsedFunction<dim>>>
53 dirichlet_value;
54 /// Convective heat transfer coefficients for Robin conditions
55 std::map<types::boundary_id,
56 std::shared_ptr<Functions::ParsedFunction<dim>>>
57 h;
58 /// Environment temperatures for Robin conditions
59 std::map<types::boundary_id,
60 std::shared_ptr<Functions::ParsedFunction<dim>>>
61 Tinf;
62 /// Emissivity coefficients for radiation boundary conditions
63 std::map<types::boundary_id,
64 std::shared_ptr<Functions::ParsedFunction<dim>>>
65 emissivity;
66 /// Heat-flux values for flux boundary conditions
67 std::map<types::boundary_id,
68 std::shared_ptr<Functions::ParsedFunction<dim>>>
69 heat_flux_bc;
70 /// Stefan-Boltzmann constant used in radiation boundary conditions
71 double Stefan_Boltzmann_constant;
