44 /**
45 * @brief DEM data structure containers

110 // Value stored per "particle one" in the adjacency containers below. The
111 // iterator to particle one is shared by all its contacts, so it is stored
112 // once here instead of being duplicated in every particle-particle contact
113 // info. second_particles maps uses each particle two id (key)
114 // to access the contact info (value) between particle one and two.
115 // (particle one iterator, <particle id, particle-particle info>)
116 struct adjacent_particle_pairs_value
117 {
118 Particles::ParticleIterator<dim> particle_one;
119 particle_contact_info second_particles;
120 };
121
122 // Periodic counterpart of adjacent_particle_pairs_value.
123 // (particle one iterator, <particle id, periodic particle-particle info>)
124 struct periodic_adjacent_particle_pairs_value

126 Particles::ParticleIterator<dim> particle_one;
127 periodic_particle_contact_info second_particles;
128 };

130 // <particle id, (particle one iterator, <particle id, particle-particle

194 using cells_total_neighbor_list = ankerl::unordered_dense::map<
195 types::global_cell_index,
196 std::vector<typename Triangulation<dim>::active_cell_iterator>>;
197
198 // <global_face_id, (tensor, point)>
199 using boundary_points_and_normal_vectors =
200 ankerl::unordered_dense::map<global_face_id,

202
203 // <unsigned int, <global_face_id, tensor>>
204 using vector_on_boundary =

208 // [cell iterators]

210 std::vector<typename Triangulation<dim>::active_cell_iterator>;

212 // [cell iterators]
213 using cell_set =

215
216 // <cell id, periodic cells info>
217 using periodic_boundaries_cells_info =
218 std::unordered_multimap<types::global_cell_index,

220
221 // <cell id, integer value>
222 using cell_index_int_map =

224 };
