30/**
31 * @brief Compute the effective electromagnetic properties of a material at a
32 * set of points. The effective electric permittivity is
33 * \f$\varepsilon' + i(\varepsilon'' + \sigma)\f$ and the effective magnetic
34 * permeability is \f$\mu' + i\mu''\f$, where each property is evaluated with
35 * the physical property models of the material. If the material is an
36 * effective medium, the properties are given by the effective medium models of
37 * its fluid-dispersed phase interaction. The conductivity of both phases is
38 * then already included in the imaginary part of the effective permittivity.
39 *
40 * @note The time-harmonic Maxwell equations do not support multiple
41 * fluids so the fluid_id is not necessary to determine the material
42 * properties. The material properties only depend on the material_id, which
43 * is used to identify the different materials in the input file and assign
44 * them their corresponding properties.
45 *
46 * @param[in] physical_properties_manager The object that manages the
47 * physical properties of the problem and provides them at any given position
48 * of the domain.
49 * @param[in] field_values_vector A map containing the vector values of the
50 * fields at the current position. This is used to compute the material
51 * properties that depend on the fields.
52 * @param[in] material_id The material id of the current position, used to
53 * determine the appropriate material properties from the input parameters.
54 * @param[out] effective_electric_permittivity Effective electric
55 * permittivity at the current position. This value is updated in place by the
56 * function.
57 * @param[out] effective_magnetic_permeability Effective magnetic
58 * permeability at the current position. This value is updated in place by the
59 * function.
60 */
61void
62compute_effective_electromagnetic_properties(
63 const PhysicalPropertiesManager &physical_properties_manager,
64 const std::map<field, std::vector<double>> &field_values_vectors,
65 const unsigned int material_id,
66 std::vector<std::complex<double>> &effective_electric_permittivities,
67 std::vector<std::complex<double>> &effective_magnetic_permeabilities);
68
69/**
70 * @brief Check if at least one electromagnetic property of a material depends
71 * on a field. If the material is an effective medium, the dependencies are
72 * those of the effective medium models of its fluid-dispersed phase
73 * interaction. Otherwise, they are those of the electric permittivity, electric
74 * conductivity and magnetic permeability models of the material.
75 *
76 * @param[in] physical_properties_manager The object that manages the
77 * physical properties of the problem.
78 * @param[in] material_id The material id of the cell.
79 * @param[in] field_id The field for which the dependency is checked.
80 * @return true if at least one electromagnetic property of the material
81 * depends on the field, false otherwise.
82 */
83bool
84electromagnetic_properties_depend_on(
85 const PhysicalPropertiesManager &physical_properties_manager,
86 const unsigned int material_id,
87 const field field_id);

90/**
91 * @brief Class that stores the information required by the assembly of the
92 * Discontinuous Petrov-Galerkin (DPG) ultraweak formulation of the
93 * time-harmonic Maxwell equations. For each cell, it evaluates the complex
94 * shape functions of the test space (\f$\mathbf{F}\f$ for the electric test
95 * functions and \f$\mathbf{I}\f$ for the magnetic ones, with their curls and
96 * complex conjugates) and of the interior trial space (\f$\mathbf{E}\f$ and
97 * \f$\mathbf{H}\f$) and the effective material properties at the quadrature
98 * points. For each face of the cell, it evaluates the test functions and the
99 * tangential traces of the skeleton trial space (\f$\hat{\mathbf{E}}\f$ and
100 * \f$\hat{\mathbf{H}}\f$) at the face quadrature points. The complex values are
101 * stored with their complex conjugates because the conjugate of a complex
102 * tensor is not implemented in deal.II.
103 *
104 * @tparam dim An integer that denotes the number of spatial dimensions. Only
105 * dim = 3 is supported.
106 *
107 * @ingroup solvers
108 */
109template <int dim>
110class TimeHarmonicMaxwellScratchData : public PhysicsScratchDataBase
111{
112public:
113 /**
114 * @brief Bit flags identifying to which field a shape function of the
115 * [E_real, E_imag, H_real, H_imag] finite element systems belongs. The
116 * composite flags is_electric and is_magnetic allow to check whether a shape
117 * function belongs to the electric or magnetic field, regardless of whether
118 * it is the real or imaginary part.
119 */
120 enum ShapeFunctionType : unsigned char
121 {
122 electric_real = 1u << 0,
123 electric_imag = 1u << 1,
124 magnetic_real = 1u << 2,
125 magnetic_imag = 1u << 3,
126
127 is_electric = electric_real | electric_imag,
128 is_magnetic = magnetic_real | magnetic_imag
129 };

131 /**
132 * @brief Constructor. Creates the FEValues and FEFaceValues objects of the
133 * interior trial, skeleton trial and test spaces and allocates the memory of
134 * the scratch. It does not do any evaluation, since this is done at the cell
135 * level.
136 *
137 * @param[in] properties_manager Manager of the physical property models.
138 *
139 * @param[in] fe_trial_interior Finite element of the interior trial space.
140 *
141 * @param[in] fe_trial_skeleton Finite element of the skeleton trial space.
142 *
143 * @param[in] fe_test Finite element of the test space.
144 *
145 * @param[in] cell_quadrature Quadrature used for the cell integrals.
146 *
147 * @param[in] face_quadrature Quadrature used for the face integrals.
148 *
149 * @param[in] mapping Mapping of the domain.
150 */
151 TimeHarmonicMaxwellScratchData(
152 const PhysicalPropertiesManager &properties_manager,
153 const FiniteElement<dim> &fe_trial_interior,
154 const FiniteElement<dim> &fe_trial_skeleton,
155 const FiniteElement<dim> &fe_test,
156 const Quadrature<dim> &cell_quadrature,
157 const Quadrature<dim - 1> &face_quadrature,
158 const Mapping<dim> &mapping)
159 : properties_manager(properties_manager)
160 , fe_values_trial_interior(mapping,
161 fe_trial_interior,
162 cell_quadrature,
163 update_values | update_quadrature_points |
164 update_JxW_values)
165 , fe_values_test(mapping,
166 fe_test,
167 cell_quadrature,
168 update_values | update_gradients)
169 , fe_face_values_trial_skeleton(mapping,
170 fe_trial_skeleton,
171 face_quadrature,
172 update_values | update_quadrature_points |
173 update_normal_vectors | update_JxW_values)
174 , fe_face_values_test(mapping, fe_test, face_quadrature, update_values)
175 {
176 gather_temperature = false;
177 gather_void_fraction = false;
178 allocate();
179 }
180
181 /**
182 * @brief Copy constructor. Same as the main constructor. It only uses the
183 * other scratch to build the FEValues objects, it does not copy the content
184 * of the other scratch since, by definition of the WorkStream mechanism, the
185 * content of the scratch is reset on a cell basis.
186 *
187 * @param[in] sd The scratch data to be copied.
188 */
189 TimeHarmonicMaxwellScratchData(const TimeHarmonicMaxwellScratchData<dim> &sd)
190 : properties_manager(sd.properties_manager)
191 , fe_values_trial_interior(sd.fe_values_trial_interior.get_mapping(),
192 sd.fe_values_trial_interior.get_fe(),
193 sd.fe_values_trial_interior.get_quadrature(),
194 update_values | update_quadrature_points |
195 update_JxW_values)
196 , fe_values_test(sd.fe_values_test.get_mapping(),
197 sd.fe_values_test.get_fe(),
198 sd.fe_values_test.get_quadrature(),
199 update_values | update_gradients)
200 , fe_face_values_trial_skeleton(
201 sd.fe_face_values_trial_skeleton.get_mapping(),
202 sd.fe_face_values_trial_skeleton.get_fe(),
203 sd.fe_face_values_trial_skeleton.get_quadrature(),
204 update_values | update_quadrature_points | update_normal_vectors |
205 update_JxW_values)
206 , fe_face_values_test(sd.fe_face_values_test.get_mapping(),
207 sd.fe_face_values_test.get_fe(),
208 sd.fe_face_values_test.get_quadrature(),
209 update_values)
210 {
211 gather_temperature = sd.gather_temperature;
212 gather_void_fraction = sd.gather_void_fraction;
213
214 allocate();
215
216 if (sd.gather_temperature)
217 enable_temperature(sd.fe_values_temperature->get_fe(),
218 sd.fe_values_temperature->get_quadrature(),
219 sd.fe_face_values_temperature->get_quadrature(),
220 sd.fe_values_temperature->get_mapping());
221
222 if (sd.gather_void_fraction)
223 enable_void_fraction(sd.fe_values_void_fraction->get_fe(),
224 sd.fe_values_void_fraction->get_quadrature(),
225 sd.fe_face_values_void_fraction->get_quadrature(),
226 sd.fe_values_void_fraction->get_mapping());
227 }

238 /**
239 * @brief Classify the shape functions of a finite element space of the
240 * time-harmonic Maxwell equations according to the field they belong to.
241 * Each shape function of the [E_real, E_imag, H_real, H_imag] finite element
242 * system receives the ShapeFunctionType flag of its field, which is
243 * determined with FiniteElement::shape_function_belongs_to(). Since every
244 * shape function belongs to exactly one field, the local indices of the
245 * electric and magnetic shape functions are then gathered in two lists.
246 *
247 * @param[in] fe Finite element system of the [E_real, E_imag, H_real,
248 * H_imag] fields (interior trial, skeleton trial or test space).
249 *
250 * @param[out] shape_function_type ShapeFunctionType flag of each shape
251 * function of the finite element.
252 *
253 * @param[out] electric_dofs Local indices of the shape functions that belong
254 * to the electric field (E_real or E_imag).
255 *
256 * @param[out] magnetic_dofs Local indices of the shape functions that belong
257 * to the magnetic field (H_real or H_imag).
258 */
259 void
260 classify_shape_functions(const FiniteElement<dim> &fe,
261 std::vector<unsigned char> &shape_function_type,
262 std::vector<unsigned int> &electric_dofs,
263 std::vector<unsigned int> &magnetic_dofs) const;

301 /**
302 * @brief Reinitialize the content of the scratch for a cell. It evaluates
303 * the shape functions of the test and interior trial spaces at the
304 * quadrature points. The temperature values are set to zero and are only
305 * overwritten by reinit_temperature when the material of the cell depends on
306 * the temperature. Likewise, the void fraction values are set to one (pure
307 * fluid) and are only overwritten by reinit_void_fraction when the material
308 * of the cell depends on the void fraction.
309 *
310 * @param[in] cell The cell of the interior trial space DoFHandler over which
311 * the assembly is carried out.
312 *
313 * @param[in] cell_test The same cell, but for the test space DoFHandler.
314 */
315 void
316 reinit(const typename DoFHandler<dim>::active_cell_iterator &cell,
317 const typename DoFHandler<dim>::active_cell_iterator &cell_test);

319 /**
320 * @brief Reinitialize the content of the scratch for a face of the current
321 * cell. It evaluates the test functions and the tangential traces of the
322 * skeleton trial space at the face quadrature points. The face temperature
323 * values are set to zero and are only overwritten by reinit_face_temperature
324 * when the material of the cell depends on the temperature. Likewise, the
325 * face void fraction values are set to one (pure fluid) and are only
326 * overwritten by reinit_face_void_fraction when the material of the cell
327 * depends on the void fraction.
328 *
329 * @param[in] cell_skeleton The current cell, for the skeleton trial space
330 * DoFHandler.
331 *
332 * @param[in] cell_test The current cell, for the test space DoFHandler.
333 *
334 * @param[in] face_no Index of the face within the cell.
335 */
336 void
337 reinit_face(

339 const typename DoFHandler<dim>::active_cell_iterator &cell_test,
340 const unsigned int face_no);
