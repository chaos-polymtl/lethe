30/**
31 * @brief Compute the effective electromagnetic properties of a material at a
32 * set of points. The effective electric permittivity is
33 * \f$\varepsilon' + i(\varepsilon'' + \sigma)\f$ and the effective magnetic
34 * permeability is \f$\mu' + i\mu''\f$, where each property is evaluated with
35 * the physical property models of the material.
36 *
37 * @note The time-harmonic Maxwell equations do not support multiple
38 * fluids so the fluid_id is not necessary to determine the material
39 * properties. The material properties only depend on the material_id, which
40 * is used to identify the different materials in the input file and assign
41 * them their corresponding properties.
42 *
43 * @param[in] physical_properties_manager The object that manages the
44 * physical properties of the problem and provides them at any given position
45 * of the domain.
46 * @param[in] field_values_vector A map containing the vector values of the
47 * fields at the current position. This is used to compute the material
48 * properties that depend on the fields.
49 * @param[in] material_id The material id of the current position, used to
50 * determine the appropriate material properties from the input parameters.
51 * @param[out] effective_electric_permittivity Effective electric
52 * permittivity at the current position. This value is updated in place by the
53 * function.
54 * @param[out] effective_magnetic_permeability Effective magnetic
55 * permeability at the current position. This value is updated in place by the
56 * function.
57 */
58void
59compute_effective_electromagnetic_properties(
60 const PhysicalPropertiesManager &physical_properties_manager,
61 const std::map<field, std::vector<double>> &field_values_vectors,
62 const unsigned int material_id,
63 std::vector<std::complex<double>> &effective_electric_permittivities,
64 std::vector<std::complex<double>> &effective_magnetic_permeabilities);

67/**
68 * @brief Class that stores the information required by the assembly of the
69 * Discontinuous Petrov-Galerkin (DPG) ultraweak formulation of the
70 * time-harmonic Maxwell equations. For each cell, it evaluates the complex
71 * shape functions of the test space (\f$\mathbf{F}\f$ for the electric test
72 * functions and \f$\mathbf{I}\f$ for the magnetic ones, with their curls and
73 * complex conjugates) and of the interior trial space (\f$\mathbf{E}\f$ and
74 * \f$\mathbf{H}\f$) and the effective material properties at the quadrature
75 * points. For each face of the cell, it evaluates the test functions and the
76 * tangential traces of the skeleton trial space (\f$\hat{\mathbf{E}}\f$ and
77 * \f$\hat{\mathbf{H}}\f$) at the face quadrature points. The complex values are
78 * stored with their complex conjugates because the conjugate of a complex
79 * tensor is not implemented in deal.II.
80 *
81 * @tparam dim An integer that denotes the number of spatial dimensions. Only
82 * dim = 3 is supported.
83 *
84 * @ingroup solvers
85 */
86template <int dim>
87class TimeHarmonicMaxwellScratchData : public PhysicsScratchDataBase
88{
89public:
90 /**
91 * @brief Bit flags identifying to which field a shape function of the
92 * [E_real, E_imag, H_real, H_imag] finite element systems belongs. The
93 * composite flags is_electric and is_magnetic allow to check whether a shape
94 * function belongs to the electric or magnetic field, regardless of whether
95 * it is the real or imaginary part.
96 */
97 enum ShapeFunctionType : unsigned char
98 {
99 electric_real = 1u << 0,
100 electric_imag = 1u << 1,
101 magnetic_real = 1u << 2,
102 magnetic_imag = 1u << 3,
103
104 is_electric = electric_real | electric_imag,
105 is_magnetic = magnetic_real | magnetic_imag
106 };
107
108 /**
109 * @brief Constructor. Creates the FEValues and FEFaceValues objects of the

113 *
114 * @param[in] properties_manager Manager of the physical property models.
115 *
116 * @param[in] fe_trial_interior Finite element of the interior trial space.
117 *
118 * @param[in] fe_trial_skeleton Finite element of the skeleton trial space.
119 *
120 * @param[in] fe_test Finite element of the test space.
121 *
122 * @param[in] cell_quadrature Quadrature used for the cell integrals.
123 *
124 * @param[in] face_quadrature Quadrature used for the face integrals.
125 *
126 * @param[in] mapping Mapping of the domain.

134 const Quadrature<dim - 1> &face_quadrature,

142 , fe_values_test(mapping,
143 fe_test,

157 /**
158 * @brief Copy constructor. Same as the main constructor. It only uses the
159 * other scratch to build the FEValues objects, it does not copy the content

184 sd.fe_face_values_test.get_quadrature(),
185 update_values)

189 allocate();

195 sd.fe_values_temperature->get_mapping());

207 /**
208 * @brief Classify the shape functions of a finite element space of the
209 * time-harmonic Maxwell equations according to the field they belong to.
210 * Each shape function of the [E_real, E_imag, H_real, H_imag] finite element
211 * system receives the ShapeFunctionType flag of its field, which is
212 * determined with FiniteElement::shape_function_belongs_to(). Since every
213 * shape function belongs to exactly one field, the local indices of the
214 * electric and magnetic shape functions are then gathered in two lists.
215 *
216 * @param[in] fe Finite element system of the [E_real, E_imag, H_real,
217 * H_imag] fields (interior trial, skeleton trial or test space).
218 *
219 * @param[out] shape_function_type ShapeFunctionType flag of each shape
220 * function of the finite element.
221 *
222 * @param[out] electric_dofs Local indices of the shape functions that belong
223 * to the electric field (E_real or E_imag).
224 *
225 * @param[out] magnetic_dofs Local indices of the shape functions that belong
226 * to the magnetic field (H_real or H_imag).
227 */
228 void
229 classify_shape_functions(const FiniteElement<dim> &fe,
230 std::vector<unsigned char> &shape_function_type,
231 std::vector<unsigned int> &electric_dofs,
232 std::vector<unsigned int> &magnetic_dofs) const;

288 /**
289 * @brief Evaluate the temperature field at the cell quadrature points.
290 *
291 * @tparam VectorType The vector type of the temperature solution.
292 *
293 * @param[in] cell_temperature The current cell, for the heat transfer
294 * DoFHandler.
295 *
296 * @param[in] temperature_solution Temperature solution of the heat transfer
297 * physics.
298 */
299 template <typename VectorType>
300 void
301 reinit_temperature(
302 const typename DoFHandler<dim>::active_cell_iterator &cell_temperature,
303 const VectorType &temperature_solution)
304 {
305 fe_values_temperature->reinit(cell_temperature);
306 fe_values_temperature->get_function_values(temperature_solution,
307 this->temperature_values);
308 }
