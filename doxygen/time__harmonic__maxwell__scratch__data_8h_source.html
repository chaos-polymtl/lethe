31/**
32 * @brief Compute the effective electromagnetic properties of a material at a
33 * set of points. The effective electric permittivity is
34 * \f$\varepsilon' + i(\varepsilon'' + \sigma)\f$ and the effective magnetic
35 * permeability is \f$\mu' + i\mu''\f$, where each property is evaluated with
36 * the physical property models of the material. If the material is an
37 * effective medium, the properties are given by the effective medium models of
38 * its fluid-dispersed phase interaction. The conductivity of both phases is
39 * then already included in the imaginary part of the effective permittivity.
40 *
41 * @note The time-harmonic Maxwell equations do not support multiple
42 * fluids so the fluid_id is not necessary to determine the material
43 * properties. The material properties only depend on the material_id, which
44 * is used to identify the different materials in the input file and assign
45 * them their corresponding properties.
46 *
47 * @param[in] physical_properties_manager The object that manages the
48 * physical properties of the problem and provides them at any given position
49 * of the domain.
50 * @param[in] field_values_vector A map containing the vector values of the
51 * fields at the current position. This is used to compute the material
52 * properties that depend on the fields.
53 * @param[in] material_id The material id of the current position, used to
54 * determine the appropriate material properties from the input parameters.
55 * @param[out] effective_electric_permittivity Effective electric
56 * permittivity at the current position. This value is updated in place by the
57 * function.
58 * @param[out] effective_magnetic_permeability Effective magnetic
59 * permeability at the current position. This value is updated in place by the
60 * function.
61 */
62void
63compute_effective_electromagnetic_properties(
64 const PhysicalPropertiesManager &physical_properties_manager,
65 const std::map<field, std::vector<double>> &field_values_vectors,
66 const unsigned int material_id,
67 std::vector<std::complex<double>> &effective_electric_permittivities,
68 std::vector<std::complex<double>> &effective_magnetic_permeabilities);

70/**
71 * @brief Check if at least one electromagnetic property of a material depends
72 * on a field. If the material is an effective medium, the dependencies are
73 * those of the effective medium models of its fluid-dispersed phase
74 * interaction. Otherwise, they are those of the electric permittivity, electric
75 * conductivity and magnetic permeability models of the material.
76 *
77 * @param[in] physical_properties_manager The object that manages the
78 * physical properties of the problem.
79 * @param[in] material_id The material id of the cell.
80 * @param[in] field_id The field for which the dependency is checked.
81 * @return true if at least one electromagnetic property of the material
82 * depends on the field, false otherwise.
83 */
84bool
85electromagnetic_properties_depend_on(
86 const PhysicalPropertiesManager &physical_properties_manager,
87 const unsigned int material_id,
88 const field field_id);

91/**
92 * @brief Class that stores the information required by the assembly of the
93 * Discontinuous Petrov-Galerkin (DPG) ultraweak formulation of the
94 * time-harmonic Maxwell equations. For each cell, it evaluates the base
95 * functions of the test space (\f$\mathbf{F}\f$ and \f$\mathbf{I}\f$) with
96 * their curls, the base functions of the interior trial space (\f$\mathbf{E}\f$
97 * and \f$\mathbf{H}\f$) and the effective material properties at the quadrature
98 * points. Additionally, for each face of the cell, it evaluates the base
99 * functions of the tangential traces of the test space and the base functions
100 * of the skeleton trial space (\f$\hat{\mathbf{E}}\f$ and
101 * \f$\hat{\mathbf{H}}\f$), with their cross products with the normal. Since the
102 * Nedelec base functions whose degrees of freedom are not located on a face
103 * have a zero tangential trace on this face, only the base functions associated
104 * with the face are evaluated.
105 *
106 * @tparam dim An integer that denotes the number of spatial dimensions. Only
107 * dim = 3 is supported.
108 *
109 * @ingroup solvers
110 */
111template <int dim>
112class TimeHarmonicMaxwellScratchData : public PhysicsScratchDataBase
113{
114public:
115 /**
116 * @brief Fields of the finite element systems, numbered in the order in which
117 * they appear in the systems following the convention: [E_real, E_imag,
118 * H_real, H_imag].
119 */
120 enum ElectromagneticFieldComponents : unsigned int
121 {
122 E_real = 0,
123 E_imag = 1,
124 H_real = 2,
125 H_imag = 3
126 };
127
128 // Number of fields of the finite element systems
129 static constexpr unsigned int n_electromagnetic_fields = 4;

136 *
137 * @param[in] properties_manager Manager of the physical property models.
138 *
139 * @param[in] omega Angular frequency of the electromagnetic fields.
140 *
141 * @param[in] fe_trial_interior Finite element of the interior trial space.
142 *
143 * @param[in] fe_trial_skeleton Finite element of the skeleton trial space.
144 *
145 * @param[in] fe_test Finite element of the test space.
146 *
147 * @param[in] cell_quadrature Quadrature used for the cell integrals.
148 *
149 * @param[in] face_quadrature Quadrature used for the face integrals.
150 *
151 * @param[in] mapping Mapping of the domain.
152 */

154 const PhysicalPropertiesManager &properties_manager,
155 const double omega,
156 const FiniteElement<dim> &fe_trial_interior,
157 const FiniteElement<dim> &fe_trial_skeleton,
158 const FiniteElement<dim> &fe_test,
159 const Quadrature<dim> &cell_quadrature,
160 const Quadrature<dim - 1> &face_quadrature,
161 const Mapping<dim> &mapping)
162 : properties_manager(properties_manager)
163 , omega(omega)
164 , fe_values_trial_interior(mapping,
165 fe_trial_interior,
166 cell_quadrature,
167 update_values | update_quadrature_points |
168 update_JxW_values)
169 , fe_values_test(mapping,
170 fe_test,
171 cell_quadrature,
172 update_values | update_gradients)
173 , fe_face_values_trial_skeleton(mapping,
174 fe_trial_skeleton,
175 face_quadrature,
176 update_values | update_quadrature_points |
177 update_normal_vectors | update_JxW_values)
178 , fe_face_values_test(mapping, fe_test, face_quadrature, update_values)
179 {
180 gather_temperature = false;
181 gather_void_fraction = false;
182 allocate();
183 }

185 /**
186 * @brief Copy constructor. Same as the main constructor. It only uses the
187 * other scratch to build the FEValues objects, it does not copy the content
188 * of the other scratch since, by definition of the WorkStream mechanism, the
189 * content of the scratch is reset on a cell basis.
190 *
191 * @param[in] sd The scratch data to be copied.
192 */
193 TimeHarmonicMaxwellScratchData(const TimeHarmonicMaxwellScratchData<dim> &sd)
194 : properties_manager(sd.properties_manager)
195 , omega(sd.omega)
196 , fe_values_trial_interior(sd.fe_values_trial_interior.get_mapping(),
197 sd.fe_values_trial_interior.get_fe(),
198 sd.fe_values_trial_interior.get_quadrature(),
199 update_values | update_quadrature_points |
200 update_JxW_values)
201 , fe_values_test(sd.fe_values_test.get_mapping(),
202 sd.fe_values_test.get_fe(),
203 sd.fe_values_test.get_quadrature(),
204 update_values | update_gradients)
205 , fe_face_values_trial_skeleton(
206 sd.fe_face_values_trial_skeleton.get_mapping(),
207 sd.fe_face_values_trial_skeleton.get_fe(),
208 sd.fe_face_values_trial_skeleton.get_quadrature(),
209 update_values | update_quadrature_points | update_normal_vectors |
210 update_JxW_values)
211 , fe_face_values_test(sd.fe_face_values_test.get_mapping(),
212 sd.fe_face_values_test.get_fe(),
213 sd.fe_face_values_test.get_quadrature(),
214 update_values)
215 {
216 gather_temperature = sd.gather_temperature;
217 gather_void_fraction = sd.gather_void_fraction;
218
219 allocate();
220
221 if (sd.gather_temperature)
222 enable_temperature(sd.fe_values_temperature->get_fe(),
223 sd.fe_values_temperature->get_quadrature(),
224 sd.fe_face_values_temperature->get_quadrature(),
225 sd.fe_values_temperature->get_mapping());
226
227 if (sd.gather_void_fraction)
228 enable_void_fraction(sd.fe_values_void_fraction->get_fe(),
229 sd.fe_values_void_fraction->get_quadrature(),
230 sd.fe_face_values_void_fraction->get_quadrature(),
231 sd.fe_values_void_fraction->get_mapping());
232 }
