17template <int dim>
18std::pair<Tensor<1, dim, std::complex<double>>,
19 Tensor<1, dim, std::complex<double>>>
20compute_waveguide_port_incident_fields(
21 [[maybe_unused]] const Parameters::TimeHarmonicMaxwell<dim>
22 &time_harmonic_maxwell_parameters,
23 [[maybe_unused]] const Point<dim> &p,
24 [[maybe_unused]] const Tensor<1, dim> &normal,
25 [[maybe_unused]] const std::complex<double> &effective_electric_permittivity,
26 [[maybe_unused]] const std::complex<double> &effective_magnetic_permeability,
27 [[maybe_unused]] const unsigned int waveguide_port_index)
28{
29 // The waveguide port excitation would be completely different in 2D as curls
30 // and cross products are not defined the same way as in 3D. Therefore, we do
31 // not implement it for now and throw an error if someone tries to use the 2D
32 // version.
33 if constexpr (dim != 3)
34 {
35 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
36 return std::pair<Tensor<1, dim, std::complex<double>>,
37 Tensor<1, dim, std::complex<double>>>();
38 }
39 // Define some constexpr values for the computation
40 static constexpr std::complex<double> imag{0., 1.};
41 static constexpr double PI = numbers::PI;
42
43 // Gather the relevant waveguide parameters
44 const double omega =
45 2.0 * PI * time_harmonic_maxwell_parameters.electromagnetic_frequency;
46 const auto &waveguide_corners =
47 time_harmonic_maxwell_parameters.waveguide_corners[waveguide_port_index];
48 const Parameters::WaveguideMode mode =
49 time_harmonic_maxwell_parameters.waveguide_mode[waveguide_port_index];
50 unsigned int m =
51 time_harmonic_maxwell_parameters.mode_order_m[waveguide_port_index];
52 unsigned int n =
53 time_harmonic_maxwell_parameters.mode_order_n[waveguide_port_index];
54
55 // We first define the transverse face vectors of the waveguide in the global
56 // system
57 Tensor<1, dim> transverse_vector_1 =
58 waveguide_corners[1] - waveguide_corners[0];
59 Tensor<1, dim> transverse_vector_2 =
60 waveguide_corners[2] - waveguide_corners[0];
61 double length_t1 = transverse_vector_1.norm();
62 double length_t2 = transverse_vector_2.norm();
63 Tensor<1, dim> e_t1 = transverse_vector_1 / length_t1;
64 Tensor<1, dim> e_t2 = transverse_vector_2 / length_t2;
65
66 // Check if the transverse vectors form a perfect rectangle (orthogonal and
67 // aligned with the axes)
68 AssertThrow(
69 std::abs(e_t1 * e_t2) < 1e-12,
70 ExcMessage(
71 "The transverse plane defined by the waveguide corners for the waveguide port at boundary ID " +
72 std::to_string(time_harmonic_maxwell_parameters
73 .waveguide_boundary_ids[waveguide_port_index]) +
74 " is not a perfect rectangle (i.e., the vector created by the waveguide corners are not orthogonal). Please check the waveguide corners definition in the input prm file."));
75
76
77 // Also verify that those transverse vectors are perpendicular to the normal
78 // of the face and form an orthogonal basis.
79 if ((std::abs(normal * e_t1) > 1e-12) || (std::abs(normal * e_t2) > 1e-12))
80 AssertThrow(
81 false,
82 ExcMessage(
83 "The transverse plane defined by the waveguide corners for the waveguide port at boundary ID " +
84 std::to_string(time_harmonic_maxwell_parameters
85 .waveguide_boundary_ids[waveguide_port_index]) +
86 " is not orthogonal to the boundary face normal. Please check the waveguide corners definition in the input prm file."));
87
88 // Create a third vector to complete the right-handed coordinate system
89 Tensor<1, dim> e_t3 = cross_product_3d(e_t1, e_t2);
90
91 // Determine if the system needs to be flipped so the e_t3 vector points in
92 // the direction opposite to the outward normal of the face boundary. For an
93 // incident wave at the inlet, the propagation direction should point into the
94 // domain (opposite to the outward normal). We use the sign of (normal · t3)
95 // to determine this:
96 // - If normal · t3 > 0: t3 points outward, need to flip entire system
97 // - If normal · t3 < 0: t3 points inward (correct for incident wave)
98 // The flip swaps e_t1 and e_t2 (and the associated lengths and mode orders,
99 // so the mode keeps the same transverse profile) and recomputes e_t3 as their
100 // cross product so the resulting system is right-handed.
101 if ((normal * e_t3) > 0)
102 {
103 std::swap(e_t1, e_t2);
104 std::swap(length_t1, length_t2);
105 std::swap(m, n);
106 e_t3 =
107 cross_product_3d(e_t1, e_t2); // Recompute t3 after swapping t1 and t2
108 // to be sure the system is right-handed
109 }
110
111 // Compute the various wavenumbers k in using this global coordinate system
112 double k_t1 = m * PI / length_t1; // Transverse wavenumber 1
113 double k_t2 = n * PI / length_t2; // Transverse wavenumber 2
114 double k_c = std::sqrt(k_t1 * k_t1 + k_t2 * k_t2); // Cutoff wavenumber
115 std::complex<double> k =
116 omega *
117 std::sqrt(effective_electric_permittivity *
118 effective_magnetic_permeability); // Wavenumber in the medium
119 std::complex<double> k_l = std::sqrt(
120 k * k - std::complex<double>(k_c * k_c, 0)); // Longitudinal wavenumber
121
122 // Verify that the mode is not evanescent, i.e. k_l is not purely imaginary
123 // (k_c^2 < k^2). std::norm computes the squared magnitude of a complex
124 // number.
125 AssertThrow(
126 std::norm(k) > (k_c * k_c),
127 ExcMessage("The chosen mode for the waveguide port at boundary ID " +
128 std::to_string(time_harmonic_maxwell_parameters
129 .waveguide_boundary_ids[waveguide_port_index]) +
130 " is evanescent at the given frequency. Please "
131 "choose another mode or increase the frequency."));
132
133 // Now we want to compute the electromagnetic field and surface admittance at
134 // point p as if the waveguide center was at the origin. So we will perform a
135 // change of basis to a local coordinate system where the waveguide center is
136 // at the origin.
137 Tensor<1, dim> origin_local =
138 0.25 * (waveguide_corners[0] + waveguide_corners[1] + waveguide_corners[2] +
139 waveguide_corners[3]);
140
141 Tensor<1, dim> p_local =
142 p - origin_local; // Coordinates of point p in this local system.
143
144 const double x_local =
145 p_local * e_t1; // Coordinate along e_t1 in the local system
146 const double y_local =
147 p_local * e_t2; // Coordinate along e_t2 in the local system
148 // We assume that the z_local coordinate is 0 since we are on the face.
149
150 // Compute the E and H field components for the TE mode in the local
151 // coordinate system {t1, t2, t3} = {x', y', z'}. We assume z' = 0 at the
152 // boundary.
153 Tensor<1, dim, std::complex<double>> E_inc_local;
154 Tensor<1, dim, std::complex<double>> H_inc_local;
155
156 // sin and cos functions that are repeatedly used in the computation of the E
157 // and H fields.
158 const double sin_k_t1_x = std::sin(k_t1 * (x_local + 0.5 * length_t1));
159 const double cos_k_t1_x = std::cos(k_t1 * (x_local + 0.5 * length_t1));
160 const double sin_k_t2_y = std::sin(k_t2 * (y_local + 0.5 * length_t2));
161 const double cos_k_t2_y = std::cos(k_t2 * (y_local + 0.5 * length_t2));
162
163 if (mode == Parameters::WaveguideMode::TE)
164 {
165 std::complex<double> factor =
166 imag * omega * effective_magnetic_permeability / (k_c * k_c);
167
168 E_inc_local[0] = -factor * k_t2 * cos_k_t1_x * sin_k_t2_y;
169 E_inc_local[1] = factor * k_t1 * sin_k_t1_x * cos_k_t2_y;
170 E_inc_local[2] = 0.0;
171
172 H_inc_local[0] =
173 -imag * k_l * k_t1 / (k_c * k_c) * sin_k_t1_x * cos_k_t2_y;
174 H_inc_local[1] =
175 -imag * k_l * k_t2 / (k_c * k_c) * cos_k_t1_x * sin_k_t2_y;
176 H_inc_local[2] = cos_k_t1_x * cos_k_t2_y;
177 }
178 else if (mode == Parameters::WaveguideMode::TM)
179 {
180 std::complex<double> factor =
181 imag * omega * effective_electric_permittivity / (k_c * k_c);
182
183 H_inc_local[0] = -factor * k_t2 * sin_k_t1_x * cos_k_t2_y;
184 H_inc_local[1] = factor * k_t1 * cos_k_t1_x * sin_k_t2_y;
185 H_inc_local[2] = 0.0;
186
187 E_inc_local[0] =
188 imag * k_l * k_t1 / (k_c * k_c) * cos_k_t1_x * sin_k_t2_y;
189 E_inc_local[1] =
190 imag * k_l * k_t2 / (k_c * k_c) * sin_k_t1_x * cos_k_t2_y;
191 E_inc_local[2] = sin_k_t1_x * sin_k_t2_y;
192 }
193 else
194 {
195 AssertThrow(false, ExcMessage("Unknown waveguide mode type."));
196 }
197
198 // Convert the E and H field components from the local coordinate system back
199 // to the global coordinate system using the basis vectors e_t1, e_t2, e_t3
200 const Tensor<1, dim, std::complex<double>> E_inc =
201 E_inc_local[0] * e_t1 + E_inc_local[1] * e_t2 + E_inc_local[2] * e_t3;
202 const Tensor<1, dim, std::complex<double>> H_inc =
203 H_inc_local[0] * e_t1 + H_inc_local[1] * e_t2 + H_inc_local[2] * e_t3;
204
205 return std::make_pair(E_inc, H_inc);
206}

208template <int dim>
209unsigned int
210get_waveguide_port_index(
211 const Parameters::TimeHarmonicMaxwell<dim> &time_harmonic_maxwell_parameters,
212 const types::boundary_id boundary_id)
213{
214 // The waveguide port index is the position of the boundary id in the list of
215 // the port boundary ids of the waveguide inlets
216 const std::vector<int> &waveguide_boundary_ids =
217 time_harmonic_maxwell_parameters.waveguide_boundary_ids;
218 const unsigned int waveguide_port_index = static_cast<unsigned int>(
219 std::distance(waveguide_boundary_ids.begin(),
220 std::ranges::find(waveguide_boundary_ids,
221 static_cast<int>(boundary_id))));
222
223 return waveguide_port_index;
224}

299template <int dim>
300std::pair<Tensor<1, dim, std::complex<double>>, std::complex<double>>
301compute_waveguide_port_excitation(
302 [[maybe_unused]] const Parameters::TimeHarmonicMaxwell<dim>
303 &time_harmonic_maxwell_parameters,
304 [[maybe_unused]] const Point<dim> &p,
305 [[maybe_unused]] const Tensor<1, dim> &normal,
306 [[maybe_unused]] const std::complex<double> &effective_electric_permittivity,
307 [[maybe_unused]] const std::complex<double> &effective_magnetic_permeability,
308 [[maybe_unused]] const unsigned int waveguide_port_index,
309 [[maybe_unused]] const double relative_amplitude)
310{
311 // The waveguide port excitation would be completely different in 2D as curls
312 // and cross products are not defined the same way as in 3D. Therefore, we do
313 // not implement it for now and throw an error if someone tries to use the 2D
314 // version.
315 if constexpr (dim != 3)
316 {
317 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
318 return std::pair<Tensor<1, dim, std::complex<double>>,
319 std::complex<double>>();
320 }
321
322 static constexpr double PI = numbers::PI;
323 const auto incident_fields =
324 compute_waveguide_port_incident_fields(time_harmonic_maxwell_parameters,
325 p,
326 normal,
327 effective_electric_permittivity,
328 effective_magnetic_permeability,
329 waveguide_port_index);
330
331 // Gather the relevant waveguide parameters
332 const double omega =
333 2.0 * PI * time_harmonic_maxwell_parameters.electromagnetic_frequency;
334 const auto &waveguide_corners =
335 time_harmonic_maxwell_parameters.waveguide_corners[waveguide_port_index];
336 const Parameters::WaveguideMode mode =
337 time_harmonic_maxwell_parameters.waveguide_mode[waveguide_port_index];
338 unsigned int m =
339 time_harmonic_maxwell_parameters.mode_order_m[waveguide_port_index];
340 unsigned int n =
341 time_harmonic_maxwell_parameters.mode_order_n[waveguide_port_index];
342
343 const double k_t1 = m * PI /
344 (waveguide_corners[1] - waveguide_corners[0])
345 .norm(); // Transverse wavenumber 1
346 const double k_t2 = n * PI /
347 (waveguide_corners[2] - waveguide_corners[0])
348 .norm(); // Transverse wavenumber 2
349 const std::complex<double> k_l =
350 std::sqrt(omega * omega * effective_electric_permittivity *
351 effective_magnetic_permeability -
352 (k_t1 * k_t1 + k_t2 * k_t2)); // Longitudinal wavenumber
353
354 std::complex<double> surface_admittance =
355 (mode == Parameters::WaveguideMode::TE) ?
356 k_l / (omega * effective_magnetic_permeability) :
357 omega * effective_electric_permittivity / k_l;
358
359 const Tensor<1, dim, std::complex<double>> &E_inc = incident_fields.first;
360 const Tensor<1, dim, std::complex<double>> &H_inc = incident_fields.second;
361
362 const Tensor<1, dim, std::complex<double>> excitation =
363 relative_amplitude *
364 (cross_product_3d(normal, H_inc) +
365 tangential_component(surface_admittance * E_inc, normal));
366
367 return std::make_pair(excitation, surface_admittance);
368}

370template <int dim>
371std::pair<Tensor<1, dim, std::complex<double>>, std::complex<double>>
372compute_robin_boundary_data(
373 const Parameters::TimeHarmonicMaxwell<dim> &time_harmonic_maxwell_parameters,
374 const BoundaryConditions::TimeHarmonicMaxwellBoundaryConditions<dim>
375 &boundary_conditions,
376 const std::vector<double> &waveguide_ports_relative_amplitudes,
377 const types::boundary_id boundary_id,
378 const Point<dim> &position,
379 const Tensor<1, dim> &normal,
380 const std::complex<double> &effective_electric_permittivity,
381 const std::complex<double> &effective_magnetic_permeability)
382{
383 static constexpr std::complex<double> imag{0., 1.};
384
385 Tensor<1, dim, std::complex<double>> g_inc;
386 std::complex<double> boundary_surface_admittance;
387
388 const BoundaryConditions::TimeHarmonicMaxwellBoundaryType bc_type =
389 boundary_conditions.type.at(boundary_id);
390
391 if (bc_type ==
392 BoundaryConditions::TimeHarmonicMaxwellBoundaryType::silver_muller)
393 {
394 boundary_surface_admittance = std::sqrt(effective_electric_permittivity /
395 effective_magnetic_permeability);
396 g_inc = 0.;
397 }
398 if (bc_type ==
399 BoundaryConditions::TimeHarmonicMaxwellBoundaryType::impedance_boundary)
400 {
401 boundary_surface_admittance =
402 boundary_conditions.surface_admittance_real.at(boundary_id)
403 ->value(position) +
404 imag * boundary_conditions.surface_admittance_imag.at(boundary_id)
405 ->value(position);
406
407 // Get the incident electromagnetic field at this face. The excitation
408 // is only defined for 3D problems.
409 if constexpr (dim == 3)
410 {
411 g_inc[0] =
412 boundary_conditions.excitation_x_real.at(boundary_id)
413 ->value(position) +
414 imag * boundary_conditions.excitation_x_imag.at(boundary_id)
415 ->value(position);
416
417 g_inc[1] =
418 boundary_conditions.excitation_y_real.at(boundary_id)
419 ->value(position) +
420 imag * boundary_conditions.excitation_y_imag.at(boundary_id)
421 ->value(position);
422
423 g_inc[2] =
424 boundary_conditions.excitation_z_real.at(boundary_id)
425 ->value(position) +
426 imag * boundary_conditions.excitation_z_imag.at(boundary_id)
427 ->value(position);
428 }
429 else
430 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
431 }
432 if (bc_type ==
433 BoundaryConditions::TimeHarmonicMaxwellBoundaryType::waveguide_port)
434 {
435 const unsigned int waveguide_port_index =
436 get_waveguide_port_index(time_harmonic_maxwell_parameters, boundary_id);
437
438 std::tie(g_inc, boundary_surface_admittance) =
439 compute_waveguide_port_excitation(
440 time_harmonic_maxwell_parameters,
441 position,
442 normal,
443 effective_electric_permittivity,
444 effective_magnetic_permeability,
445 waveguide_port_index,
446 waveguide_ports_relative_amplitudes[waveguide_port_index]);
447 }
448
449 return std::make_pair(g_inc, boundary_surface_admittance);
450}

452template <int dim>
453BoundaryPowerType
454compute_robin_power_densities(const Tensor<1, dim, std::complex<double>> &E_T,
455 const Tensor<1, dim, std::complex<double>> &g_T,
456 const std::complex<double> &Y)
457{
458 BoundaryPowerType densities;
459
460 // The power density entering the domain is the negative of the normal
461 // component of the Poynting vector 0.5 * Re(E x H*), since the normal points
462 // outward from the domain. Using (E x H*) . n = -E . (n x H)*, it is
463 // 0.5 * Re(E . (n x H)*), with n x H = g - Y E_T given by the Robin boundary
464 // condition.
465 densities.net = 0.5 * std::real(hermitian_product(E_T, g_T - Y * E_T));
466
467 // The tangential field is split into an incident wave, entering the domain,
468 // and an outgoing wave, which satisfy n x H_inc = Y E_inc and n x H_out =
469 // -Y E_out. The Robin boundary condition then gives g = 2 Y E_inc, so E_inc =
470 // g / (2Y) and E_out = E_T - E_inc. On a waveguide port, E_inc is the field
471 // of the incident mode. The normal Poynting vector of each wave gives the
472 // incident and outgoing power densities p_inc = 0.5 Re(Y) |E_inc|^2 and
473 // p_out = 0.5 Re(Y) |E_out|^2, and the net power density is
474 // p_net = p_inc - p_out + Im(Y) Im(E_out . E_inc*). The waves cannot be
475 // separated if Y = 0, in which case the whole net power density is counted
476 // as cross power density.
477 if (Y == 0.)
478 {
479 densities.cross = densities.net;
480 return densities;
481 }
482
483 const Tensor<1, dim, std::complex<double>> E_incident = g_T / (2. * Y);
484 const Tensor<1, dim, std::complex<double>> E_outgoing = E_T - E_incident;
485
486 densities.incident = 0.5 * Y.real() * E_incident.norm_square();
487 densities.outgoing = 0.5 * Y.real() * E_outgoing.norm_square();
488 densities.cross =
489 Y.imag() * std::imag(hermitian_product(E_outgoing, E_incident));
490
491 return densities;
492}

500template <int dim>
501void
502TimeHarmonicMaxwellAssemblerCore<dim>::assemble_matrix(
503 const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
504 DPGCopyData &copy_data)
505{
506 using ScratchData = TimeHarmonicMaxwellScratchData<dim>;
507
508 const unsigned int n_q_points = scratch_data.n_q_points;
509 const unsigned int n_base_functions_test = scratch_data.n_base_functions_test;
510 const unsigned int n_base_functions_trial_interior =
511 scratch_data.n_base_functions_trial_interior;
512
513 // Local dof indices of the real and imaginary parts of each base function:
514 // electric (F) and magnetic (I) test functions, and electric (E) and magnetic
515 // (H) interior trial functions
516 const std::vector<unsigned int> &test_dofs_F_real =
517 scratch_data.test_local_dof_index_map[ScratchData::E_real];
518 const std::vector<unsigned int> &test_dofs_F_imag =
519 scratch_data.test_local_dof_index_map[ScratchData::E_imag];
520 const std::vector<unsigned int> &test_dofs_I_real =
521 scratch_data.test_local_dof_index_map[ScratchData::H_real];
522 const std::vector<unsigned int> &test_dofs_I_imag =
523 scratch_data.test_local_dof_index_map[ScratchData::H_imag];
524 const std::vector<unsigned int> &trial_dofs_E_real =
525 scratch_data.trial_interior_local_dof_index_map[ScratchData::E_real];
526 const std::vector<unsigned int> &trial_dofs_E_imag =
527 scratch_data.trial_interior_local_dof_index_map[ScratchData::E_imag];
528 const std::vector<unsigned int> &trial_dofs_H_real =
529 scratch_data.trial_interior_local_dof_index_map[ScratchData::H_real];
530 const std::vector<unsigned int> &trial_dofs_H_imag =
531 scratch_data.trial_interior_local_dof_index_map[ScratchData::H_imag];
532
533 // Coefficients of the terms at the quadrature points, multiplied by JxW and
534 // computed by the scratch data, with iweps = i omega eps and
535 // iwmu = i omega mu. Since JxW is real, conj(iwmu) JxW is the complex
536 // conjugate of iwmu JxW.
537 const std::vector<double> &JxW = scratch_data.JxW;
538 const std::vector<double> &JxW_FF_container = scratch_data.JxW_FF_container;
539 const std::vector<double> &JxW_II_container = scratch_data.JxW_II_container;
540 const std::vector<std::complex<double>> &iweps_JxW_container =
541 scratch_data.iweps_JxW_container;
542 const std::vector<std::complex<double>> &iwmu_JxW_container =
543 scratch_data.iwmu_JxW_container;
544
545 LAPACKFullMatrix<double> &G_matrix = copy_data.G_matrix;
546 LAPACKFullMatrix<double> &B_matrix = copy_data.B_matrix;
547
548 // Gram matrix G (Riesz map of the test space norm). For the test base
549 // functions i (rows) and j (columns), the blocks are:
550 // - FF: (1 + |iweps|^2) (phi_i, phi_j) + (curl phi_i, curl phi_j)
551 // - II: (1 + |iwmu|^2) (phi_i, phi_j) + (curl phi_i, curl phi_j)
552 // - FI: iweps (phi_i, curl phi_j) - conj(iwmu) (curl phi_i, phi_j)
553 // - IF: since G is Hermitian, the block IF(j, i) is conj(FI(i, j)).
554 // FF and II are real and symmetric in i and j, so we only loop over the pairs
555 // i <= j and add the blocks of the pair (j, i) at the same time.
556 for (unsigned int i = 0; i < n_base_functions_test; ++i)
557 {
558 const auto phi_i = scratch_data.phi_test[i];
559 const auto curl_phi_i = scratch_data.curl_phi_test[i];
560
561 for (unsigned int j = i; j < n_base_functions_test; ++j)
562 {
563 const auto phi_j = scratch_data.phi_test[j];
564 const auto curl_phi_j = scratch_data.curl_phi_test[j];
565
566 double z_FF = 0.;
567 double z_II = 0.;
568 std::complex<double> z_FI_ij(0.);
569 std::complex<double> z_FI_ji(0.);
570 for (unsigned int q = 0; q < n_q_points; ++q)
571 {
572 // The four dot products are shared by the four blocks
573 const double phi_i_phi_j = phi_i[q] * phi_j[q];
574 const double curl_i_curl_j = curl_phi_i[q] * curl_phi_j[q];
575 const double phi_i_curl_j = phi_i[q] * curl_phi_j[q];
576 const double curl_i_phi_j = curl_phi_i[q] * phi_j[q];
577
578 z_FF +=
579 JxW_FF_container[q] * phi_i_phi_j + JxW[q] * curl_i_curl_j;
580 z_II +=
581 JxW_II_container[q] * phi_i_phi_j + JxW[q] * curl_i_curl_j;
582 z_FI_ij += iweps_JxW_container[q] * phi_i_curl_j -
583 std::conj(iwmu_JxW_container[q]) * curl_i_phi_j;
584 z_FI_ji += iweps_JxW_container[q] * curl_i_phi_j -
585 std::conj(iwmu_JxW_container[q]) * phi_i_curl_j;
586 }
587
588 // Blocks of the pair (i, j)
589 add_real_block(G_matrix,
590 test_dofs_F_real[i],
591 test_dofs_F_imag[i],
592 test_dofs_F_real[j],
593 test_dofs_F_imag[j],
594 z_FF);
595 add_real_block(G_matrix,
596 test_dofs_I_real[i],
597 test_dofs_I_imag[i],
598 test_dofs_I_real[j],
599 test_dofs_I_imag[j],
600 z_II);
601 add_complex_block(G_matrix,
602 test_dofs_F_real[i],
603 test_dofs_F_imag[i],
604 test_dofs_I_real[j],
605 test_dofs_I_imag[j],
606 z_FI_ij);
607 add_complex_block(G_matrix,
608 test_dofs_I_real[j],
609 test_dofs_I_imag[j],
610 test_dofs_F_real[i],
611 test_dofs_F_imag[i],
612 std::conj(z_FI_ij));
613
614 // Blocks of the pair (j, i), we skip the case i == j because the
615 // diagonal blocks are already filled by the pair (i, j)
616 if (j != i)
617 {
618 add_real_block(G_matrix,
619 test_dofs_F_real[j],
620 test_dofs_F_imag[j],
621 test_dofs_F_real[i],
622 test_dofs_F_imag[i],
623 z_FF);
624 add_real_block(G_matrix,
625 test_dofs_I_real[j],
626 test_dofs_I_imag[j],
627 test_dofs_I_real[i],
628 test_dofs_I_imag[i],
629 z_II);
630 add_complex_block(G_matrix,
631 test_dofs_F_real[j],
632 test_dofs_F_imag[j],
633 test_dofs_I_real[i],
634 test_dofs_I_imag[i],
635 z_FI_ji);
636 add_complex_block(G_matrix,
637 test_dofs_I_real[i],
638 test_dofs_I_imag[i],
639 test_dofs_F_real[j],
640 test_dofs_F_imag[j],
641 std::conj(z_FI_ji));
642 }
643 }
644 }
645
646 // Interior bilinear form B. For the test base function i (rows) and the
647 // interior trial base function j (columns), the blocks are:
648 // - FE: iweps (phi_i, phi_j)
649 // - FH and IE: (curl phi_i, phi_j)
650 // - IH: -iwmu (phi_i, phi_j)
651 for (unsigned int i = 0; i < n_base_functions_test; ++i)
652 {
653 const auto phi_i = scratch_data.phi_test[i];
654 const auto curl_phi_i = scratch_data.curl_phi_test[i];
655
656 for (unsigned int j = 0; j < n_base_functions_trial_interior; ++j)
657 {
658 const auto phi_j = scratch_data.phi_trial_interior[j];
659
660 std::complex<double> z_FE(0.);
661 double z_FH = 0.;
662 std::complex<double> z_IH(0.);
663 for (unsigned int q = 0; q < n_q_points; ++q)
664 {
665 const double phi_i_phi_j = phi_i[q] * phi_j[q];
666 const double curl_i_phi_j = curl_phi_i[q] * phi_j[q];
667
668 z_FE += iweps_JxW_container[q] * phi_i_phi_j;
669 z_FH += JxW[q] * curl_i_phi_j;
670 z_IH -= iwmu_JxW_container[q] * phi_i_phi_j;
671 }
672
673 add_complex_block(B_matrix,
674 test_dofs_F_real[i],
675 test_dofs_F_imag[i],
676 trial_dofs_E_real[j],
677 trial_dofs_E_imag[j],
678 z_FE);
679 add_real_block(B_matrix,
680 test_dofs_F_real[i],
681 test_dofs_F_imag[i],
682 trial_dofs_H_real[j],
683 trial_dofs_H_imag[j],
684 z_FH);
685 // This block is the addition for the result of z_IE = (curl phi_i,
686 // phi_j) which is the same as z_FH, so we can reuse it
687 add_real_block(B_matrix,
688 test_dofs_I_real[i],
689 test_dofs_I_imag[i],
690 trial_dofs_E_real[j],
691 trial_dofs_E_imag[j],
692 z_FH);
693 add_complex_block(B_matrix,
694 test_dofs_I_real[i],
695 test_dofs_I_imag[i],
696 trial_dofs_H_real[j],
697 trial_dofs_H_imag[j],
698 z_IH);
699 }
700 }

732template <int dim>
733void
734TimeHarmonicMaxwellAssemblerSkeleton<dim>::assemble_matrix(
735 const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
736 DPGCopyData &copy_data)
737{
738 using ScratchData = TimeHarmonicMaxwellScratchData<dim>;
739
740 // On the faces where a Robin boundary condition is applied, the magnetic
741 // trace term is replaced by the Robin boundary condition.
742 const bool robin_face = scratch_data.face_at_boundary &&
743 is_robin_boundary_type(boundary_conditions.type.at(
744 scratch_data.face_boundary_id));
745
746 // Local dof indices of the real and imaginary parts of each base function:
747 // electric (F) and magnetic (I) test functions, and electric (E_hat) and
748 // magnetic (H_hat) skeleton trial functions
749 const std::vector<unsigned int> &test_dofs_F_real =
750 scratch_data.test_local_dof_index_map[ScratchData::E_real];
751 const std::vector<unsigned int> &test_dofs_F_imag =
752 scratch_data.test_local_dof_index_map[ScratchData::E_imag];
753 const std::vector<unsigned int> &test_dofs_I_real =
754 scratch_data.test_local_dof_index_map[ScratchData::H_real];
755 const std::vector<unsigned int> &test_dofs_I_imag =
756 scratch_data.test_local_dof_index_map[ScratchData::H_imag];
757 const std::vector<unsigned int> &trial_dofs_E_hat_real =
758 scratch_data.trial_skeleton_local_dof_index_map[ScratchData::E_real];
759 const std::vector<unsigned int> &trial_dofs_E_hat_imag =
760 scratch_data.trial_skeleton_local_dof_index_map[ScratchData::E_imag];
761 const std::vector<unsigned int> &trial_dofs_H_hat_real =
762 scratch_data.trial_skeleton_local_dof_index_map[ScratchData::H_real];
763 const std::vector<unsigned int> &trial_dofs_H_hat_imag =
764 scratch_data.trial_skeleton_local_dof_index_map[ScratchData::H_imag];
765
766 const std::vector<double> &face_JxW = scratch_data.face_JxW;
767
768 LAPACKFullMatrix<double> &B_hat_matrix = copy_data.B_hat_matrix;
769
770 // For the test base function i (rows) and the skeleton trial base function j
771 // (columns), the blocks are:
772 // - IE_hat: <phi_i, n x e_hat_j>
773 // - FH_hat: <phi_i, n x h_hat_j>, the same value since the electric and
774 // magnetic skeleton fields share the same base functions
775 // Here we use the precomputed tangential base functions (a subset of base
776 // functions) of the test and trial spaces for the loop to avoid extra
777 // computations of non-contributing base functions.
778 for (const unsigned int i :
779 scratch_data.test_tangential_base_functions[scratch_data.face_no])
780 {
781 const auto phi_i = scratch_data.phi_test_face[i];
782
783 for (const unsigned int j :
784 scratch_data
785 .trial_skeleton_tangential_base_functions[scratch_data.face_no])
786 {
787 const auto n_cross_phi_j =
788 scratch_data.n_cross_tangential_phi_trial_skeleton[j];
789
790 double z_IE = 0.;
791 for (unsigned int q = 0; q < scratch_data.n_face_q_points; ++q)
792 z_IE += face_JxW[q] * (phi_i[q] * n_cross_phi_j[q]);
793
794 add_real_block(B_hat_matrix,
795 test_dofs_I_real[i],
796 test_dofs_I_imag[i],
797 trial_dofs_E_hat_real[j],
798 trial_dofs_E_hat_imag[j],
799 z_IE);
800
801 // Here we also add z_FH, which is the same as z_IE. However, we only
802 // do this if the Robin boundary condition is not applied, because in
803 // that case, the magnetic trace term is replaced by the Robin
804 // boundary condition on the faces where it is applied.
805 if (!robin_face)
806 add_real_block(B_hat_matrix,
807 test_dofs_F_real[i],
808 test_dofs_F_imag[i],
809 trial_dofs_H_hat_real[j],
810 trial_dofs_H_hat_imag[j],
811 z_IE);
812 }
813 }

834template <int dim>
835void
836TimeHarmonicMaxwellAssemblerRobinBC<dim>::assemble_matrix(
837 const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
838 DPGCopyData &copy_data)
839{
840 if (!scratch_data.face_at_boundary)
841 return;
842
843 const BoundaryConditions::TimeHarmonicMaxwellBoundaryType bc_type =
844 boundary_conditions.type.at(scratch_data.face_boundary_id);
845 if (!is_robin_boundary_type(bc_type))
846 return;
847
848 using ScratchData = TimeHarmonicMaxwellScratchData<dim>;
849
850 const unsigned int n_face_q_points = scratch_data.n_face_q_points;
851
852 // Base functions of the test space associated with the face
853 const std::vector<unsigned int> &test_tangential_base_functions =
854 scratch_data.test_tangential_base_functions[scratch_data.face_no];
855 const unsigned int n_test_tangential_base_functions =
856 test_tangential_base_functions.size();
857
858 // Local dof indices of the real and imaginary parts of each base function:
859 // electric (F) and magnetic (I) test functions, and electric (E_hat)
860 // skeleton trial functions
861 const std::vector<unsigned int> &test_dofs_F_real =
862 scratch_data.test_local_dof_index_map[ScratchData::E_real];
863 const std::vector<unsigned int> &test_dofs_F_imag =
864 scratch_data.test_local_dof_index_map[ScratchData::E_imag];
865 const std::vector<unsigned int> &test_dofs_I_real =
866 scratch_data.test_local_dof_index_map[ScratchData::H_real];
867 const std::vector<unsigned int> &test_dofs_I_imag =
868 scratch_data.test_local_dof_index_map[ScratchData::H_imag];
869 const std::vector<unsigned int> &trial_dofs_E_hat_real =
870 scratch_data.trial_skeleton_local_dof_index_map[ScratchData::E_real];
871 const std::vector<unsigned int> &trial_dofs_E_hat_imag =
872 scratch_data.trial_skeleton_local_dof_index_map[ScratchData::E_imag];
873
874 const std::vector<double> &face_JxW = scratch_data.face_JxW;
875
876 LAPACKFullMatrix<double> &G_matrix = copy_data.G_matrix;
877 LAPACKFullMatrix<double> &B_hat_matrix = copy_data.B_hat_matrix;
878
879 // Surface admittance Y at the face quadrature points, multiplied by JxW:
880 // |Y|^2 JxW and Y JxW. It depends on the boundary condition, so it is
881 // computed here and not in the scratch data.
882 std::vector<double> YY_JxW_container(n_face_q_points);
883 std::vector<std::complex<double>> Y_JxW_container(n_face_q_points);
884 for (unsigned int q = 0; q < n_face_q_points; ++q)
885 {
886 const std::complex<double> boundary_surface_admittance =
887 compute_robin_boundary_data(
888 time_harmonic_maxwell_parameters,
889 boundary_conditions,
890 waveguide_ports_relative_amplitudes,
891 scratch_data.face_boundary_id,
892 scratch_data.face_quadrature_points[q],
893 scratch_data.face_normals[q],
894 scratch_data.face_effective_electric_permittivities[q],
895 scratch_data.face_effective_magnetic_permeabilities[q])
896 .second;
897 YY_JxW_container[q] =
898 std::norm(boundary_surface_admittance) * face_JxW[q];
899 Y_JxW_container[q] = boundary_surface_admittance * face_JxW[q];
900 }
901
902 // Boundary terms of the Gram matrix G coming from the test norm on the Robin
903 // boundaries:
904 // (n x I + Y^* F_t, n x I + Y^* F_t)_{Gamma_R},
905 // where F_t = n x (F x n) is the tangential trace of F and the inner product
906 // is conjugated in its first argument. This term only involves the
907 // tangential traces of the test functions, which are in L^2(Gamma_R) for the
908 // test space. Since |n| = 1, F_t . F_t = (n x F) . (n x F), so for the test
909 // base functions i (rows) and j (columns), the blocks are:
910 // - FF: |Y|^2 <n x phi_i, n x phi_j>
911 // - II: <n x phi_i, n x phi_j>
912 // - FI: Y <phi_i, n x phi_j>
913 // - IF: since G is Hermitian, the block IF(j, i) is conj(FI(i, j)).
914 // As for the cell terms, we only loop over the pairs i <= j and add the
915 // blocks of the pair (j, i) at the same time.
916 for (unsigned int position_i = 0;
917 position_i < n_test_tangential_base_functions;
918 ++position_i)
919 {
920 const unsigned int i = test_tangential_base_functions[position_i];
921 const auto phi_i = scratch_data.phi_test_face[i];
922 const auto n_cross_phi_i = scratch_data.n_cross_phi_test_face[i];
923
924 for (unsigned int position_j = position_i;
925 position_j < n_test_tangential_base_functions;
926 ++position_j)
927 {
928 const unsigned int j = test_tangential_base_functions[position_j];
929 const auto phi_j = scratch_data.phi_test_face[j];
930 const auto n_cross_phi_j = scratch_data.n_cross_phi_test_face[j];
931
932 double z_FF = 0.;
933 double z_II = 0.;
934 std::complex<double> z_FI_ij(0.);
935 std::complex<double> z_FI_ji(0.);
936 for (unsigned int q = 0; q < n_face_q_points; ++q)
937 {
938 // The FF and II blocks share the dot product of the tangential
939 // traces
940 const double n_cross_i_n_cross_j =
941 n_cross_phi_i[q] * n_cross_phi_j[q];
942
943 z_FF += YY_JxW_container[q] * n_cross_i_n_cross_j;
944 z_II += face_JxW[q] * n_cross_i_n_cross_j;
945 z_FI_ij += Y_JxW_container[q] * (phi_i[q] * n_cross_phi_j[q]);
946 z_FI_ji += Y_JxW_container[q] * (phi_j[q] * n_cross_phi_i[q]);
947 }
948
949 // Blocks of the pair (i, j)
950 add_real_block(G_matrix,
951 test_dofs_F_real[i],
952 test_dofs_F_imag[i],
953 test_dofs_F_real[j],
954 test_dofs_F_imag[j],
955 z_FF);
956 add_real_block(G_matrix,
957 test_dofs_I_real[i],
958 test_dofs_I_imag[i],
959 test_dofs_I_real[j],
960 test_dofs_I_imag[j],
961 z_II);
962 add_complex_block(G_matrix,
963 test_dofs_F_real[i],
964 test_dofs_F_imag[i],
965 test_dofs_I_real[j],
966 test_dofs_I_imag[j],
967 z_FI_ij);
968 add_complex_block(G_matrix,
969 test_dofs_I_real[j],
970 test_dofs_I_imag[j],
971 test_dofs_F_real[i],
972 test_dofs_F_imag[i],
973 std::conj(z_FI_ij));
974
975 // Blocks of the pair (j, i), we skip the case i == j because the
976 // diagonal blocks are already filled by the pair (i, j)
977 if (j != i)
978 {
979 add_real_block(G_matrix,
980 test_dofs_F_real[j],
981 test_dofs_F_imag[j],
982 test_dofs_F_real[i],
983 test_dofs_F_imag[i],
984 z_FF);
985 add_real_block(G_matrix,
986 test_dofs_I_real[j],
987 test_dofs_I_imag[j],
988 test_dofs_I_real[i],
989 test_dofs_I_imag[i],
990 z_II);
991 add_complex_block(G_matrix,
992 test_dofs_F_real[j],
993 test_dofs_F_imag[j],
994 test_dofs_I_real[i],
995 test_dofs_I_imag[i],
996 z_FI_ji);
997 add_complex_block(G_matrix,
998 test_dofs_I_real[i],
999 test_dofs_I_imag[i],
1000 test_dofs_F_real[j],
1001 test_dofs_F_imag[j],
1002 std::conj(z_FI_ji));
1003 }
1004 }
1005 }
1006
1007 // Robin term of the skeleton bilinear form B_hat, which replaces the
1008 // magnetic trace term. For the test base function i (rows) and the skeleton
1009 // trial base function j (columns), the FE_hat block is
1010 // -Y < phi_i, n x (e_hat_j x n) >. It only involves the tangential trace of
1011 // the test functions, so only the base functions associated with the face
1012 // contribute.
1013 for (const unsigned int i : test_tangential_base_functions)
1014 {
1015 const auto phi_i = scratch_data.phi_test_face[i];
1016
1017 for (const unsigned int j :
1018 scratch_data
1019 .trial_skeleton_tangential_base_functions[scratch_data.face_no])
1020 {
1021 const auto tangential_phi_j =
1022 scratch_data.tangential_phi_trial_skeleton[j];
1023
1024 std::complex<double> z_FE(0.);
1025 for (unsigned int q = 0; q < n_face_q_points; ++q)
1026 z_FE -= Y_JxW_container[q] * (phi_i[q] * tangential_phi_j[q]);
1027
1028 add_complex_block(B_hat_matrix,
1029 test_dofs_F_real[i],
1030 test_dofs_F_imag[i],
1031 trial_dofs_E_hat_real[j],
1032 trial_dofs_E_hat_imag[j],
1033 z_FE);
1034 }
1035 }
1036}

1038template <int dim>
1039void
1040TimeHarmonicMaxwellAssemblerRobinBC<dim>::assemble_rhs(
1041 const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
1042 DPGCopyData &copy_data)
1043{
1044 if (!scratch_data.face_at_boundary)
1045 return;
1046
1047 const BoundaryConditions::TimeHarmonicMaxwellBoundaryType bc_type =
1048 boundary_conditions.type.at(scratch_data.face_boundary_id);
1049 if (!is_robin_boundary_type(bc_type))
1050 return;
1051
1052 using ScratchData = TimeHarmonicMaxwellScratchData<dim>;
1053
1054 const unsigned int n_face_q_points = scratch_data.n_face_q_points;
1055
1056 // Local dof indices of the real and imaginary parts of each base function of
1057 // the electric test functions (F)
1058 const std::vector<unsigned int> &test_dofs_F_real =
1059 scratch_data.test_local_dof_index_map[ScratchData::E_real];
1060 const std::vector<unsigned int> &test_dofs_F_imag =
1061 scratch_data.test_local_dof_index_map[ScratchData::E_imag];
1062
1063 Vector<double> &l_vector = copy_data.l_vector;
1064
1065 // Tangential component of the excitation g at the face quadrature points,
1066 // multiplied by JxW. The left-hand side of the Robin boundary condition
1067 // n x H + Y n x (E x n) = g is tangential to the boundary, so only the
1068 // tangential component of g is meaningful. It is also the only component
1069 // for which -<F, g> is bounded on the test space, which only controls the
1070 // tangential trace of F on the Robin boundaries.
1071 std::vector<Tensor<1, dim, std::complex<double>>> g_inc_JxW_container(
1072 n_face_q_points);

1084 scratch_data.face_effective_magnetic_permeabilities[q])

1086 scratch_data.face_normals[q]) *

1102 l_vector[test_dofs_F_real[i]] -= w.real();
1103 l_vector[test_dofs_F_imag[i]] -= w.imag();
1104 }
