8template <int dim>
9void
10QuadraturePointsDataOut<dim>::build_patches(
11 const dealii::DoFHandler<dim> &dof_handler,
12 const dealii::Mapping<dim> &mapping,
13 const dealii::Quadrature<dim> &quadrature,
14 const std::function<
15 double(const typename dealii::DoFHandler<dim>::active_cell_iterator &)>
16 &filter_radius)
17{
18 patches.clear();
19 // Upper bound: also counts the ghost and artificial cells stored locally
20 patches.reserve(dof_handler.get_triangulation().n_active_cells() *
21 quadrature.size());
22
23 dealii::FEValues<dim> fe_values(mapping,
24 dof_handler.get_fe(),
25 quadrature,
26 dealii::update_quadrature_points);
27 for (const auto &cell : dof_handler.active_cell_iterators())
28 {
29 if (!cell->is_locally_owned())
30 continue;
31
32 fe_values.reinit(cell);
33 const double radius = filter_radius(cell);
34
35 for (const auto &q : fe_values.get_quadrature_points())
36 {
37 dealii::DataOutBase::Patch<0, dim> patch;
38 patch.vertices[0] = q;
39 patch.data.reinit(dataset_names.size(), 1);
40 patch.data(0, 0) = radius;
41 patch.data(1, 0) = 2. * radius;
42 patch.data(2, 0) = cell->global_active_cell_index();
43 patches.push_back(patch);
44 }
45 }
46}
