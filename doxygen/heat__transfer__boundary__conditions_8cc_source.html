24 template <int dim>
25 void
26 HeatTransferBoundaryConditions<dim>::declare_default_entry(
27 ParameterHandler &prm,
28 const types::boundary_id default_boundary_id)
29 {
30 declare_enum_entry(prm,
31 "type",
32 BoundaryType::noflux,
33 "Type of boundary condition for heat transfer.",
34 deprecated_boundary_type_names,
35 heat_transfer_boundary_types);
36
37 prm.declare_entry("id",
38 Utilities::to_string(default_boundary_id, 2),
39 Patterns::List(Patterns::Integer()),
40 "Mesh id for boundary conditions");
41
42 Functions::ParsedFunction<dim> temporary_function;
43
44 // Expression for the temperature for an imposed temperature at bc
45 prm.enter_subsection("value");
46 temporary_function.declare_parameters(prm);
47 prm.leave_subsection();
48
49 // Expression for the h coefficient of convection-radiation-flux bc
50 prm.enter_subsection("h");
51 temporary_function.declare_parameters(prm);
52 prm.leave_subsection();
53
54 // Temperature of environment for convection-radiation-flux bc
55 prm.enter_subsection("Tinf");
56 temporary_function.declare_parameters(prm);
57 prm.leave_subsection();
58
59 // Emissivity of the boundary for convection-radiation-flux bc
60 prm.enter_subsection("emissivity");
61 temporary_function.declare_parameters(prm);
62 prm.leave_subsection();
63
64 // Heat flux (Neumann) at the boundary for convection-radiation-flux bc
65 prm.enter_subsection("heat_flux");
66 temporary_function.declare_parameters(prm);
67 prm.leave_subsection();
68
69 // Periodic boundary condition parameters for HT physics
70 prm.declare_entry(
71 "periodic id",
72 "-1",
73 Patterns::Integer(),
74 "Mesh id for periodic face matching. Default entry is -1 to ensure that the periodic id is set by the user");
75
76 prm.declare_entry("periodic direction",
77 "0",
78 Patterns::Integer(),
79 "Direction for periodic boundary condition");

117 template <int dim>
118 void
119 HeatTransferBoundaryConditions<dim>::parse_boundary(ParameterHandler &prm)
120 {
121 // Parse the list of boundary ids
122 std::vector<types::boundary_id> boundary_ids =
123 convert_string_to_vector<types::boundary_id>(prm, "id");
124
125 // Check if the list contains at least one boundary id
126 AssertThrow(
127 boundary_ids.size() > 0,
128 ExcMessage(
129 "A boundary id has not been set for one of the heat transfer boundary conditions. Please ensure that the id is set for every boundary condition."));
130
131 // Loop through all boundary ids to ensure that they are all non-negative
132 // and unique
133 for (const auto boundary_id : boundary_ids)
134 {
135 AssertThrow(this->type.find(boundary_id) == this->type.end(),
136 HeatTransferBoundaryDuplicated(boundary_id));
137
138
139 const BoundaryType boundary_type =
140 string_to_enum<BoundaryType>(prm.get("type"),
141 deprecated_boundary_type_names,
142 "type");
143 this->type[boundary_id] = boundary_type;
144 if (boundary_type == BoundaryType::convection_radiation)
145 {
146 this->has_convection_radiation_bc = true;
147
148 // Emissivity validity (0 <= emissivity <= 1) will be checked at
149 // evaluation.
150 }
151 else if (boundary_type == BoundaryType::periodic)
152 {
153 types::boundary_id periodic_boundary_id =
154 prm.get_integer("periodic id");
155
156 // We attribute a periodic neighbor boundary type to the neighbor
157 // boundary to ensure that all boundaries have a defined type
158 this->type[periodic_boundary_id] = BoundaryType::periodic_neighbor;
159
160 // We store the periodic id and direction
161 this->periodic_boundaries[boundary_id] = {
162 .neighbor_id = periodic_boundary_id,
163 .direction = static_cast<unsigned int>(
164 prm.get_integer("periodic direction"))};
165 }
166
167 // All the functions are parsed since they might be used for
168 // post-processing
169 prm.enter_subsection("value");
170 this->dirichlet_value[boundary_id] =
171 std::make_shared<Functions::ParsedFunction<dim>>();
172 this->dirichlet_value[boundary_id]->parse_parameters(prm);
173 prm.leave_subsection();
174 prm.enter_subsection("h");
175 this->h[boundary_id] =
176 std::make_shared<Functions::ParsedFunction<dim>>();
177 this->h[boundary_id]->parse_parameters(prm);
178 prm.leave_subsection();
179 prm.enter_subsection("Tinf");
180 this->Tinf[boundary_id] =
181 std::make_shared<Functions::ParsedFunction<dim>>();
182 this->Tinf[boundary_id]->parse_parameters(prm);
183 prm.leave_subsection();
184 prm.enter_subsection("emissivity");
185 this->emissivity[boundary_id] =
186 std::make_shared<Functions::ParsedFunction<dim>>();
187 this->emissivity[boundary_id]->parse_parameters(prm);
188 prm.leave_subsection();
189 prm.enter_subsection("heat_flux");
190 this->heat_flux_bc[boundary_id] =
191 std::make_shared<Functions::ParsedFunction<dim>>();
192 this->heat_flux_bc[boundary_id]->parse_parameters(prm);
193 prm.leave_subsection();
194 }
195 }

197 template <int dim>
198 void
199 HeatTransferBoundaryConditions<dim>::parse_parameters(ParameterHandler &prm)
200 {
201 prm.enter_subsection("boundary conditions heat transfer");
202 {
203 this->number_of_boundary_conditions = prm.get_integer("number");
204 this->time_dependent = prm.get_bool("time dependent");
205 for (unsigned int n = 0; n < this->number_of_boundary_conditions; n++)
206 {
207 prm.enter_subsection("bc " + std::to_string(n));
208 {
209 parse_boundary(prm);
210 }
211 prm.leave_subsection();
212 }
213 this->Stefan_Boltzmann_constant =
214 prm.get_double("Stefan-Boltzmann constant");
215 }
216 prm.leave_subsection();
217 }
