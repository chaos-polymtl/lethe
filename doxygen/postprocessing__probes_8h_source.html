17public:
18 /**
19 * @brief Object that evaluates variables (e.g., velocity) at specified points
20 * in the domain.
21 *
22 * Within every physics postprocessing attribute (either
23 * NavierStokesBase::postprocess_fd() or AuxiliaryPhysics::postprocess())
24 * the evaluation can made by calling the postprocess_probes() attribute of
25 * this object through the MultiphysicsInterface object.
26 * All evaluated values of variables of a same probe are stored within a same
27 * table. All tables are grouped in the PostprocessingProbes::probe_tables
28 * vector. Within the vector, the tables are ordered in increasing order of
29 * probe IDs.
30 *
31 * @param[in] simulation_control SimulationControl object which is used to
32 * obtain the current time and current iteration number.
33 * @param[in] probing_points_parameters Set of parameters describing the
34 * specified probing point and variables of interest.
35 * @param[in] output_frequency Output iteration frequency for postprocessing.
36 * @param[in] verbosity Verbosity level for console output.
37 *
38 * @remark At the moment, probing for only the velocity, pressure, phase, and
39 * temperature variables are implemented. Implementation for other variables
40 * will follow.
41 */
42 PostprocessingProbes(std::shared_ptr<SimulationControl> &simulation_control,
43 const Parameters::PostProcessing<dim>::ProbingPoints
44 &probing_points_parameters,
45 const unsigned int output_frequency,
46 const Parameters::Verbosity &verbosity);

100 /**
101 * @brief Writes the probe tables into output `.dat` file.
102 */
103 inline void
104 write_probing_points_tables()
105 {
106 if (simulation_control->get_iteration_number() % output_frequency == 0 ||
107 simulation_control->is_at_end())
108 if (Utilities::MPI::this_mpi_process(MPI_COMM_WORLD) == 0)
109 {
110 for (unsigned int i = 0; i < probe_tables.size(); ++i)
111 {
112 std::string file_path =
113 output_folder + "probe_" + Utilities::int_to_string(i, 2) +
114 "_" + probing_points_parameters.probing_points_output_names[i] +
115 ".dat";
116 std::ofstream output(file_path.c_str());
117
118 probe_tables[i].write_text(output);
119 }
120 }
121 }

174private:
175 /**
176 * @brief Prepares probe tables by setting the entry formats and precisions
177 * for all probes of the specified `variable`.
178 *
179 * @param[in] variable Variable of interest.
180 *
181 */
182 inline void
183 prepare_probe_tables(const Variable &variable)
184 {
185 const auto &probing_points_of_variable =
186 probing_points_parameters.probing_points_per_variable.at(variable);
187
188 // Go through tables of the variable and prepare them by declaring and
189 // formating the column(s) of the variable
190 for (const unsigned int &id : probing_points_of_variable.ids)
191 {
192 TableHandler &current_table = probe_tables[id];
193 const std::string column_name =
194 get_variable_string_with_underscores(variable);
195
196 if (implemented_scalar_variables.contains(variable))
197 {
198 current_table.declare_column(column_name);
199 current_table.set_precision(column_name, table_precision);
200 current_table.set_scientific(column_name, true);
201 }
202 else if (implemented_vector_variables.contains(variable))
203 {
204 // Only velocity is implemented at the moment
205 current_table.declare_column(column_name + "_x");
206 current_table.set_precision(column_name + "_x", table_precision);
207 current_table.set_scientific(column_name + "_x", true);
208 current_table.declare_column(column_name + "_y");
209 current_table.set_precision(column_name + "_y", table_precision);
210 current_table.set_scientific(column_name + "_y", true);
211 if constexpr (dim == 3)
212 {
213 current_table.declare_column(column_name + "_z");
214 current_table.set_precision(column_name + "_z",
215 table_precision);
216 current_table.set_scientific(column_name + "_z", true);
217 }
218 }
219 }
220 }

222 /**
223 * @brief Checks if the table of a specified probing point (@p probe_id)
224 * already has a row entry for the current time step.
225 *
226 * @param[in] probe_id Identifier of the probing point investigated.
227 *
228 * @return
229 * - @p true if there already is an entry for the current time step.
230 * - @p false if there is no row for the current time step.
231 */
232 inline bool
233 check_if_table_contains_current_time(const unsigned int &probe_id)
234 {
235 if (probe_tables_with_current_time.contains(probe_id))
236 return true;
237 else // No entry yet for the current time step
238 {
239 // Add the ID of the probing point to the set
240 probe_tables_with_current_time.insert(probe_id);
241 return false;
242 }
243 };
