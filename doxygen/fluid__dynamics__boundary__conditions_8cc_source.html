24 template <int dim>
25 void
26 FluidDynamicsBoundaryConditions<dim>::declare_default_entry(
27 ParameterHandler &prm,
28 const types::boundary_id default_boundary_id)
29 {
30 declare_enum_entry(prm,
31 "type",
32 BoundaryType::none,
33 "Type of boundary condition.",
34 deprecated_boundary_type_names,
35 fluid_dynamics_boundary_types);
36
37 prm.declare_entry("id",
38 Utilities::to_string(default_boundary_id, 2),
39 Patterns::List(Patterns::Integer()),
40 "Mesh id for boundary conditions.");
41
42 prm.declare_entry(
43 "periodic id",
44 "-1",
45 Patterns::Integer(),
46 "Mesh id for periodic face matching. Default entry is -1 to ensure that the periodic id is set by the user");
47
48 prm.declare_entry("periodic direction",
49 "0",
50 Patterns::Integer(),
51 "Direction for periodic boundary condition");
52
53 // Create a dummy NavierStokesBoundaryFunctions object to declare the
54 // appropriate parameters for this boundary condition.
55 NavierStokesBoundaryFunctions<dim> temporary_fluid_dynamics_functions;
56
57 prm.enter_subsection("u");
58 temporary_fluid_dynamics_functions.u.declare_parameters(prm);
59 prm.leave_subsection();
60
61 prm.enter_subsection("v");
62 temporary_fluid_dynamics_functions.v.declare_parameters(prm);
63 prm.leave_subsection();
64
65 prm.enter_subsection("w");
66 temporary_fluid_dynamics_functions.w.declare_parameters(prm);
67 prm.leave_subsection();
68
69 prm.enter_subsection("p");
70 temporary_fluid_dynamics_functions.p.declare_parameters(prm);
71 prm.leave_subsection();
72
73 /*
74 Vector-valued function for the traction with dim components for the
75 traction vector and 1 component for the pressure value at the boundary
76 which is dummy and is needed to properly assemble the traction term as
77 FEFaceIntegrator uses MatrixFree which in turn is based on the DoFHandler
78 which in turn is based on the FE system that has dim velocity components
79 and 1 pressure component.
80 */
81
82 prm.enter_subsection("traction fn");
83 temporary_fluid_dynamics_functions.traction_fn.declare_parameters(prm,
84 dim + 1);
85 prm.leave_subsection();
86
87 // Center of rotation of the boundary condition for torque calculation
88 prm.enter_subsection("center of rotation");
89 prm.declare_entry("x", "0", Patterns::Double(), "X COR");
90 prm.declare_entry("y", "0", Patterns::Double(), "Y COR");
91 prm.declare_entry("z", "0", Patterns::Double(), "Z COR");
92 prm.leave_subsection();
93
94 prm.declare_entry(
95 "beta",
96 "1",
97 Patterns::Double(),
98 "Penalty parameter for weak boundary condition imposed through Nitsche's method or outlets");
99
100 prm.declare_entry(
101 "boundary layer thickness",
102 "0",
103 Patterns::Double(),
104 "Thickness of the boundary layer used to calculate the penalty parameter for partial slip boundary condition in tangent direction imposed through Nitsche's method or outlets");
105 }

107 template <int dim>
108 void
109 FluidDynamicsBoundaryConditions<dim>::declare_parameters(
110 ParameterHandler &prm,
111 const unsigned int number_of_boundary_conditions)
112 {
113 prm.enter_subsection("boundary conditions");
114 {
115 prm.declare_entry("number",
116 Utilities::int_to_string(number_of_boundary_conditions),
117 Patterns::Integer(),
118 "Number of boundary conditions");
119 prm.declare_entry(
120 "time dependent",
121 "false",
122 Patterns::Bool(),
123 "Bool to define if the boundary condition is time dependent");
124
125 prm.declare_entry(
126 "fix pressure constant",
127 "false",
128 Patterns::Bool(),
129 "Bool to define if zero pressure is used as a constraint one node");
130
131 for (unsigned int n = 0; n < number_of_boundary_conditions; n++)
132 {
133 prm.enter_subsection("bc " + std::to_string(n));
134 {
135 declare_default_entry(prm, n);
136 }
137 prm.leave_subsection();
138 }
139 }
140 prm.leave_subsection();
141 }

143 template <int dim>
144 void
145 FluidDynamicsBoundaryConditions<dim>::parse_boundary(ParameterHandler &prm)
146 {
147 // Parse the list of boundary ids
148 std::vector<types::boundary_id> boundary_ids =
149 convert_string_to_vector<types::boundary_id>(prm, "id");
150
151 // Check if the list contains at least one boundary id
152 AssertThrow(
153 boundary_ids.size() > 0,
154 ExcMessage(
155 "A boundary id has not been set for one of the fluid dynamics boundary conditions. Please ensure that the id is set for every boundary condition."));
156
157 // and unique
158 for (const auto boundary_id : boundary_ids)
159 {
160 AssertThrow(this->type.find(boundary_id) == this->type.end(),
161 NavierStokesBoundaryDuplicated(boundary_id));
162
163 // Allocate the navier_stokes_functions object for every boundary
164 // condition to ensure that they have a defined function and a center of
165 // rotation.
166 navier_stokes_functions[boundary_id] =
167 std::make_shared<NavierStokesBoundaryFunctions<dim>>();
168
169 prm.enter_subsection("u");
170 navier_stokes_functions[boundary_id]->u.parse_parameters(prm);
171 prm.leave_subsection();
172
173 prm.enter_subsection("v");
174 navier_stokes_functions[boundary_id]->v.parse_parameters(prm);
175 prm.leave_subsection();
176
177 prm.enter_subsection("w");
178 navier_stokes_functions[boundary_id]->w.parse_parameters(prm);
179 prm.leave_subsection();
180
181 prm.enter_subsection("p");
182 navier_stokes_functions[boundary_id]->p.parse_parameters(prm);
183 prm.leave_subsection();
184
185 prm.enter_subsection("traction fn");
186 navier_stokes_functions[boundary_id]->traction_fn.parse_parameters(prm);
187 prm.leave_subsection();
188
189 prm.enter_subsection("center of rotation");
190 navier_stokes_functions[boundary_id]->center_of_rotation[0] =
191 prm.get_double("x");
192 navier_stokes_functions[boundary_id]->center_of_rotation[1] =
193 prm.get_double("y");
194
195 if (dim == 3)
196 navier_stokes_functions[boundary_id]->center_of_rotation[2] =
197 prm.get_double("z");
198 prm.leave_subsection();
199
200 // Establish the type of boundary condition
201 const BoundaryType boundary_type =
202 string_to_enum<BoundaryType>(prm.get("type"),
203 deprecated_boundary_type_names,
204 "type");
205 this->type[boundary_id] = boundary_type;
206 if (boundary_type == BoundaryType::periodic)
207 {
208 types::boundary_id periodic_boundary_id =
209 prm.get_integer("periodic id");
210
211 this->type[boundary_id] = BoundaryType::periodic;
212
213 // We attribute a periodic neighbor boundary type to the neighbor
214 // boundary to ensure that all boundaries have a defined type
215 this->type[periodic_boundary_id] = BoundaryType::periodic_neighbor;
216
217 // We store the periodic id and direction
218 this->periodic_boundaries[boundary_id] = {
219 .neighbor_id = periodic_boundary_id,
220 .direction = static_cast<unsigned int>(
221 prm.get_integer("periodic direction"))};
222
223 // Allocate the navier_stokes_functions object for the periodic
224 // neighbor condition to ensure that they have a defined function
225 // and a center of rotation. Parse the information of the boundary
226 // id to duplicate it.
227 navier_stokes_functions[periodic_boundary_id] =
228 std::make_shared<NavierStokesBoundaryFunctions<dim>>();
229
230 prm.enter_subsection("u");
231 navier_stokes_functions[periodic_boundary_id]->u.parse_parameters(
232 prm);
233 prm.leave_subsection();
234
235 prm.enter_subsection("v");
236 navier_stokes_functions[periodic_boundary_id]->v.parse_parameters(
237 prm);
238 prm.leave_subsection();
239
240 prm.enter_subsection("w");
241 navier_stokes_functions[periodic_boundary_id]->w.parse_parameters(
242 prm);
243 prm.leave_subsection();
244
245 prm.enter_subsection("traction fn");
246 navier_stokes_functions[periodic_boundary_id]
247 ->traction_fn.parse_parameters(prm);
248 prm.leave_subsection();
249
250 prm.enter_subsection("p");
251 navier_stokes_functions[periodic_boundary_id]->p.parse_parameters(
252 prm);
253 prm.leave_subsection();
254
255 prm.enter_subsection("center of rotation");
256 navier_stokes_functions[periodic_boundary_id]
257 ->center_of_rotation[0] = prm.get_double("x");
258 navier_stokes_functions[periodic_boundary_id]
259 ->center_of_rotation[1] = prm.get_double("y");
260
261 if (dim == 3)
262 navier_stokes_functions[periodic_boundary_id]
263 ->center_of_rotation[2] = prm.get_double("z");
264 prm.leave_subsection();
265 }
266 this->beta[boundary_id] = prm.get_double("beta");
267 this->boundary_layer_thickness[boundary_id] =
268 prm.get_double("boundary layer thickness");
269 }
270 }

272 template <int dim>
273 void
274 FluidDynamicsBoundaryConditions<dim>::parse_parameters(ParameterHandler &prm)
275 {
276 prm.enter_subsection("boundary conditions");
277 {
278 this->number_of_boundary_conditions = prm.get_integer("number");
279 this->time_dependent = prm.get_bool("time dependent");
280 this->fix_pressure_constant = prm.get_bool("fix pressure constant");
281
282 for (unsigned int n = 0; n < this->number_of_boundary_conditions; n++)
283 {
284 prm.enter_subsection("bc " + std::to_string(n));
285 {
286 parse_boundary(prm);
287 }
288 prm.leave_subsection();
289 }
290 }
291 prm.leave_subsection();
292 }

298template <int dim>
299double
300FluidDynamicsVelocityFunctionDefined<dim>::value(
301 const Point<dim> &point,
302 const unsigned int component) const
303{
304 Assert(component < this->n_components,
305 ExcIndexRange(component, 0, this->n_components));
306
307 if (component == 0)
308 {
309 return u->value(point);
310 }
311 if (component == 1)
312 {
313 return v->value(point);
314 }
315 if (component == 2)
316 {
317 return w->value(point);
318 }
319 return 0.;
320}
