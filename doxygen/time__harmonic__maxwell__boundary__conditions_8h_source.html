147/**
148 * @brief This class implements a boundary condition type for the Time Harmonic Maxwell equation
149 * where the electric field components are defined using individual functions
150 */
151template <int dim>
152class TimeHarmonicMaxwellElectricFieldDefined : public Function<dim>
153{
154 Functions::ParsedFunction<dim> *e_x_real;
155 Functions::ParsedFunction<dim> *e_y_real;
156 Functions::ParsedFunction<dim> *e_z_real;
157 Functions::ParsedFunction<dim> *e_x_imag;
158 Functions::ParsedFunction<dim> *e_y_imag;
159 Functions::ParsedFunction<dim> *e_z_imag;
160
161public:
162 /**
163 * @brief Initialize the electric-field boundary function with real and
164 * imaginary component functions.
165 *
166 * @param[in] p_e_x_real The parsed function for the real x-component.
167 * @param[in] p_e_y_real The parsed function for the real y-component.
168 * @param[in] p_e_z_real The parsed function for the real z-component.
169 * @param[in] p_e_x_imag The parsed function for the imaginary x-component.
170 * @param[in] p_e_y_imag The parsed function for the imaginary y-component.
171 * @param[in] p_e_z_imag The parsed function for the imaginary z-component.
172 */
173 TimeHarmonicMaxwellElectricFieldDefined(
174 Functions::ParsedFunction<dim> *p_e_x_real,
175 Functions::ParsedFunction<dim> *p_e_y_real,
176 Functions::ParsedFunction<dim> *p_e_z_real,
177 Functions::ParsedFunction<dim> *p_e_x_imag,
178 Functions::ParsedFunction<dim> *p_e_y_imag,
179 Functions::ParsedFunction<dim> *p_e_z_imag)
180 : Function<dim>(4 * dim)
181 , e_x_real(p_e_x_real)
182 , e_y_real(p_e_y_real)
183 , e_z_real(p_e_z_real)
184 , e_x_imag(p_e_x_imag)
185 , e_y_imag(p_e_y_imag)
186 , e_z_imag(p_e_z_imag)
187 {}
188
189 /**
190 * @brief Calculates the value of an imposed electric field equations.
191 *
192 * @param[in] point A point at which the function will be evaluated.
193 * @param[in] component The vector component of the boundary condition (0-> E
194 * x real, 1-> E y real, 2-> E z real, 3-> E x imag, 4-> E y imag, 5-> E z
195 * imag)
196 */
197 double
198 value(const Point<dim> &point, const unsigned int component) const override;
199};
