105 /**
106 * @brief Check whether a QMOM scalar name is valid. The names are used as
107 * field names in the output files, so they may only contain letters,
108 * digits and underscores.
109 *
110 * @param[in] scalar_name Name of the scalar.
111 *
112 * @return Whether the name is non-empty and only contains letters, digits
113 * and underscores.
114 */
115 bool
116 is_valid_scalar_name(const std::string &scalar_name)
117 {
118 const auto is_valid_character = [](const char character) {
119 return std::isalnum(static_cast<unsigned char>(character)) != 0 ||
120 character == '_';
121 };
122 return !scalar_name.empty() &&
123 std::ranges::all_of(scalar_name, is_valid_character);
124 }

326 {
327 this->reinitialization_method_type =

334 Parameters::ReinitializationMethodType::pde_based)

339
340 this->frequency = prm.get_integer("frequency");
341 Assert(this->frequency > 0,
342 ReinitializationMethodFrequencyError(this->frequency));
343
344 this->verbosity = string_to_enum<Verbosity>(prm.get("verbosity"),
345 deprecated_verbosity_names(),
346 "verbosity");
347
348 this->sharpening.parse_parameters(prm);
349 this->pde_based_interface_reinitialization.parse_parameters(prm);
350 this->geometric_interface_reinitialization.parse_parameters(prm);
351 }
352 prm.leave_subsection();

911void
912Parameters::QMOM::parse_parameters(ParameterHandler &prm)
913{
914 prm.enter_subsection("QMOM");
915 {
916 process = string_to_enum<QMOMProcess>(prm.get("process"));
917 number_of_scalars = prm.get_integer("number of scalars");
918 number_of_moments = prm.get_integer("number of moments");
919 scalar_names = Utilities::split_string_list(prm.get("scalar names"));
920
921 // A process other than generic imposes its scalars and its number of
922 // moments. The entries of the parameter file are ignored, but must not
923 // contradict the process.
924 std::vector<std::string> process_scalar_names;
925 unsigned int process_number_of_moments = number_of_moments;
926 switch (process)
927 {
928 case QMOMProcess::generic:
929 break;
930 case QMOMProcess::lipid_nanoparticle_production:
931 process_scalar_names = {"ethanol_concentration",
932 "free_lipids_concentration"};
933 process_number_of_moments = 8; // 4-node quadrature
934 break;
935 }
936 if (process != QMOMProcess::generic)
937 {
938 const QMOM defaults;
939 AssertThrow(number_of_scalars == defaults.number_of_scalars ||
940 number_of_scalars == process_scalar_names.size(),
941 QMOMProcessConflictError("number of scalars",
942 enum_to_string(process)));
943 AssertThrow(scalar_names.empty() ||
944 scalar_names == process_scalar_names,
945 QMOMProcessConflictError("scalar names",
946 enum_to_string(process)));
947 AssertThrow(number_of_moments == defaults.number_of_moments ||
948 number_of_moments == process_number_of_moments,
949 QMOMProcessConflictError("number of moments",
950 enum_to_string(process)));
951 number_of_scalars = process_scalar_names.size();
952 scalar_names = process_scalar_names;
953 number_of_moments = process_number_of_moments;
954 }
955 number_of_nodes = (number_of_moments + 1) / 2;
956
957 if (scalar_names.empty())
958 for (unsigned int i_scalar = 0; i_scalar < number_of_scalars; ++i_scalar)
959 scalar_names.emplace_back("scalar_" + std::to_string(i_scalar));
960 AssertThrow(scalar_names.size() == number_of_scalars,
961 QMOMScalarNamesCountError(scalar_names.size(),
962 number_of_scalars));
963 for (const std::string &scalar_name : scalar_names)
964 AssertThrow(is_valid_scalar_name(scalar_name),
965 QMOMScalarNameError(scalar_name));
966
967 splitting_scheme = string_to_enum<Parameters::QMOMSplittingScheme>(
968 prm.get("splitting scheme"));
969
970 enable_nucleation = prm.get_bool("enable nucleation");
971 enable_coalescence = prm.get_bool("enable coalescence");
972 enable_growth = prm.get_bool("enable growth");
973
974 phase_evolution.parse_parameters(prm);
975 lipid_nanoparticle_production.parse_parameters(prm);
976 physical_constants.parse_parameters(prm);
977 }
978 prm.leave_subsection();
979}

1406void
1407Parameters::TimeHarmonicMaxwell<dim>::parse_parameters(
1408 ParameterHandler &prm,
1409 const Dimensionality &dimensions)
1410{
1411 prm.enter_subsection("time harmonic maxwell");
1412 {
1413 prm.enter_subsection("time coupling strategy");
1414 {
1415 time_coupling_strategy =
1416 string_to_enum<Parameters::TimeHarmonicMaxwellCouplingStrategy>(
1417 prm.get("type"));
1418
1419 TimeHarmonicMaxwell::coupling_iteration =
1420 prm.get_integer("coupling iteration");
1421 TimeHarmonicMaxwell::coupling_time = prm.get_double("coupling time");
1422 TimeHarmonicMaxwell::coupling_threshold =
1423 prm.get_double("coupling threshold");
1424 }
1425 prm.leave_subsection();
1426
1427 TimeHarmonicMaxwell::electromagnetic_frequency =
1428 prm.get_double("electromagnetic frequency") *
1429 dimensions.electromagnetic_frequency_scaling;
1430
1431 TimeHarmonicMaxwell::electromagnetic_scaling_type =
1432 string_to_enum<Parameters::ElectromagneticScalingType>(
1433 prm.get("electromagnetic scaling type"),
1434 deprecated_electromagnetic_scaling_type_names,
1435 "electromagnetic scaling type");
1436
1437 // The user always provides the electric and magnetic field amplitudes
1438 // in dimensional units (V/m and A/m respectively). The scaling factors
1439 // will be applied in the time_harmonic_maxwell class.
1440 TimeHarmonicMaxwell::electric_field_amplitude =
1441 prm.get_double("electric field amplitude");
1442
1443 TimeHarmonicMaxwell::magnetic_field_amplitude =
1444 prm.get_double("magnetic field amplitude");
1445
1446 TimeHarmonicMaxwell::number_of_waveguide_inlets =
1447 prm.get_integer("number of waveguide inlets");
1448
1449 // Ensure that the number of waveguide inlets is smaller than the
1450 // maximum declared in declare_parameters.
1451 AssertThrow(
1452 TimeHarmonicMaxwell::number_of_waveguide_inlets <= 10,
1453 ExcMessage(
1454 "The number of waveguide inlets specified exceeds "
1455 "the maximum allowed. Please increase the value "
1456 "of max_waveguide_inlets in the code declare_parameters section of the TimeHarmonicMaxwell class."));
1457
1458 // Resize vectors to hold the correct number of inlets
1459 TimeHarmonicMaxwell::waveguide_mode.resize(number_of_waveguide_inlets);
1460 TimeHarmonicMaxwell::mode_order_m.resize(number_of_waveguide_inlets);
1461 TimeHarmonicMaxwell::mode_order_n.resize(number_of_waveguide_inlets);
1462 TimeHarmonicMaxwell::waveguide_boundary_ids.resize(
1463 number_of_waveguide_inlets);
1464 TimeHarmonicMaxwell::waveguide_power.resize(number_of_waveguide_inlets);
1465
1466 for (unsigned int inlet = 0; inlet < number_of_waveguide_inlets; ++inlet)
1467 {
1468 prm.enter_subsection("waveguide inlet " + std::to_string(inlet));
1469 {
1470 TimeHarmonicMaxwell::waveguide_boundary_ids[inlet] =
1471 prm.get_integer("port boundary id");
1472
1473 TimeHarmonicMaxwell::waveguide_power[inlet] =
1474 prm.get_double("waveguide power");
1475
1476 // Check that the waveguide power is not zero which would imply no
1477 // excitation at the inlet.
1478 AssertThrow(
1479 TimeHarmonicMaxwell::waveguide_power[inlet] > 0,
1480 ExcMessage(
1481 "The waveguide power for the inlet " + std::to_string(inlet) +
1482 " is zero. Please check the waveguide power parameter in the input prm file for this inlet. If you really want to have no excitation but an open boundary condition at this inlet, you should change the boundary condition type to the impedance boundary condition with zero excitation and the desired admittance."));
1483
1484 prm.enter_subsection("waveguide mode");
1485 {
1486 TimeHarmonicMaxwell::waveguide_mode[inlet] =
1487 string_to_enum<Parameters::WaveguideMode>(prm.get("mode type"));
1488
1489 TimeHarmonicMaxwell::mode_order_m[inlet] =
1490 prm.get_integer("mode order m");
1491
1492 TimeHarmonicMaxwell::mode_order_n[inlet] =
1493 prm.get_integer("mode order n");
1494
1495 // Check that the choice of mode orders is valid
1496 if (TimeHarmonicMaxwell::mode_order_m[inlet] == 0 &&
1497 TimeHarmonicMaxwell::mode_order_n[inlet] == 0)
1498 {
1499 throw(std::runtime_error(
1500 "Invalid waveguide mode orders m and n. "
1501 "At least one of the two mode orders must be non-zero."));
1502 }
1503 }
1504 prm.leave_subsection();
1505
1506 const unsigned int num_corners = Utilities::fixed_power<2>(dim - 1);
1507 std::array<Tensor<1, dim>, num_corners> tmp_corners;
1508
1509 for (unsigned int corner = 0; corner < num_corners; ++corner)
1510 {
1511 const std::string corner_name =
1512 "corner " + std::to_string(corner);
1513 const Tensor<1, dim> corner_value =
1514 value_string_to_tensor<dim>(prm.get(corner_name));
1515
1516 tmp_corners[corner] = corner_value;
1517 }
1518
1519 // In 3D, verify that the provided corners define a coplanar
1520 // quadrilateral. This is not useful in 2D as any 2 points are
1521 // always collinear.
1522 if constexpr (dim == 3)
1523 {
1524 const Tensor<1, dim> vec1 =
1525 tmp_corners[1] -
1526 tmp_corners[0]; // Vector from corner 1 to corner 2
1527 const Tensor<1, dim> vec2 =
1528 tmp_corners[2] -
1529 tmp_corners[0]; // Vector from corner 1 to corner 3
1530 const Tensor<1, dim> vec3 =
1531 tmp_corners[3] -
1532 tmp_corners[0]; // Vector from corner 1 to corner 4
1533
1534 // The triple scalar product (determinant) scales as the
1535 // volume of the parallelepiped defined by the three vectors.
1536 // So we make the it dimensionless by a characteristic volume
1537 // (product of the norms of the three vectors). This way we
1538 // can set a tolerance that is independent of the actual size
1539 // of the waveguide.
1540 const double determinant =
1541 scalar_product(vec3, cross_product_3d(vec1, vec2)) /
1542 (vec1.norm() * vec2.norm() * vec3.norm());
1543
1544 const double tolerance = 1e-10;
1545
1546 AssertThrow(std::abs(determinant) < tolerance,
1547 ExcMessage(
1548 "The provided corners for waveguide inlet " +
1549 std::to_string(inlet) +
1550 " do not define a coplanar quadrilateral. "
1551 "Please check the corner coordinates."));
1552 }
1553
1554 TimeHarmonicMaxwell::waveguide_corners.push_back(tmp_corners);
1555 }
1556 prm.leave_subsection();
1557 }
1558
1559 // Check that there is at least one waveguide inlet if the user use the
1560 // `power` electromagnetic scaling type.
1561 AssertThrow(
1562 !((TimeHarmonicMaxwell::electromagnetic_scaling_type ==
1563 Parameters::ElectromagneticScalingType::power) &&
1564 TimeHarmonicMaxwell::number_of_waveguide_inlets == 0),
1565 ExcMessage(
1566 "The power-based electromagnetic scaling type requires at least one waveguide inlet to be defined. Please check the number of waveguide inlets specified in the input prm file."));
1567 }
1568 prm.leave_subsection();
1569}
