21/**
22 * @brief Solver class for coupled Computational Fluid Dynamics and Discrete Element Method (CFD-DEM) simulations using a matrix-free formulation.
23 *
24 * This class implements a two-way coupled CFD-DEM solver that combines the
25 * Volume-averaged Navier-Stokes (VANS) equations for fluid flow with the
26 * soft-sphere Discrete Element Method for particle dynamics. The solver
27 * enables simulation of fluid-particle interactions in systems containing
28 * dense particle suspensions.
29 *
30 * The CFD component uses Galerkin Least Squares (GLS) stabilization for
31 * the VANS equations, while the DEM component handles particle-particle
32 * and particle-wall collisions using the soft-sphere contact model. A
33 * matrix-free formulation is used to solve the VANS equation.
34 *
35 * @tparam dim Spatial dimension of the simulation (2 or 3).
36 * @tparam PropertiesIndex Index of the properties used within the
37 * ParticleHandler.
38 * While 2D simulation are theoretically supported, they have not been tested.
39 *
40 * @ingroup solvers
41 */

114 /**
115 * @brief Set the initial condition. The version in this class does not output
116 * the solution, since the void fraction is not initialized yet.
117 *
118 * @param[in] initial_condition_type Type of method use to impose initial
119 *condition.
120 *
121 * @param[in] restart Indicator if the simulation is being restarted or not.
122 *
123 **/
124
125 void
126 set_initial_condition(
127 const Parameters::FluidDynamicsInitialConditionType initial_condition_type,
128 const bool restart = false) override
129 {
130 unsigned int ref_iter = 0;
131 do
132 {
133 if (ref_iter > 0)
134 this->refine_mesh();
135
136 this->set_initial_condition_fd(initial_condition_type, restart);
137 if (!restart)
138 {
139 this->multiphysics->set_initial_conditions();
140 }
141 ref_iter++;
142 }
143 while (
144 ref_iter <
145 (this->simulation_parameters.mesh_adaptation.initial_refinement + 1) &&
146 restart == false);
147
148 this->pcout
149 << "---------------------------------------------------------------"
150 << std::endl;
151 }
