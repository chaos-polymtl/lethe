46/**
47 * @brief Check if a boundary condition type of the time-harmonic Maxwell
48 * equations is a Robin boundary condition, which is applied weakly through the
49 * assembly.
50 *
51 * @param[in] type Boundary condition type.
52 *
53 * @return True for the silver_muller, impedance_boundary and waveguide_port
54 * boundary conditions.
55 */
56inline bool
57is_robin_boundary_type(const BoundaryConditions::BoundaryType type)
58{
59 return (type == BoundaryConditions::BoundaryType::silver_muller) ||
60 (type == BoundaryConditions::BoundaryType::impedance_boundary) ||
61 (type == BoundaryConditions::BoundaryType::waveguide_port);
62}

123/**
124 * @brief Class that assembles the cell terms of the ultraweak DPG formulation
125 * of the time-harmonic Maxwell equations. With the test functions
126 * \f$\mathbf{F}\f$ (electric) and \f$\mathbf{I}\f$ (magnetic), it assembles:
127 *
128 * - the graph norm of the test space in the Gram matrix \f$G\f$:
129 * \f$(\mathbf{F}, \mathbf{F}) + (\nabla \times \mathbf{F}, \nabla \times
130 * \mathbf{F}) + (i\omega\varepsilon \mathbf{F}, i\omega\varepsilon
131 * \mathbf{F})\f$ and the equivalent terms coupling \f$\mathbf{F}\f$ and
132 * \f$\mathbf{I}\f$ and for \f$\mathbf{I}\f$ alone;
133 *
134 * - the interior bilinear form in the matrix \f$B\f$:
135 * \f$(i\omega\varepsilon \mathbf{E}, \mathbf{F}) + (\mathbf{H}, \nabla
136 * \times \mathbf{F}) + (\mathbf{E}, \nabla \times \mathbf{I}) -
137 * (i\omega\mu \mathbf{H}, \mathbf{I})\f$;
138 *
139 * - the interior load vector \f$l\f$, which is zero since no volume source
140 * term is considered at the moment.
141 *
142 * @tparam dim An integer that denotes the number of spatial dimensions
143 *
144 * @ingroup assemblers
145 */
146template <int dim>
147class TimeHarmonicMaxwellAssemblerCore
148 : public TimeHarmonicMaxwellAssemblerBase<dim>
149{
150public:
151 /**
152 * @brief Constructor.
153 *
154 * @param[in] time_harmonic_maxwell_parameters Parameters of the
155 * time-harmonic Maxwell physics, used to get the angular frequency.
156 */
157 TimeHarmonicMaxwellAssemblerCore(const Parameters::TimeHarmonicMaxwell<dim>
158 &time_harmonic_maxwell_parameters);
159
160 /**
161 * @brief Assemble the Gram matrix and the interior bilinear form.
162 *
163 * @param[in] scratch_data (see base class)
164 *
165 * @param[in,out] copy_data (see base class)
166 */
167 void
168 assemble_matrix(const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
169 DPGCopyData &copy_data) override;
170
171 /**
172 * @brief Assemble the interior load vector, which is zero since no volume
173 * source term is considered at the moment.
174 *
175 * @param[in] scratch_data (see base class)
176 *
177 * @param[in,out] copy_data (see base class)
178 */
179 void
180 assemble_rhs(const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
181 DPGCopyData &copy_data) override;
182
183private:
184 /**
185 * Angular frequency of the electromagnetic fields.
186 */
187 const double omega;
188};

191/**
192 * @brief Class that assembles the skeleton terms of the ultraweak DPG
193 * formulation of the time-harmonic Maxwell equations. On every face, it
194 * assembles \f$\langle \mathbf{n} \times \hat{\mathbf{E}}, \mathbf{I}
195 * \rangle\f$ in the matrix \f$\hat{B}\f$ and, except on the faces where a
196 * Robin boundary condition is applied, it assembles
197 * \f$\langle \mathbf{n} \times \hat{\mathbf{H}}, \mathbf{F} \rangle\f$. On
198 * Robin faces, the latter term is replaced by the Robin boundary condition
199 * (see TimeHarmonicMaxwellAssemblerRobinBC).
200 *
201 * @tparam dim An integer that denotes the number of spatial dimensions
202 *
203 * @ingroup assemblers
204 */
205template <int dim>
206class TimeHarmonicMaxwellAssemblerSkeleton
207 : public TimeHarmonicMaxwellFaceAssemblerBase<dim>
208{
209public:
210 /**
211 * @brief Constructor.
212 *
213 * @param[in] boundary_conditions Boundary conditions of the time-harmonic
214 * Maxwell physics, used to identify the Robin faces.
215 */
216 TimeHarmonicMaxwellAssemblerSkeleton(
217 const BoundaryConditions::TimeHarmonicMaxwellBoundaryConditions<dim>
218 &boundary_conditions)
219 : boundary_conditions(boundary_conditions)
220 {}
221
222 /**
223 * @brief Assemble the skeleton terms of the bilinear form.
224 *
225 * @param[in] scratch_data (see base class)
226 *
227 * @param[in,out] copy_data (see base class)
228 */
229 void
230 assemble_matrix(const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
231 DPGCopyData &copy_data) override;
232
233 /**
234 * @brief There is no skeleton term in the load vector, since the boundary
235 * data is imposed through the constraints or by
236 * TimeHarmonicMaxwellAssemblerRobinBC.
237 *
238 * @param[in] scratch_data (see base class)
239 *
240 * @param[in,out] copy_data (see base class)
241 */
242 void
243 assemble_rhs(const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
244 DPGCopyData &copy_data) override;
245
246private:
247 /**
248 * Boundary conditions of the time-harmonic Maxwell physics.
249 */
250 const BoundaryConditions::TimeHarmonicMaxwellBoundaryConditions<dim>
251 &boundary_conditions;
252};

255/**
256 * @brief Class that assembles the Robin boundary conditions of the
257 * time-harmonic Maxwell equations (silver_muller, impedance_boundary and
258 * waveguide_port). On these faces, the magnetic trace is expressed from the
259 * electric trace with the surface admittance \f$Y\f$ and the excitation
260 * \f$\mathbf{g}\f$, which gives the terms
261 * \f$-\langle Y \hat{\mathbf{E}}, \mathbf{F} \rangle\f$ in \f$\hat{B}\f$ and
262 * \f$-\langle \mathbf{g}, \mathbf{F} \rangle\f$ in \f$l\f$. Since the energy
263 * norm that is minimized is also modified on these faces, it also assembles
264 * the boundary terms of the Gram matrix
265 * \f$G\f$:
266 *\f[\left(\mathbf{n}\times\mathbf{I}+Y*\mathbf{n}\times(\mathbf{F}\times\mathbf{n}),
267\mathbf{n}\times\tilde{\mathbf{I}}+Y*\mathbf{n}\times(\tilde{\mathbf{F}}\times\mathbf{n})\right)_{\Gamma_R}.\f]
268 *
269 * @tparam dim An integer that denotes the number of spatial dimensions
270 *
271 * @ingroup assemblers
272 */
273template <int dim>
274class TimeHarmonicMaxwellAssemblerRobinBC
275 : public TimeHarmonicMaxwellFaceAssemblerBase<dim>
276{
277public:
278 /**
279 * @brief Constructor.
280 *
281 * @param[in] time_harmonic_maxwell_parameters Parameters of the
282 * time-harmonic Maxwell physics, which contain the waveguide port
283 * definitions.
284 *
285 * @param[in] boundary_conditions Boundary conditions of the time-harmonic
286 * Maxwell physics.
287 *
288 * @param[in] waveguide_ports_electric_amplitudes Electric field amplitudes of
289 * all the waveguide ports. They are normalized by their maximum
290 * to get the relative amplitudes of the ports during initialization.
291 */
292 TimeHarmonicMaxwellAssemblerRobinBC(
293 const Parameters::TimeHarmonicMaxwell<dim>
294 &time_harmonic_maxwell_parameters,
295 const BoundaryConditions::TimeHarmonicMaxwellBoundaryConditions<dim>
296 &boundary_conditions,
297 const std::vector<double> &waveguide_ports_electric_amplitudes);
298
299 /**
300 * @brief Assemble the Robin boundary terms of the Gram matrix and of the
301 * skeleton bilinear form.
302 *
303 * @param[in] scratch_data (see base class)
304 *
305 * @param[in,out] copy_data (see base class)
306 */
307 void
308 assemble_matrix(const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
309 DPGCopyData &copy_data) override;
310
311 /**
312 * @brief Assemble the excitation of the Robin boundary conditions in the
313 * load vector.
314 *
315 * @param[in] scratch_data (see base class)
316 *
317 * @param[in,out] copy_data (see base class)
318 */
319 void
320 assemble_rhs(const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
321 DPGCopyData &copy_data) override;
322
323private:
324 /**
325 * @brief Compute the excitation and the surface admittance of the Robin
326 * boundary condition at a quadrature point of the current face.
327 *
328 * @param[in] scratch_data Scratch data reinitialized on the current face.
329 *
330 * @param[in] q Index of the face quadrature point.
331 *
332 * @param[in] bc_type Type of the Robin boundary condition of the face.
333 *
334 * @param[in] waveguide_port_index Index of the waveguide port condition of
335 * the face. Only used if bc_type is waveguide_port.
336 *
337 * @return A pair containing the excitation and the surface admittance.
338 */
339 std::pair<Tensor<1, dim, std::complex<double>>, std::complex<double>>
340 compute_robin_boundary_data(
341 const TimeHarmonicMaxwellScratchData<dim> &scratch_data,
342 const unsigned int q,
343 const BoundaryConditions::BoundaryType bc_type,
344 const unsigned int waveguide_port_index) const;
345
346 /**
347 * @brief Compute the excitation and the surface admittance of the Robin
348 * boundary condition of a rectangular waveguide port from the input
349 * parameters. The excitation used is the normalized one across all the
350 * waveguide ports. This function is only implemented for 3D problems.
351 *
352 * @param[in] p Position where the excitation is computed.
353 *
354 * @param[in] normal Unit normal vector of the waveguide port face.
355 *
356 * @param[in] effective_electric_permittivity Effective electric permittivity
357 * at the point p.
358 *
359 * @param[in] effective_magnetic_permeability Effective magnetic permeability
360 * at the point p.
361 *
362 * @param[in] waveguide_port_index Index of the waveguide port condition
363 * within the waveguide port parameters.
364 *
365 * @return A pair containing the excitation of the Robin boundary condition
366 * and the surface admittance at the point p. The excitation is scaled by the
367 * ratio of the port amplitude to the maximum amplitude of all the ports.
368 */
369 std::pair<Tensor<1, dim, std::complex<double>>, std::complex<double>>
370 compute_waveguide_port_excitation(
371 const Point<dim> &p,
372 const Tensor<1, dim> &normal,
373 const std::complex<double> &effective_electric_permittivity,
374 const std::complex<double> &effective_magnetic_permeability,
375 const unsigned int waveguide_port_index) const;
376
377 /**
378 * Parameters of the time-harmonic Maxwell physics.
379 */
380 const Parameters::TimeHarmonicMaxwell<dim> &time_harmonic_maxwell_parameters;
381
382 /**
383 * Boundary conditions of the time-harmonic Maxwell physics.
384 */
385 const BoundaryConditions::TimeHarmonicMaxwellBoundaryConditions<dim>
386 &boundary_conditions;
387
388 /**
389 * Relative amplitudes of all the waveguide ports, that is the electric field
390 * amplitude of each port divided by the maximum amplitude of all the ports.
391 */
392 std::vector<double> waveguide_ports_relative_amplitudes;
393};
