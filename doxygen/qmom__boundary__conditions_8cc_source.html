74 template <int dim>
75 void
76 QMOMBoundaryConditions<dim>::parse_boundary(ParameterHandler &prm)
77 {
78 const std::vector<types::boundary_id> boundary_ids =
79 convert_string_to_vector<types::boundary_id>(prm, "id");
80
81 AssertThrow(
82 boundary_ids.size() > 0,
83 ExcMessage(
84 "A boundary id has not been set for one of the QMOM boundary conditions. Please ensure that the id is set for every boundary condition."));
85
86 const QMOMBoundaryType boundary_type =
87 string_to_enum<QMOMBoundaryType>(prm.get("type"));
88 for (const auto boundary_id : boundary_ids)
89 {
90 AssertThrow(this->type.find(boundary_id) == this->type.end(),
91 QMOMBoundaryDuplicated(boundary_id));
92
93 this->type[boundary_id] = boundary_type;
94 if (boundary_type == QMOMBoundaryType::dirichlet)
95 {
96 prm.enter_subsection("dirichlet");
97 dirichlet[boundary_id].parse_parameters(prm);
98 prm.leave_subsection();
99 }
100 }
101 }

103 template <int dim>
104 void
105 QMOMBoundaryConditions<dim>::parse_parameters(ParameterHandler &prm)
106 {
107 prm.enter_subsection("boundary conditions qmom");
108 {
109 this->number_of_boundary_conditions = prm.get_integer("number");
110 this->time_dependent = prm.get_bool("time dependent");
111
112 for (unsigned int i_bc = 0; i_bc < this->number_of_boundary_conditions;
113 ++i_bc)
114 {
115 prm.enter_subsection("bc " + std::to_string(i_bc));
116 {
117 parse_boundary(prm);
118 }
119 prm.leave_subsection();
120 }
121 }
122 prm.leave_subsection();
123 }
