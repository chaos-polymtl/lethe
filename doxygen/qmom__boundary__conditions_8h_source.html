27 /**
28 * @brief Boundary conditions of the QMOM physics. They apply to the advective
29 * flux of all the QMOM fields (scalars and moments), which is meant to be
30 * computed with an upwind numerical flux:
31 * - dirichlet: the prescribed values enter the domain where the flow enters
32 * (u.n < 0) and the interior values leave it where the flow leaves;
33 * - outlet: the interior values leave the domain where the flow leaves and
34 * nothing enters where the flow enters (backflow);
35 * - noflux: the advective flux is zero, which is the natural choice for
36 * walls.
37 * Boundaries without a QMOM boundary condition are treated as noflux.
38 *
39 * @tparam dim Number of spatial dimensions.
40 */
41 template <int dim>
42 class QMOMBoundaryConditions : public BoundaryConditions<QMOMBoundaryType>
43 {
44 public:
45 /// Values imposed by the Dirichlet boundary conditions, for each boundary
46 /// id of type dirichlet

52 * @param[in,out] prm Parameter handler.
53 * @param[in] default_boundary_id Default boundary id of the boundary
54 * condition.
55 */
56 void
57 declare_default_entry(ParameterHandler &prm,
58 const types::boundary_id default_boundary_id);

60 /**
