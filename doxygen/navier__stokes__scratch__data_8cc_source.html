137template <int dim>
138void
139NavierStokesScratchData<dim>::allocate()
140{
141 // Initialize size of arrays
142 this->n_q_points = fe_values.get_quadrature().size();
143 this->n_dofs = fe_values.get_fe().n_dofs_per_cell();
144
145 // Initialize arrays related to quadrature
146 this->JxW.reinit(n_q_points);
147
148 // Initialize component array
149 this->components = std::vector<unsigned int>(n_dofs);
150
151 // Forcing term array
152 this->rhs_force =
153 std::vector<Vector<double>>(n_q_points, Vector<double>(dim + 1));
154 this->force = std::vector<Tensor<1, dim>>(n_q_points);
155 this->mass_source = std::vector<double>(n_q_points);
156
157 // Initialize arrays related to velocity and pressure
158 this->velocities.first_vector_component = 0;
159 this->pressure.component = dim;
160
161 // Velocity
162 this->velocity_values = std::vector<Tensor<1, dim>>(n_q_points);
163 this->velocity_divergences = std::vector<double>(n_q_points);
164 this->velocity_gradients = std::vector<Tensor<2, dim>>(n_q_points);
165 this->velocity_laplacians = std::vector<Tensor<1, dim>>(n_q_points);
166 this->velocity_hessians = std::vector<Tensor<3, dim>>(n_q_points);
167 this->velocity_gradient_divergence = std::vector<Tensor<1, dim>>(n_q_points);
168 this->shear_rate = std::vector<double>(n_q_points);
169 this->advective_velocity = std::vector<Tensor<1, dim>>(n_q_points);
170
171 // For SDIRK method: sum(a_ij * k_j)
172 if (this->simulation_control->is_sdirk())
173 {
174 this->sdirk_stage_sum = std::vector<Tensor<1, dim>>(n_q_points);
175 }
176
177 // Velocity for BDF schemes
178 this->previous_velocity_values = std::vector<std::vector<Tensor<1, dim>>>(
179 maximum_number_of_previous_solutions(),
180 std::vector<Tensor<1, dim>>(n_q_points));
181
182 // Pressure
183 this->pressure_values = std::vector<double>(n_q_points);
184 this->pressure_gradients = std::vector<Tensor<1, dim>>(n_q_points);
185 this->pressure_scaling_factor = 1;
186
187 // Pressure for BDF schemes (compressible Navier-Stokes)
188 this->previous_pressure_values =
189 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
190 std::vector<double>(n_q_points));
191 // Initialize arrays related to shape functions
192 // Velocity shape functions
193 this->phi_u.reinit(n_q_points, n_dofs);
194 this->grad_phi_u.reinit(n_q_points, n_dofs);
195 this->div_phi_u.reinit(n_q_points, n_dofs);
196 this->hess_phi_u.reinit(n_q_points, n_dofs);
197 this->laplacian_phi_u.reinit(n_q_points, n_dofs);
198 this->gradient_divergence_phi_u.reinit(n_q_points, n_dofs);
199
200 // Pressure shape functions
201 this->phi_p.reinit(n_q_points, n_dofs);
202 this->grad_phi_p.reinit(n_q_points, n_dofs);
203
204 // Physical properties
205 fields.insert(
206 std::pair<field, std::vector<double>>(field::pressure, n_q_points));
207 fields.insert(
208 std::pair<field, std::vector<double>>(field::shear_rate, n_q_points));
209
210 density = std::vector<double>(n_q_points);
211 dynamic_viscosity = std::vector<double>(n_q_points);
212 kinematic_viscosity = std::vector<double>(n_q_points);
213 dynamic_viscosity_for_stabilization = std::vector<double>(n_q_points);
214 kinematic_viscosity_for_stabilization = std::vector<double>(n_q_points);
215 thermal_expansion = std::vector<double>(n_q_points);
216 density_thermal_expansion = std::vector<double>(n_q_points);
217 grad_kinematic_viscosity_shear_rate = std::vector<double>(n_q_points);
218
219 previous_density =
220 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
221 std::vector<double>(this->n_q_points));
222}

225template <int dim>
226template <typename VectorType>
227void
228NavierStokesScratchData<dim>::reinit(
229 const typename DoFHandler<dim>::active_cell_iterator &cell,
230 const VectorType &current_solution,
231 const std::vector<VectorType> &previous_solutions,
232 const VectorType &sum_over_previous_stages,
233 const std::shared_ptr<Function<dim>> &forcing_function,
234 const Tensor<1, dim> &beta_force,
235 const double pressure_scaling_factor)
236{
237 this->fe_values.reinit(cell);
238
239 quadrature_points = this->fe_values.get_quadrature_points();
240 auto &fe = this->fe_values.get_fe();
241
242 forcing_function->vector_value_list(quadrature_points, this->rhs_force);
243
244 // Establish the force vector
245 for (unsigned int q = 0; q < n_q_points; ++q)
246 {
247 for (int d = 0; d < dim; ++d)
248 {
249 const unsigned int component_i =
250 fe.system_to_component_index(d).first;
251 this->force[q][d] = this->rhs_force[q](component_i);
252 }
253
254 const unsigned int component_mass =
255 fe.system_to_component_index(dim).first;
256 this->mass_source[q] = this->rhs_force[q](component_mass);
257
258 // Correct force to include the dynamic forcing term for flow
259 // control
260 force[q] = force[q] + beta_force;
261 }
262
263 for (const unsigned int k : fe_values.dof_indices())
264 {
265 components[k] = fe.system_to_component_index(k).first;
266 }
267
268 // Compute cell diameter
269 double cell_measure =
270 compute_cell_measure_with_JxW(this->fe_values.get_JxW_values());
271 this->cell_size = compute_cell_diameter<dim>(cell_measure, fe.degree);
272
273 // For the SDIRK methods, \sum_{j=1}^{i-1} a_{ij} k_j is needed for the
274 // assembler
275 if (this->simulation_control->is_sdirk())
276 {
277 this->fe_values[velocities].get_function_values(sum_over_previous_stages,
278 this->sdirk_stage_sum);
279 }
280
281 // Gather velocity (values, gradient and laplacian)
282 this->fe_values[velocities].get_function_values(current_solution,
283 this->velocity_values);
284 this->fe_values[velocities].get_function_gradients(current_solution,
285 this->velocity_gradients);
286 this->fe_values[velocities].get_function_laplacians(
287 current_solution, this->velocity_laplacians);
288 if (gather_hessian)
289 this->fe_values[velocities].get_function_hessians(current_solution,
290 this->velocity_hessians);
291
292 // Gather convective velocity (same as velocity_values unless ALE is
293 // enabled. The ALE correction is made within the correponding reinit
294 // function)
295 this->advective_velocity = this->velocity_values;
296
297 for (unsigned int q = 0; q < this->n_q_points; ++q)
298 {
299 this->velocity_divergences[q] = trace(this->velocity_gradients[q]);
300 // In the VANS momentum equations, the viscous stress tensor includes
301 // the gradient of the divergence of the velocity, ∇(∇·u) = ∂i(∂juj). We
302 // therefore compute this quantity here when assembling VANS-specific
303 // contributions (gated by gather_void_fraction).
304 if (gather_void_fraction)
305 {
306 // Compute term ∂j(∂iuj) = ∂i(∂juj) for use in the strong residual
307 // of the VANS momentum equations.
308 this->velocity_gradient_divergence[q] = 0;
309 for (int d1 = 0; d1 < dim; ++d1)
310 {
311 for (int d2 = 0; d2 < dim; ++d2)
312 {
313 this->velocity_gradient_divergence[q][d1] +=
314 (this->velocity_hessians[q][d2][d1][d2]);
315 }
316 }
317 }
318 }
319
320 // Gather pressure (values, gradient)
321 fe_values[pressure].get_function_values(current_solution,
322 this->pressure_values);
323 fe_values[pressure].get_function_gradients(current_solution,
324 this->pressure_gradients);
325 this->pressure_scaling_factor = pressure_scaling_factor;
326
327 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
328 {
329 this->fe_values[velocities].get_function_values(
330 previous_solutions[p], previous_velocity_values[p]);
331 }
332
333 // Only gather the pressure when a pressure history is necessary
334 // (compressible Navier-Stokes)
335 if (!this->properties_manager.density_is_constant())
336 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
337 {
338 this->fe_values[pressure].get_function_values(
339 previous_solutions[p], previous_pressure_values[p]);
340 }
341
342 for (unsigned int q = 0; q < n_q_points; ++q)
343 {
344 this->JxW[q] = this->fe_values.JxW(q);
345 for (unsigned int k = 0; k < n_dofs; ++k)
346 {
347 // Velocity
348 this->phi_u[q][k] = this->fe_values[velocities].value(k, q);
349 this->div_phi_u[q][k] = this->fe_values[velocities].divergence(k, q);
350 this->grad_phi_u[q][k] = this->fe_values[velocities].gradient(k, q);
351 this->hess_phi_u[q][k] = this->fe_values[velocities].hessian(k, q);
352 for (int d = 0; d < dim; ++d)
353 this->laplacian_phi_u[q][k][d] = trace(this->hess_phi_u[q][k][d]);
354
355 // In the VANS momentum equations, the viscous stress tensor
356 // includes the gradient of the divergence of the velocity, ∇(∇·u) =
357 // ∂i(∂juj). This results in a term ∂i(∂jδuj), which we compute here
358 // when assembling VANS-specific contributions (gated by
359 // gather_void_fraction).
360 if (gather_void_fraction)
361 {
362 this->gradient_divergence_phi_u[q][k] = 0;
363 // Compute term ∂j(∂iδuj) = ∂i(∂jδuj) to be used in the strong
364 // Jacobian of the VANS momentum equations
365
366 for (int d1 = 0; d1 < dim; ++d1)
367 {
368 for (int d2 = 0; d2 < dim; ++d2)
369 {
370 this->gradient_divergence_phi_u[q][k][d1] +=
371 (this->hess_phi_u[q][k][d2][d1][d2]);
372 }
373 }
374 }
375 // Pressure
376 this->phi_p[q][k] = this->fe_values[pressure].value(k, q);
377 this->grad_phi_p[q][k] = this->fe_values[pressure].gradient(k, q);
378 }
379 }
380
381 is_boundary_cell = cell->at_boundary();
382 if (is_boundary_cell)
383 {
384 n_faces = cell->n_faces();
385 is_boundary_face = std::vector<bool>(n_faces, false);
386 n_faces_q_points = fe_face_values.get_quadrature().size();
387 boundary_face_id = std::vector<unsigned int>(n_faces);
388
389 face_JxW.reinit(n_faces, n_faces_q_points);
390
391 // Velocity and pressure values
392 // First vector is face number, second quadrature point
393 this->face_velocity_values = std::vector<std::vector<Tensor<1, dim>>>(
394 n_faces, std::vector<Tensor<1, dim>>(n_faces_q_points));
395
396 this->face_velocity_divergences =
397 std::vector<std::vector<double>>(n_faces,
398 std::vector<double>(n_faces_q_points));
399
400 this->face_velocity_gradients = std::vector<std::vector<Tensor<2, dim>>>(
401 n_faces, std::vector<Tensor<2, dim>>(n_faces_q_points));
402
403 this->face_velocity_laplacians = std::vector<std::vector<Tensor<1, dim>>>(
404 n_faces, std::vector<Tensor<1, dim>>(n_faces_q_points));
405
406 this->face_pressure_values =
407 std::vector<std::vector<double>>(n_faces,
408 std::vector<double>(n_faces_q_points));
409
410 this->face_pressure_gradients = std::vector<std::vector<Tensor<1, dim>>>(
411 n_faces, std::vector<Tensor<1, dim>>(n_faces_q_points));
412
413 this->face_normal.reinit(n_faces, n_faces_q_points);
414
415 this->face_quadrature_points = std::vector<std::vector<Point<dim>>>(
416 n_faces, std::vector<Point<dim>>(n_faces_q_points));
417
418 this->face_div_phi_u.reinit(n_faces, n_faces_q_points, n_dofs);
419
420 this->face_phi_u.reinit(n_faces, n_faces_q_points, n_dofs);
421 this->face_hess_phi_u.reinit(n_faces, n_faces_q_points, n_dofs);
422
423 this->face_laplacian_phi_u.reinit(n_faces, n_faces_q_points, n_dofs);
424
425 this->face_grad_phi_u.reinit(n_faces, n_faces_q_points, n_dofs);
426 this->face_phi_p.reinit(n_faces, n_faces_q_points, n_dofs);
427 this->face_grad_phi_p.reinit(n_faces, n_faces_q_points, n_dofs);
428
429 for (const auto face : cell->face_indices())
430 {
431 is_boundary_face[face] = cell->face(face)->at_boundary();
432 if (is_boundary_face[face])
433 {
434 fe_face_values.reinit(cell, face);
435 n_dofs = fe_face_values.get_fe().n_dofs_per_cell();
436 boundary_face_id[face] = cell->face(face)->boundary_id();
437 // Shape functions
438 // First vector is face number, second quadrature point, third
439 // DOF
440
441 // Gather velocity (values, gradient and laplacian)
442 this->fe_face_values[velocities].get_function_values(
443 current_solution, this->face_velocity_values[face]);
444 this->fe_face_values[velocities].get_function_gradients(
445 current_solution, this->face_velocity_gradients[face]);
446 this->fe_face_values[velocities].get_function_laplacians(
447 current_solution, this->face_velocity_laplacians[face]);
448
449 // Gather pressure (values, gradient)
450 this->fe_face_values[pressure].get_function_values(
451 current_solution, this->face_pressure_values[face]);
452 this->fe_face_values[pressure].get_function_gradients(
453 current_solution, this->face_pressure_gradients[face]);
454
455 for (unsigned int q = 0; q < n_faces_q_points; ++q)
456 {
457 this->face_JxW[face][q] = this->fe_face_values.JxW(q);
458 this->face_normal[face][q] =
459 this->fe_face_values.normal_vector(q);
460 this->face_quadrature_points[face][q] =
461 this->fe_face_values.quadrature_point(q);
462 for (const unsigned int k : fe_face_values.dof_indices())
463 {
464 // Velocity
465 this->face_phi_u[face][q][k] =
466 this->fe_face_values[velocities].value(k, q);
467 this->face_div_phi_u[face][q][k] =
468 this->fe_face_values[velocities].divergence(k, q);
469 this->face_grad_phi_u[face][q][k] =
470 this->fe_face_values[velocities].gradient(k, q);
471 this->face_hess_phi_u[face][q][k] =
472 this->fe_face_values[velocities].hessian(k, q);
473 for (int d = 0; d < dim; ++d)
474 this->face_laplacian_phi_u[face][q][k][d] =
475 trace(this->face_hess_phi_u[face][q][k][d]);
476 // Pressure
477 this->face_phi_p[face][q][k] =
478 this->fe_face_values[pressure].value(k, q);
479 this->face_grad_phi_p[face][q][k] =
480 this->fe_face_values[pressure].gradient(k, q);
481 }
482 }
483 }
484 }
485 }

1082
1083template <int dim>
1084void
1085NavierStokesScratchData<dim>::calculate_physical_properties()
1086{
1087 if (properties_manager.field_is_required(field::temperature) &&
1088 gather_temperature)
1089 {
1090 set_field_vector(field::temperature,
1091 this->temperature_values,
1092 this->fields);
1093 }
1094
1095 if (properties_manager.field_is_required(field::pressure))
1096 {
1097 set_field_vector(field::pressure, this->pressure_values, this->fields);
1098 }
1099
1100 if (properties_manager.field_is_required(field::shear_rate))
1101 {
1102 for (unsigned int q = 0; q < n_q_points; ++q)
1103 {
1104 // Calculate shear rate (at each q)
1105 const Tensor<2, dim> shear_rate_tensor =
1106 velocity_gradients[q] + transpose(velocity_gradients[q]);
1107
1108 // Calculate the shear rate magnitude
1109 shear_rate[q] = calculate_shear_rate_magnitude(shear_rate_tensor);
1110 }
1111
1112 set_field_vector(field::shear_rate, shear_rate, this->fields);
1113 }
1114
1115 switch (properties_manager.get_number_of_fluids())
1116 {
1117 case 1:
1118 {
1119 // In the case of an incompressible flow, viscosity is the only
1120 // required property
1121 const auto rheology_model = properties_manager.get_rheology();
1122 rheology_model->vector_value(fields, kinematic_viscosity);
1123
1124 rheology_model->get_kinematic_viscosity_for_stabilization_vector(
1125 fields, kinematic_viscosity_for_stabilization);
1126
1127 kinematic_viscosity_scale =
1128 rheology_model->get_kinematic_viscosity_scale();
1129
1130 if (!properties_manager.density_is_constant())
1131 {
1132 // For a weakly compressible flow, density variations will play a
1133 // role
1134 const auto density_model = properties_manager.get_density();
1135 density_model->vector_value(fields, density);
1136 density_psi = density_model->get_psi();
1137 density_ref = density_model->get_density_ref();
1138 // Dynamic viscosity is also necessary for compressible flows
1139 rheology_model->get_dynamic_viscosity_vector(density_ref,
1140 fields,
1141 dynamic_viscosity);
1142 rheology_model->get_dynamic_viscosity_for_stabilization_vector(
1143 density_ref, fields, dynamic_viscosity_for_stabilization);
1144 }
1145 else
1146 {
1147 density_scale = properties_manager.get_density_scale();
1148 }
1149
1150 if (properties_manager.is_non_newtonian())
1151 {
1152 // Calculate derivative of viscosity with respect to shear rate
1153 rheology_model->vector_jacobian(
1154 fields, field::shear_rate, grad_kinematic_viscosity_shear_rate);
1155 }
1156
1157 if (gather_temperature)
1158 {
1159 const auto thermal_expansion_model =
1160 properties_manager.get_thermal_expansion();
1161 thermal_expansion_model->vector_value(fields, thermal_expansion);
1162 }
1163 break;
1164 }
1165 case 2:
1166 {
1167 // We need both density and viscosity
1168 const auto density_model_0 = properties_manager.get_density(0);
1169 const auto rheology_model_0 = properties_manager.get_rheology(0);
1170 const auto density_model_1 = properties_manager.get_density(1);
1171 const auto rheology_model_1 = properties_manager.get_rheology(1);
1172
1173 density_ref_0 = density_model_0->get_density_ref();
1174 density_ref_1 = density_model_1->get_density_ref();
1175
1176 kinematic_viscosity_scale_0 =
1177 rheology_model_0->get_kinematic_viscosity_scale();
1178 kinematic_viscosity_scale_1 =
1179 rheology_model_1->get_kinematic_viscosity_scale();
1180
1181 kinematic_viscosity_scale =
1182 std::max(kinematic_viscosity_scale_0, kinematic_viscosity_scale_1);
1183
1184 // Gather properties from material interactions if necessary
1185 if (properties_manager.get_number_of_material_interactions() > 0)
1186 {
1187 const auto material_interaction_id =
1188 properties_manager.get_material_interaction_id(
1189 material_interactions_type::fluid_fluid, 0, 1);
1190 // Gather surface tension
1191 const auto surface_tension_model =
1192 properties_manager.get_surface_tension(material_interaction_id);
1193 surface_tension_model->vector_value(fields, surface_tension);
1194 // Gather surface tension gradient only if necessary
1195 if (!properties_manager.surface_tension_is_constant())
1196 surface_tension_model->vector_jacobian(
1197 fields, field::temperature, surface_tension_gradient);
1198 }
1199
1200 density_model_0->vector_value(fields, density_0);
1201 rheology_model_0->get_dynamic_viscosity_vector(density_ref_0,
1202 fields,
1203 dynamic_viscosity_0);
1204 rheology_model_0->get_dynamic_viscosity_for_stabilization_vector(
1205 density_ref_0, fields, dynamic_viscosity_for_stabilization_0);
1206
1207 density_model_1->vector_value(fields, density_1);
1208 rheology_model_1->get_dynamic_viscosity_vector(density_ref_1,
1209 fields,
1210 dynamic_viscosity_1);
1211 rheology_model_1->get_dynamic_viscosity_for_stabilization_vector(
1212 density_ref_1, fields, dynamic_viscosity_for_stabilization_1);
1213 if (gather_temperature)
1214 {
1215 const auto thermal_expansion_model_0 =
1216 properties_manager.get_thermal_expansion(0);
1217 const auto thermal_expansion_model_1 =
1218 properties_manager.get_thermal_expansion(1);
1219 thermal_expansion_model_0->vector_value(fields,
1220 thermal_expansion_0);
1221 thermal_expansion_model_1->vector_value(fields,
1222 thermal_expansion_1);
1223 }
1224
1225 if (gather_cls && !gather_cahn_hilliard)
1226 {
1227 for (unsigned int q = 0; q < this->n_q_points; ++q)
1228 {
1229 double filtered_phase_value = this->filtered_phase_values[q];
1230
1231 density[q] = calculate_point_property(filtered_phase_value,
1232 this->density_0[q],
1233 this->density_1[q]);
1234
1235 dynamic_viscosity[q] =
1236 calculate_point_property(filtered_phase_value,
1237 this->dynamic_viscosity_0[q],
1238 this->dynamic_viscosity_1[q]);
1239
1240 kinematic_viscosity[q] = dynamic_viscosity[q] / density[q];
1241
1242 dynamic_viscosity_for_stabilization[q] =
1243 calculate_point_property(
1244 filtered_phase_value,
1245 this->dynamic_viscosity_for_stabilization_0[q],
1246 this->dynamic_viscosity_for_stabilization_1[q]);
1247
1248 thermal_expansion[q] =
1249 calculate_point_property(filtered_phase_value,
1250 this->thermal_expansion_0[q],
1251 this->thermal_expansion_1[q]);
1252
1253 // Phase interpolate the product rho*beta per-phase to avoid
1254 // cross term error from bilinear phase interpolation:
1255 // interp_lin(rho)*interp_lin(beta) != interp_lin(rho*beta)
1256 density_thermal_expansion[q] = calculate_point_property(
1257 filtered_phase_value,
1258 this->density_0[q] * this->thermal_expansion_0[q],
1259 this->density_1[q] * this->thermal_expansion_1[q]);
1260 }
1261
1262 // Gather density_psi for isothermal compressible NS equations
1263 if (!properties_manager.density_is_constant())
1264 {
1265 density_psi_0 = density_model_0->get_psi();
1266 density_psi_1 = density_model_1->get_psi();
1267
1268 for (unsigned int q = 0; q < this->n_q_points; ++q)
1269 {
1270 double filtered_phase_value =
1271 this->filtered_phase_values[q];
1272 // To avoid calculating twice in the assemblers
1273 compressibility_multiplier[q] =
1274 filtered_phase_value *
1275 (density_0[q] / density_1[q] * density_psi_1 -
1276 density_psi_0) +
1277 density_psi_0;
1278 }
1279 }
1280
1281 for (unsigned p = 0; p < previous_phase_values.size(); ++p)
1282 {
1283 // Blend the physical properties using the CLS field
1284 for (unsigned int q = 0; q < this->n_q_points; ++q)
1285 {
1286 // Calculate previous density (right now assumes constant
1287 // density model per phase)
1288 previous_density[p][q] = calculate_point_property(
1289 filter->filter_phase(this->previous_phase_values[p][q]),
1290 this->density_0[q],
1291 this->density_1[q]);
1292 }
1293 }
1294 break;
1295 }
1296 else if (gather_cahn_hilliard && !gather_cls)
1297 {
1298 // Blend the physical properties using the CahnHilliard field
1299 for (unsigned int q = 0; q < this->n_q_points; ++q)
1300 {
1301 double phase_order_cahn_hilliard_value =
1302 this->filtered_phase_order_cahn_hilliard_values[q];
1303
1304 density[q] = calculate_point_property_cahn_hilliard(
1305 phase_order_cahn_hilliard_value,
1306 this->density_0[q],
1307 this->density_1[q]);
1308
1309 dynamic_viscosity[q] = calculate_point_property_cahn_hilliard(
1310 phase_order_cahn_hilliard_value,
1311 this->dynamic_viscosity_0[q],
1312 this->dynamic_viscosity_1[q]);
1313
1314 dynamic_viscosity_for_stabilization[q] =
1315 calculate_point_property_cahn_hilliard(
1316 phase_order_cahn_hilliard_value,
1317 this->dynamic_viscosity_for_stabilization_0[q],
1318 this->dynamic_viscosity_for_stabilization_1[q]);
1319 }
1320 break;
1321 }
1322 break;
1323 }
1324 default:
1325 throw std::runtime_error("Unsupported number of fluids (>2)");
1326 }
1327}

1406{

1422 particle_properties[PropertiesIndex::v_z];

1425 total_particle_volume +=
1426 M_PI * pow(particle_properties[PropertiesIndex::dp], dim) / (2 * dim);
1427
1428 average_particle_velocity += particle_velocity[i_particle];
1429 i_particle++;
1430 }
1431 number_of_particles = i_particle;
1432 if (number_of_particles != 0)
1433 { // Calculate the average particle velocity within the cell

1435 average_particle_velocity / number_of_particles;

1445 this->fe_values_void_fraction->get_JxW_values());

1447 // The cell_void_fraction vector needs to be resized to the correct size
1448 // since the quadrature used in the FEValues for the void fraction will

1458 (cell_volume - total_particle_volume) / cell_volume;

1461 cell_void_fraction[j] = cell_void_fraction_bulk;

1582template <int dim>
1583void
1584NavierStokesScratchData<dim>::calculate_fluid_properties_at_particle_location()
1585{
1586 AssertThrow(
1587 properties_manager.density_is_constant(),
1588 ExcMessage(
1589 "The lethe-fluid-particles solver only supports constant densities"));
1590 AssertThrow(
1591 !properties_manager.get_rheology()->is_non_newtonian_rheological_model() &&
1592 !properties_manager.get_rheology()->is_non_newtonian_rheological_model(),
1593 ExcMessage(
1594 "The lethe-fluid-particles solver only supports constant rheology model"));
1595
1596 if (gather_cls)
1597 {
1598 for (unsigned int i_particle = 0; i_particle < number_of_particles;
1599 ++i_particle)
1600 {
1601 density_at_particle_location[i_particle] = calculate_point_property(
1602 filtered_phase_values_at_particle_location[i_particle],
1603 this->density_ref_0,
1604 this->density_ref_1);
1605 double dynamic_viscosity_at_particle_location =
1606 calculate_point_property(
1607 filtered_phase_values_at_particle_location[i_particle],
1608 this->kinematic_viscosity_scale_0 * this->density_ref_0,
1609 this->kinematic_viscosity_scale_1 * this->density_ref_1);
1610
1611 kinematic_viscosity_at_particle_location[i_particle] =
1612 dynamic_viscosity_at_particle_location /
1613 density_at_particle_location[i_particle];
1614 }
1615
1616 // Properties have been calculated, return
1617 return;
1618 }
1619
1620 // Regular case without CLS
1621 for (unsigned int i_particle = 0; i_particle < number_of_particles;
1622 ++i_particle)
1623 {
1624 // Gather the kinematic viscosity and density at the particle
1625 // location assuming a constant kinematic viscosity and density in a
1626 // single fluid
1627 kinematic_viscosity_at_particle_location[i_particle] =
1628 kinematic_viscosity_scale;
1629 density_at_particle_location[i_particle] = density_scale;
1630 }

1633template <int dim>
1634template <typename PropertiesIndex>
1635void
1636NavierStokesScratchData<dim>::calculate_force_parameters_at_particle_location()
1637{
1638 // Tolerance for the calculation of the particle Reynolds number.
1639 // TODO -> Revisit if that is not a too high value.
1640 const double Re_particle_tolerance = 1e-3;
1641 unsigned int i_particle = 0;
1642 average_fluid_particle_relative_velocity = 0;
1643
1644 for (auto &particle : pic)
1645 {
1646 auto particle_properties = particle.get_properties();
1647
1648 fluid_particle_relative_velocity_at_particle_location[i_particle] =
1649 fluid_velocity_at_particle_location[i_particle] -
1650 particle_velocity[i_particle];
1651 average_fluid_particle_relative_velocity +=
1652 fluid_particle_relative_velocity_at_particle_location[i_particle];
1653
1654 Re_particle[i_particle] =
1655 Re_particle_tolerance +
1656 cell_void_fraction[i_particle] *
1657 fluid_particle_relative_velocity_at_particle_location[i_particle]
1658 .norm() *
1659 particle_properties[PropertiesIndex::dp] /
1660 (kinematic_viscosity_at_particle_location[i_particle] + DBL_MIN);
1661 i_particle++;
1662 }
1663
1664 average_fluid_particle_relative_velocity =
1665 average_fluid_particle_relative_velocity / i_particle;
1666}
