4/**
5 * @file qmom_scratch_data.h
6 * @brief Scratch data used to assemble the discontinuous Galerkin transport of
7 * the QMOM components.
8 *
9 * All the QMOM components (scalars and moments) are discretized with the same
10 * scalar discontinuous element and advected by the same fluid velocity. The
11 * matrix of the transport is therefore the same for every component, while
12 * each component has its own right-hand side.
13 */

42/**
43 * @brief Scratch data of the transport of the QMOM components.
44 *
45 * The transport is linear and is solved directly for the solution at the end
46 * of the time step (linear solver strategy). Unlike the tracer physics, which
47 * assembles the residual of a Newton iteration, the assembly therefore never
48 * needs the present solution: the matrix only needs the shape functions and
49 * the fluid velocity, and the right-hand side the previous solutions of the
50 * BDF scheme, the source term and the Dirichlet values.
51 *
52 * @tparam dim Number of spatial dimensions.
53 */
54template <int dim>
55class QMOMScratchData : public PhysicsScratchDataBase
56{
57public:
58 /**
59 * @brief Constructor.
60 *
61 * @param[in] fe_qmom Scalar discontinuous element shared by all the QMOM
62 * components.
63 * @param[in] quadrature Cell quadrature.
64 * @param[in] face_quadrature Face quadrature.
65 * @param[in] mapping Mapping of the QMOM physics, also used to evaluate the
66 * fluid velocity at the same quadrature points.
67 * @param[in] fe_fluid FESystem of the fluid dynamics (velocity and
68 * pressure).
69 * @param[in] n_components Number of QMOM components (n_scalars + n_moments).
70 */
71 QMOMScratchData(const FiniteElement<dim> &fe_qmom,
72 const Quadrature<dim> &quadrature,
73 const Quadrature<dim - 1> &face_quadrature,
74 const Mapping<dim> &mapping,
75 const FiniteElement<dim> &fe_fluid,
76 const unsigned int n_components);
77
78 /**
79 * @brief Copy constructor, required by MeshWorker::mesh_loop. The FEValues
80 * objects cannot be copied and are rebuilt.

82 * @param[in] sd Scratch data to copy.

84 QMOMScratchData(const QMOMScratchData<dim> &sd);

109 void
110 reinit_previous_solutions(

113 /**
114 * @brief Evaluate the source term of every component at the quadrature

123 /**
124 * @brief Gather the fluid velocity at the quadrature points of a cell.
125 *
126 * @tparam VectorType Vector type of the fluid dynamics solution.
127 * @param[in] fluid_cell Cell of the fluid dynamics DoFHandler.
128 * @param[in] fluid_solution Present (ghosted) fluid dynamics solution. The

132 template <typename VectorType>
133 void
134 reinit_velocity(
135 const typename DoFHandler<dim>::active_cell_iterator &fluid_cell,
136 const VectorType &fluid_solution);
137
138 /**
139 * @brief Evaluate the shape functions of the two cells that share an
140 * interior face (the "here" and "there" sides), and their jump, at the face
141 * quadrature points.
142 *
143 * @param[in] cell Cell on the "here" side of the face.
144 * @param[in] face_no Face number in cell.
145 * @param[in] sub_face_no Subface number in cell, or

147 * @param[in] neighbor_cell Cell on the "there" side of the face.
148 * @param[in] neighbor_face_no Face number in neighbor_cell.
149 * @param[in] neighbor_sub_face_no Subface number in neighbor_cell.
150 */
151 void
152 reinit_internal_face(
153 const typename DoFHandler<dim>::active_cell_iterator &cell,
154 const unsigned int face_no,
155 const unsigned int sub_face_no,
156 const typename DoFHandler<dim>::active_cell_iterator &neighbor_cell,
157 const unsigned int neighbor_face_no,
158 const unsigned int neighbor_sub_face_no);
159
160 /**
161 * @brief Reinitialize the interface values on a boundary face.
162 *
163 * @param[in] cell Cell of the QMOM DoFHandler.
164 * @param[in] face_no Face number in cell.
165 */
166 void
167 reinit_boundary_face(
168 const typename DoFHandler<dim>::active_cell_iterator &cell,
169 const unsigned int face_no);

171 /**
172 * @brief Gather the fluid velocity at the quadrature points of a face, as
173 * seen from the "here" side of the face. These quadrature points are the
174 * ones of the last call to reinit_internal_face() or
175 * reinit_boundary_face().
176 *
177 * @tparam VectorType Vector type of the fluid dynamics solution.
178 * @param[in] fluid_cell Cell of the fluid dynamics DoFHandler on the "here"
179 * side of the face.
180 * @param[in] face_no Face number in fluid_cell.
181 * @param[in] sub_face_no Subface number in fluid_cell, which must be
182 * numbers::invalid_unsigned_int: with assemble_own_interior_faces_once,
183 * MeshWorker::mesh_loop() calls the face worker from the fine side of a
184 * refined face, where it is a whole face.
185 * @param[in] fluid_solution Present (ghosted) fluid dynamics solution.
186 */
187 template <typename VectorType>
188 void
189 reinit_face_velocity(
190 const typename DoFHandler<dim>::active_cell_iterator &fluid_cell,
191 const unsigned int face_no,
192 const unsigned int sub_face_no,
193 const VectorType &fluid_solution);

255
256 /// Jump of the shape functions, here minus there, at the face quadrature
257 /// points: face_jump_phi[q][i], with i an interface degree of freedom
258 Table<2, double> face_jump_phi;
259
260 /// Fluid velocity at the face quadrature points
261 std::vector<Tensor<1, dim>> face_velocity_values;
262};
