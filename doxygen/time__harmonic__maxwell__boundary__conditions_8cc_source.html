134 TimeHarmonicMaxwellBoundaryConditions<dim>::declare_parameters(

136 const unsigned int number_of_boundary_conditions)
137 {

139 {

141 "0",
142 Patterns::Integer(),
143 "Number of boundary conditions");
144 prm.declare_entry(
145 "time dependent",
146 "false",
147 Patterns::Bool(),
148 "Bool to define if the boundary condition is time-dependent");
149
150 for (unsigned int n = 0; n < number_of_boundary_conditions; n++)
151 {
152 prm.enter_subsection("bc " + std::to_string(n));
153 {
154 declare_default_entry(prm, n);
155 }
156 prm.leave_subsection();
157 }
158 }

162 template <int dim>
163 void
164 TimeHarmonicMaxwellBoundaryConditions<dim>::parse_boundary(
165 ParameterHandler &prm)
166 {
167 // Parse the list of boundary ids
168 std::vector<types::boundary_id> boundary_ids =
169 convert_string_to_vector<types::boundary_id>(prm, "id");
170
171 // Check if the list contains at least one boundary id
172 AssertThrow(
173 boundary_ids.size() > 0,
174 ExcMessage(
175 "A boundary id has not been set for one of the time harmonic maxwell boundary conditions. Please ensure that the id is set for every boundary condition."));
176
177 // Loop through all boundary ids to ensure that they are all non-negative
178 // and unique
179 for (const auto boundary_id : boundary_ids)
180 {
181 AssertThrow(this->type.find(boundary_id) == this->type.end(),
182 TimeHarmonicMaxwellDuplicated(boundary_id));
183
184 // Allocate the imposed electromagnetic field functions object for every
185 // boundary condition
186 imposed_electromagnetic_fields[boundary_id] =
187 std::make_shared<TimeHarmonicMaxwellBoundaryFunctions<dim>>();
188
189 prm.enter_subsection("E x real part");
190 imposed_electromagnetic_fields[boundary_id]->e_x_real.parse_parameters(
191 prm);
192 prm.leave_subsection();
193
194 prm.enter_subsection("E y real part");
195 imposed_electromagnetic_fields[boundary_id]->e_y_real.parse_parameters(
196 prm);
197 prm.leave_subsection();
198
199 prm.enter_subsection("E z real part");
200 imposed_electromagnetic_fields[boundary_id]->e_z_real.parse_parameters(
201 prm);
202 prm.leave_subsection();
203
204 prm.enter_subsection("E x imag part");
205 imposed_electromagnetic_fields[boundary_id]->e_x_imag.parse_parameters(
206 prm);
207 prm.leave_subsection();
208
209 prm.enter_subsection("E y imag part");
210 imposed_electromagnetic_fields[boundary_id]->e_y_imag.parse_parameters(
211 prm);
212 prm.leave_subsection();
213
214 prm.enter_subsection("E z imag part");
215 imposed_electromagnetic_fields[boundary_id]->e_z_imag.parse_parameters(
216 prm);
217 prm.leave_subsection();
218
219 prm.enter_subsection("H x real part");
220 imposed_electromagnetic_fields[boundary_id]->h_x_real.parse_parameters(
221 prm);
222 prm.leave_subsection();
223
224 prm.enter_subsection("H y real part");
225 imposed_electromagnetic_fields[boundary_id]->h_y_real.parse_parameters(
226 prm);
227 prm.leave_subsection();
228
229 prm.enter_subsection("H z real part");
230 imposed_electromagnetic_fields[boundary_id]->h_z_real.parse_parameters(
231 prm);
232 prm.leave_subsection();
233
234 prm.enter_subsection("H x imag part");
235 imposed_electromagnetic_fields[boundary_id]->h_x_imag.parse_parameters(
236 prm);
237 prm.leave_subsection();
238
239 prm.enter_subsection("H y imag part");
240 imposed_electromagnetic_fields[boundary_id]->h_y_imag.parse_parameters(
241 prm);
242 prm.leave_subsection();
243
244 prm.enter_subsection("H z imag part");
245 imposed_electromagnetic_fields[boundary_id]->h_z_imag.parse_parameters(
246 prm);
247 prm.leave_subsection();
248
249 /// The following functions are parsed individually because they will be
250 /// used for Robin boundary conditions and will not be applied on the
251 /// global solution vector.
252 prm.enter_subsection("surface admittance real part");
253 this->surface_admittance_real[boundary_id] =
254 std::make_shared<Functions::ParsedFunction<dim>>();
255 this->surface_admittance_real[boundary_id]->parse_parameters(prm);
256 prm.leave_subsection();
257
258 prm.enter_subsection("surface admittance imag part");
259 this->surface_admittance_imag[boundary_id] =
260 std::make_shared<Functions::ParsedFunction<dim>>();
261 this->surface_admittance_imag[boundary_id]->parse_parameters(prm);
262 prm.leave_subsection();
263
264 prm.enter_subsection("excitation x real part");
265 this->excitation_x_real[boundary_id] =
266 std::make_shared<Functions::ParsedFunction<dim>>();
267 this->excitation_x_real[boundary_id]->parse_parameters(prm);
268 prm.leave_subsection();
269
270 prm.enter_subsection("excitation y real part");
271 this->excitation_y_real[boundary_id] =
272 std::make_shared<Functions::ParsedFunction<dim>>();
273 this->excitation_y_real[boundary_id]->parse_parameters(prm);
274 prm.leave_subsection();
275
276 prm.enter_subsection("excitation z real part");
277 this->excitation_z_real[boundary_id] =
278 std::make_shared<Functions::ParsedFunction<dim>>();
279 this->excitation_z_real[boundary_id]->parse_parameters(prm);
280 prm.leave_subsection();
281
282 prm.enter_subsection("excitation x imag part");
283 this->excitation_x_imag[boundary_id] =
284 std::make_shared<Functions::ParsedFunction<dim>>();
285 this->excitation_x_imag[boundary_id]->parse_parameters(prm);
286 prm.leave_subsection();

308 template <int dim>
309 void
310 TimeHarmonicMaxwellBoundaryConditions<dim>::parse_parameters(
311 ParameterHandler &prm)
312 {
313 prm.enter_subsection("boundary conditions time harmonic maxwell");
314 {
315 this->number_of_boundary_conditions = prm.get_integer("number");
316 this->time_dependent = prm.get_bool("time dependent");
317
318 for (unsigned int n = 0; n < this->number_of_boundary_conditions; n++)
319 {
320 prm.enter_subsection("bc " + std::to_string(n));
321 {
322 parse_boundary(prm);
323 }
324 prm.leave_subsection();
325 }
326 }
327 prm.leave_subsection();
328 }

335template <int dim>
336double
337TimeHarmonicMaxwellElectricFieldDefined<dim>::value(
338 const Point<dim> &point,
339 const unsigned int component) const
340{
341 Assert(component < this->n_components,
342 ExcIndexRange(component, 0, this->n_components));
343
344 if (component == 0)
345 {
346 return e_x_real->value(point);
347 }
348 if (component == 1)
349 {
350 return e_y_real->value(point);
351 }
352 if (component == 2)
353 {
354 return e_z_real->value(point);
355 }
356 if (component == 3)
357 {
358 return e_x_imag->value(point);
359 }
360 if (component == 4)
361 {
362 return e_y_imag->value(point);
363 }
364 if (component == 5)
365 {
366 return e_z_imag->value(point);
367 }
368 return 0.;
369}
