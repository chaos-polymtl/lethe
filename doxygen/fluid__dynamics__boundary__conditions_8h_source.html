64 /**
65 * @brief This class manages the functions associated with function-type boundary conditions
66 * of the Navier-Stokes equations
67 */
68 template <int dim>
69 class NavierStokesBoundaryFunctions
70 {
71 public:
72 // Velocity components
73 Functions::ParsedFunction<dim> u;
74 Functions::ParsedFunction<dim> v;
75 Functions::ParsedFunction<dim> w;
76
77 /// Pressure
78 Functions::ParsedFunction<dim> p;
79
80 // Neumann Traction σ = 2με(u)− pI
81 // σ.n = g
82 Functions::ParsedFunction<dim> traction_fn;
83
84 /// Point for the center of rotation
85 Point<dim> center_of_rotation;
86
87 /*
88 Constructor required to avoid vector-valued function t of ParsedFunction
89 type which in-turn initializes the function_parser object with
90 n_components = 1 by default if n_components is not passed.
91
92 Moreover, the test and trial function is dim+1 components including
93 pressure. Hence, we have the traction function as a vector-valued function
94 with dim components for the traction vector and 1 component for the
95 pressure value at the boundary which is dummy and is needed to properly
96 assemble the traction term as FEFaceIntegrator uses MatrixFree which in
97 turn is based on the DoFHandler which in turn is based on the FE system
98 that has dim velocity components and 1 pressure component.
99 t = (t₁, t₂, …, t_dim, 0.0 (dummy pressure component))
100 φ = (φ₁, φ₂, …, φ_dim, φ_p)
101 ∫_{Γ_t} = t · φ ds
102 */
103
104 NavierStokesBoundaryFunctions()
105 : traction_fn(dim + 1) // The traction function has dim components for the
106 // traction vector plus 1 for the pressure.
107 {}
108 };

198template <int dim>
199class FluidDynamicsVelocityFunctionDefined : public Function<dim>
200{
201 Functions::ParsedFunction<dim> *u;
202 Functions::ParsedFunction<dim> *v;
203 Functions::ParsedFunction<dim> *w;
204
205public:
206 /**
207 * @brief Initialize the fluid dynamics boundary function with its velocity
208 * components.
209 *
210 * @param[in] p_u The parsed function defining the x-velocity component.
211 * @param[in] p_v The parsed function defining the y-velocity component.
212 * @param[in] p_w The parsed function defining the z-velocity component.
213 */
214 FluidDynamicsVelocityFunctionDefined(Functions::ParsedFunction<dim> *p_u,
215 Functions::ParsedFunction<dim> *p_v,
216 Functions::ParsedFunction<dim> *p_w)
217 : Function<dim>(dim + 1)
218 , u(p_u)
219 , v(p_v)
220 , w(p_w)
221 {}
222
223 /**
224 * @brief Calculates the value of a Function-type fluid dynamics equations
225 *
226 * @param[in] point A point at which the function will be evaluated
227 * @param[in] component The vector component of the boundary condition (0-x,
228 * 1-y and 2-z)
229 */
230 double
231 value(const Point<dim> &point, const unsigned int component) const override;
232};
