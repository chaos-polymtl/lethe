35 /**
36 * @brief This class manages the functions associated with function-type boundary conditions
37 * of the Navier-Stokes equations
38 */
39 template <int dim>
40 class NavierStokesBoundaryFunctions
41 {
42 public:
43 // Velocity components
44 Functions::ParsedFunction<dim> u;
45 Functions::ParsedFunction<dim> v;
46 Functions::ParsedFunction<dim> w;
47
48 /// Pressure
49 Functions::ParsedFunction<dim> p;
50
51 // Neumann Traction σ = 2με(u)− pI
52 // σ.n = g
53 Functions::ParsedFunction<dim> traction_fn;
54
55 /// Point for the center of rotation
56 Point<dim> center_of_rotation;
57
58 /*
59 Constructor required to avoid vector-valued function t of ParsedFunction
60 type which in-turn initializes the function_parser object with
61 n_components = 1 by default if n_components is not passed.
62
63 Moreover, the test and trial function is dim+1 components including
64 pressure. Hence, we have the traction function as a vector-valued function
65 with dim components for the traction vector and 1 component for the
66 pressure value at the boundary which is dummy and is needed to properly
67 assemble the traction term as FEFaceIntegrator uses MatrixFree which in
68 turn is based on the DoFHandler which in turn is based on the FE system
69 that has dim velocity components and 1 pressure component.
70 t = (t₁, t₂, …, t_dim, 0.0 (dummy pressure component))
71 φ = (φ₁, φ₂, …, φ_dim, φ_p)
72 ∫_{Γ_t} = t · φ ds
73 */
74
75 NavierStokesBoundaryFunctions()
76 : traction_fn(dim + 1) // The traction function has dim components for the
77 // traction vector plus 1 for the pressure.
78 {}
79 };

156/**
157 * @brief This class implements a boundary conditions for the fluid dynamics solver
158 * where the velocity component are defined using individual functions
159 */
160template <int dim>
161class FluidDynamicsVelocityFunctionDefined : public Function<dim>
162{
163 Functions::ParsedFunction<dim> *u;
164 Functions::ParsedFunction<dim> *v;
165 Functions::ParsedFunction<dim> *w;
166
167public:
168 /**
169 * @brief Initialize the fluid dynamics boundary function with its velocity
170 * components.
171 *
172 * @param[in] p_u The parsed function defining the x-velocity component.
173 * @param[in] p_v The parsed function defining the y-velocity component.
174 * @param[in] p_w The parsed function defining the z-velocity component.
175 */
176 FluidDynamicsVelocityFunctionDefined(Functions::ParsedFunction<dim> *p_u,
177 Functions::ParsedFunction<dim> *p_v,
178 Functions::ParsedFunction<dim> *p_w)
179 : Function<dim>(dim + 1)
180 , u(p_u)
181 , v(p_v)
182 , w(p_w)
183 {}
184
185 /**
186 * @brief Calculates the value of a Function-type fluid dynamics equations
187 *
188 * @param[in] point A point at which the function will be evaluated
189 * @param[in] component The vector component of the boundary condition (0-x,
190 * 1-y and 2-z)
191 */
192 double
193 value(const Point<dim> &point, const unsigned int component) const override;
194};
