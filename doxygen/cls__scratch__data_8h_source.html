17/**
18 * @brief Store the information required by the assembly procedure
19 * for a CLS free surface equation. Consequently, this class
20 * calculates the phase values (values, gradients, laplacians) and the shape
21 * method (values, gradients, laplacians) at all the gauss points for all
22 * degrees of freedom and stores it into arrays.
23 * This class serves as a separation between the evaluation at the gauss point
24 * of the variables of interest and their use in the assembly, which is carried
25 * out by the assembler methods.
26 *
27 * @tparam dim An integer that denotes the dimension of the space in which
28 * the flow is solved
29 *
30 * @ingroup solvers
31 **/
32
33template <int dim>
34class CLSScratchData : public PhysicsScratchDataBase
35{
36public:
37 /**
38 * @brief Constructor. The constructor creates the fe_values that will be used
39 * to fill the member variables of the scratch. It also allocated the
40 * necessary memory for all member variables. However, it does not do any
41 * evaluation, since this needs to be done at the cell level.
42 *
43 * @param simulation_control The SimulationControl object that holds
44 * information related to the control of the steady-state or transient
45 * simulation. This is used to extrapolate velocity solutions in time
46 * for transient simulation.
47 *
48 * @param properties_manager Manager to calculate the physical properties.
49 *
50 * @param fe_cls The FESystem used to solve the CLS equations.
51 *
52 * @param quadrature The quadrature to use for the assembly within the cells.
53 *
54 * @param face_quadrature The quadrature to use for the assembly on faces.
55 *
56 * @param mapping The mapping of the domain in which the Navier-Stokes
57 * equations are solved.
58 *
59 * @param fe_fd The FESystem used to solve the Fluid Dynamics equations.
60 *
61 */
62 CLSScratchData(const std::shared_ptr<SimulationControl> &simulation_control,
63 const PhysicalPropertiesManager &properties_manager,
64 const FiniteElement<dim> &fe_cls,
65 const Quadrature<dim> &quadrature,
66 const Quadrature<dim - 1> &face_quadrature,
67 const Mapping<dim> &mapping,
68 const FiniteElement<dim> &fe_fd);
69
70 /**
71 * @brief Copy Constructor. Same as the main constructor.
72 * This constructor only uses the other scratch to build the FeValues, it
73 * does not copy the content of the other scratch into itself since, by
74 * definition of the WorkStream mechanism it is assumed that the content of
75 * the scratch will be reset on a cell basis.
76 *
77 * @param sd The scratch data to be copied
78 */
79 CLSScratchData(const CLSScratchData<dim> &sd);

89 /** @brief Reinitialize the content of the scratch
90 *
91 * Using the FeValues and the content of the solutions and previous solutions,
92 * fills all of the class member of the scratch
93 *
94 * @tparam VectorType The Vector type used for the solvers
95 *
96 * @param[in] cell The cell over which the assembly is being carried.
97 * This cell must be compatible with the FE which is used to fill the
98 * FeValues.
99 *
100 * @param[in] current_solution The present value of the solution for the CLS.
101 *
102 * @param[in] previous_solutions The solutions at the previous time steps.
103 *
104 */
105
106 template <typename VectorType>
107 void
108 reinit(const typename DoFHandler<dim>::active_cell_iterator &cell,
109 const VectorType &current_solution,
110 const std::vector<VectorType> &previous_solutions);
