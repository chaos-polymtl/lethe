106/**
107 * @brief Polynomial electric conductivity.
108 */
109class PolynomialElectricConductivity : public ElectricConductivityModel
110{
111public:
112 /**
113 * @brief Default constructor
114 */
115 PolynomialElectricConductivity(
116 const std::vector<double> &p_electric_conductivity_polynomial_coefficients)
117 : electric_conductivity_polynomial_coefficients(
118 p_electric_conductivity_polynomial_coefficients)
119 {
120 this->model_depends_on[field::temperature] = true;
121 }
122
123 /**
124 * @brief Calculates the value of the electric conductivity
125 * @param[in] field_values Values of the various fields on which the electric
126 * conductivity depends.
127 * @return Value of the electric conductivity calculated with the `field_values`.
128 */
129 double
130 value(const std::map<field, double> &field_values) override
131 {
132 AssertThrow(field_values.contains(field::temperature),
133 PhysicialPropertyModelFieldUndefined(
134 "PolynomialElectricConductivity", "temperature"));
135 const double temperature = field_values.at(field::temperature);
136
137 // Compute the polynomial power using Horner's method for efficiency , i.e.
138 // P(x) = (...((a_n * x + a_{n-1}) * x + a_{n-2}) * x + ... + a_1) * x + a_0
139 double electric_conductivity = 0.0;
140 for (const auto &coefficient :
141 electric_conductivity_polynomial_coefficients)
142 {
143 electric_conductivity =
144 coefficient + temperature * electric_conductivity;
145 }
146
147 return electric_conductivity;
148 }
149
150 /**
151 * @brief Calculates, in a vector, values of of electric conductivities
152 * @param[in] field_vectors Vector of fields on which the electric
153 * conductivities depend
154 * @param[in,out] property_vector Values of the electric conductivities
155 */
156 void
157 vector_value(const std::map<field, std::vector<double>> &field_vectors,
158 std::vector<double> &property_vector) override
159 {
160 AssertThrow(field_vectors.find(field::temperature) != field_vectors.end(),
161 PhysicialPropertyModelFieldUndefined(
162 "PolynomialElectricConductivity", "temperature"));
163
164 const std::vector<double> &temperature =
165 field_vectors.at(field::temperature);
166 for (unsigned int i = 0; i < property_vector.size(); ++i)
167 {
168 // Compute the polynomial power using Horner's method for efficiency
169 // , i.e. P(x) = (...((a_n * x + a_{n-1}) * x + a_{n-2}) * x + ... +
170 // a_1) * x + a_0
171 double electric_conductivity = 0.0;
172 for (const auto &coefficient :
173 electric_conductivity_polynomial_coefficients)
174 {
175 electric_conductivity =
176 coefficient + temperature[i] * electric_conductivity;
177 }
178 property_vector[i] = electric_conductivity;
179 }
180 }
181
182 /**
183 * @brief Calculates the jacobian (the partial derivative) of the electric conductivity with respect to a field
184 * @param[in] field_values Value of the various fields on which the property
185 * may depend.
186 * @param[in] id Identifier of the field with respect to which the jacobian
187 * should be calculated
188 * @return Value of the partial derivative of the electric conductivity with respect to the field.
189 *
190 * @remark The jacobian is only implemented with respect to the temperature
191 * field.
192 */
193 double
194 jacobian(const std::map<field, double> &field_values, const field id) override
195 {
196 // The derivative can only be taken with respect to the temperature field,
197 // since the polynomial is a function of temperature.
198 if (id == field::temperature)
199 {
200 AssertThrow(field_values.find(field::temperature) != field_values.end(),
201 PhysicialPropertyModelFieldUndefined(
202 "PolynomialElectricConductivity", "temperature"));
203
204 // Use a Horner accumulator to evaluate the derivative directly.
205 unsigned int polynomial_order =
206 electric_conductivity_polynomial_coefficients.size() - 1;
207 double electric_conductivity_derivative =
208 polynomial_order * electric_conductivity_polynomial_coefficients[0];
209
210 double temperature = field_values.at(field::temperature);
211
212 for (unsigned int i = 1; i < polynomial_order; ++i)
213 {
214 electric_conductivity_derivative =
215 electric_conductivity_derivative * temperature +
216 (polynomial_order - i) *
217 electric_conductivity_polynomial_coefficients[i];
218 }
219
220 return electric_conductivity_derivative;
221 }
222 else
223 return 0;
224 }
225
226 /**
227 * @brief Calculates the derivatives of the electric conductivity with respect to a field
228 * @param[in] field_vectors Vector for the values of the fields used to
229 * evaluate the property
230 * @param[in] id Identifier of the field with respect to which a derivative
231 * should be calculated
232 * @param[in,out] jacobian_vector Vector of the values of the derivatives of
233 * the electric conductivity with respect to the specified field
234 *
235 * @remark The vector_jacobian is only implemented with respect to the
236 * temperature field.
237 */
238 void
239 vector_jacobian(const std::map<field, std::vector<double>> &field_vectors,
240 const field id,
241 std::vector<double> &jacobian_vector) override
242 {
243 if (id == field::temperature)
244 {
245 AssertThrow(
246 field_vectors.find(field::temperature) != field_vectors.end(),
247 PhysicialPropertyModelFieldUndefined("PolynomialElectricConductivity",
248 "temperature"));
249
250 const std::vector<double> &temperature =
251 field_vectors.at(field::temperature);
252 for (unsigned int i = 0; i < jacobian_vector.size(); ++i)
253 {
254 // Use a Horner accumulator to evaluate the derivative directly.
255 unsigned int polynomial_order =
256 electric_conductivity_polynomial_coefficients.size() - 1;
257 double electric_conductivity_derivative =
258 polynomial_order *
259 electric_conductivity_polynomial_coefficients[0];
260 for (unsigned int j = 1; j < polynomial_order; ++j)
261 {
262 electric_conductivity_derivative =
263 electric_conductivity_derivative * temperature[i] +
264 (polynomial_order - j) *
265 electric_conductivity_polynomial_coefficients[j];
266 }
267
268 jacobian_vector[i] = electric_conductivity_derivative;
269 }
270 }
271 else
272 {
273 std::fill(jacobian_vector.begin(), jacobian_vector.end(), 0);
274 }
275 }
276
277private:
278 const std::vector<double> electric_conductivity_polynomial_coefficients;
279};
