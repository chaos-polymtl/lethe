24 void
25 CahnHilliardBoundaryConditions<dim>::declare_default_entry(
26 ParameterHandler &prm,
27 const types::boundary_id default_boundary_id)
28 {
29 prm.declare_entry(
30 "type",
31 "none",
32 Patterns::Selection(
33 "none|noflux|dirichlet|angle_of_contact|free_angle|periodic"),
34 "Type of boundary condition for the Cahn-Hilliard equations"
35 "Choices are <none|noflux|dirichlet|angle_of_contact|free_angle|periodic>.");
36
37 prm.declare_entry("id",
38 Utilities::int_to_string(default_boundary_id, 2),
39 Patterns::List(Patterns::Integer()),
40 "Mesh id for boundary conditions");
41
42 prm.declare_entry(
43 "angle value",
44 "0",
45 Patterns::Double(),
46 "Inner angle of contact between the fluid 1 and the boundary (in degrees)");
47
48 Functions::ParsedFunction<dim> temporary_function;
49 prm.enter_subsection("phi");
50 temporary_function.declare_parameters(prm);
51 prm.leave_subsection();
52
53 // Periodic boundary condition parameters for Cahn-Hilliards physics
54 prm.declare_entry(
55 "periodic id",
56 "-1",
57 Patterns::Integer(),
58 "Mesh id for periodic face matching. Default entry is -1 to ensure that the periodic id is set by the user");
59
60 prm.declare_entry("periodic direction",
61 "0",
62 Patterns::Integer(),
63 "Direction for periodic boundary condition");
64 }

96 template <int dim>
97 void
98 CahnHilliardBoundaryConditions<dim>::parse_boundary(ParameterHandler &prm)
99 {
100 // Parse the list of boundary ids
101 std::vector<types::boundary_id> boundary_ids =
102 convert_string_to_vector<types::boundary_id>(prm, "id");
103
104 // Check if the list contains at least one boundary id
105 AssertThrow(
106 boundary_ids.size() > 0,
107 ExcMessage(
108 "A boundary id has not been set for one of the Cahn Hilliard boundary conditions. Please ensure that the id is set for every boundary condition."));
109 // Loop through all boundary ids to ensure that they are all non-negative
110 // and unique
111 for (const auto boundary_id : boundary_ids)
112 {
113 AssertThrow(this->type.find(boundary_id) == this->type.end(),
114 CahnHilliardBoundaryDuplicated(boundary_id));
115
116 // Create and parse the phase order boundary condition for all cases.
117 prm.enter_subsection("phi");
118 this->bcFunctions[boundary_id] =
119 std::make_shared<CahnHilliardBoundaryFunctions<dim>>();
120 bcFunctions[boundary_id]->phi.parse_parameters(prm);
121 prm.leave_subsection();
122
123 // Do the same with the angle of contact
124 this->angle_of_contact[boundary_id] = prm.get_double("angle value");
125
126 const std::string op = prm.get("type");
127 if (op == "none")
128 {
129 this->type[boundary_id] = BoundaryType::none;
130 }
131 else if (op == "noflux")
132 {
133 this->type[boundary_id] = BoundaryType::cahn_hilliard_noflux;
134 }
135 else if (op == "dirichlet")
136 {
137 this->type[boundary_id] =
138 BoundaryType::cahn_hilliard_dirichlet_phase_order;
139 }
140 else if (op == "angle_of_contact")
141 {
142 this->type[boundary_id] =
143 BoundaryType::cahn_hilliard_angle_of_contact;
144 }
145 else if (op == "free_angle")
146 {
147 this->type[boundary_id] = BoundaryType::cahn_hilliard_free_angle;
148 }
149 else if (op == "periodic")
150 {
151 types::boundary_id periodic_boundary_id =
152 prm.get_integer("periodic id");
153
154 this->type[boundary_id] = BoundaryType::periodic;
155
156 // We attribute a periodic neighbor boundary type to the neighbor
157 // boundary to ensure that all boundaries have a defined type
158 this->type[periodic_boundary_id] = BoundaryType::periodic_neighbor;
159
160 // We store the periodic id and direction
161 this->periodic_boundaries[boundary_id] = {
162 .neighbor_id = periodic_boundary_id,
163 .direction = static_cast<unsigned int>(
164 prm.get_integer("periodic direction"))};
165 }
166 else
167 {
168 AssertThrow(
169 false,
170 ExcMessage("Unknown boundary condition type for Cahn-Hilliard."));
171 }
172 }
173 }

175 template <int dim>
176 void
177 CahnHilliardBoundaryConditions<dim>::parse_parameters(ParameterHandler &prm)
178 {
179 prm.enter_subsection("boundary conditions cahn hilliard");
180 {
181 this->number_of_boundary_conditions = prm.get_integer("number");
182 this->time_dependent = prm.get_bool("time dependent");
183
184 for (unsigned int n = 0; n < this->number_of_boundary_conditions; n++)
185 {
186 prm.enter_subsection("bc " + std::to_string(n));
187 {
188 parse_boundary(prm);
189 }
190 prm.leave_subsection();
191 }
192 }
193 prm.leave_subsection();
194 }
