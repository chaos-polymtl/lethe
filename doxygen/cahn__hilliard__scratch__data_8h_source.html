15/**
16 * @brief Class that stores the information required by the assembly procedure
17 * for the Cahn-Hilliard equations. Consequently, this class
18 * calculates the phase field parameter Phi (values, gradients, laplacians),
19 * the chemical potential eta (values, gradients, laplacians) and the shape
20 * function (values, gradients, laplacians) at all the Gauss points for all
21 * degrees of freedom and stores it into arrays. Additionally, the user can
22 * request that this class gathers additional fields for physics which are
23 * coupled to the Cahn-Hilliard equations, such as the velocity which is
24 * required. This class serves as a separation between the evaluation at the
25 * Gauss point of the variables of interest and their use in the assembly, which
26 * is carried out by the assembler functions. For more information on this
27 * design, the reader can consult deal.II step-9
28 * "https://www.dealii.org/current/doxygen/deal.II/step_9.html". In this latter
29 * example, the scratch is a struct instead of a templated class because of the
30 * simplicity of step-9.
31 *
32 * @tparam dim An integer that denotes the dimension of the space in which
33 * the flow is solved
34 *
35 * @ingroup solvers
36 **/
37
38template <int dim>
39class CahnHilliardScratchData : public PhysicsScratchDataBase
40{
41public:
42 /**
43 * @brief Constructor. The constructor creates the fe_values that will be used
44 * to fill the member variables of the scratch. It also allocated the
45 * necessary memory for all member variables. However, it does not do any
46 * evaluation, since this needs to be done at the cell level.
47 *
48 * @param properties_manager The physical properties manager (see physical_properties_manager.h)
49 * @param fe_cahn_hilliard The FESystem used to solve the Cahn-Hilliard equations
50 * @param quadrature The quadrature to use for the assembly
51 * @param mapping The mapping of the domain in which the Cahn-Hilliard equations are solved
52 * @param fe_fd The FESystem used to solve the Fluid Dynamics equations
53 *
54 */
55 CahnHilliardScratchData(const PhysicalPropertiesManager &properties_manager,
56 const FESystem<dim> &fe_cahn_hilliard,
57 const Quadrature<dim> &quadrature,
58 const Mapping<dim> &mapping,
59 const FiniteElement<dim> &fe_fd,
60 const Quadrature<dim - 1> &face_quadrature);
61
62 /**
63 * @brief Copy Constructor. Same as the main constructor.
64 * This constructor only uses the other scratch to build the FeValues, it
65 * does not copy the content of the other scratch into itself since, by
66 * definition of the WorkStream mechanism it is assumed that the content of
67 * the scratch will be reset on a cell basis.
68 *
69 * @param sd
70 */
71 CahnHilliardScratchData(const CahnHilliardScratchData<dim> &sd);
72

76 void
77 allocate() override;

79 /** @brief Reinitializes the content of the scratch.
80 *
81 * Using the FeValues and the content of the solutions and previous solutions,
82 * fills all of the class member of the scratch.
83 *
84 * @param[in] cell The cell over which the assembly is being carried.
85 * This cell must be compatible with the fe which is used to fill the
86 * FeValues.
87 *
88 * @param[in] current_solution The present value of the solution for [Phi,eta]
89 *
90 * @param[in] previous_solutions The solutions at the previous time steps.
91 *
92 * @param[in] source_function The function describing the source term in
93 * Cahn-Hilliard equations.
94 *
95 */
96
97 template <typename VectorType>
98 void
99 reinit(const typename DoFHandler<dim>::active_cell_iterator &cell,
100 const VectorType &current_solution,
101 const std::vector<VectorType> &previous_solutions,

104 /**
105 * @brief Reinitialize the content of the scratch data for the velocity

107 * @param[in] cell The cell over which the assembly is being carried.

110 *
111 * @param[in] current_solution The present value of the solution for

114 * @param[in] ale ALE parameters.

116 template <typename VectorType>
117 void
118 reinit_velocity(const typename DoFHandler<dim>::active_cell_iterator &cell,
