37template <int dim>
38QMOMFieldFunction<dim>::QMOMFieldFunction(
39 const Parameters::QMOMFunctions<dim> &functions,
40 const unsigned int n_scalars,
41 const unsigned int n_moments,
42 const std::string &description)
43 : Function<dim>(n_scalars + n_moments)
44 , functions(functions)
45 , n_scalars(n_scalars)
46 , n_moments(n_moments)
47{
48 // A function has at least one component: without scalars, the scalars
49 // function is ignored
50 AssertThrow(n_scalars == 0 || functions.scalars->n_components == n_scalars,
51 QMOMScalarExpressionsCountError(description,
52 functions.scalars->n_components,
53 n_scalars));
54
55#ifndef LETHE_WITH_QMOMKIT
56 // The moments of a particle size distribution are computed by QMOMKit
57 AssertThrow(functions.distribution_type ==
58 Parameters::QMOMDistributionType::none,
59 QMOMWithoutQMOMKitError());
60#endif
61}

63template <int dim>
64double
65QMOMFieldFunction<dim>::value(const Point<dim> &p,
66 const unsigned int component) const
67{
68 AssertIndexRange(component, this->n_components);
69
70 // Components 0 to n_scalars - 1 are the scalars, and component
71 // n_scalars + k is the moment M_k
72 if (component < n_scalars)
73 return functions.scalars->value(p, component);
74
75 return compute_moments(p)[component - n_scalars];
76}

78template <int dim>
79void
80QMOMFieldFunction<dim>::vector_value(const Point<dim> &p,
81 Vector<double> &values) const
82{
83 AssertDimension(values.size(), this->n_components);
84
85 for (unsigned int i_scalar = 0; i_scalar < n_scalars; ++i_scalar)
86 values[i_scalar] = functions.scalars->value(p, i_scalar);
87
88 const std::vector<double> moments = compute_moments(p);
89 for (unsigned int i_moment = 0; i_moment < n_moments; ++i_moment)
90 values[n_scalars + i_moment] = moments[i_moment];
91}

124template <int dim>
125std::vector<double>
126QMOMFieldFunction<dim>::compute_moments(
127 [[maybe_unused]] const Point<dim> &p) const
128{
129 using DistributionType = Parameters::QMOMDistributionType;
130
131 std::vector<double> moments(n_moments, 0.);
132 if (functions.distribution_type == DistributionType::none)
133 return moments;
134
135#ifdef LETHE_WITH_QMOMKIT
136 // Number density followed by the parameters of the distribution
137 Vector<double> distribution_values(functions.distribution->n_components);
138 functions.distribution->vector_value(p, distribution_values);
139 const double number_density = distribution_values[0];
140
141 AssertThrow(number_density >= 0.,
142 QMOMNegativeNumberDensityError(number_density));
143
144 // Without particles, all the moments are zero whatever the distribution
145 if (number_density <= 0.)
146 return moments;
147
148 qmomkit::DistributionSpecification distribution;
149 switch (functions.distribution_type)
150 {
151 case DistributionType::normal:
152 distribution =
153 qmomkit::NormalDistribution{.mean = distribution_values[1],
154 .standard_deviation =
155 distribution_values[2]};
156 break;
157 case DistributionType::lognormal:
158 distribution =
159 qmomkit::LogNormalDistribution{.normal_mean = distribution_values[1],
160 .normal_standard_deviation =
161 distribution_values[2]};
162 break;
163 case DistributionType::gamma:
164 distribution =
165 qmomkit::GammaDistribution{.shape = distribution_values[1],
166 .scale = distribution_values[2]};
167 break;
168 case DistributionType::weibull:
169 distribution =
170 qmomkit::WeibullDistribution{.shape = distribution_values[1],
171 .scale = distribution_values[2]};
172 break;
173 case DistributionType::beta:
174 distribution =
175 qmomkit::BetaDistribution{.alpha = distribution_values[1],
176 .beta = distribution_values[2]};
177 break;
178 case DistributionType::chi:
179 distribution = qmomkit::ChiDistribution{.degrees_of_freedom =
180 distribution_values[1]};
181 break;
182 case DistributionType::chi_squared:
183 distribution = qmomkit::ChiSquaredDistribution{
184 .degrees_of_freedom = distribution_values[1]};
185 break;
186 case DistributionType::none:
187 Assert(false, ExcInternalError());
188 break;
189 }
190
191 const qmomkit::RawMomentResult result =
192 qmomkit::compute_raw_moments(distribution, n_moments - 1, number_density);
193 AssertThrow(result.ok(), QMOMKitMomentsError(result.message));
194
195 for (unsigned int i_moment = 0; i_moment < n_moments; ++i_moment)
196 moments[i_moment] = result.moments[i_moment];
197#endif
198
199 return moments;
200}
