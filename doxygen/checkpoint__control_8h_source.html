16class CheckpointControl
17{
18public:
19 /**
20 * @brief The CheckpointControl is responsible for keeping track of the
21 * next checkpoint id to use.
22 *
23 * @param param Restart struct - Controls writing and reading
24 * simulation checkpoints.
25 */
26 CheckpointControl(const Parameters::Restart &param)
27 : max_checkpoint_id(2)
28 , next_checkpoint_id(0)
29 , checkpointing_frequency(param.frequency)
30 , restart(param.checkpoint)
31 , filename(param.filename)
32 {}
33
34 /**
35 * @brief Return true if a checkpoint should be made at the current time step:
36 * at each checkpointing frequency time step and at the last time step.
37 * @param time_step_number Current time step.
38 * @param is_last_time_step Whether the current time step is the last one.
39 */
40 bool
41 is_checkpoint_time_step(const unsigned int time_step_number,
42 const bool is_last_time_step)
43 {
44 if (restart &&
45 (!(time_step_number % checkpointing_frequency) || is_last_time_step))
46 {
47 increment_checkpoint_id();
48 return true;
49 }
50 return false;
51 }
52
53 /**
54 * @brief Return the next checkpoint id.
55 */
56 unsigned int
57 get_next_checkpoint_id() const
58 {
59 return next_checkpoint_id;
60 }
61
62 /**
63 * @brief Return the prefix for the restart files.
64 */
65 std::string
66 get_filename() const
67 {
68 return filename;
69 }
70
71 /**
72 * @brief Write the id of the last checkpoint in the file
73 * <prefix>.checkpoint_controller, from which a restart reads. Only rank 0
74 * writes it. It is called once every file of the checkpoint is written.
75 *
76 * @param[in] prefix Prefix of the checkpoint files, without checkpoint slot.
77 * @param[in] mpi_communicator MPI communicator of the simulation.
78 */
79 void
80 write(const std::string &prefix, const MPI_Comm &mpi_communicator) const;
81
82 /**
83 * @brief Read the id of the last checkpoint from the file
84 * <prefix>.checkpoint_controller. It is called before the checkpoint is read.
85 *
86 * @param[in] prefix Prefix of the checkpoint files, without checkpoint slot.
87 * @param[in] mpi_communicator MPI communicator of the simulation.
88 */
89 void
90 read(const std::string &prefix, const MPI_Comm &mpi_communicator);
91
92 /**
93 * @brief Return the id of the last checkpoint recorded in the file
94 * <prefix>.checkpoint_controller. Only rank 0 reads the file, and the id is
95 * broadcast to the other ranks.
96 *
97 * @param[in] prefix Prefix of the checkpoint files, without checkpoint slot.
98 * @param[in] file_description Description of the file, and of the parameter
99 * that gives it, reported if the file cannot be read.
100 * @param[in] mpi_communicator MPI communicator of the simulation.
101 *
102 * @return Id of the last checkpoint.
103 */
104 static unsigned int
105 read_checkpoint_id(const std::string &prefix,
106 const std::string &file_description,
107 const MPI_Comm &mpi_communicator);
108
109 /**
110 * @brief Serialize the checkpoint controller object to an output archive.
111 * @param ar Output archive where the attributes are stored.
112 */
113 void
114 serialize(boost::archive::text_oarchive &ar) const

116 ar &next_checkpoint_id;

118
119 /**
120 * @brief Deserialize an input archive to the checkpoint controller object.
121 * @param ar Input archive where the attributes are stored.
122 */
123 void
124 deserialize(boost::archive::text_iarchive &ar)
125 {
126 ar &next_checkpoint_id;
127 }
128
129private:
130 /**
131 * @brief Calculate the next checkpoint id.
132 */
133 void
134 increment_checkpoint_id()
135 {
136 next_checkpoint_id = (next_checkpoint_id + 1) % max_checkpoint_id;
137 }
138
139 const unsigned int max_checkpoint_id;
140 unsigned int next_checkpoint_id;
141 const unsigned int checkpointing_frequency;
142 const bool restart;
143 const std::string filename;
144};
