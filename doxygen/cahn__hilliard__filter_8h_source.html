60/**
61 * @brief Calculate a filtered phase fraction for Cahn-Hilliard simulations.
62 * In this case, a simple clamping is performed on the phase fraction parameter
63 * for it to remain in the [-1,1] interval.
64 */
65class CahnHilliardClipFilter : public CahnHilliardFilterBase
66{
67public:
68 CahnHilliardClipFilter() = default;
69
70 /**
71 * @brief Calculate the value of the filtered phase fraction.
72 * @param[in] unfiltered_phase Value of the phase fraction before applying the
73 * filter.
74 * @return Value of the phase fraction after applying the filter.
75 */
76 virtual double
77 filter_phase(const double &unfiltered_phase) override
78 {
79 return (std::abs(unfiltered_phase) < 1) ? unfiltered_phase :
80 sgn(unfiltered_phase);
81 }
82};
83
84/**
85 * @brief Calculate a filtered phase fraction for Cahn-Hilliard simulations.
86 *
87 * The filtered phase is defined as \f$\phi_f = \tanh(\beta \phi)\f$ with
88 * \f$ \phi \f$ the unfiltered phase field.
89 */
90class CahnHilliardTanhFilter : public CahnHilliardFilterBase
91{
92public:
93 CahnHilliardTanhFilter(const double beta)
94 : beta(beta)
95 {}
96
97 /**
98 * @brief Calculate the value of the filtered phase fraction.
99 * @param[in] unfiltered_phase Value of the phase fraction before applying the
100 * filter.
101 * @return Value of the phase fraction after applying the filter.
102 */
103 virtual double
104 filter_phase(const double &unfiltered_phase) override
105 {
106 return tanh(beta * unfiltered_phase);
107 }
108
109private:
110 /**
111 * User-defined parameter that influences how sharp the filtering is.
112 */
113 const double beta;
114};
