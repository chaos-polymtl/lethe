37/**
38 * @brief Function of all the QMOM fields, built from the functions of the
39 * scalars and from the particle size distribution.
40 *
41 * It has one component per field, n_scalars + n_moments components in total.
42 * The scalars come first, then the moments:
43 * - components 0 to n_scalars - 1 are the scalars: component c is scalar c;
44 * - components n_scalars to n_scalars + n_moments - 1 are the moments:
45 * component c is the moment M_k, with k = c - n_scalars.
46 *
47 * For example, with 2 scalars and 4 moments, components 0 and 1 are the
48 * scalars 0 and 1, and components 2, 3, 4 and 5 are the moments M_0, M_1, M_2
49 * and M_3. Without scalars, component c is the moment M_c.
50 *
51 * Scalar c is given by expression c of the scalars function, so the scalar
52 * expressions must be given in the same order as the scalar names.
53 *
54 * The moments of the particle size distribution are scaled by
55 * the number density: M_k = n E[L^k], where n is the number density in
56 * particles per m^3 and L the particle size in m. They are computed
57 * analytically by QMOMKit, and are all zero if the distribution type is none.
58 *
59 * @tparam dim Number of spatial dimensions.
60 */
61template <int dim>
62class QMOMFieldFunction : public Function<dim>
63{
64public:
65 /**
66 * @brief Constructor. Verifies that the scalars function has one component
67 * per scalar, unless there are no scalars, and that QMOMKit is available if
68 * the moments are computed from a particle size distribution.
69 *
70 * @param[in] functions Functions of the scalars and particle size
71 * distribution.
72 * @param[in] n_scalars Number of scalars transported by the QMOM physics.
73 * @param[in] n_moments Number of moments transported by the QMOM physics.
74 * @param[in] description Description of the functions (e.g. "initial
75 * conditions"), used in the error message.
76 */
77 QMOMFieldFunction(const Parameters::QMOMFunctions<dim> &functions,
78 const unsigned int n_scalars,
79 const unsigned int n_moments,
80 const std::string &description);
81
82 /**
83 * @brief Evaluate one component of the function: scalar component if
84 * component < n_scalars, moment M_{component - n_scalars} otherwise.
85 *
86 * Evaluating a moment computes all the moments at p. Use vector_value() to
87 * evaluate several moments at the same point.
88 *
89 * @param[in] p Point at which the function is evaluated.
90 * @param[in] component Index of the field, between 0 and
91 * n_scalars + n_moments - 1 (see the class description).
92 *
93 * @return Value of the component at p.
94 */
95 double
96 value(const Point<dim> &p, const unsigned int component) const override;
97
98 /**
99 * @brief Evaluate all the components of the function. The moments are
100 * computed once for all of them.
101 *
102 * @param[in] p Point at which the function is evaluated.
103 * @param[out] values Values of all the components at p, of size
104 * n_scalars + n_moments: values[i] is scalar i for i < n_scalars and
105 * values[n_scalars + k] is the moment M_k.
106 */
107 void
108 vector_value(const Point<dim> &p, Vector<double> &values) const override;
109
110 /**
111 * @brief Evaluate the scalars only, indexed from 0 regardless of the
112 * moments.
113 *
114 * @param[in] p Point at which the function is evaluated.
115 * @param[out] values Values of the scalars at p, of size n_scalars:
116 * values[i] is scalar i.
117 */
118 void
119 scalar_values(const Point<dim> &p, Vector<double> &values) const;
120
121 /**
122 * @brief Evaluate the moments only, indexed from 0 regardless of the
123 * scalars. The moments are computed once for all of them.
124 *
125 * @param[in] p Point at which the function is evaluated.
126 * @param[out] values Values of the moments at p, of size n_moments:
127 * values[k] is the moment M_k.
128 */
129 void
130 moment_values(const Point<dim> &p, Vector<double> &values) const;
131
132 /**
133 * @brief Set the time of the function and of the underlying functions.
134 *
135 * @param[in] new_time New time.
136 */
137 void
138 set_time(const double new_time) override;
139
140private:
141 /**
142 * @brief Compute the moments of the particle size distribution at a
143 * point.
144 *
145 * @param[in] p Point at which the distribution is evaluated.
146 *
147 * @return Moments M_0 to M_{n_moments - 1}.
148 */
149 std::vector<double>
150 compute_moments(const Point<dim> &p) const;
151
152 /// Functions of the scalars and particle size distribution
153 Parameters::QMOMFunctions<dim> functions;
154
155 /// Number of scalars
156 const unsigned int n_scalars;
157
158 /// Number of moments
159 const unsigned int n_moments;
160};
