13void
14compute_effective_electromagnetic_properties(
15 const PhysicalPropertiesManager &physical_properties_manager,
16 const std::map<field, std::vector<double>> &field_values_vectors,
17 const unsigned int material_id,
18 std::vector<std::complex<double>> &effective_electric_permittivities,
19 std::vector<std::complex<double>> &effective_magnetic_permeabilities)
20{
21 const unsigned int n_q_points =
22 field_values_vectors.at(field::temperature).size();
23
24 // Create temporary vectors to store the real and imaginary parts of the
25 // electromagnetic properties
26 std::vector<double> permittivity_real(n_q_points);
27 std::vector<double> permittivity_imag(n_q_points);
28 std::vector<double> conductivity(n_q_points);
29 std::vector<double> permeability_real(n_q_points);
30 std::vector<double> permeability_imag(n_q_points);
31
32 physical_properties_manager.get_electric_permittivity_real(0, material_id)
33 ->vector_value(field_values_vectors, permittivity_real);
34 physical_properties_manager.get_electric_permittivity_imag(0, material_id)
35 ->vector_value(field_values_vectors, permittivity_imag);
36 physical_properties_manager.get_electric_conductivity(0, material_id)
37 ->vector_value(field_values_vectors, conductivity);
38 physical_properties_manager.get_magnetic_permeability_real(0, material_id)
39 ->vector_value(field_values_vectors, permeability_real);
40 physical_properties_manager.get_magnetic_permeability_imag(0, material_id)
41 ->vector_value(field_values_vectors, permeability_imag);
42
43 // We resize the effective electromagnetic property vectors to match the
44 // number of points because this function can be called with a different
45 // number of points (e.g. cell and face quadratures)
46 effective_electric_permittivities.resize(n_q_points);
47 effective_magnetic_permeabilities.resize(n_q_points);
48
49 for (unsigned int q = 0; q < n_q_points; ++q)
50 {
51 effective_electric_permittivities[q] = {permittivity_real[q],
52 permittivity_imag[q] +
53 conductivity[q]};
54 effective_magnetic_permeabilities[q] = {permeability_real[q],
55 permeability_imag[q]};
56 }
57}

59template <int dim>
60void
61TimeHarmonicMaxwellScratchData<dim>::allocate()
62{
63 const FiniteElement<dim> &fe_test = this->fe_values_test.get_fe();
64 const FiniteElement<dim> &fe_trial_interior =
65 this->fe_values_trial_interior.get_fe();
66 const FiniteElement<dim> &fe_trial_skeleton =
67 this->fe_face_values_trial_skeleton.get_fe();
68
69 // Initialize the sizes
70 this->n_q_points = this->fe_values_test.get_quadrature().size();
71 this->n_face_q_points = this->fe_face_values_test.get_quadrature().size();
72 this->n_dofs_test = fe_test.n_dofs_per_cell();
73 this->n_dofs_trial_interior = fe_trial_interior.n_dofs_per_cell();
74 this->n_dofs_trial_skeleton = fe_trial_skeleton.n_dofs_per_cell();
75
76 // Classify the shape functions of each space according to the field they
77 // belong to and gather their local indices in the corresponding lists. This
78 // is only done once since it only depends on the finite elements. Since the
79 // three spaces are [E_real, E_imag, H_real, H_imag] finite element systems,
80 // every shape function belongs to exactly one field.
81 classify_shape_functions(fe_test,
82 this->shape_function_type_test,
83 this->test_dofs_electric,
84 this->test_dofs_magnetic);
85 classify_shape_functions(fe_trial_interior,
86 this->shape_function_type_trial_interior,
87 this->trial_interior_dofs_electric,
88 this->trial_interior_dofs_magnetic);
89 classify_shape_functions(fe_trial_skeleton,
90 this->shape_function_type_trial_skeleton,
91 this->trial_skeleton_dofs_electric,
92 this->trial_skeleton_dofs_magnetic);
93
94 // Cell quadrature
95 this->JxW = std::vector<double>(n_q_points);
96 this->quadrature_points = std::vector<Point<dim>>(n_q_points);
97
98 // Temperature and effective properties at the cell quadrature points. The
99 // temperature is the only field on which the electromagnetic properties can
100 // depend.
101 this->temperature_values = std::vector<double>(n_q_points, 0.);
102 this->fields.clear();
103 this->fields.insert(
104 std::pair<field, std::vector<double>>(field::temperature, n_q_points));
105 this->effective_electric_permittivities =
106 std::vector<std::complex<double>>(n_q_points);
107 this->effective_magnetic_permeabilities =
108 std::vector<std::complex<double>>(n_q_points);
109
110 // Test and interior trial functions at the cell quadrature points
111 this->phi_F.reinit(n_q_points, n_dofs_test);
112 this->phi_F_conj.reinit(n_q_points, n_dofs_test);
113 this->curl_phi_F.reinit(n_q_points, n_dofs_test);
114 this->curl_phi_F_conj.reinit(n_q_points, n_dofs_test);
115 this->phi_I.reinit(n_q_points, n_dofs_test);
116 this->phi_I_conj.reinit(n_q_points, n_dofs_test);
117 this->curl_phi_I.reinit(n_q_points, n_dofs_test);
118 this->curl_phi_I_conj.reinit(n_q_points, n_dofs_test);
119 this->phi_E.reinit(n_q_points, n_dofs_trial_interior);
120 this->phi_H.reinit(n_q_points, n_dofs_trial_interior);
121
122 // Face quadrature
123 this->face_at_boundary = false;
124 this->face_boundary_id = numbers::invalid_boundary_id;
125 this->face_JxW = std::vector<double>(n_face_q_points);
126 this->face_quadrature_points = std::vector<Point<dim>>(n_face_q_points);
127 this->face_normals = std::vector<Tensor<1, dim>>(n_face_q_points);
128
129 // Temperature and effective properties at the face quadrature points
130 this->face_temperature_values = std::vector<double>(n_face_q_points, 0.);
131 this->face_fields.clear();
132 this->face_fields.insert(
133 std::pair<field, std::vector<double>>(field::temperature, n_face_q_points));
134 this->face_effective_electric_permittivities =
135 std::vector<std::complex<double>>(n_face_q_points);
136 this->face_effective_magnetic_permeabilities =
137 std::vector<std::complex<double>>(n_face_q_points);
138
139 // Test and skeleton trial functions at the face quadrature points
140 this->phi_F_face.reinit(n_face_q_points, n_dofs_test);
141 this->phi_F_face_conj.reinit(n_face_q_points, n_dofs_test);
142 this->phi_I_face_conj.reinit(n_face_q_points, n_dofs_test);
143 this->n_cross_phi_I_face.reinit(n_face_q_points, n_dofs_test);
144 this->n_cross_phi_I_face_conj.reinit(n_face_q_points, n_dofs_test);
145 this->phi_E_hat.reinit(n_face_q_points, n_dofs_trial_skeleton);
146 this->n_cross_phi_E_hat.reinit(n_face_q_points, n_dofs_trial_skeleton);
147 this->n_cross_phi_H_hat.reinit(n_face_q_points, n_dofs_trial_skeleton);
148}

150template <int dim>
151void
152TimeHarmonicMaxwellScratchData<dim>::classify_shape_functions(
153 const FiniteElement<dim> &fe,
154 std::vector<unsigned char> &shape_function_type,
155 std::vector<unsigned int> &electric_dofs,
156 std::vector<unsigned int> &magnetic_dofs) const
157{
158 shape_function_type.assign(fe.n_dofs_per_cell(), 0);
159 electric_dofs.clear();
160 magnetic_dofs.clear();
161
162 for (unsigned int k = 0; k < fe.n_dofs_per_cell(); ++k)
163 {
164 if (fe.shape_function_belongs_to(k, extractor_E_real))
165 shape_function_type[k] |= electric_real;
166 if (fe.shape_function_belongs_to(k, extractor_E_imag))
167 shape_function_type[k] |= electric_imag;
168 if (fe.shape_function_belongs_to(k, extractor_H_real))
169 shape_function_type[k] |= magnetic_real;
170 if (fe.shape_function_belongs_to(k, extractor_H_imag))
171 shape_function_type[k] |= magnetic_imag;
172
173 if (shape_function_type[k] & is_electric)
174 electric_dofs.emplace_back(k);
175 else if (shape_function_type[k] & is_magnetic)
176 magnetic_dofs.emplace_back(k);
177 }
178}

202template <int dim>
203void
204TimeHarmonicMaxwellScratchData<dim>::reinit(
205 const typename DoFHandler<dim>::active_cell_iterator &cell,
206 const typename DoFHandler<dim>::active_cell_iterator &cell_test)
207{
208 // The curl of the test functions and the cross products are only defined in
209 // 3D. The dimension is checked at compile time so the 3D operations are not
210 // compiled when dim = 2.
211 if constexpr (dim != 3)
212 {
213 (void)cell;
214 (void)cell_test;
215 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
216 }
217 else
218 {
219 static constexpr std::complex<double> imag{0., 1.};
220
221 this->material_id = cell->material_id();
222
223 // We check if the physical properties depend on the temperature field.
224 // If so, the solver will evaluate the temperature at the quadrature
225 // points using reinit_temperature.
226 this->cell_material_needs_temperature =
227 properties_manager.get_electric_conductivity(0, material_id)
228 ->depends_on(field::temperature) ||
229 properties_manager.get_electric_permittivity_real(0, material_id)
230 ->depends_on(field::temperature) ||
231 properties_manager.get_electric_permittivity_imag(0, material_id)
232 ->depends_on(field::temperature) ||
233 properties_manager.get_magnetic_permeability_real(0, material_id)
234 ->depends_on(field::temperature) ||
235 properties_manager.get_magnetic_permeability_imag(0, material_id)
236 ->depends_on(field::temperature);
237 std::ranges::fill(this->temperature_values, 0.);
238
239 this->fe_values_trial_interior.reinit(cell);
240 this->fe_values_test.reinit(cell_test);
241
242 this->quadrature_points =
243 this->fe_values_trial_interior.get_quadrature_points();
244
245 for (unsigned int q = 0; q < n_q_points; ++q)
246 {
247 this->JxW[q] = this->fe_values_trial_interior.JxW(q);
248
249 // Electric test functions F
250 for (const unsigned int k : this->test_dofs_electric)
251 {
252 phi_F[q][k] = fe_values_test[extractor_E_real].value(k, q) +
253 imag * fe_values_test[extractor_E_imag].value(k, q);
254 phi_F_conj[q][k] =
255 fe_values_test[extractor_E_real].value(k, q) -
256 imag * fe_values_test[extractor_E_imag].value(k, q);
257
258 curl_phi_F[q][k] =
259 fe_values_test[extractor_E_real].curl(k, q) +
260 imag * fe_values_test[extractor_E_imag].curl(k, q);
261 curl_phi_F_conj[q][k] =
262 fe_values_test[extractor_E_real].curl(k, q) -
263 imag * fe_values_test[extractor_E_imag].curl(k, q);
264 }
265
266 // Magnetic test functions I
267 for (const unsigned int k : this->test_dofs_magnetic)
268 {
269 phi_I[q][k] = fe_values_test[extractor_H_real].value(k, q) +
270 imag * fe_values_test[extractor_H_imag].value(k, q);
271 phi_I_conj[q][k] =
272 fe_values_test[extractor_H_real].value(k, q) -
273 imag * fe_values_test[extractor_H_imag].value(k, q);
274
275 curl_phi_I[q][k] =
276 fe_values_test[extractor_H_real].curl(k, q) +
277 imag * fe_values_test[extractor_H_imag].curl(k, q);
278 curl_phi_I_conj[q][k] =
279 fe_values_test[extractor_H_real].curl(k, q) -
280 imag * fe_values_test[extractor_H_imag].curl(k, q);
281 }
282
283 // Interior electric (E) and magnetic (H) trial functions
284 for (const unsigned int k : this->trial_interior_dofs_electric)
285 {
286 phi_E[q][k] =
287 fe_values_trial_interior[extractor_E_real].value(k, q) +
288 imag * fe_values_trial_interior[extractor_E_imag].value(k, q);
289 }
290 for (const unsigned int k : this->trial_interior_dofs_magnetic)
291 {
292 phi_H[q][k] =
293 fe_values_trial_interior[extractor_H_real].value(k, q) +
294 imag * fe_values_trial_interior[extractor_H_imag].value(k, q);
295 }
296 }
297 }
298}

300template <int dim>
301void
302TimeHarmonicMaxwellScratchData<dim>::reinit_face(
303 const typename DoFHandler<dim>::active_cell_iterator &cell_skeleton,
304 const typename DoFHandler<dim>::active_cell_iterator &cell_test,
305 const unsigned int face_no)
306{
307 if constexpr (dim != 3)
308 {
309 (void)cell_skeleton;
310 (void)cell_test;
311 (void)face_no;
312 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
313 }
314 else
315 {
316 static constexpr std::complex<double> imag{0., 1.};
317
318 const auto face = cell_skeleton->face(face_no);
319 this->face_at_boundary = face->at_boundary();
320 this->face_boundary_id = this->face_at_boundary ?
321 face->boundary_id() :
322 numbers::invalid_boundary_id;
323 std::ranges::fill(this->face_temperature_values, 0.);
324
325 this->fe_face_values_test.reinit(cell_test, face_no);
326 this->fe_face_values_trial_skeleton.reinit(cell_skeleton, face_no);
327
328 for (unsigned int q = 0; q < n_face_q_points; ++q)
329 {
330 const Tensor<1, dim> &normal =
331 this->fe_face_values_trial_skeleton.normal_vector(q);
332
333 this->face_JxW[q] = this->fe_face_values_trial_skeleton.JxW(q);
334 this->face_quadrature_points[q] =
335 this->fe_face_values_trial_skeleton.quadrature_point(q);
336 this->face_normals[q] = normal;
337
338 // Electric test functions F
339 for (const unsigned int k : this->test_dofs_electric)
340 {
341 phi_F_face[q][k] =
342 fe_face_values_test[extractor_E_real].value(k, q) +
343 imag * fe_face_values_test[extractor_E_imag].value(k, q);
344 phi_F_face_conj[q][k] =
345 fe_face_values_test[extractor_E_real].value(k, q) -
346 imag * fe_face_values_test[extractor_E_imag].value(k, q);
347 }
348
349 // Magnetic test functions I
350 for (const unsigned int k : this->test_dofs_magnetic)
351 {
352 phi_I_face_conj[q][k] =
353 fe_face_values_test[extractor_H_real].value(k, q) -
354 imag * fe_face_values_test[extractor_H_imag].value(k, q);
355
356 n_cross_phi_I_face[q][k] = cross_product_3d(
357 normal,
358 fe_face_values_test[extractor_H_real].value(k, q) +
359 imag * fe_face_values_test[extractor_H_imag].value(k, q));
360 n_cross_phi_I_face_conj[q][k] = cross_product_3d(
361 normal,
362 fe_face_values_test[extractor_H_real].value(k, q) -
363 imag * fe_face_values_test[extractor_H_imag].value(k, q));
364 }
365
366 // Skeleton trial functions. To be in H^-1/2(curl), the fields need
367 // to have the tangential trace mapping (n x (E x n)) which extracts
368 // the tangential component of the field at the face.
369 for (const unsigned int k : this->trial_skeleton_dofs_electric)
370 {
371 phi_E_hat[q][k] = tangential_component(
372 fe_face_values_trial_skeleton[extractor_E_real].value(k, q) +
373 imag *
374 fe_face_values_trial_skeleton[extractor_E_imag].value(k, q),
375 normal);
376 n_cross_phi_E_hat[q][k] =
377 cross_product_3d(normal, phi_E_hat[q][k]);
378 }
379 for (const unsigned int k : this->trial_skeleton_dofs_magnetic)
380 {
381 n_cross_phi_H_hat[q][k] = cross_product_3d(
382 normal,
383 tangential_component(
384 fe_face_values_trial_skeleton[extractor_H_real].value(k, q) +
385 imag *
386 fe_face_values_trial_skeleton[extractor_H_imag].value(k,
387 q),
388 normal));
389 }
390 }
391 }
