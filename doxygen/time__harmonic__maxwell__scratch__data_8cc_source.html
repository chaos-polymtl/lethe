13void
14compute_effective_electromagnetic_properties(
15 const PhysicalPropertiesManager &physical_properties_manager,
16 const std::map<field, std::vector<double>> &field_values_vectors,
17 const unsigned int material_id,
18 std::vector<std::complex<double>> &effective_electric_permittivities,
19 std::vector<std::complex<double>> &effective_magnetic_permeabilities)
20{
21 const unsigned int n_q_points =
22 field_values_vectors.at(field::temperature).size();
23
24 // Create temporary vectors to store the real and imaginary parts of the
25 // electromagnetic properties
26 std::vector<double> permittivity_real(n_q_points);
27 std::vector<double> permittivity_imag(n_q_points);
28 std::vector<double> conductivity(n_q_points);
29 std::vector<double> permeability_real(n_q_points);
30 std::vector<double> permeability_imag(n_q_points);
31
32 // We use the fluid_id = 0 because the time-harmonic Maxwell solver only
33 // supports a single fluid. The material_id is used to identify the different
34 // materials in the input file and assign them their corresponding properties.
35 if (physical_properties_manager.is_effective_medium(0, material_id))
36 {
37 // The material is a mixture of the fluid and of a dispersed phase. The
38 // time-harmonic Maxwell solver supports a single fluid (id 0) and a
39 // single dispersed phase (id 0).
40 const unsigned int material_interaction_id =
41 physical_properties_manager.get_material_interaction_id(
42 material_interactions_type::fluid_dispersed_phase, 0, 0);
43
44 physical_properties_manager
45 .get_effective_electric_permittivity_real(material_interaction_id)
46 ->vector_value(field_values_vectors, permittivity_real);
47 physical_properties_manager
48 .get_effective_electric_permittivity_imag(material_interaction_id)
49 ->vector_value(field_values_vectors, permittivity_imag);
50 physical_properties_manager
51 .get_effective_magnetic_permeability_real(material_interaction_id)
52 ->vector_value(field_values_vectors, permeability_real);
53 physical_properties_manager
54 .get_effective_magnetic_permeability_imag(material_interaction_id)
55 ->vector_value(field_values_vectors, permeability_imag);
56 }
57 else
58 {
59 // The mixing law is applied to the complete complex permittivity
60 // eps' + i(eps'' + sigma) of each phase, so the conductivity of both
61 // phases is already included in the imaginary part of the effective
62 // permittivity. The conductivity vector is therefore left at zero.
63 physical_properties_manager
64 .get_electric_permittivity_real(0, material_id)
65 ->vector_value(field_values_vectors, permittivity_real);
66 physical_properties_manager
67 .get_electric_permittivity_imag(0, material_id)
68 ->vector_value(field_values_vectors, permittivity_imag);
69 physical_properties_manager.get_electric_conductivity(0, material_id)
70 ->vector_value(field_values_vectors, conductivity);
71 physical_properties_manager
72 .get_magnetic_permeability_real(0, material_id)
73 ->vector_value(field_values_vectors, permeability_real);
74 physical_properties_manager
75 .get_magnetic_permeability_imag(0, material_id)
76 ->vector_value(field_values_vectors, permeability_imag);
77 }
78
79 // We resize the effective electromagnetic property vectors to match the
80 // number of points because this function can be called with a different
81 // number of points (e.g. cell and face quadratures)
82 effective_electric_permittivities.resize(n_q_points);
83 effective_magnetic_permeabilities.resize(n_q_points);
84
85 for (unsigned int q = 0; q < n_q_points; ++q)
86 {
87 effective_electric_permittivities[q] = {permittivity_real[q],
88 permittivity_imag[q] +
89 conductivity[q]};
90 effective_magnetic_permeabilities[q] = {permeability_real[q],
91 permeability_imag[q]};
92 }
93}

95bool
96electromagnetic_properties_depend_on(
97 const PhysicalPropertiesManager &physical_properties_manager,
98 const unsigned int material_id,
99 const field field_id)
100{
101 if (physical_properties_manager.is_effective_medium(0, material_id))
102 {
103 // The time-harmonic Maxwell solver supports a single fluid (id 0) and a
104 // single dispersed phase (id 0). The dependencies of the effective
105 // permittivity models include those of the conductivity models of the
106 // phases used by the mixing law.
107 const unsigned int material_interaction_id =
108 physical_properties_manager.get_material_interaction_id(
109 material_interactions_type::fluid_dispersed_phase, 0, 0);
110
111 return physical_properties_manager
112 .get_effective_electric_permittivity_real(
113 material_interaction_id)
114 ->depends_on(field_id) ||
115 physical_properties_manager
116 .get_effective_electric_permittivity_imag(
117 material_interaction_id)
118 ->depends_on(field_id) ||
119 physical_properties_manager
120 .get_effective_magnetic_permeability_real(
121 material_interaction_id)
122 ->depends_on(field_id) ||
123 physical_properties_manager
124 .get_effective_magnetic_permeability_imag(
125 material_interaction_id)
126 ->depends_on(field_id);
127 }
128
129 return physical_properties_manager.get_electric_conductivity(0, material_id)
130 ->depends_on(field_id) ||
131 physical_properties_manager
132 .get_electric_permittivity_real(0, material_id)
133 ->depends_on(field_id) ||
134 physical_properties_manager
135 .get_electric_permittivity_imag(0, material_id)
136 ->depends_on(field_id) ||
137 physical_properties_manager
138 .get_magnetic_permeability_real(0, material_id)
139 ->depends_on(field_id) ||
140 physical_properties_manager
141 .get_magnetic_permeability_imag(0, material_id)
142 ->depends_on(field_id);
143}

145template <int dim>
146void
147TimeHarmonicMaxwellScratchData<dim>::allocate()
148{
149 const FiniteElement<dim> &fe_test = this->fe_values_test.get_fe();
150 const FiniteElement<dim> &fe_trial_interior =
151 this->fe_values_trial_interior.get_fe();
152 const FiniteElement<dim> &fe_trial_skeleton =
153 this->fe_face_values_trial_skeleton.get_fe();
154
155 // Initialize the sizes
156 this->n_q_points = this->fe_values_test.get_quadrature().size();
157 this->n_face_q_points = this->fe_face_values_test.get_quadrature().size();
158 this->n_dofs_test = fe_test.n_dofs_per_cell();
159 this->n_dofs_trial_interior = fe_trial_interior.n_dofs_per_cell();
160 this->n_dofs_trial_skeleton = fe_trial_skeleton.n_dofs_per_cell();
161
162 // Classify the shape functions of each space according to the field they
163 // belong to and gather their local indices in the corresponding lists. This
164 // is only done once since it only depends on the finite elements. Since the
165 // three spaces are [E_real, E_imag, H_real, H_imag] finite element systems,
166 // every shape function belongs to exactly one field.
167 classify_shape_functions(fe_test,
168 this->shape_function_type_test,
169 this->test_dofs_electric,
170 this->test_dofs_magnetic);
171 classify_shape_functions(fe_trial_interior,
172 this->shape_function_type_trial_interior,
173 this->trial_interior_dofs_electric,
174 this->trial_interior_dofs_magnetic);
175 classify_shape_functions(fe_trial_skeleton,
176 this->shape_function_type_trial_skeleton,
177 this->trial_skeleton_dofs_electric,
178 this->trial_skeleton_dofs_magnetic);
179
180 // Cell quadrature
181 this->JxW = std::vector<double>(n_q_points);
182 this->quadrature_points = std::vector<Point<dim>>(n_q_points);
183
184 // Temperature, void fraction and effective properties at the cell quadrature
185 // points. The temperature and the void fraction are the only fields on which
186 // the electromagnetic properties can depend.
187 this->temperature_values = std::vector<double>(n_q_points, 0.);
188 this->void_fraction_values = std::vector<double>(n_q_points, 1.);
189 this->fields.clear();
190 this->fields.insert(
191 std::pair<field, std::vector<double>>(field::temperature, n_q_points));
192 this->fields.insert(
193 std::pair<field, std::vector<double>>(field::void_fraction, n_q_points));
194 this->effective_electric_permittivities =
195 std::vector<std::complex<double>>(n_q_points);
196 this->effective_magnetic_permeabilities =
197 std::vector<std::complex<double>>(n_q_points);
198
199 // Test and interior trial functions at the cell quadrature points
200 this->phi_F.reinit(n_q_points, n_dofs_test);
201 this->phi_F_conj.reinit(n_q_points, n_dofs_test);
202 this->curl_phi_F.reinit(n_q_points, n_dofs_test);
203 this->curl_phi_F_conj.reinit(n_q_points, n_dofs_test);
204 this->phi_I.reinit(n_q_points, n_dofs_test);
205 this->phi_I_conj.reinit(n_q_points, n_dofs_test);
206 this->curl_phi_I.reinit(n_q_points, n_dofs_test);
207 this->curl_phi_I_conj.reinit(n_q_points, n_dofs_test);
208 this->phi_E.reinit(n_q_points, n_dofs_trial_interior);
209 this->phi_H.reinit(n_q_points, n_dofs_trial_interior);
210
211 // Face quadrature
212 this->face_at_boundary = false;
213 this->face_boundary_id = numbers::invalid_boundary_id;
214 this->face_JxW = std::vector<double>(n_face_q_points);
215 this->face_quadrature_points = std::vector<Point<dim>>(n_face_q_points);
216 this->face_normals = std::vector<Tensor<1, dim>>(n_face_q_points);
217
218 // Temperature, void fraction and effective properties at the face quadrature
219 // points
220 this->face_temperature_values = std::vector<double>(n_face_q_points, 0.);
221 this->face_void_fraction_values = std::vector<double>(n_face_q_points, 1.);
222 this->face_fields.clear();
223 this->face_fields.insert(
224 std::pair<field, std::vector<double>>(field::temperature, n_face_q_points));
225 this->face_fields.insert(
226 std::pair<field, std::vector<double>>(field::void_fraction,
227 n_face_q_points));
228 this->face_effective_electric_permittivities =
229 std::vector<std::complex<double>>(n_face_q_points);
230 this->face_effective_magnetic_permeabilities =
231 std::vector<std::complex<double>>(n_face_q_points);
232
233 // Test and skeleton trial functions at the face quadrature points
234 this->phi_F_face.reinit(n_face_q_points, n_dofs_test);
235 this->phi_F_face_conj.reinit(n_face_q_points, n_dofs_test);
236 this->phi_I_face_conj.reinit(n_face_q_points, n_dofs_test);
237 this->n_cross_phi_I_face.reinit(n_face_q_points, n_dofs_test);
238 this->n_cross_phi_I_face_conj.reinit(n_face_q_points, n_dofs_test);
239 this->phi_E_hat.reinit(n_face_q_points, n_dofs_trial_skeleton);
240 this->n_cross_phi_E_hat.reinit(n_face_q_points, n_dofs_trial_skeleton);
241 this->n_cross_phi_H_hat.reinit(n_face_q_points, n_dofs_trial_skeleton);
242}

244template <int dim>
245void
246TimeHarmonicMaxwellScratchData<dim>::classify_shape_functions(
247 const FiniteElement<dim> &fe,
248 std::vector<unsigned char> &shape_function_type,
249 std::vector<unsigned int> &electric_dofs,
250 std::vector<unsigned int> &magnetic_dofs) const
251{
252 shape_function_type.assign(fe.n_dofs_per_cell(), 0);
253 electric_dofs.clear();
254 magnetic_dofs.clear();
255
256 for (unsigned int k = 0; k < fe.n_dofs_per_cell(); ++k)
257 {
258 if (fe.shape_function_belongs_to(k, extractor_E_real))
259 shape_function_type[k] |= electric_real;
260 if (fe.shape_function_belongs_to(k, extractor_E_imag))
261 shape_function_type[k] |= electric_imag;
262 if (fe.shape_function_belongs_to(k, extractor_H_real))
263 shape_function_type[k] |= magnetic_real;
264 if (fe.shape_function_belongs_to(k, extractor_H_imag))
265 shape_function_type[k] |= magnetic_imag;
266
267 if (shape_function_type[k] & is_electric)
268 electric_dofs.emplace_back(k);
269 else if (shape_function_type[k] & is_magnetic)
270 magnetic_dofs.emplace_back(k);
271 }
272}

318template <int dim>
319void
320TimeHarmonicMaxwellScratchData<dim>::reinit(
321 const typename DoFHandler<dim>::active_cell_iterator &cell,
322 const typename DoFHandler<dim>::active_cell_iterator &cell_test)
323{
324 // The curl of the test functions and the cross products are only defined in
325 // 3D. The dimension is checked at compile time so the 3D operations are not
326 // compiled when dim = 2.
327 if constexpr (dim != 3)
328 {
329 (void)cell;
330 (void)cell_test;
331 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
332 }
333 else
334 {
335 static constexpr std::complex<double> imag{0., 1.};
336
337 this->material_id = cell->material_id();
338
339 // We check if the physical properties depend on the temperature and
340 // void fraction fields. If so, the solver will evaluate them at the
341 // quadrature points using reinit_temperature and reinit_void_fraction.
342 this->cell_material_needs_temperature =
343 electromagnetic_properties_depend_on(properties_manager,
344 material_id,
345 field::temperature);
346 this->cell_material_needs_void_fraction =
347 electromagnetic_properties_depend_on(properties_manager,
348 material_id,
349 field::void_fraction);
350
351 this->fe_values_trial_interior.reinit(cell);
352 this->fe_values_test.reinit(cell_test);
353
354 this->quadrature_points =
355 this->fe_values_trial_interior.get_quadrature_points();
356
357 for (unsigned int q = 0; q < n_q_points; ++q)
358 {
359 this->JxW[q] = this->fe_values_trial_interior.JxW(q);
360
361 // Electric test functions F
362 for (const unsigned int k : this->test_dofs_electric)
363 {
364 phi_F[q][k] = fe_values_test[extractor_E_real].value(k, q) +
365 imag * fe_values_test[extractor_E_imag].value(k, q);
366 phi_F_conj[q][k] =
367 fe_values_test[extractor_E_real].value(k, q) -
368 imag * fe_values_test[extractor_E_imag].value(k, q);
369
370 curl_phi_F[q][k] =
371 fe_values_test[extractor_E_real].curl(k, q) +
372 imag * fe_values_test[extractor_E_imag].curl(k, q);
373 curl_phi_F_conj[q][k] =
374 fe_values_test[extractor_E_real].curl(k, q) -
375 imag * fe_values_test[extractor_E_imag].curl(k, q);
376 }
377
378 // Magnetic test functions I
379 for (const unsigned int k : this->test_dofs_magnetic)
380 {
381 phi_I[q][k] = fe_values_test[extractor_H_real].value(k, q) +
382 imag * fe_values_test[extractor_H_imag].value(k, q);
383 phi_I_conj[q][k] =
384 fe_values_test[extractor_H_real].value(k, q) -
385 imag * fe_values_test[extractor_H_imag].value(k, q);
386
387 curl_phi_I[q][k] =
388 fe_values_test[extractor_H_real].curl(k, q) +
389 imag * fe_values_test[extractor_H_imag].curl(k, q);
390 curl_phi_I_conj[q][k] =
391 fe_values_test[extractor_H_real].curl(k, q) -
392 imag * fe_values_test[extractor_H_imag].curl(k, q);
393 }
394
395 // Interior electric (E) and magnetic (H) trial functions
396 for (const unsigned int k : this->trial_interior_dofs_electric)
397 {
398 phi_E[q][k] =
399 fe_values_trial_interior[extractor_E_real].value(k, q) +
400 imag * fe_values_trial_interior[extractor_E_imag].value(k, q);
401 }
402 for (const unsigned int k : this->trial_interior_dofs_magnetic)
403 {
404 phi_H[q][k] =
405 fe_values_trial_interior[extractor_H_real].value(k, q) +
406 imag * fe_values_trial_interior[extractor_H_imag].value(k, q);
407 }
408 }
409 }
410}

412template <int dim>
413void
414TimeHarmonicMaxwellScratchData<dim>::reinit_face(
415 const typename DoFHandler<dim>::active_cell_iterator &cell_skeleton,
416 const typename DoFHandler<dim>::active_cell_iterator &cell_test,
417 const unsigned int face_no)
418{
419 if constexpr (dim != 3)
420 {
421 (void)cell_skeleton;
422 (void)cell_test;
423 (void)face_no;
424 AssertThrow(false, TimeHarmonicMaxwellDimensionNotSupported(dim));
425 }
426 else
427 {
428 static constexpr std::complex<double> imag{0., 1.};
429
430 const auto face = cell_skeleton->face(face_no);
431 this->face_at_boundary = face->at_boundary();
432 this->face_boundary_id = this->face_at_boundary ?
433 face->boundary_id() :
434 numbers::invalid_boundary_id;
435 std::ranges::fill(this->face_temperature_values, 0.);
436 std::ranges::fill(this->face_void_fraction_values, 1.);
437
438 this->fe_face_values_test.reinit(cell_test, face_no);
439 this->fe_face_values_trial_skeleton.reinit(cell_skeleton, face_no);
440
441 for (unsigned int q = 0; q < n_face_q_points; ++q)
442 {
443 const Tensor<1, dim> &normal =
444 this->fe_face_values_trial_skeleton.normal_vector(q);
445
446 this->face_JxW[q] = this->fe_face_values_trial_skeleton.JxW(q);
447 this->face_quadrature_points[q] =
448 this->fe_face_values_trial_skeleton.quadrature_point(q);
449 this->face_normals[q] = normal;
450
451 // Electric test functions F
452 for (const unsigned int k : this->test_dofs_electric)
453 {
454 phi_F_face[q][k] =
455 fe_face_values_test[extractor_E_real].value(k, q) +
456 imag * fe_face_values_test[extractor_E_imag].value(k, q);
457 phi_F_face_conj[q][k] =
458 fe_face_values_test[extractor_E_real].value(k, q) -
459 imag * fe_face_values_test[extractor_E_imag].value(k, q);
460 }
461
462 // Magnetic test functions I
463 for (const unsigned int k : this->test_dofs_magnetic)
464 {
465 phi_I_face_conj[q][k] =
466 fe_face_values_test[extractor_H_real].value(k, q) -
467 imag * fe_face_values_test[extractor_H_imag].value(k, q);
468
469 n_cross_phi_I_face[q][k] = cross_product_3d(
470 normal,
471 fe_face_values_test[extractor_H_real].value(k, q) +
472 imag * fe_face_values_test[extractor_H_imag].value(k, q));
473 n_cross_phi_I_face_conj[q][k] = cross_product_3d(
474 normal,
475 fe_face_values_test[extractor_H_real].value(k, q) -
476 imag * fe_face_values_test[extractor_H_imag].value(k, q));
477 }
478
479 // Skeleton trial functions. To be in H^-1/2(curl), the fields need
480 // to have the tangential trace mapping (n x (E x n)) which extracts
481 // the tangential component of the field at the face.
482 for (const unsigned int k : this->trial_skeleton_dofs_electric)
483 {
484 phi_E_hat[q][k] = tangential_component(
485 fe_face_values_trial_skeleton[extractor_E_real].value(k, q) +
486 imag *
487 fe_face_values_trial_skeleton[extractor_E_imag].value(k, q),
488 normal);
489 n_cross_phi_E_hat[q][k] =
490 cross_product_3d(normal, phi_E_hat[q][k]);
491 }
492 for (const unsigned int k : this->trial_skeleton_dofs_magnetic)
493 {
494 n_cross_phi_H_hat[q][k] = cross_product_3d(
495 normal,
496 tangential_component(
497 fe_face_values_trial_skeleton[extractor_H_real].value(k, q) +
498 imag *
499 fe_face_values_trial_skeleton[extractor_H_imag].value(k,
500 q),
501 normal));
502 }
503 }
504 }
