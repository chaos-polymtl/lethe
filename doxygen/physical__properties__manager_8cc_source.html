6void
7PhysicalPropertiesManager::establish_fields_required_by_model(
8 PhysicalPropertyModel &model)
9{
10 // Loop through the map. The use of or (||) is there to ensure
11 // that if a field is already required, it won't be erased.
12 for (auto &f : required_fields)
13 {
14 f.second = f.second || model.depends_on(f.first);
15 }
16}

30void
31PhysicalPropertiesManager::initialize(
32 Parameters::PhysicalProperties physical_properties)
33{
34 // Keep a copy of the physical properties used to build the physical property
35 // manager
36 physical_properties_parameters = physical_properties;
37
38 is_initialized = true;
39
40 number_of_fluids = physical_properties.number_of_fluids;
41 number_of_solids = physical_properties.number_of_solids;
42 number_of_material_interactions =
43 physical_properties.number_of_material_interactions;
44 fluid_fluid_interactions_with_material_interaction_ids =
45 physical_properties.fluid_fluid_interactions_with_material_interaction_ids;
46 fluid_solid_interactions_with_material_interaction_ids =
47 physical_properties.fluid_solid_interactions_with_material_interaction_ids;
48 fluid_dispersed_phase_interactions_with_material_interaction_ids =
49 physical_properties
50 .fluid_dispersed_phase_interactions_with_material_interaction_ids;
51 reference_temperature = physical_properties.reference_temperature;
52
53 non_newtonian_flow = false;
54 phase_change = false;
55 constant_density = true;
56 constant_surface_tension = true;
57
58 required_fields[field::temperature] = false;
59 required_fields[field::temperature_p1] = false;
60 required_fields[field::temperature_p2] = false;
61 required_fields[field::shear_rate] = false;
62 required_fields[field::pressure] = false;
63 required_fields[field::phase_order_cahn_hilliard] = false;
64 required_fields[field::levelset] = false;
65 required_fields[field::tracer_concentration] = false;
66 required_fields[field::void_fraction] = false;
67
68 // For each fluid, declare the physical properties
69 for (unsigned int f = 0; f < number_of_fluids; ++f)
70 {
71 density.push_back(
72 DensityModel::model_cast(physical_properties.fluids[f]));
73 establish_fields_required_by_model(*density[f]);
74
75 // Store an indicator for the density to indicate if it is not constant
76 // This indicator is used elsewhere in the code to throw assertions
77 // if non-constant density is not implemented in a post-processing utility
78 if (!density.back()->is_constant_density_model())
79 constant_density = false;
80
81 specific_heat.push_back(
82 SpecificHeatModel::model_cast(physical_properties.fluids[f]));
83 establish_fields_required_by_model(*specific_heat[f]);
84
85 // Store an indicator that a phase change model is present
86 phase_change = phase_change ||
87 physical_properties.fluids[f].specific_heat_model ==
88 Parameters::Material::SpecificHeatModel::phase_change ||
89 physical_properties.fluids[f].rheological_model ==
90 Parameters::Material::RheologicalModel::phase_change;
91
92 thermal_conductivity.push_back(
93 ThermalConductivityModel::model_cast(physical_properties.fluids[f]));
94 establish_fields_required_by_model(*thermal_conductivity[f]);
95
96 rheology.push_back(
97 RheologicalModel::model_cast(physical_properties.fluids[f]));
98 this->establish_fields_required_by_model(*rheology[f]);
99 rheology[f]->set_dynamic_viscosity(density[f]->get_density_ref());
100
101 tracer_diffusivity.push_back(
102 TracerDiffusivityModel::model_cast(physical_properties.fluids[f]));
103 establish_fields_required_by_model(*tracer_diffusivity[f]);
104
105 tracer_reaction_prefactor.push_back(
106 TracerReactionPrefactorModel::model_cast(
107 physical_properties.fluids[f]));
108 establish_fields_required_by_model(*tracer_reaction_prefactor[f]);
109
110 tracer_reaction_order.push_back(
111 physical_properties.fluids[f].tracer_reaction_order);
112
113 thermal_expansion.push_back(
114 ThermalExpansionModel::model_cast(physical_properties.fluids[f]));
115 establish_fields_required_by_model(*thermal_expansion[f]);
116
117 phase_change_parameters.push_back(
118 physical_properties.fluids[f].phase_change_parameters);
119
120 if (rheology.back()->is_non_newtonian_rheological_model())
121 non_newtonian_flow = true;
122
123 electric_conductivity.push_back(
124 ElectricConductivityModel::model_cast(physical_properties.fluids[f]));
125
126 electric_permittivity_real.push_back(
127 ElectricPermittivityModel::model_cast_real(
128 physical_properties.fluids[f]));
129 electric_permittivity_imag.push_back(
130 ElectricPermittivityModel::model_cast_imag(
131 physical_properties.fluids[f]));
132
133 magnetic_permeability_real.push_back(
134 MagneticPermeabilityModel::model_cast_real(
135 physical_properties.fluids[f]));
136 magnetic_permeability_imag.push_back(
137 MagneticPermeabilityModel::model_cast_imag(
138 physical_properties.fluids[f]));
139 }
140
141 // For each solid, append the physical properties
142 for (unsigned int s = 0; s < number_of_solids; ++s)
143 {
144 density.push_back(
145 DensityModel::model_cast(physical_properties.solids[s]));
146 establish_fields_required_by_model(*density[s + number_of_fluids]);
147
148 // Store an indicator for the density to indicate if it is not constant
149 // This indicator is used elsewhere in the code to throw assertions
150 // if non-constant density is not implemented in a post-processing utility
151 if (!density.back()->is_constant_density_model())
152 constant_density = false;
153
154 specific_heat.push_back(
155 SpecificHeatModel::model_cast(physical_properties.solids[s]));
156 establish_fields_required_by_model(*specific_heat[s + number_of_fluids]);
157
158 thermal_conductivity.push_back(
159 ThermalConductivityModel::model_cast(physical_properties.solids[s]));
160 establish_fields_required_by_model(
161 *thermal_conductivity[s + number_of_fluids]);
162
163 rheology.push_back(
164 RheologicalModel::model_cast(physical_properties.solids[s]));
165 this->establish_fields_required_by_model(*rheology[s + number_of_fluids]);
166 rheology.back()->set_dynamic_viscosity(density.back()->get_density_ref());
167
168 tracer_diffusivity.push_back(
169 TracerDiffusivityModel::model_cast(physical_properties.solids[s]));
170 establish_fields_required_by_model(
171 *tracer_diffusivity[s + number_of_fluids]);
172
173 tracer_reaction_prefactor.push_back(
174 TracerReactionPrefactorModel::model_cast(
175 physical_properties.solids[s]));
176 establish_fields_required_by_model(
177 *tracer_reaction_prefactor[s + number_of_fluids]);
178
179 tracer_reaction_order.push_back(
180 physical_properties.fluids[s].tracer_reaction_order);
181
182 thermal_expansion.push_back(
183 ThermalExpansionModel::model_cast(physical_properties.solids[s]));
184 establish_fields_required_by_model(
185 *thermal_expansion[s + number_of_fluids]);
186
187 electric_conductivity.push_back(
188 ElectricConductivityModel::model_cast(physical_properties.solids[s]));
189
190 electric_permittivity_real.push_back(
191 ElectricPermittivityModel::model_cast_real(
192 physical_properties.solids[s]));
193 electric_permittivity_imag.push_back(
194 ElectricPermittivityModel::model_cast_imag(
195 physical_properties.solids[s]));
196
197 magnetic_permeability_real.push_back(
198 MagneticPermeabilityModel::model_cast_real(
199 physical_properties.solids[s]));
200 magnetic_permeability_imag.push_back(
201 MagneticPermeabilityModel::model_cast_imag(
202 physical_properties.solids[s]));
203 }
204
205 // The effective medium models are indexed by the material interaction id.
206 // The interactions that do not define a mixture keep an empty entry.
207 effective_electric_permittivity_real.resize(number_of_material_interactions);
208 effective_electric_permittivity_imag.resize(number_of_material_interactions);
209 effective_magnetic_permeability_real.resize(number_of_material_interactions);
210 effective_magnetic_permeability_imag.resize(number_of_material_interactions);
211
212 material_is_effective_medium.resize(number_of_fluids + number_of_solids,
213 false);
214
215 // For each pair of interacting materials (fluid-fluid, fluid-solid or
216 // fluid-dispersed phase), append physical properties
217 for (unsigned int i = 0; i < number_of_material_interactions; ++i)
218 {
219 surface_tension.push_back(SurfaceTensionModel::model_cast(
220 physical_properties.material_interactions[i]));
221 establish_fields_required_by_model(*surface_tension[i]);
222 if (!surface_tension.back()->is_constant_surface_tension_model())
223 constant_surface_tension = false;
224
225 mobility_cahn_hilliard.push_back(MobilityCahnHilliardModel::model_cast(
226 physical_properties.material_interactions[i]));
227 establish_fields_required_by_model(*mobility_cahn_hilliard[i]);
228
229 // A fluid-dispersed phase interaction defines a mixture whose properties
230 // are given by effective medium models.
231 if (physical_properties.material_interactions[i]
232 .material_interaction_type ==
233 Parameters::MaterialInteractions::MaterialInteractionsType::
234 fluid_dispersed_phase)
235 {
236 effective_electric_permittivity_real[i] =
237 EffectiveMediumModel::model_cast(
238 physical_properties,
239 i,
240 EffectiveMediumProperty::electric_permittivity,
241 EffectiveMediumModel::ComplexComponent::real);
242
243 effective_electric_permittivity_imag[i] =
244 EffectiveMediumModel::model_cast(
245 physical_properties,
246 i,
247 EffectiveMediumProperty::electric_permittivity,
248 EffectiveMediumModel::ComplexComponent::imag);
249
250 effective_magnetic_permeability_real[i] =
251 EffectiveMediumModel::model_cast(
252 physical_properties,
253 i,
254 EffectiveMediumProperty::magnetic_permeability,
255 EffectiveMediumModel::ComplexComponent::real);
256
257 effective_magnetic_permeability_imag[i] =
258 EffectiveMediumModel::model_cast(
259 physical_properties,
260 i,
261 EffectiveMediumProperty::magnetic_permeability,
262 EffectiveMediumModel::ComplexComponent::imag);
263
264 // Here we also initialize the interaction container using the first
265 // material of the interaction, as it is the host of the dispersed
266 // phase. It's the only one that carries a global id.
267 material_is_effective_medium
268 [physical_properties.material_interactions[i]
269 .fluid_dispersed_phase_interaction_with_material_interaction_id
270 .first.first] = true;
271 }
272 }
273}
