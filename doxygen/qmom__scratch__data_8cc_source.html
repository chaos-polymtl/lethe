15template <int dim>
16QMOMScratchData<dim>::QMOMScratchData(
17 const FiniteElement<dim> &fe_qmom,
18 const Quadrature<dim> &quadrature,
19 const Quadrature<dim - 1> &face_quadrature,
20 const Mapping<dim> &mapping,
21 const FiniteElement<dim> &fe_fluid,
22 const unsigned int n_components)
23 : fe_values(mapping,
24 fe_qmom,
25 quadrature,
26 update_values | update_gradients | update_quadrature_points |
27 update_JxW_values)
28 , fe_interface_values(mapping,
29 fe_qmom,
30 face_quadrature,
31 update_values | update_quadrature_points |
32 update_JxW_values | update_normal_vectors)
33 , fe_values_fluid(mapping, fe_fluid, quadrature, update_values)
34 , fe_face_values_fluid(mapping, fe_fluid, face_quadrature, update_values)
35 , velocities(0)
36 , n_components(n_components)
37{
38 AssertThrow(fe_qmom.n_components() == 1,
39 ExcMessage("The QMOM components share a scalar element, but the "
40 "element given to the scratch data has " +
41 std::to_string(fe_qmom.n_components()) +
42 " components."));
43
44 allocate();
45}

57template <int dim>
58void
59QMOMScratchData<dim>::allocate()
60{
61 // Initialize size of arrays
62 n_dofs = fe_values.get_fe().n_dofs_per_cell();
63 n_q_points = fe_values.get_quadrature().size();
64 n_face_q_points = fe_interface_values.get_quadrature().size();
65 n_interface_dofs = 0;
66
67 // Initialize arrays related to quadrature
68 JxW.resize(n_q_points);
69 quadrature_points.resize(n_q_points);
70
71 // Shape functions
72 phi.reinit(n_q_points, n_dofs);
73 grad_phi.reinit(n_q_points, n_dofs);
74
75 // Previous solutions of the BDF schemes and source term of every component
76 previous_values.assign(
77 n_components,
78 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
79 std::vector<double>(n_q_points)));
80 source_values.assign(n_q_points, Vector<double>(n_components));
81
82 // Velocity
83 velocity_values.resize(n_q_points);
84
85 // Arrays related to the faces. An interior face has at most the degrees of
86 // freedom of its two cells.
87 const unsigned int max_n_interface_dofs = 2 * n_dofs;
88 face_phi_here.reinit(n_face_q_points, max_n_interface_dofs);
89 face_phi_there.reinit(n_face_q_points, max_n_interface_dofs);
90 face_jump_phi.reinit(n_face_q_points, max_n_interface_dofs);
91 face_velocity_values.resize(n_face_q_points);
92}

94template <int dim>
95void
96QMOMScratchData<dim>::reinit(
97 const typename DoFHandler<dim>::active_cell_iterator &cell)
98{
99 fe_values.reinit(cell);
100 quadrature_points = fe_values.get_quadrature_points();
101
102 for (unsigned int q = 0; q < n_q_points; ++q)
103 {
104 JxW[q] = fe_values.JxW(q);
105 for (unsigned int i = 0; i < n_dofs; ++i)
106 {
107 phi[q][i] = fe_values.shape_value(i, q);
108 grad_phi[q][i] = fe_values.shape_grad(i, q);
109 }
110 }
111}

113template <int dim>
114void
115QMOMScratchData<dim>::reinit_previous_solutions(
116 const std::vector<std::vector<GlobalVectorType>> &previous_solutions)
117{
118 AssertDimension(previous_solutions.size(), n_components);
119
120 for (unsigned int component = 0; component < n_components; ++component)
121 for (unsigned int p = 0; p < previous_solutions[component].size(); ++p)
122 fe_values.get_function_values(previous_solutions[component][p],
123 previous_values[component][p]);
124}

126template <int dim>
127void
128QMOMScratchData<dim>::reinit_source(
129 const QMOMFieldFunction<dim> &source_function)
130{
131 // All the components are evaluated at once at each quadrature point, since
132 // the moments are computed together from the particle size distribution
133 for (unsigned int q = 0; q < n_q_points; ++q)
134 source_function.vector_value(quadrature_points[q], source_values[q]);
135}

149template <int dim>
150void
151QMOMScratchData<dim>::reinit_internal_face(
152 const typename DoFHandler<dim>::active_cell_iterator &cell,
153 const unsigned int face_no,
154 const unsigned int sub_face_no,
155 const typename DoFHandler<dim>::active_cell_iterator &neighbor_cell,
156 const unsigned int neighbor_face_no,
157 const unsigned int neighbor_sub_face_no)
158{
159 fe_interface_values.reinit(cell,
160 face_no,
161 sub_face_no,
162 neighbor_cell,
163 neighbor_face_no,
164 neighbor_sub_face_no);
165 n_interface_dofs = fe_interface_values.n_current_interface_dofs();
166
167 for (unsigned int q = 0; q < n_face_q_points; ++q)
168 {
169 // Gather the shape function quantities once per quadrature point, since
170 // the FEInterfaceValues accessors map the interface dof to the cell dofs
171 // at every call. The jump is the value on the "here" cell minus the
172 // value on the "there" cell.
173 for (unsigned int i = 0; i < n_interface_dofs; ++i)
174 {
175 face_phi_here[q][i] = fe_interface_values.shape_value(true, i, q);
176 face_phi_there[q][i] = fe_interface_values.shape_value(false, i, q);
177 face_jump_phi[q][i] = face_phi_here[q][i] - face_phi_there[q][i];
178 }
179 }
180}

191template <int dim>
192template <typename VectorType>
193void
194QMOMScratchData<dim>::reinit_face_velocity(
195 const typename DoFHandler<dim>::active_cell_iterator &fluid_cell,
196 const unsigned int face_no,
197 [[maybe_unused]] const unsigned int sub_face_no,
198 const VectorType &fluid_solution)
199{
200 // MeshWorker::mesh_loop() calls the face worker from the fine side of a
201 // refined face, where it is a whole face: its quadrature points are those of
202 // the subface of the coarse neighbor in fe_interface_values. Only
203 // assemble_own_interior_faces_both would also call it from the coarse side.
204 Assert(sub_face_no == numbers::invalid_unsigned_int,
205 ExcMessage("The QMOM face assembly expects to be called from the "
206 "fine side of a refined face. Use "
207 "assemble_own_interior_faces_once in mesh_loop()."));
208
209 // The fluid velocity is continuous: evaluating it from the first side of
210 // the face, at the quadrature points of fe_interface_values, is sufficient
211 fe_face_values_fluid.reinit(fluid_cell, face_no);
212 fe_face_values_fluid[velocities].get_function_values(fluid_solution,
213 face_velocity_values);
214}
