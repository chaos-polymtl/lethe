60template <int dim>
61void
62CahnHilliardScratchData<dim>::allocate()
63{
64 // Initialize size of arrays
65 this->n_q_points = fe_values_cahn_hilliard.get_quadrature().size();
66 this->n_dofs = fe_values_cahn_hilliard.get_fe().n_dofs_per_cell();
67
68 // Initialize arrays related to quadrature
69 this->JxW.reinit(n_q_points);
70
71 // Forcing term array
72 this->source_phase_order = std::vector<double>(n_q_points);
73 this->source_chemical_potential = std::vector<double>(n_q_points);
74
75 // Initialize arrays related to phase order and chemical potential
76 this->phase_order.component = 0;
77 this->chemical_potential.component = 1;
78
79 // Variables for the Cahn-Hilliard equations
80 this->phase_order_values = std::vector<double>(n_q_points);
81 this->phase_order_gradients = std::vector<Tensor<1, dim>>(n_q_points);
82 this->phase_order_laplacians = std::vector<double>(n_q_points);
83 this->chemical_potential_values = std::vector<double>(n_q_points);
84 this->chemical_potential_gradients = std::vector<Tensor<1, dim>>(n_q_points);
85 this->chemical_potential_laplacians = std::vector<double>(n_q_points);
86
87 // Phase order for BDF schemes
88 this->previous_phase_order_values =
89 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
90 std::vector<double>(n_q_points));
91
92 this->previous_chemical_potential_values =
93 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
94 std::vector<double>(n_q_points));
95
96 // Initialize arrays related to shape functions
97 // Phase-order shape functions
98 this->phi_phase.reinit(n_q_points, n_dofs);
99 this->grad_phi_phase.reinit(n_q_points, n_dofs);
100 this->hess_phi_phase.reinit(n_q_points, n_dofs);
101 this->laplacian_phi_phase.reinit(n_q_points, n_dofs);
102
103 // Chemical potential shape functions
104 this->phi_potential.reinit(n_q_points, n_dofs);
105 this->grad_phi_potential.reinit(n_q_points, n_dofs);
106 this->hess_phi_potential.reinit(n_q_points, n_dofs);
107 this->laplacian_phi_potential.reinit(n_q_points, n_dofs);
108
109 // Velocity values
110 this->velocity_values = std::vector<Tensor<1, dim>>(this->n_q_points);
111 this->previous_velocity_values = std::vector<std::vector<Tensor<1, dim>>>(
112 maximum_number_of_previous_solutions(),
113 std::vector<Tensor<1, dim>>(this->n_q_points));
114 this->velocity_gradient_values =
115 std::vector<Tensor<2, dim>>(this->n_q_points);
116
117 // Allocate physical properties
118 this->surface_tension = std::vector<double>(n_q_points);
119 this->mobility_cahn_hilliard = std::vector<double>(n_q_points);
120 this->mobility_cahn_hilliard_gradient = std::vector<double>(n_q_points);
121
122 // Physical properties
123 fields.insert(
124 std::pair<field, std::vector<double>>(field::phase_order_cahn_hilliard,
125 n_q_points));
126 fields.insert(std::pair<field, std::vector<double>>(
127 field::phase_order_cahn_hilliard_filtered, n_q_points));
128}

130template <int dim>
131template <typename VectorType>
132void
133CahnHilliardScratchData<dim>::reinit(
134 const typename DoFHandler<dim>::active_cell_iterator &cell,
135 const VectorType &current_solution,
136 const std::vector<VectorType> &previous_solutions,
137 Function<dim> *source_function)
138{
139 this->phase_order.component = 0;
140 this->chemical_potential.component = 1;
141
142 this->fe_values_cahn_hilliard.reinit(cell);
143
144 quadrature_points = this->fe_values_cahn_hilliard.get_quadrature_points();
145 auto &fe_cahn_hilliard = this->fe_values_cahn_hilliard.get_fe();
146
147 source_function->value_list(quadrature_points, source_phase_order, 0);
148 source_function->value_list(quadrature_points, source_chemical_potential, 1);
149
150 // Compute cell diameter
151 double cell_measure = compute_cell_measure_with_JxW(
152 this->fe_values_cahn_hilliard.get_JxW_values());
153 this->cell_size =
154 compute_cell_diameter<dim>(cell_measure, fe_cahn_hilliard.degree);
155
156 // Gather Phi and eta (values, gradient and laplacian)
157 this->fe_values_cahn_hilliard[phase_order].get_function_values(
158 current_solution, this->phase_order_values);
159 this->fe_values_cahn_hilliard[phase_order].get_function_gradients(
160 current_solution, this->phase_order_gradients);
161 this->fe_values_cahn_hilliard[phase_order].get_function_laplacians(
162 current_solution, this->phase_order_laplacians);
163
164 this->fe_values_cahn_hilliard[chemical_potential].get_function_values(
165 current_solution, this->chemical_potential_values);
166 this->fe_values_cahn_hilliard[chemical_potential].get_function_gradients(
167 current_solution, this->chemical_potential_gradients);
168 this->fe_values_cahn_hilliard[chemical_potential].get_function_laplacians(
169 current_solution, this->chemical_potential_laplacians);
170
171 // Gather previous phase order values
172 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
173 {
174 this->fe_values_cahn_hilliard[phase_order].get_function_values(
175 previous_solutions[p], previous_phase_order_values[p]);
176 }
177
178 // Gather previous chemical potential values
179 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
180 {
181 this->fe_values_cahn_hilliard[chemical_potential].get_function_values(
182 previous_solutions[p], previous_chemical_potential_values[p]);
183 }
184
185 for (unsigned int q = 0; q < n_q_points; ++q)
186 {
187 this->JxW[q] = this->fe_values_cahn_hilliard.JxW(q);
188
189 for (unsigned int k = 0; k < n_dofs; ++k)
190 {
191 // Shape functions for the phase order
192 this->phi_phase[q][k] =
193 this->fe_values_cahn_hilliard[phase_order].value(k, q);
194 this->grad_phi_phase[q][k] =
195 this->fe_values_cahn_hilliard[phase_order].gradient(k, q);
196 this->hess_phi_phase[q][k] =
197 this->fe_values_cahn_hilliard[phase_order].hessian(k, q);
198 this->laplacian_phi_phase[q][k] = trace(this->hess_phi_phase[q][k]);
199
200 // Shape functions for the chemical potential
201 this->phi_potential[q][k] =
202 this->fe_values_cahn_hilliard[chemical_potential].value(k, q);
203 this->grad_phi_potential[q][k] =
204 this->fe_values_cahn_hilliard[chemical_potential].gradient(k, q);
205 this->hess_phi_potential[q][k] =
206 this->fe_values_cahn_hilliard[chemical_potential].hessian(k, q);
207 this->laplacian_phi_potential[q][k] =
208 trace(this->hess_phi_potential[q][k]);
209 }
210 }
211
212 this->is_boundary_cell =
213 cell->at_boundary(); // The attribute needs to be updated because the
214 // assembler for the angle of contact boundary
215 // condition needs to know if the cell is at the
216 // boundary
217 if (this->is_boundary_cell)
218 {
219 n_faces = cell->n_faces();
220 is_boundary_face = std::vector<bool>(n_faces, false);
221 n_faces_q_points = fe_face_values_cahn_hilliard.get_quadrature().size();
222 boundary_face_id = std::vector<unsigned int>(n_faces);
223
224 face_JxW.reinit(n_faces, n_faces_q_points);
225 this->grad_phi_face_phase.reinit(n_faces, n_faces_q_points, n_dofs);
226
227 this->face_phase_grad_values = std::vector<std::vector<Tensor<1, dim>>>(
228 n_faces, std::vector<Tensor<1, dim>>(n_faces_q_points));
229
230 this->face_normal = std::vector<std::vector<Tensor<1, dim>>>(
231 n_faces, std::vector<Tensor<1, dim>>(n_faces_q_points));
232
233 for (const auto face : cell->face_indices())
234 {
235 this->is_boundary_face[face] = cell->face(face)->at_boundary();
236 if (this->is_boundary_face[face])
237 {
238 fe_face_values_cahn_hilliard.reinit(cell, face);
239 boundary_face_id[face] = cell->face(face)->boundary_id();
240
241 this->fe_face_values_cahn_hilliard[phase_order]
242 .get_function_gradients(current_solution,
243 this->face_phase_grad_values[face]);
244
245 for (unsigned int q = 0; q < n_faces_q_points; ++q)
246 {
247 face_JxW[face][q] = fe_face_values_cahn_hilliard.JxW(q);
248 this->face_normal[face][q] =
249 this->fe_face_values_cahn_hilliard.normal_vector(q);
250 for (const unsigned int k :
251 fe_face_values_cahn_hilliard.dof_indices())
252 {
253 this->grad_phi_face_phase[face][q][k] =
254 this->fe_face_values_cahn_hilliard[phase_order]
255 .gradient(k, q);
256 }
257 }
258 }
259 }
260 }
261}

263template <int dim>
264template <typename VectorType>
265void
266CahnHilliardScratchData<dim>::reinit_velocity(
267 const typename DoFHandler<dim>::active_cell_iterator &cell,
268 const VectorType &current_solution,
269 const Parameters::ALE<dim> &ale)
270{
271 FEValuesExtractors::Vector velocities(0);
272 this->fe_values_fd.reinit(cell);
273
274 this->fe_values_fd[velocities].get_function_values(current_solution,
275 velocity_values);
276
277 if (!ale.enabled())
278 return;
279
280
281 // ALE enabled, so extract the ALE velocity and subtract it from the
282 // velocity obtained from the fluid dynamics
283 Tensor<1, dim> velocity_ale;
284 std::shared_ptr<Functions::ParsedFunction<dim>> velocity_ale_function =
285 ale.velocity;
286 Vector<double> velocity_ale_vector(dim);
287
288 for (unsigned int q = 0; q < n_q_points; ++q)
289 {
290 velocity_ale_function->vector_value(quadrature_points[q],
291 velocity_ale_vector);
292 for (int d = 0; d < dim; ++d)
293 velocity_ale[d] = velocity_ale_vector[d];
294
295 velocity_values[q] -= velocity_ale;
296 }

299template <int dim>
300void
301CahnHilliardScratchData<dim>::calculate_physical_properties()
302{
303 switch (properties_manager.get_number_of_fluids())
304 {
305 case 1:
306 {
307 throw std::runtime_error("Unsupported number of fluids (<2)");
308 }
309 case 2:
310 {
311 set_field_vector(field::phase_order_cahn_hilliard,
312 this->phase_order_values,
313 this->fields);
314
315 // Gather properties from material interactions
316 const auto material_interaction_id =
317 properties_manager.get_material_interaction_id(
318 material_interactions_type::fluid_fluid, 0, 1);
319
320 // Gather surface tension
321 const auto surface_tension_model =
322 properties_manager.get_surface_tension(material_interaction_id);
323 surface_tension_model->vector_value(fields, surface_tension);
324
325 // Gather mobility
326 const auto mobility_cahn_hilliard_model =
327 properties_manager.get_mobility_cahn_hilliard(
328 material_interaction_id);
329 mobility_cahn_hilliard_model->vector_value(fields,
330 mobility_cahn_hilliard);
331 mobility_cahn_hilliard_model->vector_jacobian(
332 fields,
333 field::phase_order_cahn_hilliard,
334 mobility_cahn_hilliard_gradient);
335
336 break;
337 }
338 default:
339 throw std::runtime_error("Unsupported number of fluids (>2)");
340 }
341}
