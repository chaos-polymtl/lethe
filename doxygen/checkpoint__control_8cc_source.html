36unsigned int
37CheckpointControl::read_checkpoint_id(const std::string &prefix,
38 const std::string &file_description,
39 const MPI_Comm &mpi_communicator)
40{
41 const std::string filename = prefix + ".checkpoint_controller";
42
43 // Checked on every rank, so that a missing file throws on every rank instead
44 // of leaving the other ranks waiting in the broadcast.
45 check_file_exists(filename, file_description);
46
47 unsigned int checkpoint_id = 0;
48 if (Utilities::MPI::this_mpi_process(mpi_communicator) == 0)
49 {
50 std::ifstream input(filename);
51 boost::archive::text_iarchive archive(input, boost::archive::no_header);
52 archive >> checkpoint_id;
53 }
54 return Utilities::MPI::broadcast(mpi_communicator, checkpoint_id, 0);
55}
