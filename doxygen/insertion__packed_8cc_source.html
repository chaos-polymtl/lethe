12template <int dim, typename PropertiesIndex>
13InsertionPacked<dim, PropertiesIndex>::InsertionPacked(
14 const std::vector<std::shared_ptr<Distribution>>
15 &size_distribution_object_container,
16 const parallel::distributed::Triangulation<dim> &triangulation,
17 const DEMSolverParameters<dim> &dem_parameters)
18 : Insertion<dim, PropertiesIndex>(size_distribution_object_container,
19 triangulation,
20 dem_parameters)
21 , acceptance_fct(dem_parameters.insertion_info.insertion_acceptance_fct)
22{
23 // Initializing the current inserting particle type
24 // Getting properties as local parameters
25 const InsertionInfo<dim> insertion_information =
26 dem_parameters.insertion_info;
27
28 std::vector<unsigned int> axis_list;
29 axis_list.reserve(dim);
30 axis_list = {insertion_information.direction_sequence[0],
31 insertion_information.direction_sequence[1]};
32 if constexpr (dim == 3)
33 {
34 axis_list.push_back(insertion_information.direction_sequence[2]);
35 }
36
37 axis_min.resize(3);
38 axis_max.resize(3);
39 for (int axis : axis_list)
40 {
41 AssertThrow(axis < dim,
42 ExcMessage("Insertion direction must be 0, 1 or 2"));
43
44 axis_min[axis] = insertion_information.insertion_box_point_1(axis);
45 axis_max[axis] = insertion_information.insertion_box_point_2(axis);
46 }
47}

50template <int dim, typename PropertiesIndex>
51void
52InsertionPacked<dim, PropertiesIndex>::insert(
53 Particles::ParticleHandler<dim> &particle_handler,
54 const parallel::distributed::Triangulation<dim> &triangulation,
55 const DEMSolverParameters<dim> &dem_parameters)
56{
57 // We only insert once with this method. Thus, if the particle handler already
58 // possesses particles, we should not insert again.
59 if (particle_handler.n_global_particles() != 0)
60 return;
61
62
63 // Message passing interface requirements.
64 MPI_Comm communicator = triangulation.get_mpi_communicator();
65 ConditionalOStream pcout(std::cout,
66 Utilities::MPI::this_mpi_process(communicator) == 0);
67
68 auto this_mpi_process = Utilities::MPI::this_mpi_process(communicator);
69 auto n_mpi_process = Utilities::MPI::n_mpi_processes(communicator);
70
71 // Obtaining global bounding boxes
72 const auto my_bounding_box = GridTools::compute_mesh_predicate_bounding_box(
73 triangulation, IteratorFilters::LocallyOwnedCell());
74 const auto global_bounding_boxes =
75 Utilities::MPI::all_gather(communicator, my_bounding_box);
76
77 // Vector that will get filled with the insertion locations.
78 std::vector<Point<dim>> insertion_points_on_proc;
79
80 // Initialize the Random Number Generator using the configured seed
81 std::mt19937 rng(dem_parameters.insertion_info.seed_for_insertion +
82 this_mpi_process);
83
84 // We insert all the particle for every type all at once, thus we loop on each
85 // particle type.
86 for (unsigned int particle_type = 0;
87 particle_type <
88 dem_parameters.lagrangian_physical_properties.particle_type_number;
89 particle_type++)
90 {
91 const unsigned int n_particle_current_type =
92 dem_parameters.lagrangian_physical_properties.number.at(particle_type);
93
94 if (n_particle_current_type == 0)
95 continue;
96
97 // Distributing particles between processors
98 unsigned int n_particle_current_type_this_proc =
99 static_cast<unsigned int>(
100 floor(n_particle_current_type / n_mpi_process));
101 if (this_mpi_process == (n_mpi_process - 1))
102 n_particle_current_type_this_proc = static_cast<unsigned int>(
103 n_particle_current_type -
104 (n_mpi_process - 1) * floor(n_particle_current_type / n_mpi_process));
105
106 // Clear and reserve space for insertion points for this particle type.
107 // We clear, since if this is the second particle type, this vector is
108 // already filled.
109 insertion_points_on_proc.clear();
110 insertion_points_on_proc.reserve(n_particle_current_type_this_proc);
111
112 // Find the first and the last particle id for each process
113 unsigned int first_id;
114 unsigned int last_id;
115 if (this_mpi_process == (n_mpi_process - 1))
116 { // Last proc
117 first_id =
118 n_particle_current_type - n_particle_current_type_this_proc;
119 last_id = n_particle_current_type;
120 }
121 else
122 { // Every other proc
123 first_id = this_mpi_process * n_particle_current_type_this_proc;
124 last_id = (this_mpi_process + 1) * n_particle_current_type_this_proc;
125 }
126
127 // Loop and generate valid insertion points.
128 Point<dim> insertion_location;
129 for (unsigned int id = first_id; id < last_id; ++id)
130 {
131 // Call updated rejection sampling function passing the RNG state
132 generate_insertion_location(insertion_location,
133 rng,
134 dem_parameters.insertion_info);
135
136 insertion_points_on_proc.push_back(insertion_location);
137 }
138
139 // Assigning particle properties
140 std::vector<std::vector<double>> particle_properties;
141 this->assign_particle_properties(dem_parameters,
142 n_particle_current_type_this_proc,
143 particle_type,
144 insertion_points_on_proc,
145 particle_properties);
146
147
148
149 // Insert the particles using the points and assigned properties
150 particle_handler.insert_global_particles(insertion_points_on_proc,
151 global_bounding_boxes,
152 particle_properties);
153
154 this->print_insertion_info(n_particle_current_type,
155 0,
156 particle_type,
157 pcout);
158 }
159
160 // Loop on every locally owned particle
161 double this_proc_inserted_volume = 0.;
162 for (auto &particle : particle_handler)
163 {
164 const double volume = numbers::PI *
165 Utilities::fixed_power<3>(
166 particle.get_properties()[PropertiesIndex::dp]) /
167 6.;
168 this_proc_inserted_volume += volume;
169 }
170
171 // Volume of the entire triangulation
172 const double total_triangulation_volume = GridTools::volume(triangulation);
173
174 // Total volume of inserted particle.
175 double total_inserted_volume =
176 Utilities::MPI::sum(this_proc_inserted_volume, communicator);
177
178 pcout << " The particle volume to triangulation volume ratio is: "
179 << total_inserted_volume / total_triangulation_volume << std::endl;
180}

182template <int dim, typename PropertiesIndex>
183void
184InsertionPacked<dim, PropertiesIndex>::update_previous_position(
185 Particles::ParticleHandler<dim> &particle_handler)
186{
187 for (auto &particle : particle_handler)
188 {
189 auto particle_properties = particle.get_properties();
190
191 const Point<dim> particle_previous_position = particle.get_location();
192
193 particle_properties[PropertiesIndex::v_x] = particle_previous_position[0];
194 particle_properties[PropertiesIndex::v_y] = particle_previous_position[1];
195 if constexpr (dim == 3)
196 particle_properties[PropertiesIndex::v_z] =
197 particle_previous_position[2];
198 }
199}

201template <int dim, typename PropertiesIndex>
202void
203InsertionPacked<dim, PropertiesIndex>::clamp_displacement(
204 Particles::ParticleHandler<dim> &particle_handler,
205 const double max_disp,
206 std::vector<double> &displacements)
207{
208 for (auto &particle : particle_handler)
209 {
210 auto particle_properties = particle.get_properties();
211
212 Point<dim> previous_position;
213 previous_position[0] = particle_properties[PropertiesIndex::v_x];
214 previous_position[1] = particle_properties[PropertiesIndex::v_y];
215 if constexpr (dim == 3)
216 previous_position[2] = particle_properties[PropertiesIndex::v_z];
217
218 const Tensor<1, dim> displacement_tensor =
219 particle.get_location() - previous_position;
220
221 const double disp_norm = displacement_tensor.norm();
222 if (disp_norm > max_disp)
223 {
224 const unsigned int particle_id = particle.get_local_index();
225 particle.set_location(previous_position +
226 (max_disp / disp_norm) * displacement_tensor);
227 displacements[particle_id] += max_disp;
228 }
229 else
230 {
231 const unsigned int particle_id = particle.get_local_index();
232 displacements[particle_id] += disp_norm;
233 }
234 }
235}

238template <int dim, typename PropertiesIndex>
239void
240InsertionPacked<dim, PropertiesIndex>::generate_insertion_location(
241 Point<dim> &insertion_location,
242 std::mt19937 &rng,
243 const InsertionInfo<dim> &insertion_information)
244{
245 // A continuous random distribution on the range [min, max) with equal
246 // probability throughout the range. Thus the generated points are equally
247 // likely to be located in the middle of the box or on its edges.
248 std::uniform_real_distribution<double> distribution(0.0, 1.0);
249
250 bool accepted = false;
251 while (!accepted)
252 {
253 // Generate a new candidate point within the bounding box
254 for (int d = 0; d < dim; ++d)
255 {
256 // Get the actual coordinate axis (e.g., 0 for X, 1 for Y, 2 for Z)
257 // based on the configured insertion sequence
258 const unsigned int axis =
259 insertion_information.direction_sequence.at(d);
260
261 // Pull a fresh 0.0 to 1.0 random value
262 double random_value = distribution(rng);
263
264 // Map it uniformly across the specific axis span
265 insertion_location[axis] =
266 this->axis_min[axis] +
267 random_value * (this->axis_max[axis] - this->axis_min[axis]);
268 }
269 // Reject the point if the value is negative.
270 accepted = acceptance_fct->value(insertion_location) > 0.;
271 }
272}
