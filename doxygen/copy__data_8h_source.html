15/**
16 * @brief Class responsible for storing the information calculated using the assembly of regular (meaning
17 * non-stabilized) equations. It is also used to
18 * initialize, zero (reset) and store the cell_matrix, the cell_rhs and the
19 * dof indices associated with the dofs of the cell.
20 **/
21
22class CopyData
23{
24public:
25 /**
26 * @brief Constructor. Allocates the memory for the cell_matrix, cell_rhs and the local_dof_indices
27 *
28 * @param n_dofs Number of degrees of freedom per cell in the problem
29 */
30 CopyData(const unsigned int n_dofs)
31 : local_matrix(n_dofs, n_dofs)
32 , local_rhs(n_dofs)
33 , local_dof_indices(n_dofs){};
34
35 /**
36 * @brief Resets the cell_matrix and the cell_rhs to zero
37 */
38 void
39 reset()
40 {
41 local_matrix = 0;
42 local_rhs = 0;
43 }
44
45 FullMatrix<double> local_matrix;
46 Vector<double> local_rhs;
47 std::vector<types::global_dof_index> local_dof_indices;
48
49 // Boolean used to indicate if the cell being assembled is local or not
50 // This information is used to indicate to the copy_local_to_global function
51 // if it should indeed copy or not.
52 bool cell_is_local;
53};

56/**
57 * @brief Class responsible for storing the information calculated using the assembly of stabilized
58 * scalar equations. Like the CopyData class, this class is used to initialize,
59 * zero (reset) and store the cell_matrix and the cell_rhs.
60 * Contrary to the regular CopyData class, this class
61 * also stores the strong_residual and the strong_jacobian of the equation being
62 * assembled. This is useful for equations that implement residual-based
63 * stabilization such as SUPG. This class is specialized for single component
64 * equations because the strong jacobian is stored using a Vector<double>
65 **/
66class StabilizedMethodsCopyData
67{
68public:
69 /**
70 * @brief Constructor. Allocates the memory for the cell_matrix, cell_rhs
71 * and dof-indices using the number of dofs and the strong_residual using the
72 * number of quadrature points and, the strong_jacobian using both
73 *
74 * @param n_dofs Number of degrees of freedom per cell in the problem
75 *
76 * @param n_q_points Number of quadrature points
77 */
78 StabilizedMethodsCopyData(const unsigned int n_dofs,
79 const unsigned int n_q_points)
80 : local_matrix(n_dofs, n_dofs)
81 , local_rhs(n_dofs)
82 , local_dof_indices(n_dofs)
83 , strong_residual(n_q_points)
84 , strong_jacobian(n_q_points, Vector<double>(n_dofs)){};
85
86
87 /**
88 * @brief Resets the cell_matrix, cell_rhs, strong_residual
89 * and strong_jacobian to zero
90 */
91 void
92 reset()
93 {
94 local_matrix = 0;
95 local_rhs = 0;
96
97 strong_residual = 0;
98 for (unsigned int i = 0; i < strong_jacobian.size(); ++i)
99 {
100 strong_jacobian[i] = 0;
101 }
102 }
103
104 FullMatrix<double> local_matrix;
105 Vector<double> local_rhs;
106 std::vector<types::global_dof_index> local_dof_indices;
107 Vector<double> strong_residual;
108 std::vector<Vector<double>> strong_jacobian;
109
110 // Boolean used to indicate if the cell being assembled is local or not
111 // This information is used to indicate to the copy_local_to_global function
112 // if it should indeed copy or not.
113 bool cell_is_local;
114};

152/**
153 * @brief Class responsible for storing the information calculated using the assembly of stabilized
154 * Tensor<1,dim> equations. Like the CopyData class, this class is used to
155 * initialize, zero (reset) and store the cell_matrix and the cell_rhs. Contrary
156 * to the regular CopyData class, this class also stores the strong_residual and
157 * the strong_jacobian of the equation being assembled. This is useful for
158 * equations that implement residual-based stabilization such as SUPG. This
159 * class is specialized for Tensor<1,dim> equations because the strong
160 * jacobian is stored using a Tensor<1,dim>
161 **/
162
163
164template <int dim>
165class StabilizedMethodsTensorCopyData
166{
167public:
168 /**
169 * @brief Constructor. Allocates the memory for the cell_matrix and cell_rhs
170 * using the number of dofs and the strong_residual using the number of
171 * quadrature points and, the strong_jacobian using both
172 *
173 * @param n_dofs Number of degrees of freedom per cell in the problem
174 *
175 * @param n_q_points Number of quadrature points
176 */
177 StabilizedMethodsTensorCopyData(const unsigned int n_dofs,
178 const unsigned int n_q_points)
179 : local_matrix(n_dofs, n_dofs)
180 , local_rhs(n_dofs)
181 , local_dof_indices(n_dofs)
182 , strong_residual(n_q_points)
183 , strong_jacobian(n_q_points, std::vector<Tensor<1, dim>>(n_dofs)){};
184
185 /**
186 * @brief Resets the cell_matrix, cell_rhs, strong_residual
187 * and strong_jacobian to zero
188 */
189 void
190 reset()
191 {
192 local_matrix = 0;
193 local_rhs = 0;
194
195 for (unsigned int q = 0; q < strong_jacobian.size(); ++q)
196 {
197 strong_residual[q] = 0;
198
199 for (unsigned int i = 0; i < strong_jacobian[q].size(); ++i)
200 strong_jacobian[q][i] = 0;
201 }
202 }
203
204 FullMatrix<double> local_matrix;
205 Vector<double> local_rhs;
206 std::vector<types::global_dof_index> local_dof_indices;
207 std::vector<Tensor<1, dim>> strong_residual;
208 std::vector<std::vector<Tensor<1, dim>>> strong_jacobian;
209
210
211 // Boolean used to indicate if the cell being assembled is local or not
212 // This information is used to indicate to the copy_local_to_global function
213 // if it should indeed copy or not.
214 bool cell_is_local;
215 bool cell_is_cut;
216};

219/**
220 * @brief Class responsible for storing the local system of an ultraweak Discontinuous
221 * Petrov-Galerkin (DPG) discretization. The assemblers fill the uncondensed
222 * local system: the Gram matrix of the test space \f$G\f$, the matrices of the
223 * bilinear form in the cell interior \f$B\f$ and on the skeleton \f$\hat{B}\f$,
224 * and the load vector \f$l\f$. The solver then condenses this system on the
225 * skeleton unknowns using the operators
226 * \f$M_1 = B^\dagger G^{-1} B\f$, \f$M_2 = B^\dagger G^{-1} \hat{B}\f$,
227 * \f$M_3 = \hat{B}^\dagger G^{-1} \hat{B}\f$, \f$M_4 = B^\dagger G^{-1}\f$ and
228 * \f$M_5 = \hat{B}^\dagger G^{-1}\f$, and stores either the condensed skeleton
229 * system (assembly) or the reconstructed interior solution (interior
230 * reconstruction).
231 **/
232class DPGCopyData
233{
234public:
235 /**
236 * @brief Constructor. Allocates the memory of all the local matrices and
237 * vectors from the number of degrees of freedom per cell of each finite
238 * element space.
239 *
240 * @param[in] n_dofs_test Number of degrees of freedom per cell of the test
241 * space.
242 *
243 * @param[in] n_dofs_trial_interior Number of degrees of freedom per cell of
244 * the interior trial space.
245 *
246 * @param[in] n_dofs_trial_skeleton Number of degrees of freedom per cell of
247 * the skeleton trial space.
248 */
249 DPGCopyData(const unsigned int n_dofs_test,
250 const unsigned int n_dofs_trial_interior,
251 const unsigned int n_dofs_trial_skeleton)
252 : G_matrix(n_dofs_test, n_dofs_test)
253 , B_matrix(n_dofs_test, n_dofs_trial_interior)
254 , B_hat_matrix(n_dofs_test, n_dofs_trial_skeleton)
255 , l_vector(n_dofs_test)
256 , M1_matrix(n_dofs_trial_interior, n_dofs_trial_interior)
257 , M2_matrix(n_dofs_trial_interior, n_dofs_trial_skeleton)
258 , M3_matrix(n_dofs_trial_skeleton, n_dofs_trial_skeleton)
259 , M4_matrix(n_dofs_trial_interior, n_dofs_test)
260 , M5_matrix(n_dofs_trial_skeleton, n_dofs_test)
261 , tmp_matrix_M2M1(n_dofs_trial_skeleton, n_dofs_trial_interior)
262 , tmp_matrix_M2M1M2(n_dofs_trial_skeleton, n_dofs_trial_skeleton)
263 , tmp_matrix_M2M1M4(n_dofs_trial_skeleton, n_dofs_test)
264 , local_matrix(n_dofs_trial_skeleton, n_dofs_trial_skeleton)
265 , local_rhs(n_dofs_trial_skeleton)
266 , local_dof_indices(n_dofs_trial_skeleton)
267 , local_skeleton_solution(n_dofs_trial_skeleton)
268 , local_interior_rhs(n_dofs_trial_interior)
269 , local_interior_solution(n_dofs_trial_interior)
270 , tmp_vector_interior(n_dofs_trial_interior)
271 , tmp_vector_error_indicator(n_dofs_test)
272 , local_residual(n_dofs_test)
273 , local_dof_indices_trial_interior(n_dofs_trial_interior)
274 , local_residual_norm_squared(0.)
275 , active_cell_index(0)
276 , cell_is_local(false){};

278 /**
279 * @brief Resets the uncondensed local system to zero. The \f$M_1\f$ matrix
280 * is also reset since LAPACKFullMatrix keeps track of its inverse status and
281 * forbids inverting it again if it has already been inverted.
282 */
283 void
284 reset()
285 {
286 G_matrix = 0;
287 B_matrix = 0;
288 B_hat_matrix = 0;
289 l_vector = 0;
290 M1_matrix = 0;
291 }
292

295 // Local interior bilinear form matrix
296 LAPACKFullMatrix<double> B_matrix;
297 // Local skeleton bilinear form matrix
298 LAPACKFullMatrix<double> B_hat_matrix;
299 // Local load vector
300 Vector<double> l_vector;
301
302 // Static condensation matrix 1 \f$M_1 = B^\dagger G^{-1} B\f$
303 LAPACKFullMatrix<double> M1_matrix;
304 // Static condensation matrix 2 \f$M_2 = B^\dagger G^{-1} \hat{B}\f$
305 LAPACKFullMatrix<double> M2_matrix;
306 // Static condensation matrix 3 \f$M_3 = \hat{B}^\dagger G^{-1} \hat{B}\f$
307 LAPACKFullMatrix<double> M3_matrix;
308 // Static condensation matrix 4 \f$M_4 = B^\dagger G^{-1}\f$
309 LAPACKFullMatrix<double> M4_matrix;
310 // Static condensation matrix 5 \f$M_5 = \hat{B}^\dagger G^{-1}\f$
311 LAPACKFullMatrix<double> M5_matrix;
312 // Temporary matrix products M_2^\dagger M_1^{-1}
313 LAPACKFullMatrix<double> tmp_matrix_M2M1;
314 // Temporary matrix products M_2^\dagger M_1^{-1} M_2
315 LAPACKFullMatrix<double> tmp_matrix_M2M1M2;
316 // Temporary matrix products M_2^\dagger M_1^{-1} M_4
317 LAPACKFullMatrix<double> tmp_matrix_M2M1M4;
318
319 // Cell condensed skeleton system
320 FullMatrix<double> local_matrix;
321 // Cell condensed skeleton right-hand side
322 Vector<double> local_rhs;
323 // Cell dof indices of the skeleton trial space
324 std::vector<types::global_dof_index> local_dof_indices;
325
326 // Cell skeleton solution
327 Vector<double> local_skeleton_solution;
328 // Cell interior right-hand side
329 Vector<double> local_interior_rhs;
330 // Cell interior solution
331 Vector<double> local_interior_solution;
332 // Temporary vectors used for the interior reconstruction M_2 * x_skeleton
333 Vector<double> tmp_vector_interior;
334 // Temporary vectors used for the error indicator B * x_interior + B_hat *
335 // x_skeleton
336 Vector<double> tmp_vector_error_indicator;
337 // Cell error representation function (Riesz representation of the residual)
338 Vector<double> local_residual;
339 // Cell dof indices of the interior trial space
340 std::vector<types::global_dof_index> local_dof_indices_trial_interior;
341 // Squared norm of the DPG residual, which is the DPG error indicator of the
342 // cell
343 double local_residual_norm_squared;
344 // Index of the active cell being assembled
345 unsigned int active_cell_index;
346
347 // Boolean used to indicate if the cell being assembled is local or not
348 // This information is used to indicate to the copy_local_to_global function
349 // if it should indeed copy or not. It replaces the call to the function
350 // cell->is_locally_owned().
351 bool cell_is_local;
352};
