15/**
16 * @brief Class that creates a custom geometry that combines an extruded plate
17 * with a hole and an inner cylinder to form a cube.
18 *
19 * @tparam dim The dimension of the mesh, must be 3.
20 * @tparam spacedim The dimension of the space where the mesh is defined, must be 3.
21 */
22
23template <int dim, int spacedim>
24class GridCubeMerged
25{
26public:
27 /**
28 * @brief Constructor for the GridCubeMerged
29 *
30 * @param[in] grid_arguments A string with 4 parameters, `R_cylinder :
31 * L_plate : H : n_subdivisions`, separated by colons. `R_cylinder` is the
32 * radius of the cylinder mesh and of the hole in the extruded square plate;
33 * `L_plate` is the half of the edge length of the square; `H` is the height
34 * in the z direction; `n_subdivisions` is the number of subdivisions along
35 * the z direction.
36 */
37 GridCubeMerged(const std::string &grid_arguments);
38
39 /**
40 * @brief Generate the merged cube mesh.
41 *
42 * @param[out] triangulation The triangulation to fill with the mesh.
43 */
44 void
45 make_grid(Triangulation<dim, spacedim> &triangulation);
46
47private:
48 /// Arguments used to generate the grid
49 std::string grid_arguments;
50 /// Radius of the cylindrical mesh
51 double cylinder_radius;
52 /// Half edge length of the square in the extruded plate with hole mesh
53 double plate_half_length;
54 /// Height in the extruded direction (z axis)
55 double height;
56 /// Number of subdivisions in the extruded direction (z axis)
57 unsigned int n_subdivisions;
