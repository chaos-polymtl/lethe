18/**
19 * @brief Class that stores the information required by the assembly procedure
20 * for a Tracer advection-diffusion equation. Consequently, this class
21 *calculates the tracer (values, gradients, laplacians) and the shape function
22 * (values, gradients, laplacians) at all the gauss points for all degrees
23 * of freedom and stores it into arrays. Additionnaly, the use can request
24 * that this class gathers additional fields for physics which are coupled
25 * to the Tracer equation, such as the velocity which is required. This class
26 * serves as a seperation between the evaluation at the gauss point of the
27 * variables of interest and their use in the assembly, which is carried out
28 * by the assembler functions. For more information on this design, the reader
29 * can consult deal.II step-9
30 * "https://www.dealii.org/current/doxygen/deal.II/step_9.html". In this latter
31 * example, the scratch is a struct instead of a templated class because of the
32 * simplicity of step-9.
33 *
34 * @tparam dim An integer that denotes the dimension of the space in which
35 * the flow is solved
36 * @ingroup solvers
37 **/
38
39template <int dim>
40class TracerScratchData : public PhysicsScratchDataBase

42public:
43 /**
44 * @brief Constructor. The constructor creates the fe_values that will be used
45 * to fill the member variables of the scratch. It also allocated the
46 * necessary memory for all member variables. However, it does not do any
47 * evalution, since this needs to be done at the cell level.
48 *
49 * @param fe The FESystem used to solve the Navier-Stokes equations
50 *
51 * @param quadrature The quadrature to use for the assembly
52 *
53 * @param mapping The mapping of the domain in which the Navier-Stokes equations are solved
54 *
55 * @param uses_dg If true, the scratch is used for the discontinuous Galerkin
56 * formulation. The hessians, laplacians and previous gradients, which are
57 * only required by the stabilization of the continuous formulation, are then
58 * not computed.
59 *
60 */
61 TracerScratchData(const PhysicalPropertiesManager &properties_manager,
62 const FiniteElement<dim> &fe_tracer,
63 const Quadrature<dim> &quadrature,
64 const Quadrature<dim - 1> &face_quadrature,
65 const Mapping<dim> &mapping,
66 const FiniteElement<dim> &fe_fd,
67 const bool uses_dg);
