57template <int dim>
58QMOM<dim>::QMOM(
59 MultiphysicsInterface<dim> *multiphysics_interface,
60 const SimulationParameters<dim> &p_simulation_parameters,
61 const std::shared_ptr<parallel::DistributedTriangulationBase<dim>>
62 &p_triangulation,
63 const std::shared_ptr<SimulationControl> &p_simulation_control)
64 : AuxiliaryPhysics<dim, GlobalVectorType>()
65 , n_scalars(
66 p_simulation_parameters.multiphysics.qmom_parameters.number_of_scalars)
67 , n_moments(
68 p_simulation_parameters.multiphysics.qmom_parameters.number_of_moments)
69 , n_components(n_scalars + n_moments)
70 , multiphysics(multiphysics_interface)
71 , computing_timer(p_triangulation->get_mpi_communicator(),
72 this->pcout,
73 TimerOutput::never,
74 TimerOutput::wall_times)
75 , simulation_parameters(p_simulation_parameters)
76 , triangulation(p_triangulation)
77 , simulation_control(p_simulation_control)
78 , dof_handler(std::make_shared<DoFHandler<dim>>(*triangulation))
79{
80 const Parameters::QMOM &qmom_parameters =
81 simulation_parameters.multiphysics.qmom_parameters;
82
83 // Relative tolerance used to compare parameter values
84 constexpr double parameter_relative_tolerance = 1e-12;
85 const auto values_are_equal = [](const double first_value,
86 const double second_value) {
87 return std::abs(first_value - second_value) <=
88 parameter_relative_tolerance *
89 std::max({1., std::abs(first_value), std::abs(second_value)});
90 };
91
92 // The QMOM phase properties and operating conditions are given in SI units
93 // and are not rescaled by the dimensionality subsection
94 const Parameters::Dimensionality &dimensionality =
95 simulation_parameters.dimensionality;
96 const bool dimensionality_is_si =
97 values_are_equal(dimensionality.length, 1.) &&
98 values_are_equal(dimensionality.mass, 1.) &&
99 values_are_equal(dimensionality.time, 1.) &&
100 values_are_equal(dimensionality.temperature, 1.);
101 AssertThrow(dimensionality_is_si,
102 QMOMDimensionalityError(
103 "length = " + std::to_string(dimensionality.length) +
104 ", mass = " + std::to_string(dimensionality.mass) +
105 ", time = " + std::to_string(dimensionality.time) +
106 " and temperature = " +
107 std::to_string(dimensionality.temperature)));
108
109 // The transport uses the fluid velocity without mesh velocity correction
110 AssertThrow(!simulation_parameters.ale.enabled(),
111 ExcMessage("The QMOM physics does not support the arbitrary "
112 "Lagrangian-Eulerian (ALE) formulation yet."));
113
114 // The aqueous and organic phases form a single fluid, whose QMOM properties
115 // are used by the phase-space evolution
116 AssertThrow(
117 simulation_parameters.physical_properties_manager.get_number_of_fluids() ==
118 1,
119 QMOMInvalidNumberOfFluids(simulation_parameters.physical_properties_manager
120 .get_number_of_fluids()));
121
122 // Features of the QMOM physics that are not implemented yet
123 AssertThrow(n_scalars == 0,
124 ExcMessage("The transport of the QMOM scalars is not "
125 "implemented yet, but \"number of scalars\" is " +
126 std::to_string(n_scalars) +
127 " in the QMOM subsection. Set it to 0."));
128 // The steady state of the Lie splitting depends on the time step, and the
129 // phase-space evolution needs a physical time step
130 using TimeSteppingMethod = Parameters::SimulationControl::TimeSteppingMethod;
131 const TimeSteppingMethod method =
132 simulation_parameters.simulation_control.method;
133 AssertThrow(method == TimeSteppingMethod::bdf1 ||
134 method == TimeSteppingMethod::bdf2 ||
135 method == TimeSteppingMethod::bdf3,
136 ExcMessage("The QMOM physics only supports the bdf1, bdf2 and "
137 "bdf3 time-stepping methods, but \"method\" is " +
138 enum_to_string(method) +
139 " in the simulation control subsection. Set it to "
140 "bdf1, bdf2 or bdf3."));
141 AssertThrow(qmom_parameters.splitting_scheme ==
142 Parameters::QMOMSplittingScheme::lie,
143 ExcMessage("The strang splitting scheme of the QMOM physics is "
144 "not implemented yet. Set \"splitting scheme\" to lie "
145 "in the QMOM subsection."));
146 AssertThrow(!qmom_parameters.enable_nucleation &&
147 !qmom_parameters.enable_coalescence &&
148 !qmom_parameters.enable_growth,
149 ExcMessage("The evolution of the moments in phase space is not "
150 "implemented yet. Set \"enable nucleation\", \"enable "
151 "coalescence\" and \"enable growth\" to false in the "
152 "QMOM subsection."));
153
154 this->pcout << std::setprecision(simulation_control->get_log_precision())
155 << std::scientific;
156
157 // Names of the components, used for the output and the error table
158 component_names = qmom_parameters.scalar_names;
159 for (unsigned int i_moment = 0; i_moment < n_moments; ++i_moment)
160 component_names.emplace_back("moment_" + std::to_string(i_moment));
161
162 // All the components share the same scalar discontinuous element
163 const unsigned int qmom_degree =
164 simulation_parameters.fem_parameters.qmom_degree;
165 if (simulation_parameters.mesh.simplex)
166 {
167 // Simplex meshes (triangles and tetrahedra)
168 fe = std::make_unique<FE_SimplexDGP<dim>>(qmom_degree);
169 mapping = std::make_shared<MappingFE<dim>>(FE_SimplexP<dim>(1));
170 cell_quadrature = std::make_unique<QGaussSimplex<dim>>(qmom_degree + 1);
171 face_quadrature =
172 std::make_unique<QGaussSimplex<dim - 1>>(qmom_degree + 1);
173 }
174 else
175 {
176 // Quadrilateral and hexahedral meshes. A degree of 0 still requires a
177 // linear mapping.
178 fe = std::make_unique<FE_DGQ<dim>>(qmom_degree);
179 mapping = std::make_shared<MappingQ<dim>>(std::max(1U, qmom_degree));
180 cell_quadrature = std::make_unique<QGauss<dim>>(qmom_degree + 1);
181 face_quadrature = std::make_unique<QGauss<dim - 1>>(qmom_degree + 1);
182 }
183
184 // Functions of all the components. The scalars are given by one expression
185 // each, while the moments are computed from the particle size distribution,
186 // which always gives n_moments moments: the constructors therefore only
187 // verify the number of expressions given for the scalars.
188 initial_conditions = std::make_shared<QMOMFieldFunction<dim>>(
189 simulation_parameters.initial_condition->qmom,
190 n_scalars,
191 n_moments,
192 "initial conditions");
193 for (const auto &[boundary_id, functions] :
194 simulation_parameters.qmom_boundary_conditions.dirichlet)
195 dirichlet_functions[boundary_id] = std::make_shared<QMOMFieldFunction<dim>>(
196 functions,
197 n_scalars,
198 n_moments,
199 "Dirichlet boundary condition of boundary id " +
200 std::to_string(boundary_id));
201 source_term = std::make_shared<QMOMFieldFunction<dim>>(
202 simulation_parameters.source_term.qmom_source,
203 n_scalars,
204 n_moments,
205 "source term");
206 analytical_solution = std::make_shared<QMOMFieldFunction<dim>>(
207 simulation_parameters.analytical_solution->qmom,
208 n_scalars,
209 n_moments,
210 "analytical solution");
211
212 present_solutions.resize(n_components);
213 previous_solutions.assign(n_components,
214 std::vector<GlobalVectorType>(
215 maximum_number_of_previous_solutions()));
216 system_rhs.resize(n_components);
217
218 // Solution transfer, for mesh adaptation and restarts
219 solution_transfer =
220 std::make_shared<SolutionTransfer<dim, GlobalVectorType>>(*dof_handler);
221}

223template <int dim>
224std::vector<OutputStruct<dim, GlobalVectorType>>
225QMOM<dim>::gather_output_hook()
226{
227 std::vector<OutputStruct<dim, GlobalVectorType>> solution_output_structs;
228 solution_output_structs.reserve(n_components);
229
230 // Each component is stored in its own vector, which is output as a scalar
231 // named after the component
232 for (unsigned int component = 0; component < n_components; ++component)
233 solution_output_structs.emplace_back(

239 DataComponentInterpretation::component_is_scalar});
240
241 return solution_output_structs;

274template <int dim>
275void
276QMOM<dim>::percolate_time_vectors()
277{
278 for (unsigned int component = 0; component < n_components; ++component)
279 {
280 std::vector<GlobalVectorType> &component_previous_solutions =
281 previous_solutions[component];
282 for (unsigned int i_previous_solution =
283 component_previous_solutions.size() - 1;
284 i_previous_solution > 0;
285 --i_previous_solution)
286 component_previous_solutions[i_previous_solution] =
287 component_previous_solutions[i_previous_solution - 1];
288 component_previous_solutions[0] = present_solutions[component];
289 }

342template <int dim>
343void
344QMOM<dim>::pre_mesh_adaptation()
345{
346 // The present and previous solutions of all the components are transferred
347 // together: present solution of component 0, previous solutions of
348 // component 0, present solution of component 1, etc.
349 std::vector<const GlobalVectorType *> sol_set_transfer;
350 for (unsigned int component = 0; component < n_components; ++component)
351 {
352 sol_set_transfer.emplace_back(&present_solutions[component]);
353 for (const GlobalVectorType &previous_solution :
354 previous_solutions[component])
355 sol_set_transfer.emplace_back(&previous_solution);
356 }
357 solution_transfer->prepare_for_coarsening_and_refinement(sol_set_transfer);
358}

360template <int dim>
361void
362QMOM<dim>::post_mesh_adaptation()
363{
364 const MPI_Comm mpi_communicator = triangulation->get_mpi_communicator();
365
366 // Set up the vectors for the transfer, in the order of pre_mesh_adaptation()
367 std::vector<GlobalVectorType> distributed_solutions(
368 n_components * (1 + maximum_number_of_previous_solutions()),
369 GlobalVectorType(locally_owned_dofs, mpi_communicator));
370 std::vector<GlobalVectorType *> output_vectors;
371 output_vectors.reserve(distributed_solutions.size());
372 for (GlobalVectorType &distributed_solution : distributed_solutions)
373 output_vectors.emplace_back(&distributed_solution);
374
375 // Interpolate the present and previous solutions
376 solution_transfer->interpolate(output_vectors);
377
378 // Fix on the new mesh
379 unsigned int i_vector = 0;
380 for (unsigned int component = 0; component < n_components; ++component)
381 {
382 present_solutions[component] = distributed_solutions[i_vector++];
383 for (GlobalVectorType &previous_solution : previous_solutions[component])
384 previous_solution = distributed_solutions[i_vector++];
385 }
386}

467template <int dim>
468void
469QMOM<dim>::setup_dofs()
470{
471 dof_handler->distribute_dofs(*fe);
472
473 const MPI_Comm mpi_communicator = triangulation->get_mpi_communicator();
474
475 locally_owned_dofs = dof_handler->locally_owned_dofs();
476 locally_relevant_dofs = DoFTools::extract_locally_relevant_dofs(*dof_handler);
477
478 for (unsigned int component = 0; component < n_components; ++component)
479 {
480 present_solutions[component].reinit(locally_owned_dofs,
481 locally_relevant_dofs,
482 mpi_communicator);
483 for (GlobalVectorType &previous_solution : previous_solutions[component])
484 previous_solution.reinit(locally_owned_dofs,
485 locally_relevant_dofs,
486 mpi_communicator);
487 system_rhs[component].reinit(locally_owned_dofs, mpi_communicator);
488 }
489
490 define_constraints();
491
492 // Sparse matrix initialization. The upwind fluxes couple the degrees of
493 // freedom of neighboring cells, hence the flux sparsity pattern.
494 DynamicSparsityPattern dsp(locally_relevant_dofs);
495 DoFTools::make_flux_sparsity_pattern(*dof_handler,
496 dsp,
497 constraints,
498 /*keep_constrained_dofs = */ true);
499 SparsityTools::distribute_sparsity_pattern(dsp,
500 locally_owned_dofs,
501 mpi_communicator,
502 locally_relevant_dofs);
503 system_matrix.reinit(locally_owned_dofs,
504 locally_owned_dofs,
505 dsp,
506 mpi_communicator);
507
508 this->pcout << " Number of QMOM degrees of freedom: "
509 << n_components * dof_handler->n_dofs() << " (" << n_components
510 << " components of " << dof_handler->n_dofs() << ")" << std::endl;
511
512 // Provide the QMOM dof_handler and mapping to the multiphysics interface.
513 // The solution vectors, one per component, are not provided.
514 multiphysics->set_dof_handler(PhysicsID::qmom, dof_handler);
515 multiphysics->set_mapping(PhysicsID::qmom, mapping);
516}

518template <int dim>
519void
520QMOM<dim>::set_initial_conditions()
521{
522 initial_conditions->set_time(simulation_control->get_current_time());
523
524 GlobalVectorType interpolated_solution(locally_owned_dofs,
525 triangulation->get_mpi_communicator());
526 for (unsigned int component = 0; component < n_components; ++component)
527 {
528 // Each component is stored in its own vector, with the scalar element:
529 // the component of the initial conditions is interpolated as a scalar
530 // function
531 const ScalarFunctionFromFunctionObject<dim> component_initial_condition(
532 [&](const Point<dim> &p) {
533 return initial_conditions->value(p, component);
534 });
535 VectorTools::interpolate(*mapping,
536 *dof_handler,
537 component_initial_condition,
538 interpolated_solution);
539 constraints.distribute(interpolated_solution);
540 present_solutions[component] = interpolated_solution;
541 }
542
543 percolate_time_vectors();
544}

595template <int dim>
596void
597QMOM<dim>::assemble_system_matrix()
598{
599 TimerOutput::Scope t(computing_timer, "Assemble matrix");
600
601 system_matrix = 0.;
602
603 const DoFHandler<dim> &dof_handler_fluid =
604 multiphysics->get_dof_handler(PhysicsID::fluid_dynamics);
605
606 // Get the cell of the fluid dynamics that corresponds to a QMOM cell
607 const auto get_velocity_cell =
608 [&](const typename DoFHandler<dim>::active_cell_iterator &cell) {
609 return typename DoFHandler<dim>::active_cell_iterator(&(*triangulation),
610 cell->level(),
611 cell->index(),
612 &dof_handler_fluid);
613 };
614
615 // Coefficient of the present solution in the BDF time derivative, which
616 // includes the inverse of the time step
617 const double bdf_coefficient_present =
618 simulation_control->get_bdf_coefficients()[0];
619
620 QMOMScratchData<dim> scratch_data(*fe,
621 *cell_quadrature,
622 *face_quadrature,
623 *mapping,
624 dof_handler_fluid.get_fe(),
625 n_components);
626 QMOMCopyData copy_data(fe->n_dofs_per_cell(), n_components);
627
628 const auto cell_worker =
629 [&](const typename DoFHandler<dim>::active_cell_iterator &cell,
630 QMOMScratchData<dim> &scratch_data,
631 QMOMCopyData &copy_data) {
632 // The copy data is reset for every cell, including ghost cells, whose
633 // faces are assembled when they are finer than an owned neighbor
634 copy_data.local_matrix = 0.;
635 copy_data.face_data.clear();
636 copy_data.cell_is_local = cell->is_locally_owned();
637 if (!copy_data.cell_is_local)
638 return;
639
640 cell->get_dof_indices(copy_data.local_dof_indices);
641 scratch_data.reinit(cell);
642 reinit_velocity_with_adequate_solution(get_velocity_cell(cell),
643 scratch_data);
644
645 for (unsigned int q = 0; q < scratch_data.n_q_points; ++q)
646 {
647 const Tensor<1, dim> &velocity = scratch_data.velocity_values[q];
648 const double JxW = scratch_data.JxW[q];
649
650 for (unsigned int i = 0; i < scratch_data.n_dofs; ++i)
651 {
652 const double phi_i = scratch_data.phi[q][i];
653 const Tensor<1, dim> &grad_phi_i = scratch_data.grad_phi[q][i];
654
655 for (unsigned int j = 0; j < scratch_data.n_dofs; ++j)
656 {
657 const double phi_j = scratch_data.phi[q][j];
658
659 // Time derivative with the BDF scheme:
660 // \phi_i * \alpha_0 * \phi_j
661 copy_data.local_matrix(i, j) +=
662 phi_i * bdf_coefficient_present * phi_j * JxW;
663
664 // Advection in conservative form, integrated by parts:
665 // - (u . \nabla \phi_i) * \phi_j
666 copy_data.local_matrix(i, j) -=
667 velocity * grad_phi_i * phi_j * JxW;
668 }
669 }
670 }
671 };
672
673 const auto boundary_worker =
674 [&](const typename DoFHandler<dim>::active_cell_iterator &cell,
675 const unsigned int face_no,
676 QMOMScratchData<dim> &scratch_data,
677 QMOMCopyData &copy_data) {
678 // Only the outlet and Dirichlet boundaries contribute to the matrix: no
679 // flux crosses a boundary of type noflux
680 if (simulation_parameters.qmom_boundary_conditions.type.at(
681 cell->face(face_no)->boundary_id()) ==
682 BoundaryConditions::QMOMBoundaryType::noflux)
683 return;
684
685 scratch_data.reinit_boundary_face(cell, face_no);
686 reinit_face_velocity_with_adequate_solution(get_velocity_cell(cell),
687 face_no,
688 numbers::invalid_unsigned_int,
689 scratch_data);
690
691 const FEFaceValuesBase<dim> &fe_face =
692 scratch_data.fe_interface_values.get_fe_face_values(0);
693
694 for (unsigned int q = 0; q < scratch_data.n_face_q_points; ++q)
695 {
696 const double velocity_dot_n =
697 scratch_data.face_velocity_values[q] * fe_face.normal_vector(q);
698
699 // The advection term is only applied to the outflow part of the
700 // boundary. The inflow through the Dirichlet boundaries is known and
701 // assembled in the right-hand side.
702 if (velocity_dot_n <= 0.)
703 continue;
704
705 const double advection_factor = velocity_dot_n * fe_face.JxW(q);
706
707 for (unsigned int i = 0; i < scratch_data.n_dofs; ++i)
708 {
709 const double phi_i = fe_face.shape_value(i, q);
710 for (unsigned int j = 0; j < scratch_data.n_dofs; ++j)
711 {
712 // Outflow of the interior values:
713 // \phi_i * \phi_j * (u . n)
714 copy_data.local_matrix(i, j) +=
715 phi_i * fe_face.shape_value(j, q) * advection_factor;
716 }
717 }
718 }
719 };
720
721 const auto face_worker =
722 [&](const typename DoFHandler<dim>::active_cell_iterator &cell,
723 const unsigned int face_no,
724 const unsigned int sub_face_no,
725 const typename DoFHandler<dim>::active_cell_iterator &neighbor_cell,
726 const unsigned int neighbor_face_no,
727 const unsigned int neighbor_sub_face_no,
728 QMOMScratchData<dim> &scratch_data,
729 QMOMCopyData &copy_data) {
730 scratch_data.reinit_internal_face(cell,
731 face_no,
732 sub_face_no,
733 neighbor_cell,
734 neighbor_face_no,
735 neighbor_sub_face_no);
736 reinit_face_velocity_with_adequate_solution(get_velocity_cell(cell),
737 face_no,
738 sub_face_no,
739 scratch_data);
740
741 const unsigned int n_interface_dofs = scratch_data.n_interface_dofs;
742
743 // Pad copy_data memory for the internal faces elementary matrices
744 QMOMCopyData::FaceData &copy_data_face =
745 copy_data.face_data.emplace_back();
746 copy_data_face.joint_dof_indices =
747 scratch_data.fe_interface_values.get_interface_dof_indices();
748 copy_data_face.face_matrix.reinit(n_interface_dofs, n_interface_dofs);
749
750 const std::vector<double> &JxW =
751 scratch_data.fe_interface_values.get_JxW_values();
752 const std::vector<Tensor<1, dim>> &normals =
753 scratch_data.fe_interface_values.get_normal_vectors();
754
755 for (unsigned int q = 0; q < scratch_data.n_face_q_points; ++q)
756 {
757 // The normal points from the "here" cell to the "there" cell
758 const double velocity_dot_n =
759 scratch_data.face_velocity_values[q] * normals[q];
760
761 // Shape functions of the upwind cell
762 const auto &phi_upwind = (velocity_dot_n > 0.) ?
763 scratch_data.face_phi_here[q] :
764 scratch_data.face_phi_there[q];
765 const auto &jump_phi = scratch_data.face_jump_phi[q];
766
767 const double advection_factor = velocity_dot_n * JxW[q];
768
769 for (unsigned int i = 0; i < n_interface_dofs; ++i)
770 {
771 const double jump_phi_i = jump_phi[i];
772 for (unsigned int j = 0; j < n_interface_dofs; ++j)
773 {
774 // Advection term with upwinding:
775 // [\phi_i] * \phi_j^{upwind} * (u . n)
776 copy_data_face.face_matrix(i, j) +=
777 jump_phi_i * phi_upwind[j] * advection_factor;
778 }
779 }
780 }
781 };
782
783 // A face between a locally owned and a ghost cell is assembled by the two
784 // processes that own its cells. Each process only adds the rows that it
785 // owns, so that every row receives the flux of each of its faces exactly
786 // once.
787 std::vector<types::global_dof_index> row_columns;
788 std::vector<double> row_values;
789 const auto copier = [&](const QMOMCopyData &copy_data) {
790 if (copy_data.cell_is_local)
791 constraints.distribute_local_to_global(copy_data.local_matrix,
792 copy_data.local_dof_indices,
793 system_matrix);
794
795 for (const QMOMCopyData::FaceData &copy_data_face : copy_data.face_data)
796 {
797 const unsigned int n_interface_dofs =
798 copy_data_face.joint_dof_indices.size();
799 for (unsigned int i = 0; i < n_interface_dofs; ++i)
800 {
801 const types::global_dof_index row =
802 copy_data_face.joint_dof_indices[i];
803 if (!locally_owned_dofs.is_element(row))
804 continue;
805
806 row_columns.clear();
807 row_values.clear();
808 for (unsigned int j = 0; j < n_interface_dofs; ++j)
809 {
810 row_columns.push_back(copy_data_face.joint_dof_indices[j]);
811 row_values.push_back(copy_data_face.face_matrix(i, j));
812 }
813 system_matrix.add(row, row_columns, row_values);
814 }
815 }
816 };
817
818 MeshWorker::mesh_loop(dof_handler->begin_active(),
819 dof_handler->end(),
820 cell_worker,
821 copier,
822 scratch_data,
823 copy_data,
824 MeshWorker::assemble_own_cells |
825 MeshWorker::assemble_ghost_cells |
826 MeshWorker::assemble_boundary_faces |
827 MeshWorker::assemble_own_interior_faces_once |
828 MeshWorker::assemble_ghost_faces_both,
829 boundary_worker,
830 face_worker);
831
832 system_matrix.compress(VectorOperation::add);
833}

835template <int dim>
836void
837QMOM<dim>::assemble_system_rhs()
838{
839 TimerOutput::Scope t(computing_timer, "Assemble RHS");
840
841 for (GlobalVectorType &component_rhs : system_rhs)
842 component_rhs = 0.;
843
844 // Update the source term time
845 source_term->set_time(simulation_control->get_current_time());
846
847 const DoFHandler<dim> &dof_handler_fluid =
848 multiphysics->get_dof_handler(PhysicsID::fluid_dynamics);
849
850 // Get the cell of the fluid dynamics that corresponds to a QMOM cell
851 const auto get_velocity_cell =
852 [&](const typename DoFHandler<dim>::active_cell_iterator &cell) {
853 return typename DoFHandler<dim>::active_cell_iterator(&(*triangulation),
854 cell->level(),
855 cell->index(),
856 &dof_handler_fluid);
857 };
858
859 // Number of previous solutions of the BDF time derivative
860 const unsigned int n_previous_solutions =
861 number_of_previous_solutions(simulation_control->get_assembly_method());
862 const Vector<double> &bdf_coefficients =
863 simulation_control->get_bdf_coefficients();
864
865 QMOMScratchData<dim> scratch_data(*fe,
866 *cell_quadrature,
867 *face_quadrature,
868 *mapping,
869 dof_handler_fluid.get_fe(),
870 n_components);
871 QMOMCopyData copy_data(fe->n_dofs_per_cell(), n_components);
872
873 const auto cell_worker =
874 [&](const typename DoFHandler<dim>::active_cell_iterator &cell,
875 QMOMScratchData<dim> &scratch_data,
876 QMOMCopyData &copy_data) {
877 for (Vector<double> &component_rhs : copy_data.local_rhs)
878 component_rhs = 0.;
879 copy_data.cell_is_local = cell->is_locally_owned();
880 if (!copy_data.cell_is_local)
881 return;
882
883 cell->get_dof_indices(copy_data.local_dof_indices);
884 scratch_data.reinit(cell);
885 scratch_data.reinit_previous_solutions(previous_solutions);
886 scratch_data.reinit_source(*source_term);
887
888 for (unsigned int q = 0; q < scratch_data.n_q_points; ++q)
889 {
890 const double JxW = scratch_data.JxW[q];
891
892 for (unsigned int component = 0; component < n_components;
893 ++component)
894 {
895 // Previous solutions of the BDF scheme:
896 // sum_{p >= 1} \alpha_p * \zeta_c^{n+1-p}
897 double previous_terms = 0.;
898 for (unsigned int p = 0; p < n_previous_solutions; ++p)
899 previous_terms += bdf_coefficients[p + 1] *
900 scratch_data.previous_values[component][p][q];
901
902 const double source = scratch_data.source_values[q][component];
903
904 Vector<double> &local_rhs = copy_data.local_rhs[component];
905 for (unsigned int i = 0; i < scratch_data.n_dofs; ++i)
906 {
907 const double phi_i = scratch_data.phi[q][i];
908
909 // Time derivative with the BDF scheme, previous solutions:
910 // - \phi_i * sum_{p >= 1} \alpha_p * \zeta_c^{n+1-p}
911 local_rhs(i) -= phi_i * previous_terms * JxW;
912
913 // Source term: \phi_i * S_c
914 local_rhs(i) += phi_i * source * JxW;
915 }
916 }
917 }
918 };
919
920 const auto boundary_worker =
921 [&](const typename DoFHandler<dim>::active_cell_iterator &cell,
922 const unsigned int face_no,
923 QMOMScratchData<dim> &scratch_data,
924 QMOMCopyData &copy_data) {
925 // Only the inflow through the Dirichlet boundaries contributes to the
926 // right-hand side
927 const types::boundary_id boundary_id = cell->face(face_no)->boundary_id();
928 if (simulation_parameters.qmom_boundary_conditions.type.at(boundary_id) !=
929 BoundaryConditions::QMOMBoundaryType::dirichlet)
930 return;
931
932 scratch_data.reinit_boundary_face(cell, face_no);
933 reinit_face_velocity_with_adequate_solution(get_velocity_cell(cell),
934 face_no,
935 numbers::invalid_unsigned_int,
936 scratch_data);
937
938 const FEFaceValuesBase<dim> &fe_face =
939 scratch_data.fe_interface_values.get_fe_face_values(0);
940 const QMOMFieldFunction<dim> &dirichlet_function =
941 *dirichlet_functions.at(boundary_id);
942 Vector<double> dirichlet_values(n_components);
943
944 for (unsigned int q = 0; q < scratch_data.n_face_q_points; ++q)
945 {
946 const double velocity_dot_n =
947 scratch_data.face_velocity_values[q] * fe_face.normal_vector(q);
948
949 // The Dirichlet values only enter the domain through the inflow part
950 // of the boundary. The outflow is assembled in the matrix.
951 if (velocity_dot_n >= 0.)
952 continue;
953
954 // All the components are evaluated at once, since the moments are
955 // computed together from the particle size distribution
956 dirichlet_function.vector_value(fe_face.quadrature_point(q),
957 dirichlet_values);
958
959 const double advection_factor = velocity_dot_n * fe_face.JxW(q);
960
961 for (unsigned int i = 0; i < scratch_data.n_dofs; ++i)
962 {
963 const double phi_i = fe_face.shape_value(i, q);
964 for (unsigned int component = 0; component < n_components;
965 ++component)
966 {
967 // Inflow of the Dirichlet values:
968 // - \phi_i * \zeta_{c,D} * (u . n)
969 copy_data.local_rhs[component](i) -=
970 phi_i * dirichlet_values[component] * advection_factor;
971 }
972 }
973 }
974 };
975
976 const auto copier = [&](const QMOMCopyData &copy_data) {
977 if (!copy_data.cell_is_local)
978 return;
979
980 for (unsigned int component = 0; component < n_components; ++component)
981 constraints.distribute_local_to_global(copy_data.local_rhs[component],
982 copy_data.local_dof_indices,
983 system_rhs[component]);
984 };
985
986 // The right-hand side has no interior face terms: the upwind fluxes between
987 // cells only involve the unknown solution, and are assembled in the matrix
988 MeshWorker::mesh_loop(dof_handler->begin_active(),
989 dof_handler->end(),
990 cell_worker,
991 copier,
992 scratch_data,
993 copy_data,
994 MeshWorker::assemble_own_cells |
995 MeshWorker::assemble_boundary_faces,
996 boundary_worker);
997
998 for (GlobalVectorType &component_rhs : system_rhs)
999 component_rhs.compress(VectorOperation::add);
1000}

1112void
1113QMOM<dim>::advance_physical_space()
1114{

1116
1117 const Parameters::LinearSolver &linear_solver_parameters =
1118 simulation_parameters.linear_solver.at(PhysicsID::qmom);
1119 const bool verbose =
1120 linear_solver_parameters.verbosity != Parameters::Verbosity::quiet;
1121 const double rescale_metric = this->get_residual_rescale_metric();
1122
1123 GlobalVectorType completely_distributed_solution(
1124 locally_owned_dofs, triangulation->get_mpi_communicator());

1126 triangulation->get_mpi_communicator());
1127

1129

1144 system_rhs[component]) /

1155 linear_solver_tolerance * rescale_metric,

1157 true);

1159 false, linear_solver_parameters.max_krylov_vectors);
1160 TrilinosWrappers::SolverGMRES solver(solver_control, solver_parameters);
1161
1162 if (preconditioner_type ==
1163 Parameters::LinearSolver::PreconditionerType::ilu)
1164 solver.solve(system_matrix,
1165 completely_distributed_solution,
1166 system_rhs[component],

1170 completely_distributed_solution,
1171 system_rhs[component],
1172 *amg_preconditioner);

1174 if (verbose)
1175 this->pcout << " -Iterative solver of " << component_names[component]
1176 << " took : " << solver_control.last_step()
1177 << " steps to reach a residual norm of "
1178 << solver_control.last_value() / rescale_metric
1179 << std::endl;

1181 constraints.distribute(completely_distributed_solution);
1182 present_solutions[component] = completely_distributed_solution;
