64template <int dim>
65void
66CLSScratchData<dim>::allocate()
67{
68 // Initialize size of arrays
69 this->n_q_points = fe_values_cls.get_quadrature().size();
70 this->n_dofs = fe_values_cls.get_fe().n_dofs_per_cell();
71
72 // Initialize arrays related to quadrature
73 this->JxW.reinit(n_q_points);
74
75 // Initialize arrays related to velocity and pressure
76 this->velocities_fd.first_vector_component = 0;
77 // Velocity
78 this->velocity_values = std::vector<Tensor<1, dim>>(this->n_q_points);
79 this->previous_velocity_values = std::vector<std::vector<Tensor<1, dim>>>(
80 maximum_number_of_previous_solutions(),
81 std::vector<Tensor<1, dim>>(this->n_q_points));
82 this->velocity_gradient_values =
83 std::vector<Tensor<2, dim>>(this->n_q_points);
84 this->velocity_divergences = std::vector<double>(n_q_points);
85
86 this->present_phase_values = std::vector<double>(this->n_q_points);
87 this->phase_gradients = std::vector<Tensor<1, dim>>(this->n_q_points);
88 this->previous_phase_gradients =
89 std::vector<Tensor<1, dim>>(this->n_q_points);
90
91 this->phase_laplacians = std::vector<double>(this->n_q_points);
92
93 // Velocity for BDF schemes
94 this->previous_phase_values =
95 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
96 std::vector<double>(this->n_q_points));
97
98 // Initialize arrays related to shape functions
99 this->phi.reinit(n_q_points, n_dofs);
100 this->grad_phi.reinit(n_q_points, n_dofs);
101 this->hess_phi.reinit(n_q_points, n_dofs);
102 this->laplacian_phi.reinit(n_q_points, n_dofs);
103}

105template <int dim>
106template <typename VectorType>
107void
108CLSScratchData<dim>::reinit(
109 const typename DoFHandler<dim>::active_cell_iterator &cell,
110 const VectorType &current_solution,
111 const std::vector<VectorType> &previous_solutions)
112{
113 fe_values_cls.reinit(cell);
114 this->quadrature_points = fe_values_cls.get_quadrature_points();
115 auto &fe_cls = fe_values_cls.get_fe();
116
117 // Compute cell diameter
118 double cell_measure =
119 compute_cell_measure_with_JxW(this->fe_values_cls.get_JxW_values());
120 this->cell_size = compute_cell_diameter<dim>(cell_measure, fe_cls.degree);
121
122 fe_values_cls.get_function_values(current_solution,
123 this->present_phase_values);
124 fe_values_cls.get_function_gradients(current_solution, this->phase_gradients);
125 fe_values_cls.get_function_laplacians(current_solution,
126 this->phase_laplacians);
127 fe_values_cls.get_function_gradients(previous_solutions[0],
128 this->previous_phase_gradients);
129
130
131 // Gather previous cls values
132 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
133 {
134 fe_values_cls.get_function_values(previous_solutions[p],
135 this->previous_phase_values[p]);
136 }
137
138 for (unsigned int q = 0; q < this->n_q_points; ++q)
139 {
140 this->JxW[q] = fe_values_cls.JxW(q);
141
142 for (unsigned int k = 0; k < this->n_dofs; ++k)
143 {
144 // Shape function
145 this->phi[q][k] = fe_values_cls.shape_value(k, q);
146 this->grad_phi[q][k] = fe_values_cls.shape_grad(k, q);
147 this->hess_phi[q][k] = fe_values_cls.shape_hessian(k, q);
148 this->laplacian_phi[q][k] = trace(this->hess_phi[q][k]);
149 }
150 }
151}

154template <typename VectorType>
155void
156CLSScratchData<dim>::reinit_internal_face(
157 const typename DoFHandler<dim>::active_cell_iterator &cell,
158 const unsigned int &face_no,
159 const unsigned int &sub_face_no,
160 const typename DoFHandler<dim>::active_cell_iterator &neighbor_cell,
161 const unsigned int &neighbor_face_no,
162 const unsigned int &neighbor_sub_face_no,
163 const VectorType &current_solution)
164{
165 fe_interface_values_cls.reinit(cell,
166 face_no,
167 sub_face_no,
168 neighbor_cell,
169 neighbor_face_no,
170 neighbor_sub_face_no);
171 face_quadrature_points = fe_interface_values_cls.get_quadrature_points();
172
173 n_interface_dofs = fe_interface_values_cls.n_current_interface_dofs();
174
175 // BB TODO : Preallocate memory here
176 values_here.resize(face_quadrature_points.size());
177 values_there.resize(face_quadrature_points.size());
178 phase_value_jump.resize(face_quadrature_points.size());
179
180 fe_interface_values_cls.get_fe_face_values(0).get_function_values(
181 current_solution, values_here);
182 fe_interface_values_cls.get_fe_face_values(1).get_function_values(
183 current_solution, values_there);
184
185 fe_interface_values_cls.get_jump_in_function_values(current_solution,
186 phase_value_jump);
187}

189template <int dim>
190template <typename VectorType>
191void
192CLSScratchData<dim>::reinit_velocity(
193 const typename DoFHandler<dim>::active_cell_iterator &cell,
194 const VectorType &current_solution,
195 const std::vector<VectorType> &previous_solutions,
196 const Parameters::ALE<dim> &ale)
197{
198 fe_values_fd.reinit(cell);
199
200 fe_values_fd[velocities_fd].get_function_values(current_solution,
201 velocity_values);
202 fe_values_fd[velocities_fd].get_function_gradients(current_solution,
203 velocity_gradient_values);
204
205 for (unsigned int q = 0; q < this->n_q_points; ++q)
206 {
207 this->velocity_divergences[q] = trace(this->velocity_gradient_values[q]);
208 }
209
210 // Gather previous velocity values
211 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
212 {
213 fe_values_fd[velocities_fd].get_function_values(
214 previous_solutions[p], this->previous_velocity_values[p]);
215 }
216
217 if (ale.enabled())
218 {
219 // ALE enabled, so extract the ALE velocity and subtract it from the
220 // velocity obtained from the fluid dynamics
221 Tensor<1, dim> velocity_ale;
222 std::shared_ptr<Functions::ParsedFunction<dim>> velocity_ale_function =
223 ale.velocity;
224 Vector<double> velocity_ale_vector(dim);
225
226 for (unsigned int q = 0; q < n_q_points; ++q)
227 {
228 velocity_ale_function->vector_value(quadrature_points[q],
229 velocity_ale_vector);
230 for (int d = 0; d < dim; ++d)
231 velocity_ale[d] = velocity_ale_vector[d];
232
233 velocity_values[q] -= velocity_ale;
234
235 for (unsigned int p = 0; p < previous_solutions.size(); ++p)
236 {
237 this->previous_velocity_values[p][q] -= velocity_ale;
238 }
239 }
240 }
241
242 // Extrapolate velocity to t+dt using the BDF scheme if the simulation is
243 // transient
244 const auto method = this->simulation_control->get_assembly_method();
245 if (time_stepping_is_bdf(method))
246 {
247 // Extrapolate velocity
248 std::vector<double> time_vector =
249 this->simulation_control->get_simulation_times();
250 bdf_extrapolate(time_vector,
251 this->previous_velocity_values,
252 number_of_previous_solutions(method),
253 this->velocity_values);
254 }
255}

257template <int dim>
258template <typename VectorType>
259void
260CLSScratchData<dim>::reinit_face_velocity(
261 const typename DoFHandler<dim>::active_cell_iterator &velocity_cell,
262 const unsigned int &face_no,
263 const VectorType &velocity_solution,
264 const Parameters::ALE<dim> &ale)
265{
266 fe_face_values_fd.reinit(velocity_cell, face_no);
267
268 // BB note : Array could be pre-allocated
269 face_velocity_values.resize(face_quadrature_points.size());
270
271 fe_face_values_fd[velocities_fd].get_function_values(velocity_solution,
272 face_velocity_values);
273
274 if (!ale.enabled())
275 return;
276
277 // ALE enabled, so extract the ALE velocity and subtract it from the
278 // velocity obtained from the fluid dynamics
279 Tensor<1, dim> velocity_ale;
280 std::shared_ptr<Functions::ParsedFunction<dim>> velocity_ale_function =
281 ale.velocity;
282 Vector<double> velocity_ale_vector(dim);
283
284 for (unsigned int q = 0; q < face_quadrature_points.size(); ++q)
285 {
286 velocity_ale_function->vector_value(face_quadrature_points[q],
287 velocity_ale_vector);
288 for (int d = 0; d < dim; ++d)
289 velocity_ale[d] = velocity_ale_vector[d];
290
291 face_velocity_values[q] -= velocity_ale;
292 }
293}
