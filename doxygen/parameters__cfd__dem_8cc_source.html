191 template <int dim>
192 void
193 VoidFractionParameters<dim>::parse_parameters(ParameterHandler &prm)
194 {
195 prm.enter_subsection("void fraction");
196 const std::string op = prm.get("mode");
197 if (op == "function")
198 mode = Parameters::VoidFractionMode::function;
199 else if (op == "pcm")
200 mode = Parameters::VoidFractionMode::pcm;
201 else if (op == "qcm")
202 mode = Parameters::VoidFractionMode::qcm;
203 else if (op == "spm")
204 mode = Parameters::VoidFractionMode::spm;
205 else
206 throw(std::runtime_error("Invalid void fraction calculation scheme"));
207 prm.enter_subsection("function");
208 void_fraction.parse_parameters(prm);
209 prm.leave_subsection();
210
211 read_dem = prm.get_bool("read dem");
212 dem_file_name = prm.get("dem file name");
213 l2_smoothing_length = prm.get_double("l2 smoothing length");
214 particle_refinement_factor = prm.get_integer("particle refinement factor");
215 qcm_smoothing_length = prm.get_double("qcm smoothing length");
216 qcm_sphere_equal_cell_volume = prm.get_bool("qcm sphere equal cell volume");
217
218 const std::string qcm_filter_type_op = prm.get("qcm filter type");
219 if (qcm_filter_type_op == "spherical")
220 qcm_filter_type = Parameters::QCMFilterType::spherical;
221 else if (qcm_filter_type_op == "gaussian")
222 qcm_filter_type = Parameters::QCMFilterType::gaussian;
223 else
224 throw(std::runtime_error(
225 "Invalid QCM filter type. Options are 'spherical' or 'gaussian'"));
226
227 const std::string quadrature_rule_op = prm.get("quadrature rule");
228
229 if (quadrature_rule_op == "gauss")
230 quadrature_rule = Parameters::VoidFractionQuadratureRule::gauss;
231 else if (quadrature_rule_op == "gauss-lobatto")
232 quadrature_rule = Parameters::VoidFractionQuadratureRule::gauss_lobatto;
233 else
234 throw(std::runtime_error(
235 "Invalid quadrature rule for the void fraction calculation scheme. Options are 'gauss' or 'gauss-lobatto'"));
236
237 n_quadrature_points = prm.get_integer("n quadrature points");
238
239 project_particle_velocity = prm.get_bool("project particle velocity");
240
241 prm.leave_subsection();
242 }

366 void
367 CFDDEM::parse_parameters(ParameterHandler &prm)
368 {
369 prm.enter_subsection("cfd-dem");
370 grad_div = prm.get_bool("grad div");
371 void_fraction_time_derivative =
372 prm.get_bool("void fraction time derivative");
373 interpolated_void_fraction = prm.get_bool("interpolated void fraction");
374 drag_force = prm.get_bool("drag force");
375 buoyancy_force = prm.get_bool("buoyancy force");
376 shear_force = prm.get_bool("shear force");
377 pressure_force = prm.get_bool("pressure force");
378 saffman_lift_force = prm.get_bool("saffman lift force");
379 magnus_lift_force = prm.get_bool("magnus lift force");
380 rotational_viscous_torque = prm.get_bool("rotational viscous torque");
381 vortical_viscous_torque = prm.get_bool("vortical viscous torque");
382 coupling_frequency = prm.get_integer("coupling frequency");
383 fraction_of_rayleigh_time = prm.get_double("fraction rayleigh time");
384 cstar = prm.get_double("grad-div length scale");
385 implicit_stabilization = prm.get_bool("implicit stabilization");
386 particle_statistics = prm.get_bool("particle statistics");
387 project_particle_forces = prm.get_bool("project particle forces");
388
389 const std::string it_ctrl = prm.get("dem iteration control");
390 if (it_ctrl == "number of iterations")
391 dem_iteration_control = SubSimulationControlDEM::DEMSubIterationLogic::
392 fixed_number_of_iterations;
393 else if (it_ctrl == "fraction of rayleigh time")
394 dem_iteration_control = SubSimulationControlDEM::DEMSubIterationLogic::
395 fixed_fraction_of_rayleigh_time_step;
396 else
397 AssertThrow(
398 false,
399 ExcMessage(
400 "An invalid dem iteration control strategy was parsed. Simulation will now stop."));
401
402 const std::string op = prm.get("drag model");
403 if (op == "difelice")
404 drag_model = Parameters::DragModel::difelice;
405 else if (op == "rong")
406 drag_model = Parameters::DragModel::rong;
407 else if (op == "dallavalle")
408 drag_model = Parameters::DragModel::dallavalle;
409 else if (op == "kochhill")
410 drag_model = Parameters::DragModel::kochhill;
411 else if (op == "beetstra")
412 drag_model = Parameters::DragModel::beetstra;
413 else if (op == "gidaspow")
414 drag_model = Parameters::DragModel::gidaspow;
415 else
416 AssertThrow(false, ExcMessage("Invalid drag model"));
417
418 const std::string drag_coupling_str = prm.get("drag coupling");
419 if (drag_coupling_str == "implicit")
420 drag_coupling = Parameters::DragCoupling::fully_implicit;
421 else if (drag_coupling_str == "explicit")
422 drag_coupling = Parameters::DragCoupling::fully_explicit;
423 else if (drag_coupling_str == "semi-implicit")
424 drag_coupling = Parameters::DragCoupling::semi_implicit;
425 else
426 AssertThrow(false, ExcMessage("Drag coupling formulation"));
427
428 const std::string op1 = prm.get("vans model");
429 if (op1 == "modelA")
430 vans_model = Parameters::VANSModel::modelA;
431 else if (op1 == "modelB")
432 vans_model = Parameters::VANSModel::modelB;
433 else
434 throw(std::runtime_error(
435 "Invalid vans model. Valid choices are modelA and modelB."));
436 prm.leave_subsection();
437 }
