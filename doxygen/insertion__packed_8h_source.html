13template <int dim, typename PropertiesIndex>
14class InsertionPacked : public Insertion<dim, PropertiesIndex>
15{
16public:
17 /**
18 * @brief Insert particles using the packed insertion method.
19 *
20 * Particles are inserted by randomly generating points inside the insertion
21 * region and assigning them particle properties (such as their diameter).
22 * Because the generated configuration may contain particle-particle and
23 * particle-wall overlaps, particles are iteratively displaced along the
24 * corresponding contact normal directions until all overlaps are resolved.
25 *
26 * @param size_distribution_object_container Contains all distribution for each
27 * particle type
28 * @param triangulation Triangulation to access the cells in which the
29 * particles are inserted
30 * @param dem_parameters DEM parameters declared in the .prm file
31 */
32 InsertionPacked(
33 const std::vector<std::shared_ptr<Distribution>>
34 &size_distribution_object_container,
35 const parallel::distributed::Triangulation<dim> &triangulation,
36 const DEMSolverParameters<dim> &dem_parameters);
37
38 /**
39 * @brief Carries out the packed insertion of particles.
40 *
41 * @param particle_handler The particle handler of particles which are being
42 * inserted
43 * @param triangulation Triangulation to access the cells in which the
44 * particles are inserted
45 * @param dem_parameters DEM parameters declared in the .prm file
46 *
47 */
48 virtual void
49 insert(Particles::ParticleHandler<dim> &particle_handler,
50 const parallel::distributed::Triangulation<dim> &triangulation,
51 const DEMSolverParameters<dim> &dem_parameters) override;

79 /**
80 * @brief Updates the previous positions of particles in all locally owned cells.
81 *
82 * This method iterates through all active cells in the triangulation,
83 * identifying locally owned cells. For each particle in these cells, it
84 * retrieves the current particle position and stores it into the particle's
85 * properties. The position is stored within certain indices that are
86 * conventionally designated as velocity components (v_x, v_y, and v_z for 3D
87 * configurations). This allows the tracking of previous particle positions
88 * without additional data structures. This function is required for the
89 * packed insertion method.
90 *
91 * @param[out] particle_handler The particle handler of particles which are
92 * being inserted.
93 *
94 */
95 static void
96 update_previous_position(Particles::ParticleHandler<dim> &particle_handler);
97
98 /**
99 * @brief Restricts the displacement of particles to a maximum allowable
100 * value, preventing excessive movement during the simulation.
101 *
102 * This method iterates over all particles and checks their displacement
103 * norm relative to their previous positions. If the displacement exceeds
104 * the maximum particle diameter, their positions are clamped such that
105 * the displacement does not exceed this threshold. The displacement
106 * values are updated accordingly for each particle.
107 *
108 * @param[in,out] particle_handler The particle handler of particles which are
109 * being inserted.
110 * @param[in] max_disp Maximum displacement that a particle is able to move
111 * during a single pseudo time-step.
112 * @param[in] displacements Vector that stored the displacement of each
113 * particle since the last contact detection time step.
114 */
115 static void
116 clamp_displacement(Particles::ParticleHandler<dim> &particle_handler,
117 const double max_disp,
118 std::vector<double> &displacements);
119
120private:
121 /**
122 * @brief Generate an insertion point inside an insertion box.

126 * @param insertion_information Insertion parameters read from the .prm file.
127 */
128 void
129 generate_insertion_location(Point<dim> &insertion_location,
130 std::mt19937 &rng,
131 const InsertionInfo<dim> &insertion_information);

133

137 std::vector<double> axis_min, axis_max;

139 /// Function used to insert particles inside or outside an arbitrary shape.
