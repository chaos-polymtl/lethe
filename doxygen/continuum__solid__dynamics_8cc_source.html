32template <int dim>
33ContinuumSolidDynamics<dim>::ContinuumSolidDynamics(
34 MultiphysicsInterface<dim> *multiphysics_interface,
35 const SimulationParameters<dim> &p_simulation_parameters,
36 const std::shared_ptr<parallel::DistributedTriangulationBase<dim>>
37 &p_triangulation,
38 const std::shared_ptr<SimulationControl> &p_simulation_control)
39 : AuxiliaryPhysics<dim, GlobalVectorType>(
40 p_simulation_parameters.physics_solving_strategy.at(
41 PhysicsID::continuum_solid_dynamics))
42 , multiphysics(multiphysics_interface)
43 , computing_timer(p_triangulation->get_mpi_communicator(),
44 this->pcout,
45 TimerOutput::never,
46 TimerOutput::wall_times)
47 , simulation_parameters(p_simulation_parameters)
48 , triangulation(p_triangulation)
49 , simulation_control(p_simulation_control)
50 , dof_handler(std::make_shared<DoFHandler<dim>>(*triangulation))
51 , solid_velocity_extractor(0)
52 , solid_volume_fraction_extractor(dim)
53{
54 AssertThrow(!simulation_parameters.mesh.simplex,
55 ExcMessage("The solid phase physics does not support simplex "
56 "meshes yet."));
57
58 this->pcout << std::setprecision(simulation_control->get_log_precision())
59 << std::scientific;
60
61 const unsigned int velocity_degree =
62 simulation_parameters.fem_parameters.solid_velocity_degree;
63 const unsigned int volume_fraction_degree =
64 simulation_parameters.fem_parameters.solid_volume_fraction_degree;
65
66 // Solid velocity (dim components) followed by the solid volume fraction
67 fe = std::make_shared<FESystem<dim>>(FE_Q<dim>(velocity_degree),
68 dim,
69 FE_Q<dim>(volume_fraction_degree),
70 1);
71 // fe->degree is the highest degree of the two fields
72 mapping = std::make_shared<MappingQ<dim>>(fe->degree);
73 cell_quadrature = std::make_shared<QGauss<dim>>(fe->degree + 1);
74 face_quadrature = std::make_shared<QGauss<dim - 1>>(fe->degree + 1);
75
76 present_solution = std::make_shared<GlobalVectorType>();
77 solution_transfer =
78 std::make_shared<SolutionTransfer<dim, GlobalVectorType>>(*dof_handler);
79
80 previous_solutions = std::make_shared<std::vector<GlobalVectorType>>(
81 maximum_number_of_previous_solutions());
82 previous_solutions_transfer.reserve(previous_solutions->size());
83 for (unsigned int i = 0; i < previous_solutions->size(); ++i)
84 previous_solutions_transfer.emplace_back(
85 SolutionTransfer<dim, GlobalVectorType>(*dof_handler));
86}

110}

160template <int dim>
161void
162ContinuumSolidDynamics<dim>::post_mesh_adaptation()
163{
164 const auto mpi_communicator = triangulation->get_mpi_communicator();
165
166 GlobalVectorType tmp(locally_owned_dofs, mpi_communicator);
167 solution_transfer->interpolate(tmp);
168 nonzero_constraints.distribute(tmp);
169 *present_solution = tmp;
170
171 for (unsigned int i = 0; i < previous_solutions->size(); ++i)
172 {
173 GlobalVectorType tmp_previous_solution(locally_owned_dofs,
174 mpi_communicator);
175 previous_solutions_transfer[i].interpolate(tmp_previous_solution);
176 nonzero_constraints.distribute(tmp_previous_solution);
177 (*previous_solutions)[i] = tmp_previous_solution;
178 }
179}

195void
196ContinuumSolidDynamics<dim>::write_checkpoint()
197{
198 std::vector<const GlobalVectorType *> sol_set_transfer;
199
200 solution_transfer =
201 std::make_shared<SolutionTransfer<dim, GlobalVectorType>>(*dof_handler);
202
203 sol_set_transfer.emplace_back(&(*present_solution));
204 for (const auto &previous_solution : *previous_solutions)
205 sol_set_transfer.emplace_back(&previous_solution);
206
207 solution_transfer->prepare_for_serialization(sol_set_transfer);
208}

210template <int dim>
211void
212ContinuumSolidDynamics<dim>::read_checkpoint()
213{
214 const auto mpi_communicator = triangulation->get_mpi_communicator();
215 this->pcout << "Reading solid phase checkpoint" << std::endl;
216
217 std::vector<GlobalVectorType *> input_vectors(1 + previous_solutions->size());
218 GlobalVectorType distributed_system(locally_owned_dofs, mpi_communicator);
219 input_vectors[0] = &distributed_system;
220
221 std::vector<GlobalVectorType> distributed_previous_solutions;
222 distributed_previous_solutions.reserve(previous_solutions->size());
223 for (unsigned int i = 0; i < previous_solutions->size(); ++i)
224 {
225 distributed_previous_solutions.emplace_back(
226 GlobalVectorType(locally_owned_dofs, mpi_communicator));
227 input_vectors[i + 1] = &distributed_previous_solutions[i];
228 }
229
230 solution_transfer->deserialize(input_vectors);
231
232 *present_solution = distributed_system;
233 for (unsigned int i = 0; i < previous_solutions->size(); ++i)
234 (*previous_solutions)[i] = distributed_previous_solutions[i];
235}

245void
246ContinuumSolidDynamics<dim>::define_constraints()
247{
248 nonzero_constraints.clear();
249 nonzero_constraints.reinit(locally_owned_dofs, locally_relevant_dofs);
250 DoFTools::make_hanging_node_constraints(*dof_handler, nonzero_constraints);
251 nonzero_constraints.close();
252
253 zero_constraints.clear();
254 zero_constraints.reinit(locally_owned_dofs, locally_relevant_dofs);
255 DoFTools::make_hanging_node_constraints(*dof_handler, zero_constraints);
256 zero_constraints.close();

259template <int dim>
260void
261ContinuumSolidDynamics<dim>::setup_dofs()
262{
263 dof_handler->distribute_dofs(*fe);
264 DoFRenumbering::Cuthill_McKee(*dof_handler);
265
266 const auto mpi_communicator = triangulation->get_mpi_communicator();
267
268 locally_owned_dofs = dof_handler->locally_owned_dofs();
269 locally_relevant_dofs = DoFTools::extract_locally_relevant_dofs(*dof_handler);
270
271 present_solution->reinit(locally_owned_dofs,
272 locally_relevant_dofs,
273 mpi_communicator);
274 for (auto &solution : *previous_solutions)
275 solution.reinit(locally_owned_dofs,
276 locally_relevant_dofs,
277 mpi_communicator);
278
279 evaluation_point.reinit(locally_owned_dofs,
280 locally_relevant_dofs,
281 mpi_communicator);
282 system_rhs.reinit(locally_owned_dofs, mpi_communicator);
283 newton_update.reinit(locally_owned_dofs, mpi_communicator);
284 local_evaluation_point.reinit(locally_owned_dofs, mpi_communicator);
285
286 define_constraints();
287
288 DynamicSparsityPattern dsp(locally_relevant_dofs);
289 DoFTools::make_sparsity_pattern(*dof_handler,
290 dsp,
291 nonzero_constraints,
292 /*keep_constrained_dofs = */ true);
293 SparsityTools::distribute_sparsity_pattern(dsp,
294 locally_owned_dofs,
295 mpi_communicator,
296 locally_relevant_dofs);
297 system_matrix.reinit(locally_owned_dofs,
298 locally_owned_dofs,
299 dsp,
300 mpi_communicator);
301
302 this->pcout << " Number of solid phase degrees of freedom: "
303 << dof_handler->n_dofs() << std::endl;
304
305 // Share the solid phase fields with the other physics (e.g. VANS)
306 multiphysics->set_dof_handler(PhysicsID::continuum_solid_dynamics,
307 dof_handler);
308 multiphysics->set_mapping(PhysicsID::continuum_solid_dynamics, mapping);
309 multiphysics->set_solution(PhysicsID::continuum_solid_dynamics,
310 present_solution);
311 multiphysics->set_previous_solutions(PhysicsID::continuum_solid_dynamics,
312 previous_solutions);
313}
